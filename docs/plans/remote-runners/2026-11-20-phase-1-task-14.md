# Remote Runners — Phase 1 Task 14: Trace placement fields (RR-16 phase 1)

**Goal:** Every node-execution trace record shows which runner and credential source ran it and why, with the full per-attempt placement history. This includes failed attempts and the self-written first turn of an interactive node, and it is exposed identically over REST and MCP.

**Architecture:**
- Migration `042_node_execution_placement` adds four nullable `node_executions` columns. It also widens the `notifications.type` CHECK to admit P1 T13's `source_logged_out`.
- `instance.NodeExecution` gains `RunnerID`/`SourceID`/`PlacementReason`/`PlacementHistory`.
- `Engine.executeWithRetry` and `InteractiveExecutor.Execute` copy the attempt's `engine.PlacementSink` facts through one engine helper, `applyPlacement`.
- `NodeExecutionRepo` writes and reads the columns with hand-written SQL around the sqlc-generated queries. The REST DTO adds the fields; the MCP trace tools marshal `instance.NodeExecution` directly, so they gain the fields through its json tags.

**Tech Stack:** Go 1.25, SQLite (modernc), hand-written SQL, testify, chi, mcp-go.

	if err != nil {
		return nil, fmt.Errorf("listing node executions: %w", err)

	executions := make([]instance.NodeExecution, 0, len(rows))

// placementRow is the hand-written read of migration 042's columns; the
// sqlc-generated queries predate them and stay unchanged.
type placementRow struct {
	runnerID, sourceID, reason sql.NullString
	history                    sql.NullString
}

func (p placementRow) applyTo(exec *instance.NodeExecution) {
	exec.RunnerID = nullStringToPtr(p.runnerID)
	exec.SourceID = nullStringToPtr(p.sourceID)
	exec.PlacementReason = nullStringToPtr(p.reason)
	if p.history.Valid && p.history.String != "" {
		var hist []instance.PlacementRecord
		if err := json.Unmarshal([]byte(p.history.String), &hist); err == nil {
			exec.PlacementHistory = hist
		}
	}
}

// writePlacement persists the placement columns of a freshly created row. A
// row with no placement (non-dispatching node) issues no statement.
func (r *NodeExecutionRepo) writePlacement(ctx context.Context, exec *instance.NodeExecution) error {
	if exec.RunnerID == nil && exec.SourceID == nil && exec.PlacementReason == nil && len(exec.PlacementHistory) == 0 {
		return nil
	}
	var hist sql.NullString
	if len(exec.PlacementHistory) > 0 {
		b, err := json.Marshal(exec.PlacementHistory)
		if err != nil {
			return fmt.Errorf("marshalling placement history: %w", err)
		}
		hist = sql.NullString{String: string(b), Valid: true}
	}
	if _, err := r.db.ExecContext(ctx,
		`UPDATE node_executions SET runner_id = ?, source_id = ?, placement_reason = ?, placement_history = ? WHERE id = ?`,
		ptrToNullString(exec.RunnerID), ptrToNullString(exec.SourceID), ptrToNullString(exec.PlacementReason), hist, uuidToString(exec.ID),
	); err != nil {
		return fmt.Errorf("writing node execution placement: %w", err)
	}
	return nil
}

// readPlacements loads placement columns for the rows matching where/arg,
// keyed by node-execution id. Rows without any placement are omitted.
func (r *NodeExecutionRepo) readPlacements(ctx context.Context, where string, arg string) (map[uuid.UUID]placementRow, error) {
	rows, err := r.db.QueryContext(ctx,
		`SELECT id, runner_id, source_id, placement_reason, placement_history FROM node_executions
		 WHERE `+where+` AND (runner_id IS NOT NULL OR placement_history IS NOT NULL)`, arg)
	if err != nil {
		return nil, fmt.Errorf("reading node execution placement: %w", err)
	}
	defer rows.Close()
	out := map[uuid.UUID]placementRow{}
	for rows.Next() {
		var idStr string
		var p placementRow
		if err := rows.Scan(&idStr, &p.runnerID, &p.sourceID, &p.reason, &p.history); err != nil {
			return nil, fmt.Errorf("scanning node execution placement: %w", err)
		}
		id, err := parseUUID(idStr)
		if err != nil {
			return nil, fmt.Errorf("parsing node execution id: %w", err)
		}
		out[id] = p
	}
	if err := rows.Err(); err != nil {
		return nil, fmt.Errorf("iterating node execution placement: %w", err)
	}
	return out, nil
}

package engine

import "github.com/arbitron/arbitron/domain/instance"

// applyPlacement copies an attempt's placement facts onto its trace record:
// the LAST fact names where the attempt finally ran; every fact (re-routes
// after exhaustion included) goes to PlacementHistory. No facts = no change.
func applyPlacement(exec *instance.NodeExecution, facts []PlacementFact) {
	if len(facts) == 0 {
		return
	}
	last := facts[len(facts)-1]
	runner := last.RunnerID
	exec.RunnerID = &runner
	exec.SourceID = nil
	if last.SourceID != "" {
		src := last.SourceID
		exec.SourceID = &src
	}
	if last.Reason != "" {
		reason := last.Reason
		exec.PlacementReason = &reason
	}
	hist := make([]instance.PlacementRecord, 0, len(facts))
	for _, f := range facts {
		hist = append(hist, instance.PlacementRecord{
			JobID: f.JobID.String(), RunnerID: f.RunnerID, SourceID: f.SourceID, Reason: f.Reason,
			Outcome: f.Outcome, Generation: f.Generation, At: f.At,
		})
	}
	exec.PlacementHistory = hist
}

// placementRecordResponse mirrors instance.PlacementRecord with the API time format.
type placementRecordResponse struct {
	JobID      string `json:"jobId"`
	RunnerID   string `json:"runnerId"`
	SourceID   string `json:"sourceId,omitempty"`
	Reason     string `json:"reason"`
	Outcome    string `json:"outcome"`
	Generation int64  `json:"generation"`
	At         string `json:"at"`
}

func toNodeExecutionResponse(exec *instance.NodeExecution) nodeExecutionResponse {
	resp := nodeExecutionResponse{
		ID:              exec.ID.String(),
		InstanceID:      exec.InstanceID.String(),
		NodeID:          exec.NodeID,
		Attempt:         exec.Attempt,
		Status:          string(exec.Status),
		ResolvedInputs:  exec.ResolvedInputs,
		Output:          exec.Output,
		Error:           exec.Error,
		LLMConversation: exec.LLMConversation,
		LLMModel:        exec.LLMModel,
		LLMTokensIn:     exec.LLMTokensIn,
		LLMTokensOut:    exec.LLMTokensOut,
		ScriptStdout:    exec.ScriptStdout,
		ScriptStderr:    exec.ScriptStderr,
		ScriptExitCode:  exec.ScriptExitCode,
		StartedAt:       exec.StartedAt.Format(APITimeFormat),
		RunnerID:        exec.RunnerID,
		SourceID:        exec.SourceID,
		PlacementReason: exec.PlacementReason,
	}
	if exec.CompletedAt != nil {
		s := exec.CompletedAt.Format(APITimeFormat)
		resp.CompletedAt = &s
	}
	for _, p := range exec.PlacementHistory {
		resp.PlacementHistory = append(resp.PlacementHistory, placementRecordResponse{
			JobID: p.JobID, RunnerID: p.RunnerID, SourceID: p.SourceID, Reason: p.Reason,
			Outcome: p.Outcome, Generation: p.Generation, At: p.At.UTC().Format(APITimeFormat),
		})
	}
	return resp
}

## Alignment acceptance requirements (2026-11-20)

Trace provenance includes follow-up turns and group-resolution first/follow-up turns. Add a fresh PlacementSink in those entry points (P1 T2 owns identity), copy facts to the actual session record on success **and failure**, and append operation history rather than replacing the first turn's placement history. Add `TestConversation_FollowupAppendsPlacement`, `TestGroupResolution_FirstTurnCarriesPlacement` and `TestGroupResolution_FollowupPlacement` before these writes. Failed turn facts are observable while failed user/assistant text remains uncommitted. `applyPlacement` must clear stale SourceID when a latest fact has no credential source.

`NodeExecutionRepo.Create` must atomically write the base trace and placement columns using one transaction (`r.queries.WithTx(tx)` plus placement update on the same `tx`), or one INSERT containing all columns. `Update` and `UpdateIfSessionOpen` must persist appended placement history in the same statement/transaction as their existing guarded session mutation. Keep the existing endedAt CAS: a concurrently closed session cannot receive placement or messages from a stale turn. Add injected placement-write failure before implementation and assert no half-created trace row or partial session mutation remains. The engine must propagate trace persistence failures (P1 T2); Phase 3 owns coupling this to checkpoint/continuation commit.
