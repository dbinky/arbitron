# Remote Runners — Phase 5 Task 6: Queue Runner column and execution-status badges

**Goal:** Show in the queue where each run's work executes and why it waits. Concretely:
- A **Runner** column: the active operation's runner first, then the workspace home and the interactive-session pin as separate labels. Simultaneous operations (parallel branch or child work reported on the same row) stay distinct.
- An execution-status badge next to the run state: queued · slot; waiting · capacity with its reason, exact model and countdown; moving workspace with progress; recovering.

**Architecture:**
- **Data source.** Rows already carry `EnrichedInstanceSummary.execution` (Task 1 additive field) from the REST list.
- **Pure helpers.** `components/executionStatus.ts` holds the helpers: ordering, runner summary, badge text and row patching.
- **Presentation.** `RunnerCell` and `ExecutionStatusBadge` render those helpers. `queueColumns.ts` declares the new column, and `QueueTableRow` renders both new cells.
- **Live updates.** `QueueView` patches rows live from `execution.status` events, ordered per job by the server timestamp, and prunes them on node completion. It reuses `upsertExecution`/`pruneExecutions` from Task 2.
- **Unchanged.** Instance state itself is unchanged: capacity, slot, move and recovery are separate projections, not new `InstanceState` values.

**Tech Stack:** React 19, CSS modules, Vitest 4 + @testing-library/react.
**Spec / Design:** spec "Administration and visibility" (Queue: execution location and distinct states for capacity waits, slot waits, workspace moves with progress, and recovery); design phase 5 §3.3 (Queue), §5 rows "Active/queued operations…", "Run state and current-node baseline", "Per-operation wait/move/recovery state/reason/reset/progress", §8 (success "The queue shows a waiting · capacity badge with its reason"; read-model regressions "active chat runner differs visibly from workspace home", "simultaneous branch assignments remain distinct"). Contracts: `docs/plans/remote-runners/2026-11-20-contracts.md` §10 (`ExecutionStatus` json: `runId, nodeId, operationId, jobId, state, runnerId, sourceId, group, model, reason, nextReset, progress{bytesDone,bytesTotal}, homeRunnerId, sessionPinRunnerId`), §9 (`execution.status`), §18.

```ts
import type { LiveEvent } from '../api/runnerDecode'
import type { EnrichedInstanceSummary, ExecutionState, ExecutionStatus, MoveProgress } from '../api/types'
import { pruneExecutions, upsertExecution } from '../hooks/poolState'
import { formatCountdown } from '../utils/countdown'

const STATE_ORDER: Record<ExecutionState, number> = {
  running: 0,
  moving_workspace: 1,
  recovering: 2,
  waiting_capacity: 3,
  queued_slot: 4,
}

/** Active operations first, then waits; ties by node id so parallel branches have a stable order. */
export function orderedStatuses(list: ExecutionStatus[] | undefined): ExecutionStatus[] {
  return [...(list ?? [])].sort((a, b) => STATE_ORDER[a.state] - STATE_ORDER[b.state] || a.nodeId.localeCompare(b.nodeId))
}

export interface ActiveAssignment {
  jobId: string
  nodeId: string
  runnerId: string | null
  state: ExecutionState
}
export interface RunnerSummary {
  active: ActiveAssignment[]
  /** Workspace home: may differ from where a hermetic chat step executes. */
  home: string | null
  /** Interactive-session pin. */
  pin: string | null
}

export function runnerSummary(list: ExecutionStatus[] | undefined): RunnerSummary {
  const ordered = orderedStatuses(list)
  return {
    active: ordered.map((s) => ({ jobId: s.jobId, nodeId: s.nodeId, runnerId: s.runnerId ?? null, state: s.state })),
    home: ordered.find((s) => s.homeRunnerId)?.homeRunnerId ?? null,
    pin: ordered.find((s) => s.sessionPinRunnerId)?.sessionPinRunnerId ?? null,
  }
}

/** Accessor/sort text for the Runner column: the active operation's runner. */
export function runnerCellText(row: EnrichedInstanceSummary): string {
  return orderedStatuses(row.execution).find((s) => s.runnerId)?.runnerId ?? '—'
}

export function humanBytes(n: number): string {
  if (n < 1024) return `${n} B`
  if (n < 1024 ** 2) return `${(n / 1024).toFixed(1)} KiB`
  if (n < 1024 ** 3) return `${(n / 1024 ** 2).toFixed(1)} MiB`
  return `${(n / 1024 ** 3).toFixed(1)} GiB`
}

export function moveProgressText(p: MoveProgress | undefined): string {
  if (!p || p.bytesTotal <= 0) return 'progress unknown'
  const done = Math.min(p.bytesDone, p.bytesTotal)
  return `${Math.floor((done / p.bytesTotal) * 100)}% (${humanBytes(p.bytesDone)} of ${humanBytes(p.bytesTotal)})`
}

/** Badge text for a non-running execution state; null when the run state alone says it. */
export function executionBadgeLabel(s: ExecutionStatus, now: number): string | null {
  switch (s.state) {
    case 'queued_slot':
      return 'queued · slot'
    case 'waiting_capacity': {
      const detail = [s.reason, s.model ? `model ${s.model}` : ''].filter(Boolean).join(' · ')
      return `waiting · capacity${detail ? ` — ${detail}` : ''} · ${formatCountdown(s.nextReset, now)}`
    }
    case 'moving_workspace':
      return `moving workspace · ${moveProgressText(s.progress)}`
    case 'recovering':
      return 'recovering'
    default:
      return null
  }
}

/** Applies a decoded execution/node event to the rows that show it; unrelated rows keep identity. */
export function patchRowExecution(rows: EnrichedInstanceSummary[], live: LiveEvent): EnrichedInstanceSummary[] {
  if (live.kind === 'execution') {
    const idx = rows.findIndex((r) => r.id === live.status.runId)
    if (idx === -1) return rows
    const next = [...rows]
    next[idx] = { ...rows[idx], execution: upsertExecution(rows[idx].execution ?? [], live.status) }
    return next
  }
  if (live.kind === 'node_done') {
    const idx = rows.findIndex((r) => r.id === live.instanceId)
    const current = idx === -1 ? undefined : rows[idx].execution
    if (!current) return rows
    const pruned = pruneExecutions(current, live.instanceId, live.nodeId)
    if (pruned === current) return rows
    const next = [...rows]
    next[idx] = { ...rows[idx], execution: pruned }
    return next
  }
  return rows
}
```

```tsx
/** Execution-status badge beside the run state (capacity/slot/move/recovery are not InstanceState values). */
export default function ExecutionStatusBadge({ status, now, showNode }: { status: ExecutionStatus; now: number; showNode: boolean }) {

  )
}
```

```tsx
          {a.runnerId ?? 'unassigned'}
          {many ? ` (${a.nodeId})` : ''}
        </span>
      ))}
      {home && (
        <span className={styles.secondary} data-home-runner={home}>
          home {home}
        </span>
      )}
      {pin && (
        <span className={styles.secondary} data-pin-runner={pin}>
          pinned {pin}
        </span>
      )}
    </span>
  )
}
```
