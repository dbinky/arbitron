**Goal:** Show in the queue where each run's work executes and why it waits. Concretely:
- A **Runner** column: the active operation's runner first, then the workspace home and the interactive-session pin as separate labels. Simultaneous operations (parallel branch or child work reported on the same row) stay distinct.
- An execution-status badge next to the run state: queued · slot; waiting · capacity with its reason, exact model and countdown; moving workspace with progress; recovering.

**Architecture:**
- **Data source.** Rows already carry `EnrichedInstanceSummary.execution` (Task 1 additive field) from the REST list.
- **Pure helpers.** `components/executionStatus.ts` holds the helpers: ordering, runner summary and badge text. (Row patching from events is retired, unstick 2026-11-23.)
- **Presentation.** `RunnerCell` and `ExecutionStatusBadge` render those helpers. `queueColumns.ts` declares the new column, and `QueueTableRow` renders both new cells.
- **Live updates.** `QueueView` treats `execution.status`, node completion and run-scoped pool events as invalidations: it serializes `fetchAll` (one active read per `showDismissed` epoch plus one dirty flag, P5 T2 approach) and renders only the authoritative rows. Events never patch, prune or add rows. A refetch error keeps the last good rows with an inline alert.
- **Unchanged.** Instance state itself is unchanged: capacity, slot, move and recovery are separate projections, not new `InstanceState` values.

```ts
/** Badge text for a non-running execution state; null when the run state alone says it. */
export function executionBadgeLabel(s: ExecutionStatus, now: number): string | null {
  switch (s.state) {
    case 'queued_slot':
      return 'queued · slot'
    case 'waiting_capacity': {
      const detail = [s.reason, s.model ? `model ${s.model}` : ''].filter(Boolean).join(' · ')
      return `waiting · capacity${detail ? ` — ${detail}` : ''} · ${formatCountdown(s.nextReset, now)}`
    }
    case 'moving_workspace':
      return `moving workspace · ${moveProgressText(s.progress)}`
    case 'recovering':
      return 'recovering'
    default:
      return null
  }
}
```

```tsx
import type { ExecutionStatus } from '../api/types'
import { executionBadgeLabel } from './executionStatus'
import styles from './ExecutionStatus.module.css'

/** Execution-status badge beside the run state (capacity/slot/move/recovery are not InstanceState values). */
export default function ExecutionStatusBadge({ status, now, showNode }: { status: ExecutionStatus; now: number; showNode: boolean }) {
  const label = executionBadgeLabel(status, now)
  if (label === null) return null
  return (
    <span className={`${styles.badge} ${styles[status.state] ?? ''}`} data-exec-badge={status.state} title={status.reason ?? label}>
      {showNode ? `${status.nodeId}: ${label}` : label}
    </span>
  )
}
```

```tsx
import type { ExecutionStatus } from '../api/types'
import { runnerSummary } from './executionStatus'
import styles from './ExecutionStatus.module.css'

/**
 * The active operation's runner first (one line per simultaneous operation),
 * then the workspace home and the interactive pin as separate labels.
 */
export default function RunnerCell({ execution }: { execution: ExecutionStatus[] | undefined }) {
  const { active, home, pin } = runnerSummary(execution)
  if (active.length === 0 && !home && !pin) return <>—</>
  const many = active.length > 1
  return (
    <span className={styles.runnerCell}>
      {active.map((a) => (
        <span key={a.jobId} data-exec-runner={a.runnerId ?? ''} data-node={a.nodeId} title={`${a.nodeId}: ${a.state}`}>
          {a.runnerId ?? 'unassigned'}
          {many ? ` (${a.nodeId})` : ''}
        </span>
      ))}
      {home && (
        <span className={styles.secondary} data-home-runner={home}>
          home {home}
        </span>
      )}
      {pin && (
        <span className={styles.secondary} data-pin-runner={pin}>
          pinned {pin}
        </span>
      )}
    </span>
  )
}
```
