# Remote Runners — Phase 3 Task 3: Workspace homes and the single-writer resource lease

**Goal:** Implement `runner.HomeStore` over SQLite so every run's workspace has exactly one owning runner (its home) guarded by an expected-generation CAS, with a move history row for every handoff, and expose homes in `Registry.Snapshot`.
**Architecture:** `workspace_homes` (migration 041, P1 T5) holds the current owner/generation/checkpoint version/dirty flag; `home_moves` (044, P3 T1) is the append-only history. Each mutation is one SQLite transaction (`SetMaxOpenConns(1)` ⇒ single writer) with a `WHERE generation = ?` CAS. The repo never reacts to liveness: a suspect home keeps ownership until a caller (dispatcher/recovery) hands it off after loss. Events (`workspace.moved`) are emitted by the sync service (P3 T6), not here.
**Tech Stack:** Go 1.25.5, hand-written SQL (pattern `adapter/sqlite/audit_repo.go`), modernc SQLite, testify.
**Spec / Design:** spec "Recovery and session continuity" (one active execution owner); design phase 3 §3.1 (homes, generation, history, exactly one writer, suspect keeps ownership), §5 row "Home/resource identity, lease/generation, ownership and move history" (RR-08), §8 ("two workspace writers cannot overlap", "suspect homes do not move before loss"). Contracts: `docs/plans/remote-runners/2026-11-20-contracts.md` §16 (Homes), §7 (`Home`, `PoolSnapshot.Homes`), §8 (`Registry.Snapshot`).

```go
	GetHome(ctx context.Context, runID uuid.UUID) (Home, error) // no row => Home{RunID: runID} (server checkpoint only), nil error
	AcquireResource(ctx context.Context, runID uuid.UUID, owner RunnerID, expectGen int64) (Home, error)
	HandOff(ctx context.Context, runID uuid.UUID, to RunnerID, reason string, bytes int64, expectGen int64) (Home, error)
	SetCheckpoint(ctx context.Context, runID uuid.UUID, version int64) error // monotonic (never lowers)

	ListByRunner(ctx context.Context, id RunnerID) ([]Home, error)

}

// NEW
var ErrHomeGenerationConflict = errors.New("runner: workspace home generation changed (another writer won)")
var ErrHomeOwned = errors.New("runner: workspace home is owned by another runner; hand it off first")
const ServerHome RunnerID = "" // readability alias for "no runner holds a copy"
```
Semantics:
- `AcquireResource(owner)`: current owner == `owner` ⇒ returns the home unchanged (idempotent, no bump). No row or owner `""` ⇒ becomes `owner`, generation+1, history `{From:"", To: owner, Reason:"acquire"}`. Different owner ⇒ `ErrHomeOwned`. Generation ≠ `expectGen` ⇒ `ErrHomeGenerationConflict`.
- `HandOff(to)`: CAS on `expectGen`; `to` == current owner ⇒ unchanged; otherwise owner=`to`, generation+1, `dirty=0` (the new owner starts from the committed checkpoint), history row with reason/bytes.
- `adapter/sqlite/workspace_home_repo.go:WorkspaceHomeRepo` — `func NewWorkspaceHomeRepo(db *sql.DB) *WorkspaceHomeRepo`; package helper `func loadHomes(ctx context.Context, q interface{ QueryContext(context.Context, string, ...any) (*sql.Rows, error) }) (map[uuid.UUID]runner.Home, error)` used by `RunnerRegistryRepo.Snapshot`.

```go
func TestWorkspaceHomeRepo_AcquireHandOffHistory(t *testing.T) {
	ctx := context.Background()
	repo := NewWorkspaceHomeRepo(setupTestDB(t).DB)
	run := uuid.New()

	h, err := repo.GetHome(ctx, run)
	require.NoError(t, err)
	assert.Equal(t, runner.Home{RunID: run}, h, "no row = server checkpoint only")

	h, err = repo.AcquireResource(ctx, run, runner.BuiltInRunnerID, 0)
	require.NoError(t, err)
	assert.Equal(t, runner.BuiltInRunnerID, h.RunnerID)
	assert.Equal(t, int64(1), h.Generation)

	same, err := repo.AcquireResource(ctx, run, runner.BuiltInRunnerID, 1)
	require.NoError(t, err)
	assert.Equal(t, int64(1), same.Generation, "re-acquire by the owner does not bump")

	_, err = repo.AcquireResource(ctx, run, "mac-1", 1)
	require.ErrorIs(t, err, runner.ErrHomeOwned, "another runner must hand off, not acquire")

	_, err = repo.HandOff(ctx, run, "mac-1", "should_move", 4096, 0)
	require.ErrorIs(t, err, runner.ErrHomeGenerationConflict, "stale generation refused")

	require.NoError(t, repo.MarkDirty(ctx, run, true))
	h, err = repo.HandOff(ctx, run, "mac-1", "must_move: source exhausted", 4096, 1)
	require.NoError(t, err)
	assert.Equal(t, runner.RunnerID("mac-1"), h.RunnerID)
	assert.Equal(t, int64(2), h.Generation)
	assert.False(t, h.Dirty, "the new owner starts from the committed checkpoint")

	require.NoError(t, repo.Release(ctx, run, 2))
	h, err = repo.GetHome(ctx, run)
	require.NoError(t, err)
	assert.Equal(t, runner.ServerHome, h.RunnerID)
	assert.Equal(t, int64(3), h.Generation)

	hist, err := repo.History(ctx, run)
	require.NoError(t, err)
	require.Len(t, hist, 3)
	assert.Equal(t, []runner.RunnerID{"", runner.BuiltInRunnerID, "mac-1"}, []runner.RunnerID{hist[0].From, hist[1].From, hist[2].From})
	assert.Equal(t, []runner.RunnerID{runner.BuiltInRunnerID, "mac-1", ""}, []runner.RunnerID{hist[0].To, hist[1].To, hist[2].To})
	assert.Equal(t, "must_move: source exhausted", hist[1].Reason)
	assert.Equal(t, int64(4096), hist[1].Bytes)
	assert.Equal(t, []int64{1, 2, 3}, []int64{hist[0].Generation, hist[1].Generation, hist[2].Generation})
	assert.False(t, hist[1].At.IsZero())
}

// Two workspace writers cannot overlap: N concurrent handoffs with the same
// expected generation produce exactly one winner (file-backed DB, real tx).
func TestWorkspaceHomeRepo_ConcurrentHandOffSingleWinner(t *testing.T) {
	ctx := context.Background()
	db, err := Open(filepath.Join(t.TempDir(), "homes.db"))
	require.NoError(t, err)
	t.Cleanup(func() { db.Close() })
	require.NoError(t, RunMigrations(db))
	repo := NewWorkspaceHomeRepo(db)
	run := uuid.New()
	_, err = repo.AcquireResource(ctx, run, runner.BuiltInRunnerID, 0)
	require.NoError(t, err)

	const writers = 12
	var wg sync.WaitGroup
	var mu sync.Mutex
	wins, conflicts := 0, 0
	for i := 0; i < writers; i++ {
		wg.Add(1)
		go func(i int) {
			defer wg.Done()
			_, err := repo.HandOff(ctx, run, runner.RunnerID(fmt.Sprintf("mac-%d", i)), "should_move", 0, 1)
			mu.Lock()
			defer mu.Unlock()
			switch {
			case err == nil:
				wins++
			case errors.Is(err, runner.ErrHomeGenerationConflict):
				conflicts++
			default:
				t.Errorf("unexpected error: %v", err)
			}
		}(i)
	}
	wg.Wait()
	assert.Equal(t, 1, wins)
	assert.Equal(t, writers-1, conflicts)
	h, err := repo.GetHome(ctx, run)
	require.NoError(t, err)
	assert.Equal(t, int64(2), h.Generation)
	hist, err := repo.History(ctx, run)
	require.NoError(t, err)
	assert.Len(t, hist, 2, "exactly one handoff recorded")
}
```

```go
package runner

import (
	"context"
	"errors"
	"time"

	"github.com/google/uuid"
)

// ServerHome is the RunnerID of a workspace no runner holds a copy of: only the
// committed server checkpoint exists.
const ServerHome RunnerID = ""

// HomeMove is one entry of a workspace's home history.
type HomeMove struct {
	RunID      uuid.UUID `json:"runId"`
	From       RunnerID  `json:"from"`
	To         RunnerID  `json:"to"`
	Reason     string    `json:"reason"`
	Bytes      int64     `json:"bytes"`
	Generation int64     `json:"generation"`
	At         time.Time `json:"at"`
}

// HomeStore is the durable single-writer resource lease over a run's
// workspace. Every ownership change is a compare-and-swap on Generation and
// appends one HomeMove.
type HomeStore interface {
	GetHome(ctx context.Context, runID uuid.UUID) (Home, error)
	AcquireResource(ctx context.Context, runID uuid.UUID, owner RunnerID, expectGen int64) (Home, error)
	HandOff(ctx context.Context, runID uuid.UUID, to RunnerID, reason string, bytes int64, expectGen int64) (Home, error)
	SetCheckpoint(ctx context.Context, runID uuid.UUID, version int64) error
	MarkDirty(ctx context.Context, runID uuid.UUID, dirty bool) error
	History(ctx context.Context, runID uuid.UUID) ([]HomeMove, error)
	ListByRunner(ctx context.Context, id RunnerID) ([]Home, error)
	Release(ctx context.Context, runID uuid.UUID, expectGen int64) error
}

var (
	ErrHomeGenerationConflict = errors.New("runner: workspace home generation changed (another writer won)")
	ErrHomeOwned              = errors.New("runner: workspace home is owned by another runner; hand it off first")
)
```

```go
// adapter/sqlite/workspace_home_repo.go
package sqlite

import (
	"context"
	"database/sql"
	"errors"
	"fmt"
	"time"

	"github.com/arbitron/arbitron/domain/runner"
	"github.com/google/uuid"
)

var _ runner.HomeStore = (*WorkspaceHomeRepo)(nil)

// WorkspaceHomeRepo implements runner.HomeStore over workspace_homes (041) and
// home_moves (044). Hand-written SQL; every mutation is one transaction.
type WorkspaceHomeRepo struct {
	db  *sql.DB
	now func() time.Time
}

// NewWorkspaceHomeRepo creates the repo.
func NewWorkspaceHomeRepo(db *sql.DB) *WorkspaceHomeRepo {
	return &WorkspaceHomeRepo{db: db, now: func() time.Time { return time.Now().UTC() }}
}

type homeQueryer interface {
	QueryRowContext(ctx context.Context, query string, args ...any) *sql.Row
}

func getHome(ctx context.Context, q homeQueryer, runID uuid.UUID) (runner.Home, bool, error) {
	var owner string
	var gen, ver int64
	var dirty int
	err := q.QueryRowContext(ctx,
		`SELECT runner_id, generation, checkpoint_version, dirty FROM workspace_homes WHERE run_id = ?`,
		uuidToString(runID)).Scan(&owner, &gen, &ver, &dirty)
	if errors.Is(err, sql.ErrNoRows) {
		return runner.Home{RunID: runID}, false, nil
	}
	if err != nil {
		return runner.Home{}, false, fmt.Errorf("workspace home: get: %w", err)
	}
	return runner.Home{RunID: runID, RunnerID: runner.RunnerID(owner), Generation: gen, CheckpointVersion: ver, Dirty: dirty != 0}, true, nil
}

func (r *WorkspaceHomeRepo) GetHome(ctx context.Context, runID uuid.UUID) (runner.Home, error) {
	h, _, err := getHome(ctx, r.db, runID)
	return h, err
}

func (r *WorkspaceHomeRepo) AcquireResource(ctx context.Context, runID uuid.UUID, owner runner.RunnerID, expectGen int64) (runner.Home, error) {
	return r.transfer(ctx, runID, owner, "acquire", 0, expectGen, true)
}

func (r *WorkspaceHomeRepo) HandOff(ctx context.Context, runID uuid.UUID, to runner.RunnerID, reason string, bytes int64, expectGen int64) (runner.Home, error) {
	return r.transfer(ctx, runID, to, reason, bytes, expectGen, false)
}

func (r *WorkspaceHomeRepo) Release(ctx context.Context, runID uuid.UUID, expectGen int64) error {
	_, err := r.transfer(ctx, runID, runner.ServerHome, "release", 0, expectGen, false)
	return err
}

// transfer is the single CAS path. acquire=true refuses to take a home another
// runner owns (ErrHomeOwned); acquire=false (hand-off) moves it.
func (r *WorkspaceHomeRepo) transfer(ctx context.Context, runID uuid.UUID, to runner.RunnerID, reason string, bytes int64, expectGen int64, acquire bool) (runner.Home, error) {
	tx, err := r.db.BeginTx(ctx, nil)
	if err != nil {
		return runner.Home{}, fmt.Errorf("workspace home: begin: %w", err)
	}
	defer tx.Rollback()
	cur, exists, err := getHome(ctx, tx, runID)
	if err != nil {
		return runner.Home{}, err
	}
	if cur.Generation != expectGen {
		return runner.Home{}, fmt.Errorf("%w: run %s at generation %d, caller expected %d", runner.ErrHomeGenerationConflict, runID, cur.Generation, expectGen)
	}
	if cur.RunnerID == to && exists {
		return cur, nil
	}
	if acquire && cur.RunnerID != runner.ServerHome && cur.RunnerID != to {
		return runner.Home{}, fmt.Errorf("%w: run %s is homed on %s", runner.ErrHomeOwned, runID, cur.RunnerID)
	}
	next := cur.Generation + 1
	if !exists {
		if _, err := tx.ExecContext(ctx,
			`INSERT INTO workspace_homes (run_id, runner_id, generation, checkpoint_version, dirty) VALUES (?, ?, ?, 0, 0)`,
			uuidToString(runID), string(to), next); err != nil {
			return runner.Home{}, fmt.Errorf("workspace home: insert: %w", err)
		}
	} else {
		res, err := tx.ExecContext(ctx,
			`UPDATE workspace_homes SET runner_id = ?, generation = ?, dirty = 0 WHERE run_id = ? AND generation = ?`,
			string(to), next, uuidToString(runID), expectGen)
		if err != nil {
			return runner.Home{}, fmt.Errorf("workspace home: update: %w", err)
		}
		if n, _ := res.RowsAffected(); n != 1 {
			return runner.Home{}, fmt.Errorf("%w: run %s", runner.ErrHomeGenerationConflict, runID)
		}
	}
	if _, err := tx.ExecContext(ctx,
		`INSERT INTO home_moves (run_id, from_runner, to_runner, reason, bytes, generation, moved_at) VALUES (?, ?, ?, ?, ?, ?, ?)`,
		uuidToString(runID), string(cur.RunnerID), string(to), reason, bytes, next, timeToString(r.now())); err != nil {
		return runner.Home{}, fmt.Errorf("workspace home: history: %w", err)
	}
	if err := tx.Commit(); err != nil {
		return runner.Home{}, fmt.Errorf("workspace home: commit: %w", err)
	}
	return runner.Home{RunID: runID, RunnerID: to, Generation: next, CheckpointVersion: cur.CheckpointVersion, Dirty: false}, nil
}

func (r *WorkspaceHomeRepo) SetCheckpoint(ctx context.Context, runID uuid.UUID, version int64) error {
	_, err := r.db.ExecContext(ctx,
		`INSERT INTO workspace_homes (run_id, runner_id, generation, checkpoint_version, dirty) VALUES (?, '', 0, ?, 0)
		 ON CONFLICT(run_id) DO UPDATE SET checkpoint_version = MAX(checkpoint_version, excluded.checkpoint_version)`,
		uuidToString(runID), version)
	if err != nil {
		return fmt.Errorf("workspace home: set checkpoint: %w", err)
	}
	return nil
}

func (r *WorkspaceHomeRepo) MarkDirty(ctx context.Context, runID uuid.UUID, dirty bool) error {
	d := 0
	if dirty {
		d = 1
	}
	_, err := r.db.ExecContext(ctx,
		`INSERT INTO workspace_homes (run_id, runner_id, generation, checkpoint_version, dirty) VALUES (?, '', 0, 0, ?)
		 ON CONFLICT(run_id) DO UPDATE SET dirty = excluded.dirty`,
		uuidToString(runID), d)
	if err != nil {
		return fmt.Errorf("workspace home: mark dirty: %w", err)
	}
	return nil
}

func (r *WorkspaceHomeRepo) History(ctx context.Context, runID uuid.UUID) ([]runner.HomeMove, error) {
	rows, err := r.db.QueryContext(ctx,
		`SELECT from_runner, to_runner, reason, bytes, generation, moved_at FROM home_moves WHERE run_id = ? ORDER BY generation`,
		uuidToString(runID))
	if err != nil {
		return nil, fmt.Errorf("workspace home: history: %w", err)
	}
	defer rows.Close()
	var out []runner.HomeMove
	for rows.Next() {
		var from, to, reason, at string
		var bytes, gen int64
		if err := rows.Scan(&from, &to, &reason, &bytes, &gen, &at); err != nil {
			return nil, err
		}
		t, err := parseTime(at)
		if err != nil {
			return nil, err
		}
		out = append(out, runner.HomeMove{RunID: runID, From: runner.RunnerID(from), To: runner.RunnerID(to), Reason: reason, Bytes: bytes, Generation: gen, At: t})
	}
	return out, rows.Err()
}

func (r *WorkspaceHomeRepo) ListByRunner(ctx context.Context, id runner.RunnerID) ([]runner.Home, error) {
	rows, err := r.db.QueryContext(ctx,
		`SELECT run_id, runner_id, generation, checkpoint_version, dirty FROM workspace_homes WHERE runner_id = ? ORDER BY run_id`,
		string(id))
	if err != nil {
		return nil, fmt.Errorf("workspace home: list: %w", err)
	}
	defer rows.Close()
	return scanHomes(rows)
}

func scanHomes(rows *sql.Rows) ([]runner.Home, error) {
	var out []runner.Home
	for rows.Next() {
		var run, owner string
		var gen, ver int64
		var dirty int
		if err := rows.Scan(&run, &owner, &gen, &ver, &dirty); err != nil {
			return nil, err
		}
		id, err := parseUUID(run)
		if err != nil {
			return nil, err
		}
		out = append(out, runner.Home{RunID: id, RunnerID: runner.RunnerID(owner), Generation: gen, CheckpointVersion: ver, Dirty: dirty != 0})
	}
	return out, rows.Err()
}

// loadHomes reads every home for PoolSnapshot.Homes.
func loadHomes(ctx context.Context, q interface {
	QueryContext(context.Context, string, ...any) (*sql.Rows, error)
}) (map[uuid.UUID]runner.Home, error) {
	rows, err := q.QueryContext(ctx, `SELECT run_id, runner_id, generation, checkpoint_version, dirty FROM workspace_homes`)
	if err != nil {
		return nil, fmt.Errorf("workspace home: snapshot: %w", err)
	}
	defer rows.Close()
	list, err := scanHomes(rows)
	if err != nil {
		return nil, err
```
