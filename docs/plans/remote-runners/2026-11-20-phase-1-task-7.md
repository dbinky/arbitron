# Remote Runners — Phase 1 Task 7: Pool events, notifier, execution-status projection and scoped streaming

**Goal:** Give the pool its observable surfaces: one `EventRecorder` that persists and publishes every pool event, a coalescing `PoolNotifier` that wakes waiting dispatches, an `ExecutionStatusProjector` that exposes per-run queued · slot / waiting · capacity / running status through the instance enricher (REST and MCP lists) and `execution.status` bus events, and a system-stream rule that delivers system-scoped pool events to every authenticated user while run-scoped events stay visibility-filtered.

**Architecture:** `domain/runner/events.go` (recorder + event kinds + `PoolEventView`), `domain/runner/notifier.go` (`PoolNotifier`) and `domain/runner/execution_status.go` (`ExecutionStatusProjector`, reading the ledger port) are pure domain services over ports. `domain/instance/execution_status.go` holds the wire types; `domain/engine/enricher.go` gains the `ExecutionStatusProvider` port (Null Object default, nil rejected) and fills `EnrichedInstanceSummary.Execution` for the already-authorized page, so hidden runs can never appear. `adapter/http/system_stream_handler.go` lets `uuid.Nil`-scoped `runner.`/`source.`/`pool.` events bypass the ownership lookup; everything else (including `execution.status`, always run-scoped) still goes through `canSee`.

**Tech Stack:** Go 1.25.5, `log/slog`, `sync`, testify, `nhooyr.io/websocket` + `adapter/eventbus` in the HTTP test.

**Spec / Design:** spec "Operator visibility" (pool events with cause; run status names the wait), "Security and visibility" (hidden-run data never leaks); design phase 1 §3.9 (event log fed through one recorder), §3.12 (execution-status projection via `InstanceEnricher`, MCP/REST, `execution.status` events; `InstanceState` unchanged — D6), §5 rows for execution status and pool events, §8 Failure "the step waits · capacity naming the group and reset" (surface half), cross-phase "hidden-run data is absent from REST/MCP/events". Contracts: `docs/plans/remote-runners/2026-11-20-contracts.md` §9 (event vocabulary, recorder, system-stream rule), §10 (execution status types, port, projector), §8 (PoolEvent, EventLog, Ledger).

func (r *EventRecorder) Record(ctx context.Context, ev PoolEvent) {

	} else if err := r.log.Prune(ctx, r.retention); err != nil {
		slog.Warn("runner: pool event prune failed", "error", err)
	}
	scope := uuid.Nil
	if ev.RunID != nil {
		scope = *ev.RunID
	}
	r.bus.Publish(scope, engine.Event{Type: ev.Kind, Timestamp: ev.At, Data: NewPoolEventView(ev)})
}

package runner

import "sync"

// PoolNotifier signals "pool state changed" (health, config, slots) to
// waiting dispatches. Each subscriber has a 1-buffered channel, so Notify never
// blocks and bursts coalesce into one pending wake.
type PoolNotifier struct {
	mu   sync.Mutex
	next int
	subs map[int]chan struct{}
}

// NewPoolNotifier returns a notifier with no subscribers.
func NewPoolNotifier() *PoolNotifier { return &PoolNotifier{subs: map[int]chan struct{}{}} }

// Notify wakes every subscriber without blocking.
func (n *PoolNotifier) Notify() {
	n.mu.Lock()
	defer n.mu.Unlock()
	for _, ch := range n.subs {
		select {
		case ch <- struct{}{}:
		default:
		}
	}
}

// Subscribe returns a wake channel and its cancel function.
func (n *PoolNotifier) Subscribe() (<-chan struct{}, func()) {
	n.mu.Lock()
	defer n.mu.Unlock()
	id := n.next
	n.next++
	ch := make(chan struct{}, 1)
	n.subs[id] = ch
	return ch, func() {
		n.mu.Lock()
		defer n.mu.Unlock()
		delete(n.subs, id)
	}
}

// StatusFromJob projects one job in the given state.
func StatusFromJob(j Job, state instance.ExecutionState) instance.ExecutionStatus {
	s := instance.ExecutionStatus{RunID: j.Key.RunID, NodeID: j.NodeID, OperationID: j.Key.OperationID, JobID: j.ID, State: state,
		RunnerID: string(j.RunnerID), SourceID: string(j.SourceID), Group: j.Group.String(), Model: j.Model, Reason: j.Reason}
	if j.Wait != nil && (state == instance.ExecQueuedSlot || state == instance.ExecWaitingCapacity) {
		s.Reason, s.NextReset, s.Group = j.Wait.Detail, j.Wait.NextReset, j.Wait.Group.String()
	}
	return s
}

func stateOf(j Job) (instance.ExecutionState, bool) {
	switch j.State {
	case JobQueued:
		if j.Wait != nil && j.Wait.Kind == WaitCapacity {
			return instance.ExecWaitingCapacity, true
		}
		return instance.ExecQueuedSlot, true
	case JobOffered, JobRunning:
		return instance.ExecRunning, true
	}
	return "", false
}

// ExecutionStatuses implements engine.ExecutionStatusProvider: one ledger read
// for the whole (already authorized) page, filtered to runIDs.
func (p *ExecutionStatusProjector) ExecutionStatuses(ctx context.Context, runIDs []uuid.UUID) (map[uuid.UUID][]instance.ExecutionStatus, error) {
	out := map[uuid.UUID][]instance.ExecutionStatus{}
	if len(runIDs) == 0 {
		return out, nil
	}
	want := make(map[uuid.UUID]bool, len(runIDs))
	for _, id := range runIDs {
		want[id] = true
	}
	jobs, err := p.ledger.ListActive(ctx)
	if err != nil {
		return nil, err
	}
	for _, j := range jobs {
		if !want[j.Key.RunID] {
			continue
		}
		if st, ok := stateOf(j); ok {
			out[j.Key.RunID] = append(out[j.Key.RunID], StatusFromJob(j, st))
		}
	}
	return out, nil
}

// Publish emits execution.status for job, scoped to its run. Without an
// attached bus it is a no-op.
func (p *ExecutionStatusProjector) Publish(_ context.Context, job Job, state instance.ExecutionState) {
	bus := p.attachedBus()
	if bus == nil {
		return
	}
	bus.Publish(job.Key.RunID, engine.Event{Type: EventExecutionStatus, Timestamp: p.now().UTC(), Data: StatusFromJob(job, state)})
}

func dialSystemStream(t *testing.T, url string) (*websocket.Conn, context.Context) {
	t.Helper()
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	t.Cleanup(cancel)
	conn, _, err := websocket.Dial(ctx, "ws"+url[len("http"):]+"/ws", nil)
	require.NoError(t, err)
	t.Cleanup(func() { conn.Close(websocket.StatusNormalClosure, "") })
	time.Sleep(50 * time.Millisecond) // let the handler subscribe
	return conn, ctx
}

func readWire(t *testing.T, conn *websocket.Conn, ctx context.Context) wireEvent {
	t.Helper()
	_, msg, err := conn.Read(ctx)
	require.NoError(t, err)
	var ev wireEvent
	require.NoError(t, json.Unmarshal(msg, &ev))
	return ev
}
