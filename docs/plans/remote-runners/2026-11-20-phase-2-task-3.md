# Remote Runners — Phase 2 Task 3: Runner-key operator surface (MCP + REST) and runner-key refusal on user surfaces

**Goal:** Admins create (once-shown), list and revoke runner keys over MCP (`runner_key_create|list|revoke`) and REST (`GET|POST /api/v1/runners/keys`, `DELETE /api/v1/runners/keys/{id}` with `{"acknowledge":true}`), with identical JSON shapes, no hash ever in a response, and a runner key presented on any user-auth surface (REST, WS, MCP, SPA) refused as unauthenticated in every auth mode.

**Architecture:** Thin handlers over `runner.KeyService` (P2 T2), which owns admin authz and audit. The REST handler sits inside the existing user-auth `/api/v1` group; the MCP tools follow the `tools_token.go` pattern. `AuthMiddleware.Handler` gains one guard that rejects any request carrying a runner key (`Authorization: Runner …` or `Bearer ark_…`) before identity resolution, so local mode's frictionless admin and an org-mode session cookie can never be combined with a runner key.

**Tech Stack:** Go 1.25.5, chi v5, mark3labs/mcp-go, testify, SQLite (real `RunnerKeyRepo`/`RunnerRegistryRepo` in tests).

**Spec / Design:** Spec "Administration and visibility" (unique key shown once with setup guidance; list unclaimed/associated/created/last use; admin-only, audited; revoke confirmation recommending drain); design phase 2 §3.3 (Interfaces: MCP + REST thin over the service; scope: never a user identity), §2 (user auth middleware row), §7 ("A runner key used on REST or MCP is refused as unauthenticated"), §8 Edge-case ("A runner key is refused on REST and MCP") and regression ("runner-key requests cannot reach operator/key REST routes"). Contracts: `docs/plans/remote-runners/2026-11-20-contracts.md` §14 (MCP tool names, `Deps.RunnerKeys`, REST routes, camelCase DTOs), §15 (Keys).

```go
// adapter/http/runner_key_auth.go
package http

import (
	"net/http"
	"strings"

	"github.com/arbitron/arbitron/domain/runner"
)

// RunnerAuthScheme is the only way a runner presents its key:
// "Authorization: Runner ark_…". Keys never travel in query parameters.
const RunnerAuthScheme = "Runner"

// runnerKeyFromHeader extracts a key presented with the Runner scheme.
func runnerKeyFromHeader(h http.Header) (string, bool) {
	scheme, rest, ok := strings.Cut(strings.TrimSpace(h.Get("Authorization")), " ")
	if !ok || !strings.EqualFold(scheme, RunnerAuthScheme) {
		return "", false
	}
	key := strings.TrimSpace(rest)
	return key, key != ""
}

// presentsRunnerKey reports whether the request carries a runner key in any
// Authorization form. User-auth surfaces refuse such requests outright: a
// runner key authorizes only the runner channel and is never a user identity.
func presentsRunnerKey(r *http.Request) bool {
	if _, ok := runnerKeyFromHeader(r.Header); ok {
		return true
	}
	scheme, rest, ok := strings.Cut(strings.TrimSpace(r.Header.Get("Authorization")), " ")
	return ok && strings.EqualFold(scheme, "Bearer") && strings.HasPrefix(strings.TrimSpace(rest), runner.RunnerKeyPrefix)
}
```

```go
func (m *AuthMiddleware) Handler(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		// A runner key authorizes only /api/v1/runners/connect (and Phase 3
		// transfers), which are mounted outside this middleware. On every user
		// surface it is refused in EVERY mode — even local mode's implicit admin
		// and even alongside a valid session cookie.
		if presentsRunnerKey(r) {
			WriteError(w, http.StatusUnauthorized, "UNAUTHORIZED", "runner keys are not accepted on this endpoint")
			return
		}
		id, err := m.resolve(r)
		if err != nil {
			if errors.Is(err, errNoCredential) {
				m.reject(w, r)
				return
			}
			// Server-state failure (e.g. cannot establish the local default
			// identity): this is a 500, not a 401. WriteError logs it.
			WriteError(w, http.StatusInternalServerError, "INTERNAL_ERROR", "could not establish caller identity")
			return
		}
		next.ServeHTTP(w, r.WithContext(identity.WithIdentity(r.Context(), id)))
	})
}
```

```go
func writeRunnerKeyError(w http.ResponseWriter, err error) {
	switch {
	case errors.Is(err, identity.ErrUnauthenticated):
		WriteError(w, http.StatusUnauthorized, "UNAUTHORIZED", "authentication required")
	case errors.Is(err, identity.ErrForbidden):
		WriteError(w, http.StatusForbidden, "FORBIDDEN", "admin role required")
	case errors.Is(err, runner.ErrInvalidKeyName):
		WriteError(w, http.StatusBadRequest, "INVALID_NAME", err.Error())
	case errors.Is(err, runner.ErrKeyNameTaken):
		WriteError(w, http.StatusConflict, "NAME_TAKEN", "a runner key with this name already exists")
	case errors.Is(err, runner.ErrKeyNotFound):
		WriteError(w, http.StatusNotFound, "NOT_FOUND", "runner key not found")
	case errors.Is(err, runner.ErrRevokeNotAcknowledged):
		WriteError(w, http.StatusBadRequest, "ACKNOWLEDGEMENT_REQUIRED", err.Error())
	default:
		WriteError(w, http.StatusInternalServerError, "INTERNAL_ERROR", "runner key operation failed: "+err.Error())
	}
}

// Create handles POST /api/v1/runners/keys {"name": "..."}.
func (h *RunnerKeyHandler) Create(w http.ResponseWriter, r *http.Request) {
	r.Body = http.MaxBytesReader(w, r.Body, maxRunnerKeyBodyBytes)

		Name string `json:"name"`
	}

		WriteError(w, http.StatusBadRequest, "BAD_REQUEST", "invalid request body")
		return
	}

	if err != nil {
		writeRunnerKeyError(w, err)
		return
	}
	WriteJSON(w, http.StatusCreated, runnerKeyCreateResponse{Key: toRunnerKeyView(k), Plaintext: plaintext, Setup: runner.KeySetupGuidance})
}

		writeRunnerKeyError(w, err)
		return
	}
	out := make([]runnerKeyView, 0, len(keys))
	for _, k := range keys {
		out = append(out, toRunnerKeyView(k))
	}
	WriteJSON(w, http.StatusOK, out)
}

// Revoke handles DELETE /api/v1/runners/keys/{id} {"acknowledge": true}.
func (h *RunnerKeyHandler) Revoke(w http.ResponseWriter, r *http.Request) {
	id, err := uuid.Parse(chi.URLParam(r, "id"))
	if err != nil {
		WriteError(w, http.StatusBadRequest, "BAD_REQUEST", "invalid runner key id")
		return
	}
	r.Body = http.MaxBytesReader(w, r.Body, maxRunnerKeyBodyBytes)
	var req struct {
		Acknowledge bool `json:"acknowledge"`
	}
	if r.ContentLength != 0 {
		if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
			WriteError(w, http.StatusBadRequest, "BAD_REQUEST", "invalid request body")
			return
		}
	}
	if err := h.svc.Revoke(r.Context(), id, runner.RevokeAck{Acknowledged: req.Acknowledge}); err != nil {
		writeRunnerKeyError(w, err)
		return
	}
	WriteJSON(w, http.StatusOK, map[string]string{"status": "revoked", "id": id.String()})
}
```
