## Alignment correction — one-time secret lifetime at real sinks

Guard create/revoke completion by modal lifetime/request epoch. Closing while create is in flight must not later repopulate secret state, call an unmounted callback, log response, or put plaintext in the key list. A close after successful create still clears component memory and metadata reload remains authorized; do not automatically retry a non-idempotent create request after network uncertainty. Key name/claimed runner/setup fields come from P2 actual server DTO, including workspace-root config producer, not invented sample text.

**Goal:** Let administrators list runner keys, create one (its secret shown once, obscured, with Copy and setup guidance), and revoke one after an explicit confirmation. These are the dashboard's only writing controls.
**Architecture:** `RunnerKeysSection` renders only for `me.isAdmin` (from `GET /api/v1/me`). It owns the key-metadata list (`listRunnerKeys`) and opens two modals. `RunnerKeyAddModal` keeps the one-time secret only in its own component state: it is never passed to a callback, hook cache, storage, URL or log. Closing the modal clears that state and unmounts it. `RunnerKeyRevokeModal` sends the required `{"acknowledge": true}` and leaves the list unchanged on failure. The server enforces admin authorization regardless of the UI flag.
**Tech Stack:** React 19, React Router 7, CSS modules, Vitest 4 + @testing-library/react (`fireEvent`; real timers).
**Spec / Design:** spec "Administration and visibility" (add a runner with a uniquely named key shown only once with copy and setup guidance; see unclaimed keys, runners, creation and last use; restricted to administrators; revoke after confirming disconnect, automatic recovery that may repeat steps, and drain-first recommendation; dashboard read-only except key management); design phase 5 §3.1 (Runner keys), §3.4, §5 rows "Key metadata…", "Revocation confirmation and setup snippet", "Current viewer's admin flag", §7 (create/copy/revoke failures), §8 (success "Create a key…"; error "failed revoke", "failed create shows no secret"; edge "Cancelling… makes no API call", "unclaimed key", "non-admin sees no key controls", "the secret never enters persistent browser storage"). import { useId, useState } from 'react'
import { createRunnerKey } from '../api/runners'
import type { RunnerKey } from '../api/types'
import styles from './RunnerKeys.module.css'

/** Unique, DNS-label-like runner key name (the server enforces uniqueness). */
export const RUNNER_KEY_NAME = /^[a-z0-9][a-z0-9-]{1,62}$/

/** runner.env contents (contracts §15). The secret is pasted by the operator, never templated in. */
export function runnerEnvSnippet(serverUrl: string): string {
  return `ARBITRON_SERVER_URL=${serverUrl}\nARBITRON_RUNNER_KEY=<paste the key you copied>`
}

export function workspaceRootGuidance(root: string | null): string {
  const target = root ? root : "the server's workspace root (see Settings → System)"
  return `Set workspace_root in ~/.arbitron-runner/config.yaml to ${target}. A runner whose workspace root differs from the server's runs chat steps only.`
}

type CopyState = 'idle' | 'copied' | 'failed'

/**
 * Add runner: name → Create → the one-time secret. The secret lives only in
 * this component's state; onCreated receives key metadata only. Closing clears
 * the state before the dialog unmounts.
 */
export default function RunnerKeyAddModal({
  serverUrl,
  workspaceRoot,
  onCreated,
  onClose,
}: {
  serverUrl: string
  workspaceRoot: string | null
  onCreated: (key: RunnerKey) => void
  onClose: () => void
}) {
  const titleId = useId()
  const [name, setName] = useState('')
  const [submitting, setSubmitting] = useState(false)
  const [error, setError] = useState<string | null>(null)
  const [secret, setSecret] = useState<string | null>(null)
  const [reveal, setReveal] = useState(false)
  const [copy, setCopy] = useState<CopyState>('idle')

  const valid = RUNNER_KEY_NAME.test(name)

  async function handleCreate() {
    if (!valid || submitting) return
    setSubmitting(true)
    setError(null)
    try {
      const created = await createRunnerKey(name)
      setSecret(created.plaintext)
      onCreated(created.key)
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to create the key')
    } finally {
      setSubmitting(false)
    }
  }

  async function handleCopy() {
    if (secret === null) return
    try {
      if (!navigator.clipboard?.writeText) throw new Error('clipboard unavailable')
      await navigator.clipboard.writeText(secret)
      setCopy('copied')
    } catch {
      setCopy('failed')
      setReveal(true)
    }
  }

  function close() {
    setSecret(null)
    setReveal(false)
    onClose()
  }

  return (
    <div className={styles.overlay}>
      <div className={styles.modal} role="dialog" aria-modal="true" aria-labelledby={titleId}>
        <h2 id={titleId} className={styles.title}>
          Add runner
        </h2>
        {secret === null ? (
          <>
            <label className={styles.label}>
              Runner name
              <input
                className={styles.input}
                value={name}
                onChange={(e) => setName(e.target.value)}
                placeholder="mac-mini"
                autoComplete="off"
                spellCheck={false}
              />
            </label>
            <p className={styles.hint}>Lowercase letters, digits and dashes; must be unique.</p>
            {error && (
              <div className={styles.error} role="alert">
                Create failed: {error}
              </div>
            )}
            <div className={styles.actions}>
              <button type="button" className={styles.button} onClick={close}>
                Cancel
              </button>
              <button type="button" className={styles.button} disabled={!valid || submitting} onClick={() => void handleCreate()}>
                Create
              </button>
            </div>
          </>
        ) : (
          <>
            <div className={styles.warning} role="alert">
              This key won't be shown again. Copy it now and store it only in the runner's runner.env.
            </div>
            <div className={styles.label}>
              <span>Runner key</span>
              <span className={styles.secretRow}>
                <input
                  aria-label="Runner key"
                  className={styles.input}
                  type={reveal ? 'text' : 'password'}
                  value={secret}
                  readOnly
                  autoComplete="off"
                  spellCheck={false}
                />
                <button type="button" className={styles.button} onClick={() => setReveal((r) => !r)}>
                  {reveal ? 'Hide' : 'Show'}
                </button>
                <button type="button" className={styles.button} onClick={() => void handleCopy()}>
                  Copy
                </button>
              </span>
            </div>
            {copy !== 'idle' && (
              <div className={styles.status} role="status">
                {copy === 'copied' ? 'Copied' : 'Copy failed — select the key and copy it manually. It stays visible until you close this dialog.'}
              </div>
            )}
            <p className={styles.hint}>Put these lines in ~/.arbitron-runner/runner.env on the new machine (mode 0600):</p>
            <pre className={styles.snippet} data-testid="runner-env">
              {runnerEnvSnippet(serverUrl)}
            </pre>
            <p className={styles.hint} data-testid="workspace-root">
              {workspaceRootGuidance(workspaceRoot)}
            </p>
            <div className={styles.actions}>
              <button type="button" className={styles.button} onClick={close}>
                Done
              </button>
            </div>
          </>
        )}
      </div>
    </div>
  )
}

import { useId, useState } from 'react'
import { revokeRunnerKey } from '../api/runners'
import type { RunnerKey } from '../api/types'
import styles from './RunnerKeys.module.css'

/** Spec-required confirmation copy (design P5 §3.4). */
export const REVOKE_WARNINGS = [
  'The runner using this key disconnects immediately.',
  'Work it was running recovers automatically on other runners and may repeat interrupted steps, including their external effects.',
  'Recommended: drain the runner first so its work finishes or moves at safe boundaries.',
] as const

export default function RunnerKeyRevokeModal({ runnerKey, onRevoked, onClose }: { runnerKey: RunnerKey; onRevoked: () => void; onClose: () => void }) {
  const titleId = useId()
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState<string | null>(null)

  async function handleRevoke() {
    setBusy(true)
    setError(null)
    try {
      await revokeRunnerKey(runnerKey.id)
      onRevoked()
    } catch (err) {

            Revoke failed: {error}
          </div>

function AdminRunnerKeys() {
  const [keys, setKeys] = useState<RunnerKey[] | null>(null)
  const [error, setError] = useState<string | null>(null)
