**Goal:** Every Phase 4 component is constructed and injected by the real composition roots:
- `cmd/arbitron/main.go`, `cmd/arbitron/model_dependencies.go`
- `cmd/arbitron-runner/build.go`

The wiring then proves itself end to end through the built server stack:
- A multi-turn `interactive_agent` session stays pinned, with a session checkpoint after every turn.
- Exhaustion between turns moves the session to another runner, and the next turn continues with a loss notice.
- Mid-turn exhaustion resends the turn automatically, with an `interactive.turn_restarted` event.
- Two concurrent submissions from two separately built stacks over one database yield exactly one rejection. Removing the `SetTurnLeases` wiring makes that test fail.
- A turn interrupted by a server restart is resent by the startup scan.

**Architecture:**
- **`adapter/sessionports`.** This package builds the `runner.SessionPortabilityRegistry` (claudecli + codex adapters, P4 T4/T5) and the harness-version functions those adapters require. Both binaries link the same constructor.
- **`cmd/arbitron/runner_sessions.go`** groups the server's Phase 4 wiring into three steps:
  1. `wireInteractiveSessions` builds the turn lease and intent repos, the session checkpoint store, the per-turn commit repo, the export router, the workspace checkpointer, the session mover, the pin index and the session bundle staging. It also enables hub session jobs.
  2. `(*interactiveSessionStack).install` injects the same instances into the conversation handler, the `InteractiveExecutor` and the `GroupResolutionBridge`.
  3. `(*interactiveSessionStack).newSessionRecovery` builds the `SessionRecoveryController`. It subscribes the controller to pool events and makes it the runner service's session drainer.
- **`main.go`** adds `Sessions` to the built-in runner deps and `SessionPins` to the dispatcher deps. It calls the three steps at their ordering points and starts recovery from `startRuntime`.
- **Runner side.** `cmd/arbitron-runner/sessions.go:newSessionExec` builds the runner's interactive stream selector and portability registry, and hands them with the P3 T9 syncer and transfer client to the executor config. The runner advertises the two session ops.
- `domain/*` is not touched here.

**Tech Stack:** Go 1.25.5, testify, modernc SQLite, `net/http/httptest`, the real `initDomain`/`Start` composition root, POSIX shell fake harness binaries (existing `cmd/arbitron` test pattern).

// VersionFunc reports the local harness version a bundle is stamped with.
type VersionFunc = func(context.Context) (string, error)

// New registers the Claude Code and Codex session adapters.
func New(claudeVersion, codexVersion VersionFunc) (*runner.SessionPortabilityRegistry, error) {
	if claudeVersion == nil || codexVersion == nil {
		return nil, errors.New("sessionports: claude and codex harness version funcs are required")
	}
	claudePort, err := claudecli.NewSessionPortability(claudecli.SessionPortabilityOptions{HarnessVersion: claudeVersion})
	if err != nil {
		return nil, fmt.Errorf("sessionports: claude: %w", err)
	}
	codexPort, err := codex.NewSessionPortability(codex.SessionPortabilityOptions{HarnessVersion: codexVersion})
	if err != nil {
		return nil, fmt.Errorf("sessionports: codex: %w", err)
	}
	return runner.NewSessionPortabilityRegistry(map[string]engine.SessionPortability{
		workflow.BackendClaudeCode: claudePort,
		workflow.BackendCodex:      codexPort,
	}), nil
}

// ClaudeVersion probes `<bin> --version`.
func ClaudeVersion(r claudecli.CommandRunner, bin string) VersionFunc {
	return func(ctx context.Context) (string, error) {
		if strings.TrimSpace(bin) == "" {
			return "", errors.New("sessionports: claude binary not found")
		}
		return claudecli.ProbeVersion(ctx, r, bin)
	}
}

// InventoryVersion reads one harness version from local inventory detection.
func InventoryVersion(detect func(context.Context) (runner.InventoryInput, error), pick func(runner.InventoryInput) *runner.HarnessPresence, name string) VersionFunc {
	return func(ctx context.Context) (string, error) {
		in, err := detect(ctx)
		if err != nil {
			return "", fmt.Errorf("sessionports: detect %s: %w", name, err)
		}
		p := pick(in)
		if p == nil || strings.TrimSpace(p.Version) == "" {
			return "", fmt.Errorf("sessionports: %s harness not found", name)
		}
		return strings.TrimSpace(p.Version), nil
	}
}

// MapVersion reads one harness version from a detected-versions map (the
// runner's source inventory).
func MapVersion(versions func() map[string]string, key string) VersionFunc {
	return func(context.Context) (string, error) {
		v := strings.TrimSpace(versions()[key])
		if v == "" {
			return "", fmt.Errorf("sessionports: %s harness not detected on this runner", key)
		}
		return v, nil
	}
}

type turnSessionSetters interface {

// sessionConsumers are the three interactive turn owners (the conversation
// handler, the InteractiveExecutor and the GroupResolutionBridge) plus the
// restorer group staging sessions use.
type sessionConsumers struct {
	Handler, Executor, Bridge any
	Staging                   engine.StagingRestorer
}

// install gives every turn owner the same lease store, intent store, mover and
// checkpoint deps. A consumer without a setter is a wiring defect: boot fails.
func (s *interactiveSessionStack) install(c sessionConsumers) error {
	if c.Staging == nil {
		return errors.New("interactive sessions: the group staging restorer is required")
	}
	owners := []struct {
		name string
		v    any
	}{{"conversation handler", c.Handler}, {"interactive executor", c.Executor}, {"group-resolution bridge", c.Bridge}}
	for _, o := range owners {
		set, ok := o.v.(turnSessionSetters)
		if !ok {
			return fmt.Errorf("interactive sessions: the %s cannot hold turn leases, intents and a session mover", o.name)
		}
		set.SetTurnLeases(s.Leases, s.LeaseTTL)
		set.SetTurnIntents(s.Intents)
		set.SetSessionMover(s.Mover, s.MaxMoves)
	}
	for _, o := range owners[:2] {
		set, ok := o.v.(sessionCheckpointSetter)
		if !ok {
			return fmt.Errorf("interactive sessions: the %s cannot checkpoint sessions", o.name)
		}
		set.SetSessionCheckpoints(s.Checkpoints)
	}
	b, ok := c.Bridge.(stagingCheckpointSetter)
	if !ok {
		return errors.New("interactive sessions: the group-resolution bridge cannot checkpoint staging sessions")
	}
	b.SetSessionCheckpoints(s.Checkpoints, c.Staging)
	return nil
}

// newSessionRecovery builds the between-turn recovery controller, subscribes
// it to recorded pool events and makes it the drain's session mover.
func (s *interactiveSessionStack) newSessionRecovery(handler any, invocations engine.InvocationStore,
	drains interface{ SetSessionDrainer(runner.SessionDrainer) }) (*runner.SessionRecoveryController, error) {
	rec, ok := handler.(engine.SessionRecoverer)
	if !ok {
		return nil, errors.New("interactive sessions: the conversation handler cannot recover sessions")
	}
	if invocations == nil || drains == nil {
		return nil, errors.New("interactive sessions: recovery needs the invocation store and the runner service")
	}
	c := runner.NewSessionRecoveryController(runner.SessionRecoveryDeps{
		Registry: s.in.Registry, Ledger: s.in.Ledger, Pins: s.PinIndex, Intents: s.Intents, Recoverer: rec,
		Invocations: invocations, Adopter: s.in.Hub, Notifier: s.in.Notifier, Clock: s.in.Clock,
	})
	s.in.Events.AddListener(c.HandlePoolEvent)
	drains.SetSessionDrainer(c)
	return c, nil
}

// runEvents records every bus event of one run.
type runEvents struct {
	mu  sync.Mutex
	evs []engine.Event
}

func watchRun(t *testing.T, ds *domainStack, id uuid.UUID) *runEvents {
	t.Helper()
	ch, unsubscribe := ds.bus.Subscribe(id)
	r := &runEvents{}
	done := make(chan struct{})
	go func() {
		defer close(done)
		for ev := range ch {
			r.mu.Lock()
			r.evs = append(r.evs, ev)
			r.mu.Unlock()
		}
	}()
	t.Cleanup(func() { unsubscribe(); <-done })
	return r
}

func (r *runEvents) ofType(typ string) []engine.Event {
	r.mu.Lock()
	defer r.mu.Unlock()
	var out []engine.Event
	for _, ev := range r.evs {
		if ev.Type == typ {
			out = append(out, ev)
		}
	}
	return out
}

}

// sessRemoteExec is a remote runner that hosts chat sessions. Session bundle
// bytes move through the REAL transfer client against the served router, so
// the hub's session staging and the /sessions routes are on the path.
type sessRemoteExec struct {
	xfer    *transfer.Client
	mu      sync.Mutex
	streams []runneragent.Dispatch
	imports [][]byte
	exports int
}

func (e *sessRemoteExec) Supports(d runneragent.Dispatch) (bool, string) {
	switch {
	case d.Op == runneragent.OpSessionImport && d.Session != nil && d.Session.Bundle != nil,
		d.Op == runneragent.OpSessionExport && d.Session != nil,
		d.Op == runneragent.OpStream && d.Chat != nil && d.Chat.WorkDir != "":
		return true, ""
	}
	return false, "the session e2e runner hosts chat sessions only"
}

func (e *sessRemoteExec) Execute(ctx context.Context, d runneragent.Dispatch, emit runneragent.EmitFunc) runneragent.Result {
	fail := func(err error) runneragent.Result {
		return runneragent.Result{OK: false, ErrorText: err.Error(), Outcome: runner.SourceOutcome{Class: runner.OutcomeOther}}
	}
	switch d.Op {
	case runneragent.OpSessionImport:
		data, err := e.xfer.GetSession(ctx, d.Session.TransferToken, d.JobID, d.Session.Bundle.Size)
		if err != nil {
			return fail(err)
		}
		if engine.BundleChecksum(data) != d.Session.Bundle.Checksum {
			return fail(errors.New("session bundle checksum mismatch"))
		}
		e.mu.Lock()
		e.imports = append(e.imports, data)
		e.mu.Unlock()
		return runneragent.Result{OK: true, Session: &runneragent.SessionResult{SessionID: "remote-sess"}}
	case runneragent.OpSessionExport:
		e.mu.Lock()
		e.exports++
		data := []byte(fmt.Sprintf("remote-bundle-%d", e.exports))
		e.mu.Unlock()
		if err := e.xfer.PutSession(ctx, d.Session.TransferToken, d.JobID, data); err != nil {
			return fail(err)
		}
		return runneragent.Result{OK: true, Session: &runneragent.SessionResult{SessionID: d.Session.SessionID, Bundle: &runneragent.SessionBundleMeta{
			Backend: d.Session.Backend, SessionID: d.Session.SessionID, HarnessVersion: sessClaudeVersion, OriginWorkDir: d.Session.WorkDir,
			Checksum: engine.BundleChecksum(data), Size: int64(len(data))}}}
	}
	e.mu.Lock()
	e.streams = append(e.streams, d)
	e.mu.Unlock()
	for _, tok := range []string{"remote-", "answer"} {
		if err := emit(ctx, tok); err != nil {
			return fail(err)
		}
	}
	return runneragent.Result{OK: true, Chat: &runneragent.ChatResult{Output: "remote-answer", Model: d.Chat.Model, SessionID: "remote-sess", TokensIn: 1, TokensOut: 1}}
}

func (e *sessRemoteExec) snapshot() (streams []runneragent.Dispatch, imports [][]byte, exports int) {
	e.mu.Lock()
	defer e.mu.Unlock()
	return append([]runneragent.Dispatch(nil), e.streams...), append([][]byte(nil), e.imports...), e.exports
}
