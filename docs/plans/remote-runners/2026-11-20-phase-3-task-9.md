---

**Goal:** Let `arbitron-runner` execute `script`, `tool_call` and `agent_call` dispatches against a local copy of the run's workspace that it pulls (only differing files) from the committed server checkpoint and pushes back (changed files plus deletions, in hash-verified parts) after every successful step.

**Architecture:** The manifest algorithm is shared, not re-implemented: the runner walks its copy with `adapter/checkpoint.ScanEntries`, diffs with `runner.DiffManifests`/`runner.SameEntries` and encodes pushes with `checkpoint.WriteDelta` — exactly the functions the server store uses (P3 T2). The runner-agent domain (`domain/runneragent`, P2 T8) gains mirror types for process/agent/workspace jobs and a per-run copy record; adapters add a copy-record file store, an HTTP transfer client for `/api/v1/runners/transfers/*`, a `Syncer` (pull / mark-dirty / push), a hash-addressed bundle cache and `WorkspaceJobs` (process jobs through the P1 `LocalProcessRunner`, agent jobs through the linked `claudecli`/`codex` adapters at the canonical workspace path). The P2 T9 executor delegates `process`/`agent` ops to `WorkspaceJobs`; the P2 T9 `wsclient` translates the new wire fields.

**Tech Stack:** Go 1.25.5, stdlib `net/http`, `crypto/sha256`, testify, `httptest`.

**Spec / Design:** `docs/specs/2026-11-19-remote-runners-spec.md` (Recovery: "A workspace-changing step counts as complete only when its resulting workspace is recoverable elsewhere"); `docs/designs/2026-11-19-remote-runners-design-phase-3.md` §3.3 (Manifest, Pull, Push, Transport), §3.4 (runner env, script cwd), §5 "Deferred to plans → Runner binary", §8 Happy path (zero pulls after the first, a push after each step), Edge-case ("A lib script finds its siblings on a remote runner", "Passthrough env values come from the runner"), Error ("An interrupted upload never marks the step done"). Contracts: `docs/plans/remote-runners/2026-11-20-contracts.md` §5, §15 (runner binary, `~/.arbitron-runner/`), §16 (Manifest, CheckpointStore, transfer routes and headers).

```go
package runneragent

import (
	"context"
	"time"

	"github.com/arbitron/arbitron/domain/runner"
	"github.com/google/uuid"
)

// CopyState is the runner's record of its local copy of one run's workspace.
// Base is the committed manifest the copy was last synced to (the push diff
// base); Dirty is set before a step runs and cleared by a pull or a push.
type CopyState struct {
	RunID             uuid.UUID       `json:"runId"`
	WorkDir           string          `json:"workDir"`
	HomeGeneration    int64           `json:"homeGeneration"`
	CheckpointVersion int64           `json:"checkpointVersion"`
	ManifestHash      string          `json:"manifestHash"`
	Dirty             bool            `json:"dirty"`
	Base              runner.Manifest `json:"base"`
	UpdatedAt         time.Time       `json:"updatedAt"`
}

// CopyStateStore persists copy records across runner restarts.
type CopyStateStore interface {
	Get(ctx context.Context, runID uuid.UUID) (CopyState, bool, error)
	Put(ctx context.Context, s CopyState) error
	Delete(ctx context.Context, runID uuid.UUID) error
	List(ctx context.Context) ([]CopyState, error)
}

// PullTarget is what the server says the copy must equal before a step.
type PullTarget struct {
	RunID             uuid.UUID
	HomeGeneration    int64
	CheckpointVersion int64
	ManifestHash      string
}

// ReuseLocalCopy: a copy is reused untouched only when the home generation
// AND the committed checkpoint version (and its manifest) match and no
// interrupted or failed attempt dirtied it (design §3.3 Pull).
func ReuseLocalCopy(s CopyState, t PullTarget) bool {
	return s.RunID == t.RunID && !s.Dirty &&
		s.HomeGeneration == t.HomeGeneration &&
		s.CheckpointVersion == t.CheckpointVersion &&
		s.ManifestHash == t.ManifestHash
}
```

```go
package copies

import (

// FileStore keeps one JSON record per run under dir (0700), files 0600.
type FileStore struct{ dir string }

var _ runneragent.CopyStateStore = (*FileStore)(nil)

func NewFileStore(dir string) (*FileStore, error) {
	if err := os.MkdirAll(dir, 0o700); err != nil {
		return nil, fmt.Errorf("copies: %w", err)
	}
	return &FileStore{dir: dir}, nil
}

func (s *FileStore) path(id uuid.UUID) string { return filepath.Join(s.dir, id.String()+".json") }

func (s *FileStore) Get(_ context.Context, id uuid.UUID) (runneragent.CopyState, bool, error) {
	b, err := os.ReadFile(s.path(id))
	if errors.Is(err, fs.ErrNotExist) {
		return runneragent.CopyState{}, false, nil
	}
	if err != nil {
		return runneragent.CopyState{}, false, err
	}
	var st runneragent.CopyState
	if err := json.Unmarshal(b, &st); err != nil {
		return runneragent.CopyState{}, false, fmt.Errorf("copies: %s: %w", id, err)
	}
	return st, true, nil
}

func (s *FileStore) Put(_ context.Context, st runneragent.CopyState) error {
	b, err := json.Marshal(st)
	if err != nil {
		return err
	}
	tmp, err := os.CreateTemp(s.dir, ".copy-*")
	if err != nil {
		return err
	}
	defer os.Remove(tmp.Name())
	if err := tmp.Chmod(0o600); err != nil {
		tmp.Close()
		return err
	}
	if _, err := tmp.Write(b); err != nil {
		tmp.Close()
		return err
	}
	if err := tmp.Close(); err != nil {
		return err
	}
	return os.Rename(tmp.Name(), s.path(st.RunID))
}

func (s *FileStore) Delete(_ context.Context, id uuid.UUID) error {
	if err := os.Remove(s.path(id)); err != nil && !errors.Is(err, fs.ErrNotExist) {
		return err
	}
	return nil
}

func (s *FileStore) List(ctx context.Context) ([]runneragent.CopyState, error) {
	entries, err := os.ReadDir(s.dir)
	if err != nil {
		return nil, err
	}
	var out []runneragent.CopyState
	for _, e := range entries {
		name := e.Name()
		if !strings.HasSuffix(name, ".json") || strings.HasPrefix(name, ".") {
			continue
		}
		id, err := uuid.Parse(strings.TrimSuffix(name, ".json"))
		if err != nil {
			continue
		}
		st, ok, err := s.Get(ctx, id)
		if err != nil {
			return nil, err
		}
		if ok {
			out = append(out, st)
		}
	}
	return out, nil
}
```

```go
type Transfer interface {
	Manifest(ctx context.Context, token string) (runner.Manifest, error)
	FetchFile(ctx context.Context, token, path string) (io.ReadCloser, error)
	PutPart(ctx context.Context, token string, index int, data []byte) error

	FetchBundle(ctx context.Context, token, hash string) (io.ReadCloser, error)
}

// Client implements Transfer over HTTPS. The runner key travels only in
// `Authorization: Runner <key>`; extra headers (Cloudflare Access) go on
// every request.
type Client struct {
	base  string
	key   string
	hc    *http.Client
	extra http.Header
}

var _ Transfer = (*Client)(nil)

func NewClient(serverURL, runnerKey string, hc *http.Client, extraHeaders http.Header) *Client {
	if hc == nil {
		hc = http.DefaultClient
	}
	return &Client{base: strings.TrimRight(serverURL, "/") + transfersPath, key: runnerKey, hc: hc, extra: extraHeaders.Clone()}
}

func (c *Client) do(ctx context.Context, method, path, token string, body []byte, hdr map[string]string) (*http.Response, error) {
	route := strings.SplitN(path, "?", 2)[0]
	var rd io.Reader
	if body != nil {
		rd = bytes.NewReader(body)
	}
	req, err := http.NewRequestWithContext(ctx, method, c.base+path, rd)
	if err != nil {
		return nil, err
	}
	for k, values := range c.extra {
		for _, value := range values { req.Header.Add(k, value) }
	}
	for k, v := range hdr {
		req.Header.Set(k, v)
	}
	req.Header.Set("Authorization", "Runner "+c.key)
	req.Header.Set("X-Arbitron-Transfer", token)
	resp, err := c.hc.Do(req)
	if err != nil {
		return nil, fmt.Errorf("transfer %s %s: %w", method, route, err)
	}
	if resp.StatusCode/100 != 2 {
		msg, _ := io.ReadAll(io.LimitReader(resp.Body, 512))
		resp.Body.Close()
		return nil, fmt.Errorf("transfer %s %s: status %d: %s", method, route, resp.StatusCode, strings.TrimSpace(string(msg)))
	}
	return resp, nil
}
```

```go
func TestSyncer_PushSendsChangesAndDeletionsThenNextPullIsFree(t *testing.T) {
	ctx := context.Background()
	f := newFixture(t)
	ws, err := f.srv.Plan(ctx, f.workDir, 1)
	require.NoError(t, err)
	ws.PartBytes = 64
	local, err := f.sync.Pull(ctx, ws)
	require.NoError(t, err)

	require.NoError(t, f.sync.MarkDirty(ctx, ws))
	require.NoError(t, os.WriteFile(filepath.Join(local, "a.txt"), []byte(strings.Repeat("x", 3000)), 0o600))
	require.NoError(t, os.Chmod(filepath.Join(local, "a.txt"), 0o600))
	require.NoError(t, os.Remove(filepath.Join(local, "src", "b.go")))
	require.NoError(t, os.Symlink("a.txt", filepath.Join(local, "link")))

	ref, err := f.sync.Push(ctx, ws)
	require.NoError(t, err)
	assert.Equal(t, ws.CheckpointVersion+1, ref.Version)
	require.Equal(t, 1, f.srv.Commits())
	assert.Greater(t, f.srv.LastCommitParts(), 1, "a delta larger than PartBytes is split into parts")
	base, err := f.srv.LatestManifest(ctx)
	require.NoError(t, err)
	assert.Equal(t, ws.CheckpointVersion, base.Version, "the push only prepares; Latest moves in the validated commit")
	require.NoError(t, f.srv.Promote(ctx, ref))

	man, err := f.srv.LatestManifest(ctx)
	require.NoError(t, err)
	byPath := map[string]uint32{}
	for _, e := range man.Entries {
		byPath[e.Path] = e.Mode
	}
	assert.NotContains(t, byPath, "src/b.go", "the deletion reached the server checkpoint")
	assert.Equal(t, uint32(0o600), byPath["a.txt"], "chmod reached the server checkpoint")
	assert.Contains(t, byPath, "link", "the symlink reached the server checkpoint")

	next, err := f.srv.Plan(ctx, f.workDir, 1)
	require.NoError(t, err)
	f.srv.ResetCounters()
	_, err = f.sync.Pull(ctx, next)
	require.NoError(t, err)
	assert.Empty(t, f.srv.Fetches(), "after our own push is committed the copy equals the new checkpoint: zero pulls")
	assert.Zero(t, f.srv.ManifestCalls())
}

func TestSyncer_PushPreparesWithoutClaimingCommittedVersion(t *testing.T) {
	ctx := context.Background()
	f := newFixture(t)
	ws, err := f.srv.Plan(ctx, f.workDir, 1)
	require.NoError(t, err)
	local, err := f.sync.Pull(ctx, ws)
	require.NoError(t, err)
	require.NoError(t, f.sync.MarkDirty(ctx, ws))
	require.NoError(t, os.WriteFile(filepath.Join(local, "a.txt"), []byte("beta"), 0o644))

	require.NoError(t, err)
	prepared, found, err := f.srv.PreparedFor(ctx)
	require.NoError(t, err)
	require.True(t, found, "the push prepared exactly this job's candidate")
	assert.Equal(t, ref, prepared)
	latest, err := f.srv.LatestManifest(ctx)
	require.NoError(t, err)
	assert.Equal(t, ws.CheckpointVersion, latest.Version, "Latest stays on the pinned base after the push")

	st, ok, err := f.copies.Get(ctx, f.runID)
	require.NoError(t, err)
	require.True(t, ok)
	assert.Equal(t, ws.CheckpointVersion, st.CheckpointVersion, "the copy never records the prepared version as committed")
	assert.Equal(t, ws.ManifestHash, st.ManifestHash)
	assert.True(t, st.Dirty, "until the commit is acknowledged the copy is not a reusable committed base")
	require.NotNil(t, st.Prepared, "the prepared receipt is recorded")
	assert.Equal(t, ref, st.Prepared.Ref)
	assert.Equal(t, ws.BaseRef, st.Prepared.BaseRef)
	assert.Equal(t, ws.FenceKey, st.Prepared.FenceKey)
	assert.Equal(t, ws.FenceGeneration, st.Prepared.FenceGeneration)
	assert.Equal(t, ws.FenceToken, st.Prepared.FenceToken)

	// The simulated validated commit transaction (P3 T5) promotes the candidate.
	require.NoError(t, f.srv.Promote(ctx, ref))
	latest, err = f.srv.LatestManifest(ctx)
	require.NoError(t, err)
	assert.Equal(t, ref.Version, latest.Version, "Latest becomes the candidate only after the validated commit")

	next, err := f.srv.Plan(ctx, f.workDir, 1)
	require.NoError(t, err)
	f.srv.ResetCounters()
	_, err = f.sync.Pull(ctx, next)
	require.NoError(t, err)
	assert.Empty(t, f.srv.Fetches())
	assert.Zero(t, f.srv.ManifestCalls(), "the committed candidate is exactly the local tree: adopted with no transfer")
	st, _, err = f.copies.Get(ctx, f.runID)
	require.NoError(t, err)
	assert.Equal(t, ref.Version, st.CheckpointVersion)
	assert.Equal(t, ref.ManifestHash, st.ManifestHash)
	assert.False(t, st.Dirty)
	assert.Nil(t, st.Prepared)
}

	require.True(t, ok)
	assert.Equal(t, ws.CheckpointVersion, st.CheckpointVersion)

}
```

```go
// adapter/runneragent/transfer/transfertest/server.go
// Package transfertest provides a transfer.Transfer backed by a real
// checkpoint.Store, for runner-agent tests. Like P3 T7's handler, a push
// commit only PREPARES the job's exact candidate on the token's pinned base;
// Promote is the test's stand-in for the validated commit transaction (P3 T5).
package transfertest

import (
	"bytes"
	"context"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"fmt"
	"io"
	"sync"

	"github.com/arbitron/arbitron/adapter/checkpoint"
	"github.com/arbitron/arbitron/adapter/runneragent/transfer"
	"github.com/arbitron/arbitron/domain/engine"
	"github.com/arbitron/arbitron/domain/runner"
	"github.com/arbitron/arbitron/domain/runneragent"
	"github.com/google/uuid"
)

const (
	PullToken   = "pull-token"
	PushToken   = "push-token"
	BundleToken = "bundle-token"
)

// Server serves one run, written by one job fence under one resource claim.
// Tokens are checked on every call (pull token for reads, push token for
// writes, bundle token for bundles).
type Server struct {
	store *checkpoint.Store
	cat   *memCatalog
	runID uuid.UUID
	job   runner.Fence
	res   engine.ResourceFenceRef

	mu            sync.Mutex
	pinned        runner.CheckpointRef
	bundles       map[string][]byte
	parts         map[int][]byte
	fetches       []string
	manifests     int
	commits       int
	lastParts     int
	bundleFetches int
	failPart      int
	failTimes     int
}

var _ transfer.Transfer = (*Server)(nil)

func NewServer(storeDir string, runID uuid.UUID) *Server {
	cat := newMemCatalog()
	return &Server{
		store: checkpoint.NewStore(storeDir, checkpoint.WithCatalog(cat)),
		cat:   cat, runID: runID,
		job:     runner.Fence{JobID: uuid.New(), RunnerID: "mac-1", Generation: 1},
		res:     engine.ResourceFenceRef{Key: "run:" + runID.String(), Owner: "mac-1", Generation: 1, Token: uuid.New()},
		bundles: map[string][]byte{}, parts: map[int][]byte{}, failPart: -1,
	}
}

// Seed bootstraps dir as the run's first committed baseline (P3 T6 EnsureBaseline).
func (s *Server) Seed(ctx context.Context, dir string) (runner.CheckpointRef, error) {
	ref, _, err := s.store.Commit(ctx, s.runID, dir)
	return ref, err
}

// Plan returns the sync plan the hub would send for the committed checkpoint,
// with its pinned base and resource claim (P3 T8 BuildWorkspaceJob), and pins
// the transfer tokens to that base.
func (s *Server) Plan(ctx context.Context, workDir string, homeGen int64) (runneragent.WorkspaceJob, error) {
	ref, err := s.latest(ctx)
	if err != nil {
		return runneragent.WorkspaceJob{}, err
	}
	s.mu.Lock()
	s.pinned = ref
	s.mu.Unlock()
	return runneragent.WorkspaceJob{RunID: s.runID.String(), WorkDir: workDir, HomeGeneration: homeGen,
		CheckpointVersion: ref.Version, ManifestHash: ref.ManifestHash, BaseRef: ref.String(),
		FenceKey: s.res.Key, FenceOwner: s.res.Owner, FenceGeneration: s.res.Generation, FenceToken: s.res.Token.String(),
		PullToken: PullToken, PushToken: PushToken, PartBytes: 1 << 20}, nil
}

func (s *Server) latest(ctx context.Context) (runner.CheckpointRef, error) {
	ref, ok, err := s.store.Latest(ctx, s.runID)
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	if !ok {
		return runner.CheckpointRef{}, errors.New("transfertest: no checkpoint")
	}
	return ref, nil
}

func (s *Server) pinnedBase() (runner.CheckpointRef, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	if s.pinned.IsZero() {
		return runner.CheckpointRef{}, errors.New("transfertest: status 403: no plan issued")
	}
	return s.pinned, nil
}

// LatestManifest is the COMMITTED checkpoint's manifest.
func (s *Server) LatestManifest(ctx context.Context) (runner.Manifest, error) {
	ref, err := s.latest(ctx)
	if err != nil {
		return runner.Manifest{}, err
	}
	return s.store.Manifest(ctx, ref)
}

// ReadLatest reads one file of the COMMITTED checkpoint.
func (s *Server) ReadLatest(ctx context.Context, path string) ([]byte, error) {
	ref, err := s.latest(ctx)
	if err != nil {
		return nil, err
	}
	rc, err := s.store.ReadFile(ctx, ref, path)
	if err != nil {
		return nil, err
	}
	defer rc.Close()
	return io.ReadAll(rc)
}

// PreparedFor returns this job's exact prepared candidate, if any.
func (s *Server) PreparedFor(ctx context.Context) (runner.CheckpointRef, bool, error) {
	return s.store.PreparedFor(ctx, s.runID, s.job, s.res)
}

// Promote is the validated commit transaction: only the exact prepared
// candidate on the committed base becomes Latest.
func (s *Server) Promote(_ context.Context, ref runner.CheckpointRef) error {
	return s.cat.commit(ref)
}

// PromotePrepared promotes this job's single prepared candidate.
func (s *Server) PromotePrepared(ctx context.Context) (runner.CheckpointRef, error) {
	ref, ok, err := s.PreparedFor(ctx)
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	if !ok {
		return runner.CheckpointRef{}, errors.New("transfertest: no prepared candidate")
	}
	return ref, s.Promote(ctx, ref)
}

func (s *Server) AddBundle(hash string, data []byte) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.bundles[hash] = data
}

// FailPart makes PutPart(index) fail the next `times` calls.
func (s *Server) FailPart(index, times int) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.failPart, s.failTimes = index, times
}

func (s *Server) ResetCounters() {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.fetches, s.manifests, s.bundleFetches = nil, 0, 0
}

func (s *Server) Fetches() []string {
	s.mu.Lock()
	defer s.mu.Unlock()
	return append([]string(nil), s.fetches...)
}

func (s *Server) ManifestCalls() int {
	s.mu.Lock()
	defer s.mu.Unlock()
	return s.manifests
}

// Commits counts accepted push commits (each PREPARES one candidate).
func (s *Server) Commits() int {
	s.mu.Lock()
	defer s.mu.Unlock()
	return s.commits
}

func (s *Server) LastCommitParts() int {
	s.mu.Lock()
	defer s.mu.Unlock()
	return s.lastParts
}

func (s *Server) BundleFetches() int {
	s.mu.Lock()
	defer s.mu.Unlock()
	return s.bundleFetches
}

var errToken = errors.New("transfertest: status 403: wrong transfer token")

func (s *Server) Manifest(ctx context.Context, token string) (runner.Manifest, error) {
	if token != PullToken {
		return runner.Manifest{}, errToken
	}
	base, err := s.pinnedBase()
	if err != nil {
		return runner.Manifest{}, err
	}
	s.mu.Lock()
	s.manifests++
	s.mu.Unlock()
	return s.store.Manifest(ctx, base)
}

func (s *Server) FetchFile(ctx context.Context, token, path string) (io.ReadCloser, error) {
	if token != PullToken {
		return nil, errToken
	}
	base, err := s.pinnedBase()
	if err != nil {
		return nil, err
	}
	s.mu.Lock()
	s.fetches = append(s.fetches, path)
	s.mu.Unlock()
	return s.store.ReadFile(ctx, base, path)
}

func (s *Server) PutPart(_ context.Context, token string, i int, data []byte) error {
	if token != PushToken {
		return errToken
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if i == s.failPart && s.failTimes > 0 {
		s.failTimes--
		return errors.New("transfertest: connection reset mid-transfer")
	}
	s.parts[i] = append([]byte(nil), data...)
	return nil
}

// CommitPush verifies the whole stream, stages it on the pinned base and
// PREPARES the job's candidate. It never moves Latest.
func (s *Server) CommitPush(ctx context.Context, token string, c transfer.PushCommit) (runner.CheckpointRef, error) {
	if token != PushToken {
		return runner.CheckpointRef{}, errToken
	}
	base, err := s.pinnedBase()
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	s.mu.Lock()
	var all []byte
	for i := 0; i < c.Parts; i++ {
		p, ok := s.parts[i]
		if !ok {
			s.mu.Unlock()
			return runner.CheckpointRef{}, fmt.Errorf("transfertest: status 409: missing part %d", i)
		}
		all = append(all, p...)
	}
	s.parts = map[int][]byte{}
	s.mu.Unlock()
	sum := sha256.Sum256(all)
	if hex.EncodeToString(sum[:]) != c.SHA256 {
		return runner.CheckpointRef{}, errors.New("transfertest: status 422: delta hash mismatch")
	}
	staged, err := s.store.StageDelta(ctx, s.runID, base, bytes.NewReader(all))
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	staged.Resource = s.res // from the durable session, never the archive
	ref, err := s.store.Publish(ctx, staged, s.job)
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	s.mu.Lock()
	s.commits++
	s.lastParts = c.Parts
	s.mu.Unlock()
	return ref, nil
}

func (s *Server) FetchBundle(_ context.Context, token, hash string) (io.ReadCloser, error) {
	if token != BundleToken {
		return nil, errToken
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	s.bundleFetches++
	b, ok := s.bundles[hash]
	if !ok {
		return nil, errors.New("transfertest: status 404: bundle not found")
	}
	return io.NopCloser(bytes.NewReader(b)), nil
}

// memCatalog is an in-memory runner.CheckpointCatalog with prepared and
// committed rows. It checks the committed base, unique versions and the exact
// job/resource identity of PreparedFor. Liveness of the job and resource
// fences is P3 T2/T3's concern and is tested against SQLite there.
type memCatalog struct {
	mu      sync.Mutex
	rows    map[uuid.UUID]map[int64]*catRow
	pointer map[uuid.UUID]int64
}

type catRow struct {
	c         runner.CheckpointCandidate
	committed bool
}

var _ runner.CheckpointCatalog = (*memCatalog)(nil)

func newMemCatalog() *memCatalog {
	return &memCatalog{rows: map[uuid.UUID]map[int64]*catRow{}, pointer: map[uuid.UUID]int64{}}
}

func (c *memCatalog) committedLocked(run uuid.UUID) runner.CheckpointRef {
	v := c.pointer[run]
	if v == 0 {
		return runner.CheckpointRef{}
	}
	return c.rows[run][v].c.Ref
}

func (c *memCatalog) insertLocked(cand runner.CheckpointCandidate, committed bool) error {
	run := c.rows[cand.Ref.RunID]
	if run == nil {
		run = map[int64]*catRow{}
		c.rows[cand.Ref.RunID] = run
	}
	if _, ok := run[cand.Ref.Version]; ok {
		return fmt.Errorf("%w: v%d exists", runner.ErrCheckpointConflict, cand.Ref.Version)
	}
	run[cand.Ref.Version] = &catRow{c: cand, committed: committed}
	return nil
}

func (c *memCatalog) Publish(_ context.Context, cand runner.CheckpointCandidate) error {
	if err := cand.ValidatePrepared(); err != nil {
		return err
	}
	c.mu.Lock()
	defer c.mu.Unlock()
	if cur := c.committedLocked(cand.Ref.RunID); cur != cand.Base {
		return fmt.Errorf("%w: base %q is not committed %q", runner.ErrCheckpointConflict, cand.Base, cur)
	}
	return c.insertLocked(cand, false)
}

func (c *memCatalog) Bootstrap(_ context.Context, cand runner.CheckpointCandidate) error {
	if err := cand.ValidateBootstrap(); err != nil {
		return err
	}
	c.mu.Lock()
	defer c.mu.Unlock()
	if c.pointer[cand.Ref.RunID] != 0 {
		return fmt.Errorf("%w: run already has committed progress", runner.ErrCheckpointConflict)
	}
	if err := c.insertLocked(cand, true); err != nil {
		return err
	}
	c.pointer[cand.Ref.RunID] = cand.Ref.Version
	return nil
}

// commit is the validated commit transaction's promotion.
func (c *memCatalog) commit(ref runner.CheckpointRef) error {
	c.mu.Lock()
	defer c.mu.Unlock()
	row, ok := c.rows[ref.RunID][ref.Version]
	if !ok || row.committed || row.c.Ref != ref {
		return fmt.Errorf("%w: no prepared candidate %s", runner.ErrCheckpointNotFound, ref)
	}
	if c.committedLocked(ref.RunID) != row.c.Base {
		return runner.ErrCheckpointConflict
	}
	row.committed = true
	c.pointer[ref.RunID] = ref.Version
	return nil
}

func (c *memCatalog) Latest(_ context.Context, runID uuid.UUID) (runner.CheckpointRef, string, bool, error) {
	c.mu.Lock()
	defer c.mu.Unlock()
	v := c.pointer[runID]
	if v == 0 {
		return runner.CheckpointRef{}, "", false, nil
	}
	row := c.rows[runID][v]
	return row.c.Ref, row.c.ArchivePath, true, nil
}

func (c *memCatalog) Get(_ context.Context, runID uuid.UUID, v int64) (runner.CheckpointRef, string, error) {
	c.mu.Lock()
	defer c.mu.Unlock()
	row, ok := c.rows[runID][v]
	if !ok || !row.committed {
		return runner.CheckpointRef{}, "", runner.ErrCheckpointNotFound
	}
	return row.c.Ref, row.c.ArchivePath, nil
}

func (c *memCatalog) DeleteRun(_ context.Context, runID uuid.UUID) error {
	c.mu.Lock()
	defer c.mu.Unlock()
	delete(c.rows, runID)
	delete(c.pointer, runID)
	return nil
}

func (c *memCatalog) Retention(_ context.Context, runID uuid.UUID) (runner.CheckpointRetention, error) {
	c.mu.Lock()
	defer c.mu.Unlock()
	r := runner.CheckpointRetention{Versions: map[int64]bool{}, Committed: c.pointer[runID], Held: true}
	for v := range c.rows[runID] {
		r.Versions[v] = true
	}
	return r, nil
}

func (c *memCatalog) PreparedFor(_ context.Context, runID uuid.UUID, job runner.Fence, res engine.ResourceFenceRef) (runner.CheckpointRef, bool, error) {
	c.mu.Lock()
	defer c.mu.Unlock()
	var found []runner.CheckpointRef
	for _, row := range c.rows[runID] {
		if !row.committed && row.c.Job != nil && *row.c.Job == job && row.c.Resource == res {
			found = append(found, row.c.Ref)
		}
	}
	switch len(found) {
	case 0:
		return runner.CheckpointRef{}, false, nil
	case 1:
		return found[0], true, nil
	}
	return runner.CheckpointRef{}, false, fmt.Errorf("%w: %d prepared candidates for one job", runner.ErrCheckpointConflict, len(found))
}
```

```go
// adapter/runneragent/workspace/sync.go
// Package workspace keeps a runner's local copies of run workspaces equal to
// the committed server checkpoints (design §3.3). A push only PREPARES a
// candidate; the copy's committed base moves only when the server names that
// exact candidate as committed (contracts: prepared-push receipt).
package workspace

import (
	"bytes"
	"context"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"fmt"
	"io"
	"os"
	"path/filepath"
	"sort"
	"strings"
	"time"

	"github.com/arbitron/arbitron/adapter/checkpoint"
	"github.com/arbitron/arbitron/adapter/runneragent/transfer"
	"github.com/arbitron/arbitron/domain/runner"
	"github.com/arbitron/arbitron/domain/runneragent"
	"github.com/google/uuid"
)

var (
	// ErrOutsideWorkspaceRoot: the dispatch names a path outside the runner's
	// configured (server-equal) root; the runner never touches it.
	ErrOutsideWorkspaceRoot = errors.New("runner workspace: path is outside the configured workspace root")
	// ErrPullVerifyFailed: the server manifest is not the assigned one, or the
	// local tree does not equal it after the pull.
	ErrPullVerifyFailed = errors.New("runner workspace: pulled copy does not match the server checkpoint")
	// ErrStaleBase: the local copy is not on the job's pinned base, so its
	// diff would not apply to the base the push token is pinned to.
	ErrStaleBase = errors.New("runner workspace: local copy is not on the job's pinned base")
	// ErrReceiptMismatch: the server's prepared ref does not name the tree
	// this runner pushed.
	ErrReceiptMismatch = errors.New("runner workspace: prepared receipt does not match the pushed tree")
)

const (
	minPartBytes = 64
	partAttempts = 3
)

// SyncerDeps: WorkspaceRoot is the canonical root (equal to the server's, D2);
// LocalRoot is where copies physically live: equal to WorkspaceRoot in
// production; tests point it elsewhere so runner and server share a machine.
type SyncerDeps struct {
	WorkspaceRoot string
	LocalRoot     string
	Transfer      transfer.Transfer
	Copies        runneragent.CopyStateStore
	Clock         func() time.Time
}

type Syncer struct{ d SyncerDeps }

func NewSyncer(d SyncerDeps) (*Syncer, error) {
	if d.WorkspaceRoot == "" || d.Transfer == nil || d.Copies == nil {
		return nil, errors.New("runner workspace: WorkspaceRoot, Transfer and Copies are required")
	}
	if d.LocalRoot == "" {
		d.LocalRoot = d.WorkspaceRoot
	}
	if d.Clock == nil {
		d.Clock = time.Now
	}
	d.WorkspaceRoot = filepath.Clean(d.WorkspaceRoot)
	d.LocalRoot = filepath.Clean(d.LocalRoot)
	return &Syncer{d: d}, nil
}

// LocalRoot is where local copies live (WorkspaceRoot in production).
func (s *Syncer) LocalRoot() string { return s.d.LocalRoot }

// LocalPath maps a canonical {{workspace}} path to the local copy path.
func (s *Syncer) LocalPath(workDir string) (string, error) {
	rel, err := filepath.Rel(s.d.WorkspaceRoot, filepath.Clean(workDir))
	if err != nil || rel == "." || rel == ".." || strings.HasPrefix(rel, ".."+string(filepath.Separator)) || filepath.IsAbs(rel) {
		return "", fmt.Errorf("%w: %s", ErrOutsideWorkspaceRoot, workDir)
	}
	return filepath.Join(s.d.LocalRoot, rel), nil
}

// scope checks the path and the whole scoped descriptor (Cycle 1b) before any
// transfer or local mutation; nothing is defaulted.
func (s *Syncer) scope(ws runneragent.WorkspaceJob) (string, runneragent.PullTarget, error) {
	local, err := s.LocalPath(ws.WorkDir)
	if err != nil {
		return "", runneragent.PullTarget{}, err
	}
	if err := ws.Validate(); err != nil {
		return "", runneragent.PullTarget{}, fmt.Errorf("runner workspace: %w", err)
	}
	id, err := uuid.Parse(ws.RunID)
	if err != nil {
		return "", runneragent.PullTarget{}, fmt.Errorf("runner workspace: run id: %w", err)
	}
	return local, runneragent.PullTarget{RunID: id, HomeGeneration: ws.HomeGeneration, CheckpointVersion: ws.CheckpointVersion, ManifestHash: ws.ManifestHash}, nil
}

// Pull makes the local copy equal the committed checkpoint pinned by ws and
// returns the local path. A clean copy at the same home generation and
// checkpoint is reused with no transfer. A copy whose prepared receipt names
// exactly the pinned checkpoint, and whose tree is still that candidate, is
// adopted with no transfer. Otherwise the copy is compared with the server
// manifest, only differing paths are fetched (each hash-checked), paths
// absent from the checkpoint are removed, the result is verified to equal the
// checkpoint exactly, and any uncommitted receipt is discarded.
func (s *Syncer) Pull(ctx context.Context, ws runneragent.WorkspaceJob) (string, error) {
	local, tgt, err := s.scope(ws)
	if err != nil {
		return "", err
	}
	st, ok, err := s.d.Copies.Get(ctx, tgt.RunID)
	if err != nil {
		return "", err
	}
	if ok {
		if _, err := os.Stat(local); err == nil {
			if runneragent.ReuseLocalCopy(st, tgt) {
				return local, nil
			}
			adopted, err := s.adoptPrepared(ctx, local, st, tgt)
			if err != nil || adopted {
				return local, err
			}
		}
	}

	server, err := s.d.Transfer.Manifest(ctx, ws.PullToken)
	if err != nil {
		return "", err
	}
	if err := server.Verify(); err != nil {
		return "", fmt.Errorf("%w: %v", ErrPullVerifyFailed, err)
	}
	if server.Hash != ws.ManifestHash || server.Version != ws.CheckpointVersion {
		return "", fmt.Errorf("%w: server has v%d %s, pinned base is %s", ErrPullVerifyFailed, server.Version, server.Hash, ws.BaseRef)
	}
	if err := os.MkdirAll(local, 0o755); err != nil {
		return "", err
	}
	have, err := checkpoint.ScanEntries(ctx, local)
	if err != nil {
		// Unsupported content left by an interrupted step: start from empty.
		if err := os.RemoveAll(local); err != nil {
			return "", err
		}
		if err := os.MkdirAll(local, 0o755); err != nil {
			return "", err
		}
		have = nil
	}
	changed, deleted := runner.DiffManifests(have, server.Entries)
	if err := s.apply(ctx, ws.PullToken, local, changed, deleted); err != nil {
		return "", err
	}
	got, err := checkpoint.ScanEntries(ctx, local)
	if err != nil {
		return "", err
	}
	if !runner.SameEntries(got, server.Entries) {
		return "", fmt.Errorf("%w: local tree differs after pull", ErrPullVerifyFailed)
	}
	err = s.d.Copies.Put(ctx, runneragent.CopyState{ // Prepared is nil: an uncommitted receipt is discarded
		RunID: tgt.RunID, WorkDir: ws.WorkDir, HomeGeneration: ws.HomeGeneration,
		CheckpointVersion: server.Version, ManifestHash: server.Hash, Base: server, UpdatedAt: s.d.Clock(),
	})
	return local, err
}

// adoptPrepared promotes the local record to the committed checkpoint when the
// server committed exactly the candidate this runner pushed: same home
// generation, same version and manifest hash (the hash covers every entry),
// and the local tree is still that candidate. It reports false otherwise.
func (s *Syncer) adoptPrepared(ctx context.Context, local string, st runneragent.CopyState, tgt runneragent.PullTarget) (bool, error) {
	p := st.Prepared
	if p == nil || st.HomeGeneration != tgt.HomeGeneration ||
		p.Ref.Version != tgt.CheckpointVersion || p.Ref.ManifestHash != tgt.ManifestHash {
		return false, nil
	}
	base := runner.NewManifest(p.Ref.Version, p.Entries)
	if base.Hash != p.Ref.ManifestHash {
		return false, nil
	}
	have, err := checkpoint.ScanEntries(ctx, local)
	if err != nil || !runner.SameEntries(have, p.Entries) {
		return false, nil // edited since the push: verify and restore instead
	}
	st.CheckpointVersion, st.ManifestHash, st.Base = base.Version, base.Hash, base
	st.Dirty, st.Prepared, st.UpdatedAt = false, nil, s.d.Clock()
	return true, s.d.Copies.Put(ctx, st)
}

func (s *Syncer) apply(ctx context.Context, token, local string, changed []runner.ManifestEntry, deleted []string) error {
	sort.Sort(sort.Reverse(sort.StringSlice(deleted))) // children before parents
	for _, p := range deleted {
		if err := os.RemoveAll(filepath.Join(local, filepath.FromSlash(p))); err != nil {
			return err
		}
	}
	var dirs []runner.ManifestEntry
	for _, e := range changed { // sorted by path: parents first
		p := filepath.Join(local, filepath.FromSlash(e.Path))
		mode := os.FileMode(e.Mode)
		if info, err := os.Lstat(p); err == nil && (info.IsDir() != mode.IsDir() || info.Mode()&os.ModeSymlink != 0 || mode&os.ModeSymlink != 0) {
			if err := os.RemoveAll(p); err != nil {
				return err
			}
		}
		if err := os.MkdirAll(filepath.Dir(p), 0o755); err != nil {
			return err
		}
		switch {
		case mode&os.ModeSymlink != 0:
			if err := os.Symlink(e.Symlink, p); err != nil {
				return err
			}
		case mode.IsDir():
			if err := os.MkdirAll(p, 0o755); err != nil {
				return err
			}
			dirs = append(dirs, e)
		default:
			if err := s.fetchFile(ctx, token, e, p); err != nil {
				return err
			}
		}
	}
	for i := len(dirs) - 1; i >= 0; i-- { // directory modes last, deepest first
		if err := os.Chmod(filepath.Join(local, filepath.FromSlash(dirs[i].Path)), os.FileMode(dirs[i].Mode).Perm()); err != nil {
			return err
		}
	}
	return nil
}

func (s *Syncer) fetchFile(ctx context.Context, token string, e runner.ManifestEntry, dst string) error {
	rc, err := s.d.Transfer.FetchFile(ctx, token, e.Path)
	if err != nil {
		return err
	}
	defer rc.Close()
	tmp, err := os.CreateTemp(filepath.Dir(dst), ".pull-*")
	if err != nil {
		return err
	}
	defer os.Remove(tmp.Name())
	h := sha256.New()
	if _, err := io.Copy(io.MultiWriter(tmp, h), rc); err != nil {
		tmp.Close()
		return err
	}
	if err := tmp.Close(); err != nil {
		return err
	}
	if hex.EncodeToString(h.Sum(nil)) != e.SHA256 {
		return fmt.Errorf("%w: %s content hash mismatch", ErrPullVerifyFailed, e.Path)
	}
	if err := os.Chmod(tmp.Name(), os.FileMode(e.Mode).Perm()); err != nil {
		return err
	}
	return os.Rename(tmp.Name(), dst)
}

// MarkDirty records that a step is about to mutate the copy.
func (s *Syncer) MarkDirty(ctx context.Context, ws runneragent.WorkspaceJob) error {
	_, tgt, err := s.scope(ws)
	if err != nil {
		return err
	}
	st, ok, err := s.d.Copies.Get(ctx, tgt.RunID)
	if err != nil {
		return err
	}
	if !ok {
		return fmt.Errorf("runner workspace: no local copy for run %s", tgt.RunID)
	}
	st.Dirty, st.UpdatedAt = true, s.d.Clock()
	return s.d.Copies.Put(ctx, st)
}

// Push sends the copy's changes and deletions since its pinned base (one
// checkpoint.WriteDelta stream carrying DeltaControl{Deleted, Expected}) in
// parts of ws.PartBytes, then commits the upload. The server only PREPARES a
// candidate: Push records that exact receipt with its base and claim identity
// and keeps the copy dirty on the pinned base. It never records the candidate
// as the committed checkpoint. On any error nothing is recorded and the copy
// stays dirty, so the next pull restores the committed checkpoint.
func (s *Syncer) Push(ctx context.Context, ws runneragent.WorkspaceJob) (runner.CheckpointRef, error) {
	local, tgt, err := s.scope(ws)
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	st, ok, err := s.d.Copies.Get(ctx, tgt.RunID)
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	if !ok {
		return runner.CheckpointRef{}, fmt.Errorf("runner workspace: no local copy for run %s", tgt.RunID)
	}
	if st.CheckpointVersion != ws.CheckpointVersion || st.ManifestHash != ws.ManifestHash {
		return runner.CheckpointRef{}, fmt.Errorf("%w: copy is on v%d, job is pinned to %s", ErrStaleBase, st.CheckpointVersion, ws.BaseRef)
	}
	now, err := checkpoint.ScanEntries(ctx, local)
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	changed, deleted := runner.DiffManifests(st.Base.Entries, now)
	paths := make([]string, 0, len(changed))
	for _, e := range changed {
		paths = append(paths, e.Path)
	}
	var delta bytes.Buffer
	if err := checkpoint.WriteDelta(&delta, local, paths, runner.DeltaControl{Deleted: deleted, Expected: now}); err != nil {
		return runner.CheckpointRef{}, err
	}
	part := int(ws.PartBytes)
	if part < minPartBytes {
		part = minPartBytes
	}
	data := delta.Bytes()
	n := 0
	for off := 0; off < len(data); off += part {
		end := off + part
		if end > len(data) {
			end = len(data)
		}
		if err := s.putPart(ctx, ws.PushToken, n, data[off:end]); err != nil {
			return runner.CheckpointRef{}, err
		}
		n++
	}
	sum := sha256.Sum256(data)
	ref, err := s.d.Transfer.CommitPush(ctx, ws.PushToken, transfer.PushCommit{Parts: n, SHA256: hex.EncodeToString(sum[:])})
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	if ref.RunID != tgt.RunID || ref.Version <= ws.CheckpointVersion || runner.NewManifest(ref.Version, now).Hash != ref.ManifestHash {
		return runner.CheckpointRef{}, fmt.Errorf("%w: got %s", ErrReceiptMismatch, ref)
	}
	st.Dirty = true
	st.Prepared = &runneragent.PreparedCopy{Ref: ref, BaseRef: ws.BaseRef, FenceKey: ws.FenceKey,
		FenceGeneration: ws.FenceGeneration, FenceToken: ws.FenceToken, Entries: now}
	st.UpdatedAt = s.d.Clock()
	if err := s.d.Copies.Put(ctx, st); err != nil {
		return runner.CheckpointRef{}, err
	}
	return ref, nil
}

// putPart retries a part on error (a part whose hash the server rejects is
// retried, design §7); the third failure fails the push.
func (s *Syncer) putPart(ctx context.Context, token string, i int, b []byte) error {
	var err error
	for attempt := 0; attempt < partAttempts; attempt++ {
		if err = s.d.Transfer.PutPart(ctx, token, i, b); err == nil {
			return nil
		}
		if ctx.Err() != nil {
			return ctx.Err()
		}
	}
	return fmt.Errorf("runner workspace: push part %d: %w", i, err)
}
```

```go
// BundleCache keeps verified bundle archives by hash under dir
// (~/.arbitron-runner/bundles) and unpacks a fresh copy per job, so a script
// that writes into its cwd never alters the cached bundle.
type BundleCache struct {
	dir string
	t   transfer.Transfer
	mu  sync.Mutex
}

func NewBundleCache(dir string, t transfer.Transfer) (*BundleCache, error) {
	if err := os.MkdirAll(dir, 0o700); err != nil {
		return nil, err
	}
	return &BundleCache{dir: dir, t: t}, nil
}

func (c *BundleCache) archive(ctx context.Context, hash, token string) (string, error) {
	if !runner.ValidBundleHash(hash) {
		return "", errors.New("bundle cache: malformed hash")
	}
	c.mu.Lock()
	defer c.mu.Unlock()
	p := filepath.Join(c.dir, hash+".tar.gz")
	if _, err := os.Stat(p); err == nil {
		return p, nil
	}
	rc, err := c.t.FetchBundle(ctx, token, hash)
	if err != nil {
		return "", err
	}
	defer rc.Close()
	data, err := io.ReadAll(io.LimitReader(rc, process.MaxBundleBytes+1))
	if err != nil {
		return "", err
	}
	sum := sha256.Sum256(data)
	if hex.EncodeToString(sum[:]) != hash {
		return "", fmt.Errorf("bundle cache: %w", runner.ErrBundleHashMismatch)
	}
	tmp, err := os.CreateTemp(c.dir, ".bundle-*")
	if err != nil {
		return "", err
	}
	defer os.Remove(tmp.Name())
	if _, err := tmp.Write(data); err != nil {
		tmp.Close()
		return "", err
	}
	if err := tmp.Close(); err != nil {
		return "", err
	}
	return p, os.Rename(tmp.Name(), p)
}

// Unpack returns a fresh per-job directory holding the bundle's files.
func (c *BundleCache) Unpack(ctx context.Context, hash, token string) (string, func(), error) {
	p, err := c.archive(ctx, hash, token)
	if err != nil {
		return "", nil, err
	}
	data, err := os.ReadFile(p)
	if err != nil {
		return "", nil, err
	}
	dir, err := os.MkdirTemp(c.dir, "run-"+hash[:12]+"-*")
	if err != nil {
		return "", nil, err
	}
	cleanup := func() { _ = os.RemoveAll(dir) }
	if err := process.ExtractScriptBundle(bytes.NewReader(data), dir); err != nil {
		cleanup()
		return "", nil, err
	}
	return dir, cleanup, nil
}
```

```go
type WorkspaceJobsDeps struct {

// pull → mark dirty → run → push → result. A failed step is not pushed (the

type WorkspaceJobs struct{ d WorkspaceJobsDeps }

func NewWorkspaceJobs(d WorkspaceJobsDeps) (*WorkspaceJobs, error) {
	if d.Syncer == nil || d.Bundles == nil || d.Process == nil {

// Supports is the accept-time check (runneragent.JobExecutor contract): a
// dispatch this runner cannot execute is rejected so the dispatcher re-places it.
func (w *WorkspaceJobs) Supports(d runneragent.Dispatch) (bool, string) {
	if err := w.check(d); err != nil {
		return false, err.Error()
	}
	return true, ""
}

func (w *WorkspaceJobs) check(d runneragent.Dispatch) error {
	if d.Workspace == nil {
		return errors.New("workspace job without a workspace plan")
	}
	if _, err := w.d.Syncer.LocalPath(d.Workspace.WorkDir); err != nil {
		return err
	}
	switch d.Op {
	case runneragent.OpProcess:
		if d.Process == nil {
			return errors.New("process dispatch without payload")
		}
		if d.Process.Kind == string(engine.ProcessScript) && (d.Process.BundleHash == "" || d.Process.BundleEntrypoint == "") {
			return runner.ErrBundleRequired
		}
		// RR-09 decision 2026-11-22: interpreter/command first, then every
		// declared `requires` name, in order; the first miss rejects. LookPath
		// resolves against the PATH of the environment this runner builds for
		// the child (procenv.BaseEnv on the runner), never the server's.
		for _, name := range append([]string{d.Process.Command}, d.Process.Requires...) {
			if _, err := w.d.LookPath(name); err != nil {
				return fmt.Errorf("%w: %s", ErrCommandUnavailable, name)
			}
		}
	case runneragent.OpAgent:
		if d.Agent == nil {
			return errors.New("agent dispatch without payload")
		}
		bin, err := w.harness(d.Agent)
		if err != nil {
			return err
		}
		if _, err := w.d.LookPath(bin); err != nil {
			return fmt.Errorf("%w: %s", ErrCommandUnavailable, bin)
		}
	default:
		return fmt.Errorf("workspace jobs: op %q is not handled here", d.Op)
	}
	return nil
}

func (w *WorkspaceJobs) harness(a *runneragent.AgentJob) (string, error) {
	switch {
	case a.CredentialKind == string(runner.CredAnthropicSubscription) && (a.Backend == "" || a.Backend == "claude-code"):
		return w.d.ClaudeBinary, nil
	case a.CredentialKind == string(runner.CredOpenAISubscription) && a.Backend == "codex":
		return w.d.CodexBinary, nil
	}
	return "", fmt.Errorf("%w: %s/%s", ErrUnsupportedAgentCredential, a.CredentialKind, a.Backend)
}

// Execute runs one workspace dispatch. Errors are classified into the
// result's Outcome (runner.ClassifyError); the text is the error's own
// (adapters sanitize before returning).
func (w *WorkspaceJobs) Execute(ctx context.Context, d runneragent.Dispatch) runneragent.Result {
	if err := w.check(d); err != nil {
		return jobFailure(err)
	}
	local, err := w.d.Syncer.Pull(ctx, *d.Workspace)
	if err != nil {
		return jobFailure(fmt.Errorf("pull: %w", err))
	}
	if err := w.d.Syncer.MarkDirty(ctx, *d.Workspace); err != nil {
		return jobFailure(err)
	}
	var res runneragent.Result
	switch d.Op {
	case runneragent.OpProcess:
		pr, err := w.runProcess(ctx, d.Process, local)
		if err != nil {
			return jobFailure(err)
		}
		res = runneragent.Result{OK: true, Process: &runneragent.ProcessResult{Stdout: pr.Stdout, Stderr: pr.Stderr, ExitCode: pr.ExitCode}}
		if pr.ExitCode != 0 {
			return res // completed but failed: nothing is pushed
		}
	case runneragent.OpAgent:
		ar, err := w.runAgent(ctx, d.Agent, local)
		if err != nil {
			return jobFailure(err)
		}
		res = runneragent.Result{OK: true, Agent: toAgentResult(ar)}
	}
	if _, err := w.d.Syncer.Push(ctx, *d.Workspace); err != nil {
		return jobFailure(fmt.Errorf("push: %w", err))
	}
	return res
}

func (w *WorkspaceJobs) runProcess(ctx context.Context, p *runneragent.ProcessJob, local string) (engine.ProcessResult, error) {
	spec := engine.ProcessSpec{
		Kind: engine.ProcessKind(p.Kind), Command: p.Command, Args: p.Args, Stdin: p.Stdin,
		NodeEnv: p.NodeEnv, Passthrough: p.Passthrough, TimeoutSecs: p.TimeoutSecs, WorkDir: p.WorkDir,
	}
	switch spec.Kind {
	case engine.ProcessScript:
		dir, cleanup, err := w.d.Bundles.Unpack(ctx, p.BundleHash, p.BundleToken)
		if err != nil {
			return engine.ProcessResult{}, fmt.Errorf("bundle: %w", err)
		}
		defer cleanup()
		spec.Dir = dir // the script's own folder, exactly as on the server
		spec.Bundle = &engine.ScriptBundleRef{Hash: p.BundleHash, Entrypoint: p.BundleEntrypoint}
	case engine.ProcessToolCall:
		spec.Dir = local // the workspace copy
	default:
		return engine.ProcessResult{}, fmt.Errorf("unknown process kind %q", p.Kind)
	}
	return w.d.Process.RunProcess(ctx, spec)
}

func (w *WorkspaceJobs) runAgent(ctx context.Context, a *runneragent.AgentJob, local string) (*engine.AgentResult, error) {
	binding := provider.ResolvedBinding{
		Kind: provider.ProfileKindAgent, Provider: a.Provider, Model: a.Model, Backend: a.Backend,
		Pinned: a.RequireExecutedModel, AllowedTools: append([]string(nil), a.ToolScope...),
		Effort: provider.Effort(a.Effort), ContextWindow: provider.ContextWindow(a.ContextWindow),
	}
	rules := w.d.Rules // the same runner-local rules used to advertise capabilities
	if rules.Empty() {
		return nil, errors.New("workspace jobs: local capability rules required")
	}
	mission := engine.AgentMission{
		Model: a.Model, Prompt: a.Prompt, WorkDir: local, ToolScope: a.ToolScope, SessionID: a.SessionID,
		Timeout: time.Duration(a.TimeoutSecs) * time.Second,
	}
	switch a.CredentialKind {
	case string(runner.CredAnthropicSubscription):
		ad := claudecli.NewAgentAdapterForBinding(claudecli.Config{
			BinaryPath: w.d.ClaudeBinary, WorkspaceRoot: w.d.Syncer.LocalRoot(), MaxResponseBytes: w.d.MaxResponseBytes,
			Runner: w.d.HarnessRunner, CapabilityRules: rules,
		}, binding)
		return ad.RunAgent(ctx, mission)
	case string(runner.CredOpenAISubscription):
		p := codex.New(codex.Config{Binding: binding, BinaryPath: w.d.CodexBinary, Runner: w.d.HarnessRunner,
			MaxResponseBytes: w.d.MaxResponseBytes, CapabilityRules: rules})
		return p.RunAgent(ctx, mission)
	}
	return nil, fmt.Errorf("%w: %s", ErrUnsupportedAgentCredential, a.CredentialKind)
}

func toAgentResult(r *engine.AgentResult) *runneragent.AgentResult {
	if r == nil {
		return nil
	}
	out := &runneragent.AgentResult{Summary: r.Summary, Model: r.Model, DroppedTools: r.DroppedTools,
		TokensIn: r.TokensIn, TokensOut: r.TokensOut, SessionID: r.SessionID}
	for _, s := range r.Steps {
		out.Steps = append(out.Steps, runneragent.AgentStep{Kind: s.Kind, Summary: s.Summary})
	}
	for _, e := range r.ModelEvidence {
		out.ModelEvidence = append(out.ModelEvidence, runneragent.ModelEvidence{Model: e.Model, Provider: e.Provider, Source: e.Source, Kind: string(e.Kind)})
	}
	return out
}

// jobFailure is named so it does not collide with executor.go's existing
// failure(runner.SourceOutcome, string) helper (P2 T9).
func jobFailure(err error) runneragent.Result {
	o := runner.ClassifyError(err)
	if o.Class == runner.OutcomeNone {
		o.Class = runner.OutcomeOther
	}
	return runneragent.Result{OK: false, ErrorText: err.Error(), Outcome: o}
}
```
