## Alignment repair — required replacement work (2026-11-20)

**Prepared-push receipt:** `Syncer.Push` records the exact prepared candidate returned by the server with its base/job/resource identity. It may update local copy metadata to "prepared for this job" but cannot claim that it is the committed server checkpoint until the durable commit acknowledgment is received. A server crash or executor output-validation failure restores the previous committed version before reuse. P3 T6/P3 T5 own server promotion; retain the result in the spool until P2 T4's durable inbox ACK, and the server retains it through the final validated commit/replay.

Extend `WorkspaceJob`, wire dispatch and local copy inventory with the resource claim and pinned base ref produced by P3 T6/P3 T7. The runner validates the entire scoped descriptor, and rejects incomplete evidence rather than inventing defaults. The source capability producer from P2 T9 must advertise agent/process capabilities only when this actual executor, command lookup and feature evidence support them. RR-09 decision 2026-11-22: that is op-level support only. Runners advertise no per-command inventory; individual commands (interpreter/command plus the node's `requires`) are checked only at accept time in `Supports`.

Before implementation add `TestSyncer_PushPreparesWithoutClaimingCommittedVersion`, `TestSyncer_RestartDiscardsUncommittedLocalCandidate`, `TestWorkspaceExecutor_MissingResourceClaimRejectedBeforeAccept`, and `TestWorkspaceExecutor_UnsupportedCommandNeverMutatesWorkspace`. In the real transfer fixture assert `Latest` remains the base after push and becomes the candidate only after a simulated validated commit transaction.

---

**Goal:** Let `arbitron-runner` execute `script`, `tool_call` and `agent_call` dispatches against a local copy of the run's workspace that it pulls (only differing files) from the committed server checkpoint and pushes back (changed files plus deletions, in hash-verified parts) after every successful step.

**Architecture:** The manifest algorithm is shared, not re-implemented: the runner walks its copy with `adapter/checkpoint.ScanEntries`, diffs with `runner.DiffManifests`/`runner.SameEntries` and encodes pushes with `checkpoint.WriteDelta` — exactly the functions the server store uses (P3 T2). The runner-agent domain (`domain/runneragent`, P2 T8) gains mirror types for process/agent/workspace jobs and a per-run copy record; adapters add a copy-record file store, an HTTP transfer client for `/api/v1/runners/transfers/*`, a `Syncer` (pull / mark-dirty / push), a hash-addressed bundle cache and `WorkspaceJobs` (process jobs through the P1 `LocalProcessRunner`, agent jobs through the linked `claudecli`/`codex` adapters at the canonical workspace path). The P2 T9 executor delegates `process`/`agent` ops to `WorkspaceJobs`; the P2 T9 `wsclient` translates the new wire fields.

**Tech Stack:** Go 1.25.5, stdlib `net/http`, `crypto/sha256`, testify, `httptest`.

**Spec / Design:** `docs/specs/2026-11-19-remote-runners-spec.md` (Recovery: "A workspace-changing step counts as complete only when its resulting workspace is recoverable elsewhere"); `docs/designs/2026-11-19-remote-runners-design-phase-3.md` §3.3 (Manifest, Pull, Push, Transport), §3.4 (runner env, script cwd), §5 "Deferred to plans → Runner binary", §8 Happy path (zero pulls after the first, a push after each step), Edge-case ("A lib script finds its siblings on a remote runner", "Passthrough env values come from the runner"), Error ("An interrupted upload never marks the step done"). package runneragent

import (
	"context"
	"time"

	"github.com/arbitron/arbitron/domain/runner"
	"github.com/google/uuid"
)

// CopyState is the runner's record of its local copy of one run's workspace.
// Base is the committed manifest the copy was last synced to (the push diff
// base); Dirty is set before a step runs and cleared by a pull or a push.
type CopyState struct {
	RunID             uuid.UUID       `json:"runId"`
	WorkDir           string          `json:"workDir"`
	HomeGeneration    int64           `json:"homeGeneration"`
	CheckpointVersion int64           `json:"checkpointVersion"`
	ManifestHash      string          `json:"manifestHash"`
	Dirty             bool            `json:"dirty"`
	Base              runner.Manifest `json:"base"`
	UpdatedAt         time.Time       `json:"updatedAt"`
}

// CopyStateStore persists copy records across runner restarts.
type CopyStateStore interface {
	Get(ctx context.Context, runID uuid.UUID) (CopyState, bool, error)
	Put(ctx context.Context, s CopyState) error
	Delete(ctx context.Context, runID uuid.UUID) error
	List(ctx context.Context) ([]CopyState, error)
}

// PullTarget is what the server says the copy must equal before a step.
type PullTarget struct {
	RunID             uuid.UUID
	HomeGeneration    int64
	CheckpointVersion int64
	ManifestHash      string
}

// ReuseLocalCopy: a copy is reused untouched only when the home generation
// AND the committed checkpoint version (and its manifest) match and no
// interrupted or failed attempt dirtied it (design §3.3 Pull).
func ReuseLocalCopy(s CopyState, t PullTarget) bool {
	return s.RunID == t.RunID && !s.Dirty &&
		s.HomeGeneration == t.HomeGeneration &&
		s.CheckpointVersion == t.CheckpointVersion &&
		s.ManifestHash == t.ManifestHash
}

	if err := os.MkdirAll(dir, 0o700); err != nil {
		return nil, fmt.Errorf("copies: %w", err)
	}
	return &FileStore{dir: dir}, nil
}

func (s *FileStore) path(id uuid.UUID) string { return filepath.Join(s.dir, id.String()+".json") }

func (s *FileStore) Get(_ context.Context, id uuid.UUID) (runneragent.CopyState, bool, error) {
	b, err := os.ReadFile(s.path(id))
	if errors.Is(err, fs.ErrNotExist) {
		return runneragent.CopyState{}, false, nil
	}
	if err != nil {
		return runneragent.CopyState{}, false, err
	}
	var st runneragent.CopyState
	if err := json.Unmarshal(b, &st); err != nil {
		return runneragent.CopyState{}, false, fmt.Errorf("copies: %s: %w", id, err)
	}
	return st, true, nil
}

func (s *FileStore) Put(_ context.Context, st runneragent.CopyState) error {
	b, err := json.Marshal(st)
	if err != nil {
		return err
	}
	tmp, err := os.CreateTemp(s.dir, ".copy-*")
	if err != nil {
		return err
	}
	defer os.Remove(tmp.Name())
	if err := tmp.Chmod(0o600); err != nil {
		tmp.Close()
		return err
	}
	if _, err := tmp.Write(b); err != nil {
		tmp.Close()
		return err
	}
	if err := tmp.Close(); err != nil {
		return err
	}
	return os.Rename(tmp.Name(), s.path(st.RunID))
}

func (s *FileStore) Delete(_ context.Context, id uuid.UUID) error {
	if err := os.Remove(s.path(id)); err != nil && !errors.Is(err, fs.ErrNotExist) {
		return err
	}
	return nil
}

func (s *FileStore) List(ctx context.Context) ([]runneragent.CopyState, error) {
	entries, err := os.ReadDir(s.dir)
	if err != nil {
		return nil, err
	}
	var out []runneragent.CopyState
	for _, e := range entries {
		name := e.Name()
		if !strings.HasSuffix(name, ".json") || strings.HasPrefix(name, ".") {
			continue
		}
		id, err := uuid.Parse(strings.TrimSuffix(name, ".json"))
		if err != nil {
			continue
		}
		st, ok, err := s.Get(ctx, id)
		if err != nil {
			return nil, err
		}
		if ok {
			out = append(out, st)
		}
	}
	return out, nil
}

// adapter/runneragent/transfer/client.go
// Package transfer is the runner side of /api/v1/runners/transfers/*.
package transfer

import (
	"bytes"
	"context"
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"net/http"
	"net/url"
	"strconv"
	"strings"

	"github.com/arbitron/arbitron/domain/runner"
)

const transfersPath = "/api/v1/runners/transfers"

// PushCommit finalises a push: the number of parts and the SHA-256 of the
// whole delta stream (the expected manifest travels inside the delta's
// DeltaControl entry, P3 T2).
type PushCommit struct {
	Parts  int    `json:"parts"`
	SHA256 string `json:"sha256"`
}

// Transfer is what the runner's sync and bundle cache need from the server.
type Transfer interface {
	Manifest(ctx context.Context, token string) (runner.Manifest, error)
	FetchFile(ctx context.Context, token, path string) (io.ReadCloser, error)
	PutPart(ctx context.Context, token string, index int, data []byte) error
	CommitPush(ctx context.Context, token string, c PushCommit) (runner.CheckpointRef, error)
	FetchBundle(ctx context.Context, token, hash string) (io.ReadCloser, error)
}

// Client implements Transfer over HTTPS. The runner key travels only in
// `Authorization: Runner <key>`; extra headers (Cloudflare Access) go on
// every request.
type Client struct {
	base  string
	key   string
	hc    *http.Client
	extra http.Header
}

var _ Transfer = (*Client)(nil)

func NewClient(serverURL, runnerKey string, hc *http.Client, extraHeaders http.Header) *Client {
	if hc == nil {
		hc = http.DefaultClient
	}
	return &Client{base: strings.TrimRight(serverURL, "/") + transfersPath, key: runnerKey, hc: hc, extra: extraHeaders.Clone()}
}

func (c *Client) do(ctx context.Context, method, path, token string, body []byte, hdr map[string]string) (*http.Response, error) {
	route := strings.SplitN(path, "?", 2)[0]
	var rd io.Reader
	if body != nil {
		rd = bytes.NewReader(body)
	}
	req, err := http.NewRequestWithContext(ctx, method, c.base+path, rd)
	if err != nil {
		return nil, err
	}
	for k, values := range c.extra {
		for _, value := range values { req.Header.Add(k, value) }
	}
	for k, v := range hdr {
		req.Header.Set(k, v)
	}
	req.Header.Set("Authorization", "Runner "+c.key)
	req.Header.Set("X-Arbitron-Transfer", token)
	resp, err := c.hc.Do(req)
	if err != nil {
		return nil, fmt.Errorf("transfer %s %s: %w", method, route, err)
	}
	if resp.StatusCode/100 != 2 {
		msg, _ := io.ReadAll(io.LimitReader(resp.Body, 512))
		resp.Body.Close()
		return nil, fmt.Errorf("transfer %s %s: status %d: %s", method, route, resp.StatusCode, strings.TrimSpace(string(msg)))
	}
	return resp, nil
}

func (c *Client) Manifest(ctx context.Context, token string) (runner.Manifest, error) {
	resp, err := c.do(ctx, http.MethodGet, "/manifest", token, nil, nil)
	if err != nil {
		return runner.Manifest{}, err
	}
	defer resp.Body.Close()
	var m runner.Manifest
	if err := json.NewDecoder(resp.Body).Decode(&m); err != nil {
		return runner.Manifest{}, fmt.Errorf("transfer manifest: %w", err)
	}
	return m, nil
}

func (c *Client) FetchFile(ctx context.Context, token, path string) (io.ReadCloser, error) {
	resp, err := c.do(ctx, http.MethodGet, "/files?path="+url.QueryEscape(path), token, nil, nil)
	if err != nil {
		return nil, err
	}
	return resp.Body, nil
}

func (c *Client) PutPart(ctx context.Context, token string, index int, data []byte) error {
	sum := sha256.Sum256(data)
	resp, err := c.do(ctx, http.MethodPut, "/push/"+strconv.Itoa(index), token, data,
		map[string]string{"X-Part-SHA256": hex.EncodeToString(sum[:]), "Content-Type": "application/octet-stream"})
	if err != nil {
		return err
	}
	return resp.Body.Close()
}

func (c *Client) CommitPush(ctx context.Context, token string, pc PushCommit) (runner.CheckpointRef, error) {
	body, err := json.Marshal(pc)
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	resp, err := c.do(ctx, http.MethodPost, "/push/commit", token, body, map[string]string{"Content-Type": "application/json"})
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	defer resp.Body.Close()
	var ref runner.CheckpointRef
	if err := json.NewDecoder(resp.Body).Decode(&ref); err != nil {
		return runner.CheckpointRef{}, fmt.Errorf("transfer commit: %w", err)
	}
	return ref, nil
}

func (c *Client) FetchBundle(ctx context.Context, token, hash string) (io.ReadCloser, error) {
	if !runner.ValidBundleHash(hash) {
		return nil, errors.New("transfer bundle: malformed hash")
	}
	resp, err := c.do(ctx, http.MethodGet, "/bundles/"+hash, token, nil, nil)
	if err != nil {
		return nil, err
	}
	return resp.Body, nil
}

func (s *Server) FetchFile(ctx context.Context, token, path string) (io.ReadCloser, error) {
	if token != PullToken {
		return nil, errToken
	}
	s.mu.Lock()
	s.fetches = append(s.fetches, path)
	s.mu.Unlock()
	b, err := s.ReadLatest(ctx, path)
	if err != nil {
		return nil, err
	}
	return io.NopCloser(bytes.NewReader(b)), nil

func (s *Server) PutPart(_ context.Context, token string, i int, data []byte) error {
	if token != PushToken {
		return errToken
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if i == s.failPart && s.failTimes > 0 {
		s.failTimes--
		return errors.New("transfertest: connection reset mid-transfer")
	}
	s.parts[i] = append([]byte(nil), data...)
	return nil
}

func (s *Server) CommitPush(ctx context.Context, token string, c transfer.PushCommit) (runner.CheckpointRef, error) {
	if token != PushToken {
		return runner.CheckpointRef{}, errToken
	}
	s.mu.Lock()
	var all []byte
	for i := 0; i < c.Parts; i++ {
		p, ok := s.parts[i]
		if !ok {
			s.mu.Unlock()
			return runner.CheckpointRef{}, fmt.Errorf("transfertest: missing part %d", i)
		}
		all = append(all, p...)
	}
	s.parts = map[int][]byte{}
	s.mu.Unlock()
	sum := sha256.Sum256(all)
	if hex.EncodeToString(sum[:]) != c.SHA256 {
		return runner.CheckpointRef{}, errors.New("transfertest: delta hash mismatch")
	}
	base, err := s.latest(ctx)
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	staged, err := s.store.StageDelta(ctx, s.runID, base, bytes.NewReader(all))
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	ref, err := s.store.Publish(ctx, staged, runner.Fence{JobID: uuid.New(), RunnerID: "transfertest", Generation: 1})
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	s.mu.Lock()
	s.commits++
	s.lastParts = c.Parts
	s.mu.Unlock()
	return ref, nil
}

func (s *Server) FetchBundle(_ context.Context, token, hash string) (io.ReadCloser, error) {
	if token != BundleToken {
		return nil, errToken
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	s.bundleFetches++
	b, ok := s.bundles[hash]
	if !ok {
		return nil, errors.New("transfertest: status 404: bundle not found")
	}
	return io.NopCloser(bytes.NewReader(b)), nil
}

// memCatalog is an in-memory runner.CheckpointCatalog. Fences are accepted
// unchecked: fencing is P3 T2/T6's concern and is tested there.
type memCatalog struct {
	mu   sync.Mutex
	refs map[uuid.UUID][]catRow
}

type catRow struct {
	ref  runner.CheckpointRef
	path string
}

func newMemCatalog() *memCatalog { return &memCatalog{refs: map[uuid.UUID][]catRow{}} }

func (c *memCatalog) Publish(_ context.Context, ref runner.CheckpointRef, p string, _ int64, _ *runner.Fence) error {
	c.mu.Lock()
	defer c.mu.Unlock()
	rows := c.refs[ref.RunID]
	if int64(len(rows))+1 != ref.Version {
		return runner.ErrCheckpointConflict
	}
	c.refs[ref.RunID] = append(rows, catRow{ref, p})
	return nil
}

func (c *memCatalog) Latest(_ context.Context, runID uuid.UUID) (runner.CheckpointRef, string, bool, error) {
	c.mu.Lock()
	defer c.mu.Unlock()
	rows := c.refs[runID]
	if len(rows) == 0 {
		return runner.CheckpointRef{}, "", false, nil
	}
	r := rows[len(rows)-1]
	return r.ref, r.path, true, nil
}

func (c *memCatalog) Get(_ context.Context, runID uuid.UUID, v int64) (runner.CheckpointRef, string, error) {
	c.mu.Lock()
	defer c.mu.Unlock()
	rows := c.refs[runID]
	if v < 1 || v > int64(len(rows)) {
		return runner.CheckpointRef{}, "", runner.ErrCheckpointNotFound
	}
	return rows[v-1].ref, rows[v-1].path, nil
}

func (c *memCatalog) DeleteRun(_ context.Context, runID uuid.UUID) error {
	c.mu.Lock()
	defer c.mu.Unlock()
	delete(c.refs, runID)
	return nil
}

// adapter/runneragent/executor/bundle_cache.go
package executor

import (
	"bytes"
	"context"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"fmt"
	"io"
	"os"
	"path/filepath"
	"sync"

	"github.com/arbitron/arbitron/adapter/process"
	"github.com/arbitron/arbitron/adapter/runneragent/transfer"
	"github.com/arbitron/arbitron/domain/runner"
)

// BundleCache keeps verified bundle archives by hash under dir
// (~/.arbitron-runner/bundles) and unpacks a fresh copy per job, so a script
// that writes into its cwd never alters the cached bundle.
type BundleCache struct {
	dir string
	t   transfer.Transfer
	mu  sync.Mutex
}

func NewBundleCache(dir string, t transfer.Transfer) (*BundleCache, error) {
	if err := os.MkdirAll(dir, 0o700); err != nil {
		return nil, err
	}
	return &BundleCache{dir: dir, t: t}, nil
}

func (c *BundleCache) archive(ctx context.Context, hash, token string) (string, error) {
	if !runner.ValidBundleHash(hash) {
		return "", errors.New("bundle cache: malformed hash")
	}
	c.mu.Lock()
	defer c.mu.Unlock()
	p := filepath.Join(c.dir, hash+".tar.gz")
	if _, err := os.Stat(p); err == nil {
		return p, nil
	}
	rc, err := c.t.FetchBundle(ctx, token, hash)
	if err != nil {
		return "", err
	}
	defer rc.Close()
	data, err := io.ReadAll(io.LimitReader(rc, process.MaxBundleBytes+1))
	if err != nil {
		return "", err
	}
	sum := sha256.Sum256(data)
	if hex.EncodeToString(sum[:]) != hash {
		return "", fmt.Errorf("bundle cache: %w", runner.ErrBundleHashMismatch)
	}
	tmp, err := os.CreateTemp(c.dir, ".bundle-*")
	if err != nil {
		return "", err
	}
	defer os.Remove(tmp.Name())
	if _, err := tmp.Write(data); err != nil {
		tmp.Close()
		return "", err
	}
	if err := tmp.Close(); err != nil {
		return "", err
	}
	return p, os.Rename(tmp.Name(), p)
}

// Unpack returns a fresh per-job directory holding the bundle's files.
func (c *BundleCache) Unpack(ctx context.Context, hash, token string) (string, func(), error) {
	p, err := c.archive(ctx, hash, token)
	if err != nil {
		return "", nil, err
	}
	data, err := os.ReadFile(p)
	if err != nil {
		return "", nil, err
	}
	dir, err := os.MkdirTemp(c.dir, "run-"+hash[:12]+"-*")
	if err != nil {
		return "", nil, err
	}
	cleanup := func() { _ = os.RemoveAll(dir) }
	if err := process.ExtractScriptBundle(bytes.NewReader(data), dir); err != nil {
		cleanup()
		return "", nil, err
	}
	return dir, cleanup, nil
}

// adapter/runneragent/executor/workspace_jobs.go
package executor

import (
	"context"
	"errors"
	"fmt"
	"os/exec"
	"time"

	"github.com/arbitron/arbitron/adapter/claudecli"
	"github.com/arbitron/arbitron/adapter/codex"
	"github.com/arbitron/arbitron/adapter/runneragent/workspace"
	"github.com/arbitron/arbitron/domain/engine"
	"github.com/arbitron/arbitron/domain/provider"
	"github.com/arbitron/arbitron/domain/runner"
	"github.com/arbitron/arbitron/domain/runneragent"
)

var (
	// ErrCommandUnavailable: the interpreter / command / harness is not on
	// this runner (RR-09: only a PATH lookup is verified).
	ErrCommandUnavailable = errors.New("command_unavailable")
	// ErrUnsupportedAgentCredential: remote agent jobs run on the runner's own
	// Claude or Codex subscription only.
	ErrUnsupportedAgentCredential = errors.New("agent credential kind is not supported on remote runners")
)

type WorkspaceJobsDeps struct {
	Syncer           *workspace.Syncer
	Bundles          *BundleCache
	Process          engine.ProcessRunner // process.NewLocalProcessRunner(): env built HERE from passthrough names
	ClaudeBinary     string
	CodexBinary      string
	Rules            provider.CapabilityRules // loaded locally, required in production
	MaxResponseBytes int64
	HarnessRunner    claudecli.CommandRunner // nil: production os/exec runner (claude and codex)
	LookPath         func(string) (string, error)
}

// WorkspaceJobs executes OpProcess and OpAgent dispatches:
// pull → mark dirty → run → push → result. A failed step is not pushed (the
// copy stays dirty and the next pull restores the committed checkpoint).
type WorkspaceJobs struct{ d WorkspaceJobsDeps }

func NewWorkspaceJobs(d WorkspaceJobsDeps) (*WorkspaceJobs, error) {
	if d.Syncer == nil || d.Bundles == nil || d.Process == nil {
		return nil, errors.New("workspace jobs: Syncer, Bundles and Process are required")
	}
	if d.LookPath == nil {
		d.LookPath = exec.LookPath
	}
	if d.ClaudeBinary == "" {
		d.ClaudeBinary = "claude"
	}
	if d.CodexBinary == "" {
		d.CodexBinary = "codex"
	}
	return &WorkspaceJobs{d: d}, nil
}

// Supports is the accept-time check (runneragent.JobExecutor contract): a
// dispatch this runner cannot execute is rejected so the dispatcher re-places it.
func (w *WorkspaceJobs) Supports(d runneragent.Dispatch) (bool, string) {
	if err := w.check(d); err != nil {
		return false, err.Error()
	}
	return true, ""
}

func (w *WorkspaceJobs) check(d runneragent.Dispatch) error {
	if d.Workspace == nil {
		return errors.New("workspace job without a workspace plan")
	}
	if _, err := w.d.Syncer.LocalPath(d.Workspace.WorkDir); err != nil {
		return err
	}
	switch d.Op {
	case runneragent.OpProcess:
		if d.Process == nil {
			return errors.New("process dispatch without payload")
		}
		if d.Process.Kind == string(engine.ProcessScript) && (d.Process.BundleHash == "" || d.Process.BundleEntrypoint == "") {
			return runner.ErrBundleRequired
		}
		// RR-09 decision 2026-11-22: interpreter/command first, then every
		// declared `requires` name, in order; the first miss rejects. LookPath
		// resolves against the PATH of the environment this runner builds for
		// the child (procenv.BaseEnv on the runner), never the server's.
		for _, name := range append([]string{d.Process.Command}, d.Process.Requires...) {
			if _, err := w.d.LookPath(name); err != nil {
				return fmt.Errorf("%w: %s", ErrCommandUnavailable, name)
			}
		}
	case runneragent.OpAgent:
		if d.Agent == nil {
			return errors.New("agent dispatch without payload")
		}
		bin, err := w.harness(d.Agent)
		if err != nil {
			return err
		}
		if _, err := w.d.LookPath(bin); err != nil {
			return fmt.Errorf("%w: %s", ErrCommandUnavailable, bin)
		}
	default:
		return fmt.Errorf("workspace jobs: op %q is not handled here", d.Op)
	}
	return nil
}

func (w *WorkspaceJobs) harness(a *runneragent.AgentJob) (string, error) {
	switch {
	case a.CredentialKind == string(runner.CredAnthropicSubscription) && (a.Backend == "" || a.Backend == "claude-code"):
		return w.d.ClaudeBinary, nil
	case a.CredentialKind == string(runner.CredOpenAISubscription) && a.Backend == "codex":
		return w.d.CodexBinary, nil
	}
	return "", fmt.Errorf("%w: %s/%s", ErrUnsupportedAgentCredential, a.CredentialKind, a.Backend)
}

// Execute runs one workspace dispatch. Errors are classified into the
// result's Outcome (runner.ClassifyError); the text is the error's own
// (adapters sanitize before returning).
func (w *WorkspaceJobs) Execute(ctx context.Context, d runneragent.Dispatch) runneragent.Result {
	if err := w.check(d); err != nil {
		return failure(err)
	}
	local, err := w.d.Syncer.Pull(ctx, *d.Workspace)
	if err != nil {
		return failure(fmt.Errorf("pull: %w", err))
	}
	if err := w.d.Syncer.MarkDirty(ctx, *d.Workspace); err != nil {
		return failure(err)
	}
	var res runneragent.Result
	switch d.Op {
	case runneragent.OpProcess:
		pr, err := w.runProcess(ctx, d.Process, local)
		if err != nil {
			return failure(err)
		}
		res = runneragent.Result{OK: true, Process: &runneragent.ProcessResult{Stdout: pr.Stdout, Stderr: pr.Stderr, ExitCode: pr.ExitCode}}
		if pr.ExitCode != 0 {
			return res // completed but failed: nothing is pushed
		}
	case runneragent.OpAgent:
		ar, err := w.runAgent(ctx, d.Agent, local)
		if err != nil {
			return failure(err)
		}
		res = runneragent.Result{OK: true, Agent: toAgentResult(ar)}
	}
	if _, err := w.d.Syncer.Push(ctx, *d.Workspace); err != nil {
		return failure(fmt.Errorf("push: %w", err))
	}
	return res
}

func (w *WorkspaceJobs) runProcess(ctx context.Context, p *runneragent.ProcessJob, local string) (engine.ProcessResult, error) {
	spec := engine.ProcessSpec{
		Kind: engine.ProcessKind(p.Kind), Command: p.Command, Args: p.Args, Stdin: p.Stdin,
		NodeEnv: p.NodeEnv, Passthrough: p.Passthrough, TimeoutSecs: p.TimeoutSecs, WorkDir: p.WorkDir,
	}
	switch spec.Kind {
	case engine.ProcessScript:
		dir, cleanup, err := w.d.Bundles.Unpack(ctx, p.BundleHash, p.BundleToken)
		if err != nil {
			return engine.ProcessResult{}, fmt.Errorf("bundle: %w", err)
		}
		defer cleanup()
		spec.Dir = dir // the script's own folder, exactly as on the server
		spec.Bundle = &engine.ScriptBundleRef{Hash: p.BundleHash, Entrypoint: p.BundleEntrypoint}
	case engine.ProcessToolCall:
		spec.Dir = local // the workspace copy
	default:
		return engine.ProcessResult{}, fmt.Errorf("unknown process kind %q", p.Kind)
	}
	return w.d.Process.RunProcess(ctx, spec)
}

func (w *WorkspaceJobs) runAgent(ctx context.Context, a *runneragent.AgentJob, local string) (*engine.AgentResult, error) {
	binding := provider.ResolvedBinding{
		Kind: provider.ProfileKindAgent, Provider: a.Provider, Model: a.Model, Backend: a.Backend,
		Pinned: a.RequireExecutedModel, AllowedTools: append([]string(nil), a.ToolScope...),
		Effort: provider.Effort(a.Effort), ContextWindow: provider.ContextWindow(a.ContextWindow),
	}
	rules := w.d.Rules // the same runner-local rules used to advertise capabilities
	if rules.Empty() { return nil, errors.New("workspace jobs: local capability rules required") }
	mission := engine.AgentMission{
		Model: a.Model, Prompt: a.Prompt, WorkDir: local, ToolScope: a.ToolScope, SessionID: a.SessionID,
		Timeout: time.Duration(a.TimeoutSecs) * time.Second,
	}
	switch a.CredentialKind {
	case string(runner.CredAnthropicSubscription):
		ad := claudecli.NewAgentAdapterForBinding(claudecli.Config{
			BinaryPath: w.d.ClaudeBinary, WorkspaceRoot: w.d.Syncer.LocalRoot(), MaxResponseBytes: w.d.MaxResponseBytes,
			Runner: w.d.HarnessRunner, CapabilityRules: rules,
		}, binding)
		return ad.RunAgent(ctx, mission)
	case string(runner.CredOpenAISubscription):
		p := codex.New(codex.Config{Binding: binding, BinaryPath: w.d.CodexBinary, Runner: w.d.HarnessRunner,
			MaxResponseBytes: w.d.MaxResponseBytes, CapabilityRules: rules})
		return p.RunAgent(ctx, mission)
	}
	return nil, fmt.Errorf("%w: %s", ErrUnsupportedAgentCredential, a.CredentialKind)
}

func toAgentResult(r *engine.AgentResult) *runneragent.AgentResult {
	if r == nil {
		return nil
	}
	out := &runneragent.AgentResult{Summary: r.Summary, Model: r.Model, DroppedTools: r.DroppedTools,
		TokensIn: r.TokensIn, TokensOut: r.TokensOut, SessionID: r.SessionID}
	for _, s := range r.Steps {
		out.Steps = append(out.Steps, runneragent.AgentStep{Kind: s.Kind, Summary: s.Summary})
	}
	for _, e := range r.ModelEvidence {
		out.ModelEvidence = append(out.ModelEvidence, runneragent.ModelEvidence{Model: e.Model, Provider: e.Provider, Source: e.Source, Kind: string(e.Kind)})
	}
	return out
}

func failure(err error) runneragent.Result {
	o := runner.ClassifyError(err)
	if o.Class == runner.OutcomeNone {
		o.Class = runner.OutcomeOther
	}
	return runneragent.Result{OK: false, ErrorText: err.Error(), Outcome: o}
}

		w = &v
	}
