# Remote Runners — Phase 3 Task 6: Workspace sync service and dispatcher integration

**Goal:** Keep the committed server checkpoint authoritative for every workspace step: take (or move) home ownership before the step runs, snapshot the built-in runner's workspace after success and roll it back after interruption or failure, accept a remote runner's pushed checkpoint only under the job fence, refuse to complete a remote step whose push never landed, and re-place a step whose pull failed with that runner ineligible.

**Architecture:** `domain/runner/sync.go` is a pure domain service (`WorkspaceSync`) over the `CheckpointStore`, `HomeStore`, `Ledger`, `TransferTokens`, `EventRecorder` and `ExecutionStatusProjector` ports. The dispatcher calls it through a narrow `StepWorkspace` interface around each reserved attempt of an `AffinityWorkspace` step (`BeginAttempt` → runner → `FinishSuccess`/`FinishFailure`); the hub (P3 T8) calls `PreparePull` for remote runners and the transfer endpoint (P3 T7) calls `AcceptPush` and `ReportProgress`. Agent steps with a workspace now request `AffinityWorkspace`; process steps get theirs from P3 T8's `ProcessAffinity`.

**Tech Stack:** Go 1.25, `github.com/google/uuid`, testify; tests use the real SQLite ledger and home repos (`adapter/sqlite`) from an external test package (precedent `domain/layout/superseded_sqlite_test.go`) and an in-memory `CheckpointStore` fake that honours run IDs, versions and fences.

**Spec / Design:** spec §"Workspace mobility" (home, stay-versus-move, sync, completion boundary); design phase 3 §3.1, §3.3 (pull/push/progress/built-in runner/completion boundary), §4 (data flow "a workspace step that moves"), §7 (push failure, pull failure, disk full), §8 (Error: "a pull failure re-places the step", "an interrupted upload never marks the step done"; regressions: "built-in interrupted edits roll back", "two successful steps on one home publish distinct manifests", "failed pushes restore the previous checkpoint before retry", "two workspace writers cannot overlap"). Contracts: `2026-11-20-contracts.md` §6 (Affinity, ExecRequirements), §7 (Decision.Tier), §8 (Ledger, Fence, PoolEvent), §9 (`workspace.moved`, `execution.status`), §10 (`moving_workspace`, `MoveProgress`), §11 (StepJob, Assignment, DispatcherDeps), §16 (Workspace sync service, transfer tokens).

// wsVersion is one committed checkpoint held by wsStore.
type wsVersion struct {
	ref   runner.CheckpointRef
	files map[string]string
}

// wsStore is an in-memory CheckpointStore. It honours run IDs, versions,
// manifest hashes and fences: Restore requires the exact ref, Publish requires
// the latest version as base and a live fence.
type wsStore struct {
	mu        sync.Mutex
	ledger    runner.Ledger
	runs      map[uuid.UUID][]wsVersion
	restores  []runner.CheckpointRef
	discarded int
}

func newWSStore(l runner.Ledger) *wsStore {
	return &wsStore{ledger: l, runs: map[uuid.UUID][]wsVersion{}}
}

func wsReadTree(dir string) (map[string]string, error) {
	files := map[string]string{}
	err := filepath.WalkDir(dir, func(p string, d fs.DirEntry, err error) error {
		if err != nil || d.IsDir() {
			return err
		}
		b, err := os.ReadFile(p)
		if err != nil {
			return err
		}
		rel, err := filepath.Rel(dir, p)
		if err != nil {
			return err
		}
		files[filepath.ToSlash(rel)] = string(b)
		return nil
	})
	return files, err
}

func wsHash(files map[string]string) string {
	keys := make([]string, 0, len(files))
	for k := range files {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	h := sha256.New()
	for _, k := range keys {
		fmt.Fprintf(h, "%s=%s\n", k, files[k])
	}
	return hex.EncodeToString(h.Sum(nil))
}

func wsManifest(ref runner.CheckpointRef, files map[string]string) runner.Manifest {
	m := runner.Manifest{Version: ref.Version, Hash: ref.ManifestHash}
	for p, c := range files {
		m.Entries = append(m.Entries, runner.ManifestEntry{Path: p, Size: int64(len(c)), Mode: 0o644})
	}
	sort.Slice(m.Entries, func(i, j int) bool { return m.Entries[i].Path < m.Entries[j].Path })
	return m
}

func (s *wsStore) appendLocked(runID uuid.UUID, files map[string]string) runner.CheckpointRef {
	vs := s.runs[runID]
	ref := runner.CheckpointRef{RunID: runID, Version: int64(len(vs) + 1), ManifestHash: wsHash(files)}
	s.runs[runID] = append(vs, wsVersion{ref: ref, files: files})
	return ref
}

func (s *wsStore) findLocked(ref runner.CheckpointRef) (wsVersion, error) {
	for _, v := range s.runs[ref.RunID] {
		if v.ref == ref {
			return v, nil
		}
	}
	return wsVersion{}, fmt.Errorf("%w: %s", runner.ErrCheckpointNotFound, ref)
}

func (s *wsStore) Commit(_ context.Context, runID uuid.UUID, srcDir string) (runner.CheckpointRef, runner.Manifest, error) {
	files, err := wsReadTree(srcDir)
	if err != nil {
		return runner.CheckpointRef{}, runner.Manifest{}, err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	ref := s.appendLocked(runID, files)
	return ref, wsManifest(ref, files), nil
}

func (s *wsStore) Restore(_ context.Context, ref runner.CheckpointRef, dst string) error {
	s.mu.Lock()
	v, err := s.findLocked(ref)
	s.restores = append(s.restores, ref)
	s.mu.Unlock()
	if err != nil {
		return err
	}
	entries, err := os.ReadDir(dst)
	if err != nil {
		return err
	}
	for _, e := range entries {
		if err := os.RemoveAll(filepath.Join(dst, e.Name())); err != nil {
			return err
		}
	}
	for p, c := range v.files {
		full := filepath.Join(dst, filepath.FromSlash(p))
		if err := os.MkdirAll(filepath.Dir(full), 0o755); err != nil {
			return err
		}
		if err := os.WriteFile(full, []byte(c), 0o644); err != nil {
			return err
		}
	}
	return nil
}

func (s *wsStore) Manifest(_ context.Context, ref runner.CheckpointRef) (runner.Manifest, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	v, err := s.findLocked(ref)
	if err != nil {
		return runner.Manifest{}, err
	}
	return wsManifest(v.ref, v.files), nil
}

func (s *wsStore) Latest(_ context.Context, runID uuid.UUID) (runner.CheckpointRef, bool, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	vs := s.runs[runID]
	if len(vs) == 0 {
		return runner.CheckpointRef{}, false, nil
	}
	return vs[len(vs)-1].ref, true, nil
}

func (s *wsStore) StageDelta(context.Context, uuid.UUID, runner.CheckpointRef, io.Reader) (runner.StagedCheckpoint, error) {
	return runner.StagedCheckpoint{}, errors.New("wsStore: delta staging belongs to the adapter tests (P3 T2/T7)")
}

func (s *wsStore) Publish(ctx context.Context, staged runner.StagedCheckpoint, f runner.Fence) (runner.CheckpointRef, error) {
	if err := s.ledger.FenceCheck(ctx, f); err != nil {
		return runner.CheckpointRef{}, err
	}
	files, err := wsReadTree(staged.Dir)
	if err != nil {
		return runner.CheckpointRef{}, err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	vs := s.runs[staged.RunID]
	if len(vs) == 0 || vs[len(vs)-1].ref != staged.Base || staged.Version != staged.Base.Version+1 {
		return runner.CheckpointRef{}, runner.ErrCheckpointConflict
	}
	return s.appendLocked(staged.RunID, files), nil
}

func (s *wsStore) Discard(context.Context, runner.StagedCheckpoint) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.discarded++
	return nil
}

func (s *wsStore) ReadFile(_ context.Context, ref runner.CheckpointRef, path string) (io.ReadCloser, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	v, err := s.findLocked(ref)
	if err != nil {
		return nil, err
	}
	c, ok := v.files[path]
	if !ok {
		return nil, fs.ErrNotExist
	}
	return io.NopCloser(strings.NewReader(c)), nil
}

func (s *wsStore) Delete(_ context.Context, runID uuid.UUID) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	delete(s.runs, runID)
	return nil
}

func (s *wsStore) restoreCount() int {
	s.mu.Lock()
	defer s.mu.Unlock()
	return len(s.restores)
}

// wsTokens issues tokens that encode every scoping argument.
type wsTokens struct {
	mu     sync.Mutex
	issued []runner.TransferToken
}

func (t *wsTokens) Issue(_ context.Context, f runner.Fence, runID uuid.UUID, direction string, ttl time.Duration) (runner.TransferToken, error) {
	t.mu.Lock()
	defer t.mu.Unlock()

	t.mu.Lock()
	defer t.mu.Unlock()
	for _, tok := range t.issued {
		if tok.Token == token && tok.RunnerID == r {
			return tok, nil
		}
	}
	return runner.TransferToken{}, errors.New("wsTokens: unknown token")
}

package runner

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/arbitron/arbitron/domain/instance"
	"github.com/google/uuid"
)

// EventWorkspaceMoved is the run-scoped pool event recorded when a workspace
// home is handed to another runner.
const EventWorkspaceMoved = "workspace.moved"

// OutcomePullFailed classifies an attempt whose workspace could not be
// prepared on the placed runner. The dispatcher re-places the step with that
// runner ineligible for the rest of the step.
const OutcomePullFailed OutcomeClass = "pull_failed"

// MaxPullFailures bounds how many runners one step may exclude for pull
// failures before the step fails.
const MaxPullFailures = 3

// DefaultTransferTokenTTL is the lifetime of pull/push tokens.
const DefaultTransferTokenTTL = 15 * time.Minute

var (
	// ErrPushMissing: a remote workspace step finished without its pushed
	// checkpoint being published. The step is a failed attempt.
	ErrPushMissing = errors.New("runner: workspace step finished without a published checkpoint")
	// ErrNoWorkspace: a workspace-affinity step carried no workspace path.
	ErrNoWorkspace = errors.New("runner: workspace step has no workspace directory")
	// ErrTransfersUnavailable: no transfer token service is configured.
	ErrTransfersUnavailable = errors.New("runner: workspace transfers are not configured")
)

// WorkspaceSizer reports the size of a run's server-local workspace.
type WorkspaceSizer interface {
	InstanceDirStats(instanceID uuid.UUID) (bytes int64, files int, err error)
}

// SyncDeps are WorkspaceSync's collaborators.
type SyncDeps struct {
	Checkpoints CheckpointStore
	Homes       HomeStore
	Ledger      Ledger
	Tokens      TransferTokens
	Events      *EventRecorder
	Status      *ExecutionStatusProjector
	Clock       func() time.Time
	// Sizes reports server-local workspace sizes; nil => unknown (-1).
	Sizes WorkspaceSizer
	// TokenTTL is the transfer token lifetime; <= 0 => DefaultTransferTokenTTL.
	TokenTTL time.Duration
}

// PullPlan is what a remote runner needs to bring its copy to the committed
// checkpoint and push its result.
type PullPlan struct {
	RunID     uuid.UUID
	Ref       CheckpointRef
	Home      Home
	PullToken TransferToken
	PushToken TransferToken
}

// WorkspaceAttempt is one reserved attempt's workspace context.
type WorkspaceAttempt struct {
	RunID       uuid.UUID
	WorkDir     string
	Runner      RunnerID
	BaseVersion int64 // committed checkpoint version the attempt started from
	Moved       bool
}

// StepWorkspace is the dispatcher's port onto workspace sync.
type StepWorkspace interface {
	WorkspaceBytes(ctx context.Context, runID uuid.UUID) int64
	BeginAttempt(ctx context.Context, a Assignment, req ExecRequirements, d Decision, job *StepJob) (WorkspaceAttempt, error)
	FinishSuccess(ctx context.Context, a Assignment, att WorkspaceAttempt) (CheckpointRef, error)
	FinishFailure(ctx context.Context, a Assignment, att WorkspaceAttempt) error
}

// WorkspaceSync keeps the committed server checkpoint authoritative.
type WorkspaceSync struct{ d SyncDeps }

var _ StepWorkspace = (*WorkspaceSync)(nil)

// NewWorkspaceSync builds the service. Missing core collaborators panic: a
// workspace step without them could not be made safe.
func NewWorkspaceSync(d SyncDeps) *WorkspaceSync {
	if d.Checkpoints == nil || d.Homes == nil || d.Ledger == nil || d.Events == nil || d.Status == nil {
		panic("runner: NewWorkspaceSync requires Checkpoints, Homes, Ledger, Events and Status")
	}
	if d.Clock == nil {
		d.Clock = time.Now
	}
	if d.TokenTTL <= 0 {
		d.TokenTTL = DefaultTransferTokenTTL
	}
	return &WorkspaceSync{d: d}
}

func jobWorkDir(job *StepJob) string {
	switch {
	case job == nil:
		return ""
	case job.Agent != nil:
		return job.Agent.WorkDir
	case job.Process != nil:
		return job.Process.WorkDir
	}
	return ""
}

func isServerLocal(id RunnerID) bool { return id == ServerHome || id == BuiltInRunnerID }

// BeginAttempt takes (or moves) home ownership for the placed runner, makes
// sure a committed checkpoint exists, restores the built-in workspace when its
// local copy is not the committed one, and marks the home dirty.
func (s *WorkspaceSync) BeginAttempt(ctx context.Context, a Assignment, req ExecRequirements, dec Decision, job *StepJob) (WorkspaceAttempt, error) {
	runID, workDir := req.RunID, jobWorkDir(job)
	if runID == uuid.Nil || workDir == "" {
		return WorkspaceAttempt{}, ErrNoWorkspace
	}
	if err := s.d.Ledger.FenceCheck(ctx, a.Fence); err != nil {
		return WorkspaceAttempt{}, err
	}
	target := a.Fence.RunnerID
	home, err := s.d.Homes.GetHome(ctx, runID)
	if err != nil {
		return WorkspaceAttempt{}, fmt.Errorf("runner: reading workspace home: %w", err)
	}
	prevOwner, prevDirty := home.RunnerID, home.Dirty

	base, ok, err := s.d.Checkpoints.Latest(ctx, runID)
	if err != nil {
		return WorkspaceAttempt{}, fmt.Errorf("runner: reading latest checkpoint: %w", err)
	}
	baselined := false
	if !ok {
		if !isServerLocal(prevOwner) {
			return WorkspaceAttempt{}, fmt.Errorf("%w: run %s is homed on %q without a committed checkpoint", ErrCheckpointNotFound, runID, prevOwner)
		}
		// First workspace step: the server-local workspace is the truth.
		if base, _, err = s.d.Checkpoints.Commit(ctx, runID, workDir); err != nil {
			return WorkspaceAttempt{}, fmt.Errorf("runner: baseline checkpoint: %w", err)
		}
		if err := s.d.Homes.SetCheckpoint(ctx, runID, base.Version); err != nil {
			return WorkspaceAttempt{}, fmt.Errorf("runner: recording baseline checkpoint: %w", err)
		}
		baselined = true
	}

	moved := false
	switch {
	case prevOwner == target:
	case prevOwner == ServerHome && target == BuiltInRunnerID:
		if home, err = s.d.Homes.AcquireResource(ctx, runID, target, home.Generation); err != nil {
			return WorkspaceAttempt{}, fmt.Errorf("runner: acquiring workspace: %w", err)
		}
	default:
		reason := dec.Reason
		if dec.Tier != "" {
			reason = dec.Tier + ": " + dec.Reason
		}
		s.d.Status.Publish(ctx, a.Job, instance.ExecMovingWorkspace)
		if home, err = s.d.Homes.HandOff(ctx, runID, target, reason, req.WorkspaceBytes, home.Generation); err != nil {
			return WorkspaceAttempt{}, fmt.Errorf("runner: moving workspace: %w", err)
		}
		moved = true
		rid := runID
		s.d.Events.Record(ctx, PoolEvent{
			Kind: EventWorkspaceMoved, RunnerID: target, RunID: &rid, Cause: Cause{Kind: "placement"},
			Detail: map[string]any{
				"from": string(prevOwner), "to": string(target), "reason": reason, "tier": dec.Tier,
				"bytes": req.WorkspaceBytes, "generation": home.Generation, "jobId": a.Job.ID.String(),
			},
		})
	}

	if target == BuiltInRunnerID && !baselined && (prevOwner != BuiltInRunnerID || prevDirty) {
		if err := s.d.Checkpoints.Restore(ctx, base, workDir); err != nil {
			return WorkspaceAttempt{}, fmt.Errorf("runner: restoring built-in workspace: %w", err)
		}
	}
	if err := s.d.Homes.MarkDirty(ctx, runID, true); err != nil {
		return WorkspaceAttempt{}, fmt.Errorf("runner: marking workspace dirty: %w", err)
	}
	return WorkspaceAttempt{RunID: runID, WorkDir: workDir, Runner: target, BaseVersion: base.Version, Moved: moved}, nil
}

// FinishSuccess publishes the attempt's result: a snapshot of the built-in
// workspace, or confirmation that the remote runner's push was published.
func (s *WorkspaceSync) FinishSuccess(ctx context.Context, a Assignment, att WorkspaceAttempt) (CheckpointRef, error) {
	if att.Runner == BuiltInRunnerID {
		return s.CommitBuiltIn(ctx, a.Fence, att.RunID, att.WorkDir)
	}
	ref, ok, err := s.d.Checkpoints.Latest(ctx, att.RunID)
	if err != nil {
		return CheckpointRef{}, fmt.Errorf("runner: reading latest checkpoint: %w", err)
	}
	if !ok || ref.Version <= att.BaseVersion {
		return CheckpointRef{}, fmt.Errorf("%w: run %s is still at version %d", ErrPushMissing, att.RunID, att.BaseVersion)
	}
	return ref, nil
}

// FinishFailure rolls back a failed attempt. The built-in workspace is
// restored to the committed checkpoint; a remote runner's copy stays dirty and
// is restored by that runner before reuse (the server checkpoint never moved).
func (s *WorkspaceSync) FinishFailure(ctx context.Context, a Assignment, att WorkspaceAttempt) error {
	if att.Runner != BuiltInRunnerID {
		return nil
	}
	return s.RestoreBuiltIn(ctx, att.RunID, att.WorkDir)
}

// CommitBuiltIn snapshots the built-in workspace as the next checkpoint under
// the job fence; the built-in runner must own the home.
func (s *WorkspaceSync) CommitBuiltIn(ctx context.Context, f Fence, runID uuid.UUID, dir string) (CheckpointRef, error) {
	if f.RunnerID != BuiltInRunnerID {
		return CheckpointRef{}, fmt.Errorf("runner: CommitBuiltIn for runner %q", f.RunnerID)
	}
	if err := s.d.Ledger.FenceCheck(ctx, f); err != nil {
		return CheckpointRef{}, err
	}
	if err := s.requireOwner(ctx, runID, BuiltInRunnerID); err != nil {
		return CheckpointRef{}, err
	}
	ref, _, err := s.d.Checkpoints.Commit(ctx, runID, dir)
	if err != nil {
		return CheckpointRef{}, fmt.Errorf("runner: committing built-in workspace: %w", err)
	}
	if err := s.d.Homes.SetCheckpoint(ctx, runID, ref.Version); err != nil {
		return CheckpointRef{}, fmt.Errorf("runner: recording checkpoint: %w", err)
	}
	if err := s.d.Homes.MarkDirty(ctx, runID, false); err != nil {
		return CheckpointRef{}, fmt.Errorf("runner: clearing dirty: %w", err)
	}
	return ref, nil
}

// RestoreBuiltIn resets dir to the run's latest committed checkpoint.
func (s *WorkspaceSync) RestoreBuiltIn(ctx context.Context, runID uuid.UUID, dir string) error {
	ref, ok, err := s.d.Checkpoints.Latest(ctx, runID)
	if err != nil {
		return fmt.Errorf("runner: reading latest checkpoint: %w", err)
	}
	if !ok {
		return fmt.Errorf("%w: run %s", ErrCheckpointNotFound, runID)
	}
	if err := s.d.Checkpoints.Restore(ctx, ref, dir); err != nil {
		return fmt.Errorf("runner: restoring built-in workspace: %w", err)
	}
	home, err := s.d.Homes.GetHome(ctx, runID)
	if err != nil {
		return fmt.Errorf("runner: reading workspace home: %w", err)
	}
	if isServerLocal(home.RunnerID) {
		if err := s.d.Homes.MarkDirty(ctx, runID, false); err != nil {
			return fmt.Errorf("runner: clearing dirty: %w", err)
		}
	}
	return nil
}

func (s *WorkspaceSync) requireOwner(ctx context.Context, runID uuid.UUID, owner RunnerID) error {
	home, err := s.d.Homes.GetHome(ctx, runID)
	if err != nil {
		return fmt.Errorf("runner: reading workspace home: %w", err)
	}
	if home.RunnerID != owner {
		return fmt.Errorf("%w: run %s is homed on %q, not %q", ErrHomeOwned, runID, home.RunnerID, owner)
	}
	return nil
}

// WorkspaceBytes is the move-size input for placement: the server-local size
// for server/built-in homes, the committed manifest size for remote homes,
// -1 when unknown (no load-driven moves).
func (s *WorkspaceSync) WorkspaceBytes(ctx context.Context, runID uuid.UUID) int64 {
	home, err := s.d.Homes.GetHome(ctx, runID)
	if err != nil {
		return -1
	}
	if !isServerLocal(home.RunnerID) {
		ref, ok, err := s.d.Checkpoints.Latest(ctx, runID)
		if err != nil || !ok {
			return -1
		}
		m, err := s.d.Checkpoints.Manifest(ctx, ref)
		if err != nil {
			return -1
		}
		return m.TotalBytes()
	}
	if s.d.Sizes == nil {
		return -1
	}
	b, _, err := s.d.Sizes.InstanceDirStats(runID)
	if err != nil {
		return -1
	}
	return b
}

// PreparePull returns the committed checkpoint and fenced transfer tokens for
// the runner that now holds the home. Called by the hub when it dispatches a
// workspace step to a remote runner.
func (s *WorkspaceSync) PreparePull(ctx context.Context, a Assignment, runID uuid.UUID) (PullPlan, error) {
	if s.d.Tokens == nil {
		return PullPlan{}, ErrTransfersUnavailable
	}
	if err := s.d.Ledger.FenceCheck(ctx, a.Fence); err != nil {
		return PullPlan{}, err
	}
	home, err := s.d.Homes.GetHome(ctx, runID)
	if err != nil {
		return PullPlan{}, fmt.Errorf("runner: reading workspace home: %w", err)
	}
	if home.RunnerID != a.Fence.RunnerID {
		return PullPlan{}, fmt.Errorf("%w: run %s is homed on %q, not %q", ErrHomeOwned, runID, home.RunnerID, a.Fence.RunnerID)
	}
	ref, ok, err := s.d.Checkpoints.Latest(ctx, runID)
	if err != nil {
		return PullPlan{}, fmt.Errorf("runner: reading latest checkpoint: %w", err)
	}
	if !ok {
		return PullPlan{}, fmt.Errorf("%w: run %s", ErrCheckpointNotFound, runID)
	}
	pull, err := s.d.Tokens.Issue(ctx, a.Fence, runID, TransferPull, s.d.TokenTTL)
	if err != nil {
		return PullPlan{}, fmt.Errorf("runner: issuing pull token: %w", err)
	}
	push, err := s.d.Tokens.Issue(ctx, a.Fence, runID, TransferPush, s.d.TokenTTL)
	if err != nil {
		return PullPlan{}, fmt.Errorf("runner: issuing push token: %w", err)
	}
	return PullPlan{RunID: runID, Ref: ref, Home: home, PullToken: pull, PushToken: push}, nil

// AcceptPush publishes a staged, hash-verified checkpoint pushed by the home
// runner under its job fence. Anything refused is discarded; the published
// reference is the only state that changes.
func (s *WorkspaceSync) AcceptPush(ctx context.Context, f Fence, runID uuid.UUID, staged StagedCheckpoint) (CheckpointRef, error) {
	reject := func(err error) (CheckpointRef, error) {
		_ = s.d.Checkpoints.Discard(context.WithoutCancel(ctx), staged)
		return CheckpointRef{}, err
	}
	if staged.RunID != runID {
		return reject(fmt.Errorf("%w: staged checkpoint belongs to run %s, not %s", ErrManifestMismatch, staged.RunID, runID))
	}
	if err := s.d.Ledger.FenceCheck(ctx, f); err != nil {
		return reject(err)
	}
	if err := s.requireOwner(ctx, runID, f.RunnerID); err != nil {
		return reject(err)
	}
	ref, err := s.d.Checkpoints.Publish(ctx, staged, f)
	if err != nil {
		return reject(fmt.Errorf("runner: publishing pushed checkpoint: %w", err))
	}
	if err := s.d.Homes.SetCheckpoint(ctx, runID, ref.Version); err != nil {
		return CheckpointRef{}, fmt.Errorf("runner: recording checkpoint: %w", err)
	}
	if err := s.d.Homes.MarkDirty(ctx, runID, false); err != nil {
		return CheckpointRef{}, fmt.Errorf("runner: clearing dirty: %w", err)
	}
	return ref, nil
}

// ReportProgress publishes the moving-workspace execution status with byte
// progress for the job's run. Best effort: an unknown job publishes nothing.
func (s *WorkspaceSync) ReportProgress(ctx context.Context, f Fence, runID uuid.UUID, done, total int64) {
	bus := s.d.Events.Bus()
	if bus == nil {
		return
	}
	job, err := s.d.Ledger.Get(ctx, f.JobID)
	if err != nil {
		return
	}
	st := StatusFromJob(job, instance.ExecMovingWorkspace)
	st.Progress = &instance.MoveProgress{BytesDone: done, BytesTotal: total}
	st.HomeRunnerID = string(f.RunnerID)
	bus.Publish(runID, engine.Event{Type: EventExecutionStatus, Timestamp: s.d.Clock().UTC(), Data: st})
}

// AgentAffinity is the placement affinity of an agent mission: workspace when
// the mission runs in the run's workspace, built-in otherwise.
func AgentAffinity(m engine.AgentMission) Affinity {
	if m.WorkDir == "" {
		return AffinityBuiltIn
	}
	return AffinityWorkspace
}

// normalizeWorkspace degrades workspace affinity to built-in when no
// workspace sync is wired (the workspace would otherwise leave the server
// without a checkpoint protocol) and sizes the workspace for placement.
func (d *StepDispatcher) normalizeWorkspace(ctx context.Context, req ExecRequirements) ExecRequirements {
	if req.Affinity != AffinityWorkspace {
		return req
	}
	if d.deps.Workspace == nil || req.RunID == uuid.Nil {
		req.Affinity = AffinityBuiltIn
		return req
	}
	req.WorkspaceBytes = d.deps.Workspace.WorkspaceBytes(ctx, req.RunID)
	return req
}

// excludeRunners removes runners that failed to prepare this step's
// workspace; placement then treats a removed home as unavailable (must move).
func excludeRunners(snap PoolSnapshot, ex map[RunnerID]bool) PoolSnapshot {
	if len(ex) == 0 {
		return snap
	}
	out := snap
	out.Runners, out.Sources = nil, nil
	for _, r := range snap.Runners {
		if !ex[r.ID] {
			out.Runners = append(out.Runners, r)
		}
	}
	for _, s := range snap.Sources {
		if !ex[s.RunnerID] {
			out.Sources = append(out.Sources, s)
		}
	}
	return out
}

// runPlaced runs one reserved attempt. Workspace steps are bracketed by the
// sync protocol: ownership and checkpoint before, publish (or rollback) after.
// A step is only successful once its workspace result is published.
func (d *StepDispatcher) runPlaced(ctx context.Context, sr StepRunner, a Assignment, req ExecRequirements, dec Decision, job StepJob) (StepOutput, error) {
	ws := d.deps.Workspace
	if ws == nil || req.Affinity != AffinityWorkspace {
		return sr.Run(ctx, a, job)
	}
	att, err := ws.BeginAttempt(ctx, a, req, dec, &job)
	if err != nil {
		return StepOutput{}, WithOutcome(fmt.Errorf("runner: preparing workspace on %s: %w", a.Fence.RunnerID, err),
			SourceOutcome{Class: OutcomePullFailed})
	}
	out, runErr := sr.Run(ctx, a, job)
	detached := context.WithoutCancel(ctx)
	if runErr != nil {
		if ferr := ws.FinishFailure(detached, a, att); ferr != nil {
			return out, errors.Join(runErr, fmt.Errorf("runner: rolling back workspace: %w", ferr))
		}
		return out, runErr
	}
	if _, err := ws.FinishSuccess(detached, a, att); err != nil {
		if ferr := ws.FinishFailure(detached, a, att); ferr != nil {
			err = errors.Join(err, fmt.Errorf("runner: rolling back workspace: %w", ferr))
		}
		return StepOutput{}, WithOutcome(fmt.Errorf("runner: workspace result not committed: %w", err),
			SourceOutcome{Class: OutcomeOther})
	}
	return out, nil
}

// handlePullFailure excludes the runner for the rest of the step and re-queues
// the job, or fails the step once MaxPullFailures runners were excluded or no
// runner of the pool (poolRunners, counted before exclusion) remains.
// retry=false means the caller returns err.
func handlePullFailure(ctx context.Context, l Ledger, ev *EventRecorder, a Assignment, excluded map[RunnerID]bool, failures *int, poolRunners int, cause error) (retry bool, err error) {
	excluded[a.Fence.RunnerID] = true
	*failures++
	if *failures >= MaxPullFailures || len(excluded) >= poolRunners {
		if ferr := l.Fail(ctx, a.Fence, OutcomePullFailed); ferr != nil {
			return false, errors.Join(cause, ferr)
		}
		return false, cause
	}
	if _, rerr := l.Requeue(ctx, a.Fence, OutcomePullFailed); rerr != nil {
		return false, errors.Join(cause, rerr)
	}
	if ev != nil {
		rid := a.Job.Key.RunID
		ev.Record(ctx, PoolEvent{
			Kind: EventJobRerouted, RunnerID: a.Fence.RunnerID, RunID: &rid,
			Cause:  Cause{Kind: "outcome", Outcome: OutcomePullFailed},
			Detail: map[string]any{"jobId": a.Job.ID.String(), "reason": "workspace pull failed; runner ineligible for this step"},
		})
	}
	return true, nil
}
