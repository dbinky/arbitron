# Remote Runners — Phase 1 Task 16: Composition-root wiring and end-to-end provenance

**Goal:** Every model and process step the server executes goes through the runner pool. The pool consists of the dispatcher, the ledger, health and the built-in runner, all assembled at the real composition root. Operators see the built-in runner and its sources over MCP and REST. Health probing starts after crash recovery. Root-level tests assert each behaviour at its sink: `runner_jobs` rows, `node_executions.runner_id`, source health, enrichment, the bus, and MCP/REST output.

**Architecture:**
- A new `cmd/arbitron/runner_wiring.go` holds three pieces:
  - `builtinSourceSync` registers the built-in runner and its sources from `inventory.DetectLocal` + connections. It runs at boot and after every connection write.
  - `syncingConnStore` is the connection-admin store wrapper that triggers that resync.
  - `newRunnerStack` builds the registry consumers, ledger, event recorder, notifier, projector, health monitor, the built-in runner, the dispatcher, the routed selectors and process runner, and `runner.Service`.
- `initDomainWithModelDependencies` builds the stack before the executors. It hands the routed selectors to `newRunModelConsumers` (new signature), the script/tool executors and the three interactive constructors. The probe, path-inventory and model-refresh paths keep the undecorated selectors.
- `domainStack.runners` feeds `mcpDeps()`/`httpDeps()`, and `startRuntime` starts the monitor after `RecoverCrashedInstancesContext`.

**Tech Stack:** Go 1.25, SQLite (modernc), testify, mark3labs mcp-go, chi v5.

Dependencies are the producer tasks named below; implement them before their consumers. Follow strict TDD for **each** happy-path, success, failure, error and edge-case behavior: add the smallest behavioral test, run it against the current implementation and record its intended failure, implement only that behavior, rerun to green, then refactor. The implementation blocks below describe the final result; do not paste later-cycle behavior into an earlier cycle. A temporary production-code mutation after implementation is optional sensitivity validation and **does not count as the required RED step**. If a later-cycle test already passes, move that test to the cycle that first adds its behavior before implementing it. Never replace a meaningful assertion or skip a required case to get green.

Production acceptance also requires the named composition-root sink tests, with the real constructor path and SQLite/API/event sink. A fake-only unit test, a constructor read-back, a nil fallback, an empty collection, or a comment promising future wiring does not establish that a Phase-1 feature is connected. Explicit Phase-2/3/4 fields stay unavailable until their owning phase supplies the real producer; they are not Phase-1 completion claims.

	mu sync.Mutex
	d  builtinSourceSyncDeps
}

func newBuiltinSourceSync(d builtinSourceSyncDeps) *builtinSourceSync {
	if d.Registry == nil || d.Inventory.Connections == nil || d.Logger == nil {
		panic("arbitron: the built-in source sync requires a registry, a connection lister and a logger")
	}
	if d.Detect == nil {
		d.Detect = runnerinventory.DetectLocal
	}
	if d.Clock == nil {
		d.Clock = time.Now
	}
	return &builtinSourceSync{d: d}

// Sync (re)registers the runner row and its sources. A harness detection
// failure degrades to connections only (subscription sources are always
// registered, annotated harness_not_found); registry or connection-list
// failures are returned.
func (s *builtinSourceSync) Sync(ctx context.Context) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	in, err := s.d.Detect(ctx, s.d.Inventory)
	if err != nil {
		s.d.Logger.Error("runner inventory detection failed; registering built-in sources from connections only", "error", err)
		conns, lerr := s.d.Inventory.Connections.List(ctx)
		if lerr != nil {
			return fmt.Errorf("listing connections for the built-in runner: %w", lerr)
		}
		in.Connections = conns // retain successfully detected harness facts from the partial inventory
	}
	harness := map[string]string{}
	if in.ClaudeCLI != nil && in.ClaudeCLI.Version != "" {
		harness["claude-code"] = in.ClaudeCLI.Version
	}
	if in.Codex != nil && in.Codex.Version != "" {
		harness["codex"] = in.Codex.Version
	}
	now := s.d.Clock().UTC()
	if err := s.d.Registry.UpsertRunner(ctx, runner.Runner{
		ID: runner.BuiltInRunnerID, Name: "server", BuiltIn: true, StepLimit: s.d.StepLimit,
		Compat:   runner.Compatibility{State: runner.CompatCompatible},
		Liveness: runner.LivenessOnline, LastContact: &now,
		OS: runtime.GOOS, RunnerVersion: s.d.RunnerVersion, HarnessVersions: harness,
		WorkspaceRoot: s.d.WorkspaceRoot,
	}); err != nil {
		return fmt.Errorf("registering the built-in runner: %w", err)
	}
	if err := s.d.Registry.ReplaceSources(ctx, runner.BuiltInRunnerID, runner.BuiltInSources(in)); err != nil {
		return fmt.Errorf("registering the built-in runner's sources: %w", err)
	}
	return nil
}

// syncingConnStore is the store behind the connection admin service: after a
// successful create/update/delete it refreshes the built-in runner's sources.
// Reads pass straight through.
type syncingConnStore struct {
	provider.ConnectionStore
	sync func(context.Context) error
}

func newSyncingConnStore(store provider.ConnectionStore, sync func(context.Context) error) syncingConnStore {
	if store == nil || sync == nil {
		panic("arbitron: the syncing connection store requires a store and a sync function")
	}
	return syncingConnStore{ConnectionStore: store, sync: sync}
}

func (s syncingConnStore) Create(ctx context.Context, c provider.Connection, apiKey string) error {
	if err := s.ConnectionStore.Create(ctx, c, apiKey); err != nil {
		return err
	}
	return s.resync(ctx)
}

func (s syncingConnStore) Update(ctx context.Context, c provider.Connection, apiKey *string) error {
	if err := s.ConnectionStore.Update(ctx, c, apiKey); err != nil {
		return err
	}
	return s.resync(ctx)
}

func (s syncingConnStore) Delete(ctx context.Context, name string) error {
	if err := s.ConnectionStore.Delete(ctx, name); err != nil {
		return err
	}
	return s.resync(ctx)
}

func (s syncingConnStore) resync(ctx context.Context) error {
	if err := s.sync(context.WithoutCancel(ctx)); err != nil {
		return fmt.Errorf("connection change saved, but the built-in runner's sources were not refreshed (repeat the change to retry): %w", err)
	}
	return nil
}

	if clock == nil {
		clock = time.Now

type rwStatus struct {
	Run   uuid.UUID
	State instance.ExecutionState

// rwCollectStatuses records every execution.status event published on the bus.
func rwCollectStatuses(t *testing.T, bus engine.EventBus) func() []rwStatus {
	t.Helper()
	ch, unsubscribe := bus.SubscribeAll()
	var mu sync.Mutex
	var got []rwStatus
	done := make(chan struct{})
	go func() {
		defer close(done)
		for ev := range ch {
			if ev.Event.Type != runner.EventExecutionStatus {
				continue
			}
			if st, ok := ev.Event.Data.(instance.ExecutionStatus); ok {
				mu.Lock()
				got = append(got, rwStatus{Run: ev.InstanceID, State: st.State})
				mu.Unlock()
			}
		}
	}()
	t.Cleanup(func() { unsubscribe(); <-done })
	return func() []rwStatus {
		mu.Lock()
		defer mu.Unlock()
		return append([]rwStatus(nil), got...)
	}
}

// rcGoroutines maps goroutine id -> stack for every goroutine whose stack
// mentions one of the markers (the actual workers, not a global count).
func rcGoroutines(markers ...string) map[string]string {
	buf := make([]byte, 1<<20)
	for {
		n := runtime.Stack(buf, true)
		if n == len(buf) {
			buf = make([]byte, len(buf)*2)
			continue
		}
		found := map[string]string{}
		for _, stack := range strings.Split(string(buf[:n]), "\n\n") {
			for _, m := range markers {
				if strings.Contains(stack, m) {
					found[strings.Fields(stack)[1]] = stack
					break
				}
			}
		}
		return found
	}
}

func rcNew(before, now map[string]string) map[string]string {
	out := map[string]string{}
	for id, s := range now {
		if _, ok := before[id]; !ok {
			out[id] = s
		}
	}
	return out
}

func rcAnyContains(gs map[string]string, s string) bool {
	for _, st := range gs {
		if strings.Contains(st, s) {
			return true
		}
	}
	return false
}

func rcAlive(pid int) bool { return syscall.Kill(pid, 0) == nil }

// rcCloseJoins closes the root while a worker is blocked in the external leaf
// (which is never released), then proves cancellation reached the leaf, every
// worker joined before Close returned, the DB is closed and Close is repeatable.
func rcCloseJoins(t *testing.T, ds *domainStack, fake string, pid int, before map[string]string, markers []string, within time.Duration) {
	t.Helper()
	owned := rcNew(before, rcGoroutines(markers...))
	start := time.Now()
	closed := make(chan struct{})
	go func() { ds.Close(context.Background()); close(closed) }()
	select {
	case <-closed:
	case <-time.After(within):
		t.Fatalf("Close did not cancel and join the in-flight worker within %s", within)
	}
	_, err := os.Stat(filepath.Join(fake, "release"))
	require.True(t, os.IsNotExist(err), "the leaf was never released: only cancellation can end it")
	t.Logf("Close returned after %s", time.Since(start))
	// Joined, not merely signalled: none of the worker goroutines observed in
	// flight outlives Close, so none can issue SQL against the closed handle.
	after := rcGoroutines(markers...)
	for id := range owned {
		_, alive := after[id]
		assert.False(t, alive, "worker goroutine %s survived Close:\n%s", id, after[id])
	}
	assert.Empty(t, rcNew(before, after), "no monitor/confirmer/dispatcher goroutine remains after Close")
	assert.False(t, rcAlive(pid), "cancellation must kill the external leaf (pid %d)", pid)
	require.ErrorContains(t, ds.db.Ping(), "closed", "the database closes only after the workers join")
	done := make(chan struct{})
	go func() { ds.Close(context.Background()); close(done) }()
	select {
	case <-done:
	case <-time.After(5 * time.Second):
		t.Fatal("a repeated Close must return")
	}
	assert.Empty(t, rcNew(before, rcGoroutines(markers...)))
}

// rcSnapshot dumps every row of a table as text, so "unchanged" means unchanged.
func rcSnapshot(t *testing.T, db *sql.DB, table string) []string {
	t.Helper()
	rows, err := db.Query(`SELECT * FROM ` + table + ` ORDER BY rowid`)
	require.NoError(t, err)
	defer rows.Close()
	cols, err := rows.Columns()
	require.NoError(t, err)
	var out []string
	for rows.Next() {
		vals := make([]sql.RawBytes, len(cols))
		ptrs := make([]any, len(cols))
		for i := range vals {
			ptrs[i] = &vals[i]
		}
		require.NoError(t, rows.Scan(ptrs...))
		parts := make([]string, len(cols))
		for i, v := range vals {
			parts[i] = cols[i] + "=" + string(v)
		}
		out = append(out, strings.Join(parts, "|"))
	}
	require.NoError(t, rows.Err())
	return out
}

}
