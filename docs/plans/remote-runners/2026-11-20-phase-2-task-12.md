# Remote Runners — Phase 2 Task 12: Runner detail recent telemetry

**Goal:** `runner_get` (MCP) and `GET /api/v1/runners/{id}` (REST) return that runner's recent telemetry window, oldest first, from the hub that heartbeats and the built-in telemetry loop feed.

**Architecture:**
- `domain/runner/service_telemetry.go` adds the read port `TelemetryWindow` (`Window(id RunnerID) []Telemetry`; `*Hub` satisfies it) and `(*Service).recentTelemetry`. `ServiceDeps` gains an optional `Window`, and `Service.Get` fills `RunnerDetail.RecentTelemetry` from it. A nil `Window` gives an empty list: no samples means "unknown", never zeros.
- The REST and MCP detail DTOs already render `RecentTelemetry` (P1 T15 `toRunnerDetailResponse` / `toRunnerDetailJSON`), so neither adapter changes.
- The composition root (`cmd/arbitron/runner_wiring.go:newRunnerStack`, which builds `remoteStack` since P2 T10) passes `Window: remoteStack.Hub`.

**Tech Stack:** Go 1.25.5, testify, the real `initDomain`/`Start` composition root with the P2 T10 remote-runner test harness (real WebSocket agent).

**Spec / Design:**
- Spec `docs/specs/2026-11-19-remote-runners-spec.md`, "Live views → Runner detail": CPU, memory use and pressure, load, free workspace disk, thermal state, uptime.
- Design `docs/designs/2026-11-19-remote-runners-design-phase-2.md`:
  - §4 step 3: heartbeat telemetry goes to the telemetry window;
  - §6: "Pool event log and telemetry window: consumed by Phase 5".
- Design `docs/designs/2026-11-19-remote-runners-design-phase-5.md` §3.2: runner detail "sparklines from the recent telemetry window".
- Contracts `docs/plans/remote-runners/2026-11-20-contracts.md`:
  - §7 (`Telemetry`);
  - §14 (`ServiceDeps`, `RunnerDetail.RecentTelemetry /*P2*/`, `runner_get`, `GET /runners/{id}`);

```go
// rdtWindow is a per-runner telemetry window. It honours the id argument
// (each runner has its own samples) and records every id it was asked for.
type rdtWindow struct {
	mu    sync.Mutex
	byID  map[RunnerID][]Telemetry
	asked []RunnerID
}

func (w *rdtWindow) Window(id RunnerID) []Telemetry {
	w.mu.Lock()
	defer w.mu.Unlock()
	w.asked = append(w.asked, id)
	return append([]Telemetry(nil), w.byID[id]...)
}

func (w *rdtWindow) askedIDs() []RunnerID {
	w.mu.Lock()
	defer w.mu.Unlock()
	return append([]RunnerID(nil), w.asked...)
}

// rdtAliasWindow hands out its own backing array, like a careless producer.
```

```go
package runner

// TelemetryWindow is the per-runner recent telemetry the runner detail view
// shows. The production *Hub satisfies it: samples come from remote
// heartbeats and the built-in telemetry loop, oldest first, bounded by
// HubConfig.TelemetryWindow.
type TelemetryWindow interface {
	Window(id RunnerID) []Telemetry
}

var _ TelemetryWindow = (*Hub)(nil)

// recentTelemetry returns id's recent samples, oldest first. A nil Window
// (Phase 1 wiring and unit tests) yields an empty, non-nil list: no samples
// means "unknown", never zero values. The result is always a copy, so a
// caller can never alias the producer's window.
func (s *Service) recentTelemetry(id RunnerID) []Telemetry {
	if s.d.Window == nil {
		return []Telemetry{}
	}
	w := s.d.Window.Window(id)
	out := make([]Telemetry, len(w))
	copy(out, w)
	return out
}
```

```go
package main

import (

// rdtRecent reads recentTelemetry from GET /api/v1/runners/{id} as the local
// admin. It never calls require, so it is safe inside Eventually.
func rdtRecent(st *rrStack, id string) []map[string]any {
	r, ok := rrRunner(st.srv, id)
	if !ok {
		return nil
	}
	raw, _ := r["recentTelemetry"].([]any)
	out := make([]map[string]any, 0, len(raw))
	for _, item := range raw {
		if m, ok := item.(map[string]any); ok {
			out = append(out, m)
		}
	}
	return out
}

// The built-in telemetry loop (P2 T7, every heartbeat = 1 s in rrRunnersYAML)
// fills the hub window, and the runner detail shows it on REST and MCP.
// Without `Window: remoteStack.Hub` in newRunnerStack the list stays [].
func TestRunnerDetailCompositionRoot_RecentTelemetryFromTheHubWindow(t *testing.T) {
	st := bootRemoteRunnerStack(t, nil, false)

	require.Eventually(t, func() bool { return len(rdtRecent(st, "builtin")) >= 2 },
		20*time.Second, 50*time.Millisecond, "SINK: GET /api/v1/runners/builtin lists the built-in loop's recent samples")

	samples := rdtRecent(st, "builtin")
	var prev time.Time
	for i, s := range samples {
		at, ok := s["sampledAt"].(string)
		require.True(t, ok, "sample %d carries sampledAt", i)
		ts, err := time.Parse(time.RFC3339, at)
		require.NoError(t, err)
		assert.False(t, ts.Before(prev), "samples are oldest first")
		prev = ts
	}

	isErr, text := mcpToolCallerAs(t, st.ds, adminCtxForLocalUser(t, st.ds))("runner_get", map[string]any{"id": "builtin"})
	require.False(t, isErr, text)
	var detail struct {
		RecentTelemetry []map[string]any `json:"recentTelemetry"`
	}
	require.NoError(t, json.Unmarshal([]byte(text), &detail))
	assert.GreaterOrEqual(t, len(detail.RecentTelemetry), 2, "SINK: MCP runner_get shows the same window")
}

// Each runner's detail shows only its own samples: a remote runner's
// heartbeat telemetry (Cores = 977, never a real machine) appears on its
// detail and never on the built-in runner's. A registered runner that has
// not connected yet has an empty (not null) window.
func TestRunnerDetailCompositionRoot_RecentTelemetryIsPerRunner(t *testing.T) {
	st := bootRemoteRunnerStack(t, nil, false)

	_, idle, _ := rrCreateKey(t, st, "mac-idle")
	r, ok := rrRunner(st.srv, idle)
	require.True(t, ok)
	assert.Equal(t, []any{}, r["recentTelemetry"], "a runner with no samples renders an empty list")

	plain, id, _ := rrCreateKey(t, st, "mac-tel")
	cores := 977
	startRemoteAgent(t, st, plain, st.workspaceDir, newRRExecutor(t, "unused", false), runner.Telemetry{Cores: &cores})
	rrAwaitLiveness(t, st, id, "online")

	require.Eventually(t, func() bool { return len(rdtRecent(st, id)) >= 2 },
		20*time.Second, 50*time.Millisecond, "SINK: the remote runner's heartbeats fill its own recentTelemetry")
	for _, s := range rdtRecent(st, id) {
		assert.EqualValues(t, 977, s["cores"], "the remote detail shows the remote machine's samples")
	}
	require.Eventually(t, func() bool { return len(rdtRecent(st, "builtin")) >= 2 },
		20*time.Second, 50*time.Millisecond, "the built-in runner has its own samples")
	for _, s := range rdtRecent(st, "builtin") {
		assert.NotEqualValues(t, 977, s["cores"], "the built-in window never shows another runner's samples")
	}
}
```
