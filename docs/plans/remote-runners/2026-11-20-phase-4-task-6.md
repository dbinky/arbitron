# Remote Runners — Phase 4 Task 6: Session checkpoint store and per-turn checkpoint commit

**Goal:** Make every interactive turn durable as a pair: the harness session bundle (exported from the runner that ran the turn, checksummed, stored) and — for an `interactive_agent` session — the workspace checkpoint, then ONE transaction that writes the conversation messages, `session_checkpoint_ref`, `session_pin_runner`, `session_generation` and the turn-intent completion, fenced on the durable turn lease. Export or store failure fails the turn with nothing persisted; a crash between upload and commit keeps the prior pair (the orphan bundle is pruned by the next successful turn).

**Architecture:** `domain/engine/session_checkpoints.go` holds the contract port `SessionCheckpointStore` plus the engine-side ports this task needs (`TurnCommitter`, `SessionExporter`, `TurnWorkspaceCheckpointer`) and the shared `checkpointAndCommit` routine used by `conversationHandler.persistTurn` (later turns) and `InteractiveExecutor.Execute` (first turn). The runner that ran the turn comes from the turn's `PlacementSink` (last fact; none = built-in). `adapter/checkpoint/sessions.go` stores bundles on disk keyed run/invocation/turn/checksum, verifying the checksum on Put and Get, and prunes everything except the committed ref plus `retain-1` older turns. `adapter/sqlite/turn_commit_repo.go` performs the single fenced transaction (reusing P4 T1 `setSessionStateTx` and P4 T3 `completeIntentTx`). `domain/runner` gains `SessionPortabilityRegistry` (backend → adapter, used in-process by the built-in runner), `SessionExportRouter` (built-in → registry; remote → `RemoteSessionExporter`, implemented by the Hub in P4 T12) and `TurnWorkspaceCheckpointer` (built-in: `CheckpointStore.Commit` staged, `HomeStore.SetCheckpoint` only after the turn commit; remote: the checkpoint the runner pushed under its job fence).

**Tech Stack:** Go 1.25, SQLite (modernc) transactions, stdlib file IO, testify.

**Spec / Design:** spec "Recovery and session continuity" (sessions restart from their last successful turn; one active owner); design phase 4 §2 (rows "Session checkpoint", "Per-turn commit"), §3.2 (checkpoint after every successful turn; turn successful only after the export is stored and committed; workspace checkpoint pair for interactive_agent), §6 (single commit, fenced), §7 (export failure fails the turn; crash between upload and commit keeps the prior pair; orphans pruned), §8 Edge "A crash between bundle upload and commit keeps the prior pair", Success "multi-turn interactive_agent session ... checkpoint after every turn" (adapter half; e2e is P4 T13). Contracts: `docs/plans/remote-runners/2026-11-20-contracts.md` §17 (`SessionKey`, `SessionCheckpointStore`, `adapter/checkpoint/sessions.go`), §4 (`PlacementSink`, `PlacementFact`), §6 (`RunnerID`, `BuiltInRunnerID`), §16 (`CheckpointStore`, `CheckpointRef`, `HomeStore.SetCheckpoint`, `Home`), §11 (`builtin.Deps`).

func (d SessionCheckpointDeps) validate() error {

		return errors.New("session checkpoints require a store, exporter, committer and workspace checkpointer")

// sessionBackend names the harness whose session file carries the turn's
// context, or "" when the conversation messages are the whole state.
func sessionBackend(b provider.ResolvedBinding, defaultAgent string) string {
	switch b.Kind {
	case provider.ProfileKindAgent:
		if be := strings.TrimSpace(b.Backend); be != "" {
			return be
		}
		return defaultAgent
	case provider.ProfileKindChat:
		if b.ConnectionRef == "" {
			return workflow.BackendClaudeCode
		}
	}
	return ""
}

// turnRunner is the runner that ran the turn: the last placement fact.
func turnRunner(ctx context.Context) string {
	if sink, ok := PlacementSinkFromContext(ctx); ok {
		if facts := sink.Facts(); len(facts) > 0 && facts[len(facts)-1].RunnerID != "" {
			return facts[len(facts)-1].RunnerID
		}
	}
	return builtInRunnerID
}

// checkpointAndCommit makes a finished turn durable: export + store the
// harness session, stage the workspace checkpoint (agent sessions outside a
// group resolution), ONE fenced commit, then publish the workspace pointer and
// prune superseded bundles. Any failure before the commit leaves every durable
// record unchanged and fails the turn.
func (d *SessionCheckpointDeps) checkpointAndCommit(ctx context.Context, exec *instance.NodeExecution, p turnCheckpointPlan) error {
	runID := exec.InstanceID
	runnerID := turnRunner(ctx)
	var invocationID uuid.UUID
	if id, ok := StepIdentityFromContext(ctx); ok {
		invocationID = id.InvocationID
	}

	var ref string
	if backend := sessionBackend(p.binding, d.DefaultAgentBackend); backend != "" {
		if sid := SessionID(exec.LLMConversation); sid != "" {
			b, err := d.Exporter.ExportSession(ctx, SessionExportRequest{RunnerID: runnerID, Backend: backend, SessionID: sid, WorkDir: p.workDir})
			if err != nil {
				return fmt.Errorf("checkpoint session: export: %w", err)
			}
			ref, err = d.Store.Put(ctx, SessionKey{RunID: runID, InvocationID: invocationID, TurnIndex: p.turnIndex}, b)
			if err != nil {
				return fmt.Errorf("checkpoint session: store: %w", err)
			}
		}
	}

	var staged *StagedTurnWorkspace
	if p.binding.Kind == provider.ProfileKindAgent && !p.group && p.workDir != "" {
		s, err := d.Workspace.StageTurn(ctx, TurnWorkspaceRequest{RunID: runID, RunnerID: runnerID, WorkDir: p.workDir})
		if err != nil {
			return fmt.Errorf("checkpoint workspace: %w", err)
		}
		staged = &s
		conv := make(map[string]any, len(exec.LLMConversation)+1)
		for k, v := range exec.LLMConversation {
			conv[k] = v
		}
		conv[ConversationKeyWorkspaceCheckpointRef] = s.Ref
		exec.LLMConversation = conv
	}

	if err := d.Committer.CommitTurn(ctx, TurnCommit{
		Execution: exec, Create: p.create, RequireOpenSession: p.group,
		CheckpointRef: ref, PinRunner: runnerID, SessionGeneration: p.lease.Generation,
		LeaseKey: p.lease.Key, LeaseToken: p.lease.Token, IntentID: p.intentID,
	}); err != nil {
		return fmt.Errorf("commit turn: %w", err)
	}

	if staged != nil {
		if err := d.Workspace.PublishTurn(ctx, *staged); err != nil {
			return fmt.Errorf("publish workspace checkpoint: %w", err)
		}
	}
	if ref != "" {
		// Retention only: a failed prune leaves extra bundles that the next
		// successful turn's prune removes.
		_ = d.Store.Prune(context.WithoutCancel(ctx), runID, invocationID, ref)

const sessionRefPrefix = "session:"

// SessionStore keeps session bundles under <dir>/<run>/<invocation>/ as
// <turn>-<checksum>.data plus a .meta.json written last (its presence marks a
// complete bundle).
type SessionStore struct {
	dir    string
	retain int
	mu     sync.Mutex
}

// NewSessionStore opens (creating) the store; retain >= 1 bundles are kept
// per invocation (the committed one plus retain-1 older turns).
func NewSessionStore(dir string, retain int) (*SessionStore, error) {
	if dir == "" {
		return nil, errors.New("checkpoint: session store requires a directory")
	}
	if retain < 1 {
		return nil, fmt.Errorf("checkpoint: session bundle retention must be >= 1 (got %d)", retain)
	}
	if err := os.MkdirAll(dir, 0o700); err != nil {
		return nil, fmt.Errorf("checkpoint: session store: %w", err)
	}
	return &SessionStore{dir: dir, retain: retain}, nil
}

type sessionMeta struct {
	Backend        string `json:"backend"`
	SessionID      string `json:"sessionId"`
	HarnessVersion string `json:"harnessVersion"`
	OriginWorkDir  string `json:"originWorkDir"`
	Checksum       string `json:"checksum"`
}

type sessionRef struct {
	run, inv uuid.UUID
	turn     int
	checksum string
}

func (r sessionRef) String() string {
	return fmt.Sprintf("%s%s/%s/%d/%s", sessionRefPrefix, r.run, r.inv, r.turn, r.checksum)
}

func (r sessionRef) base() string { return fmt.Sprintf("%06d-%s", r.turn, r.checksum) }

func validChecksum(s string) bool {
	if len(s) != 64 || strings.ToLower(s) != s {
		return false
	}
	_, err := hex.DecodeString(s)
	return err == nil
}

func parseSessionRef(ref string) (sessionRef, error) {
	rest, ok := strings.CutPrefix(ref, sessionRefPrefix)
	parts := strings.Split(rest, "/")
	if !ok || len(parts) != 4 {
		return sessionRef{}, fmt.Errorf("checkpoint: malformed session ref %q", ref)
	}
	run, err1 := uuid.Parse(parts[0])
	inv, err2 := uuid.Parse(parts[1])
	turn, err3 := strconv.Atoi(parts[2])
	if err1 != nil || err2 != nil || err3 != nil || turn < 1 || !validChecksum(parts[3]) {
		return sessionRef{}, fmt.Errorf("checkpoint: malformed session ref %q", ref)
	}
	return sessionRef{run: run, inv: inv, turn: turn, checksum: parts[3]}, nil
}

func (s *SessionStore) invDir(run, inv uuid.UUID) string {
	return filepath.Join(s.dir, run.String(), inv.String())
}

func (s *SessionStore) paths(r sessionRef) (data, meta string) {
	base := filepath.Join(s.invDir(r.run, r.inv), r.base())
	return base + ".data", base + ".meta.json"
}

func writeFileAtomic(path string, data []byte) error {
	tmp, err := os.CreateTemp(filepath.Dir(path), ".tmp-")
	if err != nil {
		return err
	}
	name := tmp.Name()
	if _, err := tmp.Write(data); err != nil {
		tmp.Close()
		os.Remove(name)
		return err
	}
	if err := tmp.Sync(); err != nil {
		tmp.Close()
		os.Remove(name)
		return err
	}
	if err := tmp.Close(); err != nil {
		os.Remove(name)
		return err
	}
	if err := os.Rename(name, path); err != nil {
		os.Remove(name)
		return err
	}
	return nil
}

// Put implements engine.SessionCheckpointStore.
func (s *SessionStore) Put(_ context.Context, key engine.SessionKey, b engine.SessionBundle) (string, error) {
	if key.RunID == uuid.Nil || key.TurnIndex < 1 {
		return "", fmt.Errorf("checkpoint: invalid session key %+v", key)
	}
	if err := b.Verify(); err != nil {
		return "", err
	}
	r := sessionRef{run: key.RunID, inv: key.InvocationID, turn: key.TurnIndex, checksum: b.Checksum}
	meta, err := json.Marshal(sessionMeta{Backend: b.Backend, SessionID: b.SessionID, HarnessVersion: b.HarnessVersion,
		OriginWorkDir: b.OriginWorkDir, Checksum: b.Checksum})
	if err != nil {
		return "", err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if err := os.MkdirAll(s.invDir(r.run, r.inv), 0o700); err != nil {
		return "", fmt.Errorf("checkpoint: put session: %w", err)
	}
	dataPath, metaPath := s.paths(r)
	if err := writeFileAtomic(dataPath, b.Data); err != nil {
		return "", fmt.Errorf("checkpoint: put session: %w", err)
	}
	if err := writeFileAtomic(metaPath, meta); err != nil {
		return "", fmt.Errorf("checkpoint: put session: %w", err)
	}
	return r.String(), nil
}

// Get implements engine.SessionCheckpointStore.
func (s *SessionStore) Get(_ context.Context, ref string) (engine.SessionBundle, error) {
	r, err := parseSessionRef(ref)
	if err != nil {
		return engine.SessionBundle{}, err
	}
	dataPath, metaPath := s.paths(r)
	s.mu.Lock()
	defer s.mu.Unlock()
	rawMeta, err := os.ReadFile(metaPath)
	if err != nil {
		return engine.SessionBundle{}, fmt.Errorf("checkpoint: get session %s: %w", ref, err)
	}
	var m sessionMeta
	if err := json.Unmarshal(rawMeta, &m); err != nil {
		return engine.SessionBundle{}, fmt.Errorf("checkpoint: get session %s: %w", ref, err)
	}
	data, err := os.ReadFile(dataPath)
	if err != nil {
		return engine.SessionBundle{}, fmt.Errorf("checkpoint: get session %s: %w", ref, err)
	}
	b := engine.SessionBundle{Backend: m.Backend, SessionID: m.SessionID, HarnessVersion: m.HarnessVersion,
		OriginWorkDir: m.OriginWorkDir, Checksum: m.Checksum, Data: data}
	if b.Checksum != r.checksum {
		return engine.SessionBundle{}, fmt.Errorf("%w: stored metadata does not match ref", engine.ErrBundleChecksum)
	}
	if err := b.Verify(); err != nil {
		return engine.SessionBundle{}, err
	}
	return b, nil
}

// Prune implements engine.SessionCheckpointStore.
func (s *SessionStore) Prune(_ context.Context, runID, invocationID uuid.UUID, keepRef string) error {
	keep, err := parseSessionRef(keepRef)
	if err != nil {
		return err
	}
	if keep.run != runID || keep.inv != invocationID {
		return fmt.Errorf("checkpoint: keep ref %s does not belong to %s/%s", keepRef, runID, invocationID)
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	dir := s.invDir(runID, invocationID)
	if _, metaPath := s.paths(keep); !fileExists(metaPath) {
		return fmt.Errorf("checkpoint: keep ref %s is not stored; refusing to prune", keepRef)
	}
	entries, err := os.ReadDir(dir)
	if err != nil {
		return fmt.Errorf("checkpoint: prune sessions: %w", err)
	}
	type stored struct {
		base string
		turn int
	}
	var older []stored
	bases := map[string]int{}
	for _, e := range entries {
		name := e.Name()
		if strings.HasPrefix(name, ".tmp-") {
			_ = os.Remove(filepath.Join(dir, name))
			continue
		}
		base := strings.TrimSuffix(strings.TrimSuffix(name, ".meta.json"), ".data")
		turnPart, _, ok := strings.Cut(base, "-")
		turn, err := strconv.Atoi(turnPart)
		if !ok || err != nil {
			continue
		}
		bases[base] = turn
	}
	for base, turn := range bases {
		if base != keep.base() && turn < keep.turn {
			older = append(older, stored{base: base, turn: turn})
		}
	}
	sort.Slice(older, func(i, j int) bool {
		if older[i].turn != older[j].turn {
			return older[i].turn > older[j].turn
		}
		return older[i].base > older[j].base
	})
	retained := map[string]bool{keep.base(): true}
	for i := 0; i < len(older) && i < s.retain-1; i++ {
		retained[older[i].base] = true
	}
	for base := range bases {
		if retained[base] {
			continue
		}
		_ = os.Remove(filepath.Join(dir, base+".meta.json")) // meta first: a half-deleted bundle reads as missing
		_ = os.Remove(filepath.Join(dir, base+".data"))
	}
	return nil
}

func fileExists(p string) bool {
	_, err := os.Stat(p)
	return err == nil
}

// CommitTurn implements engine.TurnCommitter.
func (r *TurnCommitRepo) CommitTurn(ctx context.Context, c engine.TurnCommit) error {
	if c.Execution == nil {
		return errors.New("turn commit: execution required")
	}
	tx, err := r.db.BeginTx(ctx, nil)
	if err != nil {
		return fmt.Errorf("turn commit: begin: %w", err)
	}
	defer func() { _ = tx.Rollback() }()

	if c.LeaseToken != uuid.Nil {
		var token sql.NullString
		var gen int64
		err := tx.QueryRowContext(ctx, `SELECT token, generation FROM turn_leases WHERE key = ?`, c.LeaseKey).Scan(&token, &gen)
		if errors.Is(err, sql.ErrNoRows) || (err == nil && (!token.Valid || token.String != c.LeaseToken.String() || gen != c.SessionGeneration)) {
			return fmt.Errorf("%w: lease %s", engine.ErrTurnFenced, c.LeaseKey)
		}
		if err != nil {
			return fmt.Errorf("turn commit: read lease: %w", err)
		}
	}

	execs := &NodeExecutionRepo{db: r.db, queries: generated.New(tx)}
	switch {
	case c.Create:
		if err := execs.Create(ctx, c.Execution); err != nil {
			return fmt.Errorf("turn commit: %w", err)
		}
	case c.RequireOpenSession:
		written, err := updateIfSessionOpenOn(ctx, tx, c.Execution)
		if err != nil {
			return fmt.Errorf("turn commit: %w", err)
		}
		if !written {
			return fmt.Errorf("turn commit: session %s was closed during this turn: %w", c.Execution.ID, instance.ErrNotInteractiveWait)
		}
	default:
		if err := execs.Update(ctx, c.Execution); err != nil {
			return fmt.Errorf("turn commit: %w", err)
		}
	}

	st := instance.SessionState{Generation: c.SessionGeneration}
	if c.CheckpointRef != "" {
		ref := c.CheckpointRef

	if c.PinRunner != "" {
		pin := c.PinRunner
		st.PinRunner = &pin
