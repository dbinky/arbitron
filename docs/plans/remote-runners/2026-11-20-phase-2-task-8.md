# Remote Runners — Phase 2 Task 8: Runner agent domain (`domain/runneragent`)

**Goal:** Build the transport-neutral runner-side brain: admission (challenge signature + hello), jittered reconnect, heartbeats, dispatch dedupe/accept/reject, sequenced output with a bounded never-dropping spool, fenced kill, and terminal-result replay until acknowledged.

**Architecture:** `domain/runneragent` owns ports (`Connection`/`Session`, `JobExecutor`, `LocalSources`, `Spool`, `IdentityStore`, `Clock`) and wire-neutral message mirrors; it imports only `domain/runner` (for `MachineTelemetry`, `Telemetry`, `SourceHealth`, `SourceOutcome`, `CredentialKind`) and `domain/modelidentity`, never `adapter/*`. `adapter/runneragent/*` (P2 T9) implements the ports and translates the mirrors to `adapter/runnerproto` envelopes; `cmd/arbitron-runner` (P2 T9) is the composition root.

**Tech Stack:** Go 1.25.5, stdlib `crypto/ed25519`, `log/slog`, testify; in-memory fakes and a gating fake clock.

**Spec / Design:** `docs/specs/2026-11-19-remote-runners-spec.md` (Recovery and session continuity: brief interruptions do not move work); `docs/designs/2026-11-19-remote-runners-design-phase-2.md` §3.1 (invariants), §3.7 (ports), §4 steps 1–5, §7, §8. Contracts: `docs/plans/remote-runners/2026-11-20-contracts.md` §6 (CredentialKind, SourceHealth), §6 Outcomes (`SourceOutcome`), §7 (`Telemetry`), §15 (protocol kinds, `HeldJob`, `runner.MachineTelemetry`, machine binding string, `domain/runneragent` port names).

func TestChallengeMessageFormat(t *testing.T) {
	nonce := []byte{0xff, 0x00, 0x10}
	got := runneragent.ChallengeMessage("2f0c7a2e-1111-4222-8333-944455556666", nonce)
	assert.Equal(t, "arbitron-runner-v1:2f0c7a2e-1111-4222-8333-944455556666:/wAQ", string(got))
}

func TestChallengeMessageSignatureBindsKeyAndNonce(t *testing.T) {
	seed := make([]byte, ed25519.SeedSize)
	seed[0] = 1
	priv := ed25519.NewKeyFromSeed(seed)
	pub := priv.Public().(ed25519.PublicKey)
	sig := ed25519.Sign(priv, runneragent.ChallengeMessage("k1", []byte("nonce-a")))
	require.True(t, ed25519.Verify(pub, runneragent.ChallengeMessage("k1", []byte("nonce-a")), sig))
	assert.False(t, ed25519.Verify(pub, runneragent.ChallengeMessage("k2", []byte("nonce-a")), sig), "a different key id must not verify")
	assert.False(t, ed25519.Verify(pub, runneragent.ChallengeMessage("k1", []byte("nonce-b")), sig), "a replayed signature over another nonce must not verify")
}

package runneragent

import "time"

// Backoff is exponential (base * 2^attempt, capped at max) with "equal
// jitter": the result lies in [d/2, d], so a reconnect storm spreads out but
// never retries instantly.
func Backoff(attempt int, base, max time.Duration, rnd func() float64) time.Duration {
	if attempt < 0 {
		attempt = 0
	}
	d := base
	for i := 0; i < attempt && d < max; i++ {
		d *= 2
	}
	if d > max {
		d = max
	}
	return time.Duration(float64(d) * (0.5 + 0.5*rnd()))
}

package spooltest

import (
	"context"
	"testing"

	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"

	"github.com/arbitron/arbitron/domain/runneragent"
)

// Run executes the suite. newSpool must return an EMPTY spool with the given
// unacked-frame capacity.
func Run(t *testing.T, newSpool func(t *testing.T, capacity int) runneragent.Spool) {
	ctx := context.Background()
	a := runneragent.JobRef{JobID: "job-a", Generation: 1}
	b := runneragent.JobRef{JobID: "job-b", Generation: 2}

	t.Run("full spool refuses without storing or dropping", func(t *testing.T) {
		s := newSpool(t, 2)
		require.NoError(t, s.AppendOutput(ctx, a, runneragent.OutputFrame{Seq: 1, Token: "x"}))
		require.NoError(t, s.AppendOutput(ctx, b, runneragent.OutputFrame{Seq: 1, Token: "y"}))
		err := s.AppendOutput(ctx, a, runneragent.OutputFrame{Seq: 2, Token: "z"})
		require.ErrorIs(t, err, runneragent.ErrSpoolFull)
		got, ok, err := s.Get(ctx, a)
		require.NoError(t, err)
		require.True(t, ok)
		assert.Equal(t, []runneragent.OutputFrame{{Seq: 1, Token: "x"}}, got.Frames, "the refused frame is not stored and nothing older is dropped")
	})

	t.Run("ack drops acknowledged frames, frees capacity and is monotonic", func(t *testing.T) {
		s := newSpool(t, 3)
		for i := uint64(1); i <= 3; i++ {
			require.NoError(t, s.AppendOutput(ctx, a, runneragent.OutputFrame{Seq: i, Token: "t"}))
		}

package runneragent

import (
	"context"
	"sort"
	"sync"
)

// MemorySpool is a bounded in-memory Spool. It satisfies the same
// conformance suite as the file spool; tests and the P2 T10 contract test use
// it, production uses adapter/runneragent/spool.FileSpool.
type MemorySpool struct {
	mu       sync.Mutex
	capacity int
	frames   int
	jobs     map[JobRef]*SpooledJob
}

// NewMemorySpool returns an empty spool holding at most capacity unacked frames.
func NewMemorySpool(capacity int) *MemorySpool {
	if capacity < 1 {
		capacity = 1
	}
	return &MemorySpool{capacity: capacity, jobs: map[JobRef]*SpooledJob{}}
}

func (m *MemorySpool) job(ref JobRef) *SpooledJob {
	j := m.jobs[ref]
	if j == nil {
		j = &SpooledJob{Ref: ref}
		m.jobs[ref] = j
	}
	return j
}

func lastSeq(j *SpooledJob) uint64 {
	if n := len(j.Frames); n > 0 {
		return j.Frames[n-1].Seq
	}
	return j.LastAcked
}

// AppendOutput stores f when there is room and f.Seq is contiguous.
func (m *MemorySpool) AppendOutput(_ context.Context, ref JobRef, f OutputFrame) error {
	m.mu.Lock()
	defer m.mu.Unlock()
	j := m.jobs[ref]
	last := uint64(0)
	if j != nil {
		last = lastSeq(j)
	}
	if f.Seq != last+1 {
		return ErrSpoolSequence
	}
	if m.frames >= m.capacity {
		return ErrSpoolFull
	}
	j = m.job(ref)
	j.Frames = append(j.Frames, f)
	m.frames++
	return nil
}

// PutResult retains the terminal result.
func (m *MemorySpool) PutResult(_ context.Context, ref JobRef, r Result) error {
	m.mu.Lock()
	defer m.mu.Unlock()
	res := r
	m.job(ref).Result = &res
	return nil
}

// Ack drops frames with Seq <= seq; lower acks are ignored.
func (m *MemorySpool) Ack(_ context.Context, ref JobRef, seq uint64) error {
	m.mu.Lock()
	defer m.mu.Unlock()
	j := m.jobs[ref]
	if j == nil || seq <= j.LastAcked {
		return nil
	}
	kept := j.Frames[:0]
	for _, f := range j.Frames {
		if f.Seq <= seq {
			m.frames--
			continue
		}
		kept = append(kept, f)
	}
	j.Frames = kept
	j.LastAcked = seq
	return nil
}

// AckResult forgets the job after the server durably accepted its result.
func (m *MemorySpool) AckResult(ctx context.Context, ref JobRef) error { return m.Drop(ctx, ref) }

// Drop forgets the job (fenced or acknowledged).
func (m *MemorySpool) Drop(_ context.Context, ref JobRef) error {
	m.mu.Lock()
	defer m.mu.Unlock()
	if j := m.jobs[ref]; j != nil {
		m.frames -= len(j.Frames)
		delete(m.jobs, ref)
	}
	return nil
}

func cloneJob(j *SpooledJob) SpooledJob {
	out := SpooledJob{Ref: j.Ref, LastAcked: j.LastAcked, Frames: append([]OutputFrame(nil), j.Frames...)}
	if j.Result != nil {
		r := *j.Result
		out.Result = &r
	}
	return out
}

// Get returns a copy of one job's state.
func (m *MemorySpool) Get(_ context.Context, ref JobRef) (SpooledJob, bool, error) {
	m.mu.Lock()
	defer m.mu.Unlock()
	j := m.jobs[ref]
	if j == nil {
		return SpooledJob{}, false, nil
	}
	return cloneJob(j), true, nil
}

// Pending returns copies of every retained job, sorted.
func (m *MemorySpool) Pending(_ context.Context) ([]SpooledJob, error) {
	m.mu.Lock()
	defer m.mu.Unlock()
	out := make([]SpooledJob, 0, len(m.jobs))
	for _, j := range m.jobs {
		out = append(out, cloneJob(j))
	}
	sort.Slice(out, func(i, k int) bool {
		if out[i].Ref.JobID != out[k].Ref.JobID {
			return out[i].Ref.JobID < out[k].Ref.JobID
		}
		return out[i].Ref.Generation < out[k].Ref.Generation
	})
	return out, nil
}

// Config is the runner's admission metadata and timing policy.
type Config struct {
	OS              string
	Arch            string
	ProtocolVersion int
	RunnerVersion   string
	HarnessVersions map[string]string
	WorkspaceRoot   string
	ScratchRoot     string
	Capabilities    []string
	Heartbeat       time.Duration // default 5s
	ProbeInterval   time.Duration // default 60s
	BackoffBase     time.Duration // default 500ms
	BackoffCap      time.Duration // default 30s
}

// Deps are the agent's ports. Every field except Logger, Clock and Rand is required.
type Deps struct {
	Conn      Connection
	Exec      JobExecutor
	Sources   LocalSources
	Telemetry runner.MachineTelemetry
	Spool     Spool
	Identity  IdentityStore
	Clock     Clock
	Rand      func() float64
	Logger    *slog.Logger
}

type jobState struct {
	ref     JobRef
	cancel  context.CancelFunc
	lastSeq uint64
	done    bool
	fenced  bool
}

// Agent is the runner's connection and job lifecycle.
type Agent struct {
	cfg Config
	d   Deps
	log *slog.Logger

	sendMu  sync.Mutex // serializes Session.Send
	flushMu sync.Mutex // serializes spool->session flushing

	mu         sync.Mutex
	runCtx     context.Context
	sess       Session
	jobs       map[string]*jobState
	seen       map[JobRef]bool
	sent       map[JobRef]uint64
	resultSent map[JobRef]bool
	ackWake    chan struct{}
}

// New validates the ports and applies timing defaults.
func New(cfg Config, d Deps) (*Agent, error) {
	var missing []string
	if d.Conn == nil {
		missing = append(missing, "Conn")
	}
	if d.Exec == nil {
		missing = append(missing, "Exec")
	}
	if d.Sources == nil {
		missing = append(missing, "Sources")
	}
	if d.Telemetry == nil {
		missing = append(missing, "Telemetry")
	}
	if d.Spool == nil {
		missing = append(missing, "Spool")
	}
	if d.Identity == nil {
		missing = append(missing, "Identity")
	}
	if len(missing) > 0 {
		return nil, fmt.Errorf("runneragent: missing ports: %s", strings.Join(missing, ", "))
	}
	if d.Clock == nil {
		d.Clock = SystemClock{}
	}
	if d.Rand == nil {
		d.Rand = func() float64 { return 1 }
	}
	if d.Logger == nil {
		d.Logger = slog.Default()
	}
	if cfg.Heartbeat <= 0 {
		cfg.Heartbeat = 5 * time.Second
	}
	if cfg.ProbeInterval <= 0 {
		cfg.ProbeInterval = 60 * time.Second
	}
	if cfg.BackoffBase <= 0 {
		cfg.BackoffBase = 500 * time.Millisecond
	}
	if cfg.BackoffCap <= 0 {
		cfg.BackoffCap = 30 * time.Second
	}
	return &Agent{
		cfg: cfg, d: d, log: d.Logger,
		jobs: map[string]*jobState{}, seen: map[JobRef]bool{},
		sent: map[JobRef]uint64{}, resultSent: map[JobRef]bool{},
		ackWake: make(chan struct{}),
	}, nil
}

// Run connects, serves and reconnects until ctx ends. Jobs keep running
// across connection drops; their output is spooled and replayed.
func (a *Agent) Run(ctx context.Context) error {
	a.mu.Lock()
	a.runCtx = ctx
	a.mu.Unlock()
	if err := a.pruneOrphans(ctx); err != nil {
		return err
	}
	a.d.Sources.Probe(ctx)
	go a.probeLoop(ctx)
	attempt := 0
	for {
		admitted, err := a.serveSession(ctx)
		if ctx.Err() != nil {
			return ctx.Err()
		}
		if admitted {
			attempt = 0
		}
		delay := Backoff(attempt, a.cfg.BackoffBase, a.cfg.BackoffCap, a.d.Rand)
		attempt++
		a.log.Warn("runner session ended; reconnecting", "error", err, "delay", delay)
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-a.d.Clock.After(delay):
		}
	}
}

// pruneOrphans drops spooled frames of jobs that died with a previous
// process (no terminal result); terminal results are kept for replay.
func (a *Agent) pruneOrphans(ctx context.Context) error {
	pending, err := a.d.Spool.Pending(ctx)
	if err != nil {
		return fmt.Errorf("runneragent: read spool: %w", err)
	}
	for _, p := range pending {
		if p.Result == nil {
			if err := a.d.Spool.Drop(ctx, p.Ref); err != nil {
				return fmt.Errorf("runneragent: prune orphan %s: %w", p.Ref.JobID, err)
			}
		}
	}
	return nil
}

func (a *Agent) probeLoop(ctx context.Context) {
	for {
		select {
		case <-ctx.Done():
			return
		case <-a.d.Clock.After(a.cfg.ProbeInterval):
		}
		a.d.Sources.Probe(ctx)
	}
}

func (a *Agent) serveSession(ctx context.Context) (bool, error) {
	s, err := a.d.Conn.Dial(ctx)
	if err != nil {
		return false, fmt.Errorf("dial: %w", err)
	}
	defer s.Close()
	sctx, cancel := context.WithCancel(ctx)
	defer cancel()
	every, err := a.admit(sctx, s)
	if err != nil {
		return false, err
	}
	a.attach(s)
	defer a.detach(s)
	errc := make(chan error, 2)
	go func() { errc <- a.recvLoop(sctx, s) }()
	go func() { errc <- a.heartbeatLoop(sctx, s, every) }()
	a.replay(sctx)
	err = <-errc
	cancel()
	_ = s.Close()
	<-errc
	return true, err
}

// admit runs the handshake and returns the session's heartbeat interval.
func (a *Agent) admit(ctx context.Context, s Session) (time.Duration, error) {
	in, err := s.Recv(ctx)
	if err != nil {
		return 0, fmt.Errorf("await challenge: %w", err)
	}
	if in.Kind == KindRefused && in.Refused != nil {
		return 0, &RefusedError{Code: in.Refused.Code, Reason: in.Refused.Reason}
	}
	if in.Kind != KindChallenge || in.Challenge == nil {
		return 0, fmt.Errorf("runneragent: expected challenge, got %q", in.Kind)
	}
	key, err := a.d.Identity.Key(ctx)
	if err != nil {
		return 0, fmt.Errorf("runneragent: machine key: %w", err)
	}
	pub, ok := key.Public().(ed25519.PublicKey)
	if !ok {
		return 0, errors.New("runneragent: machine key is not ed25519")
	}
	sig := ed25519.Sign(key, ChallengeMessage(in.Challenge.KeyID, in.Challenge.Nonce))
	if err := a.send(ctx, s, Outbound{Kind: KindChallengeResponse, ChallengeResponse: &ChallengeResponse{PublicKey: pub, Signature: sig}}); err != nil {
		return 0, err
	}
	hello := Hello{
		OS: a.cfg.OS, Arch: a.cfg.Arch, ProtocolVersion: a.cfg.ProtocolVersion,
		RunnerVersion: a.cfg.RunnerVersion, HarnessVersions: a.cfg.HarnessVersions,
		WorkspaceRoot: a.cfg.WorkspaceRoot, ScratchRoot: a.cfg.ScratchRoot,
		Capabilities: a.cfg.Capabilities, Sources: a.d.Sources.Sources(ctx), Held: a.held(ctx),
	}
	if err := a.send(ctx, s, Outbound{Kind: KindHello, Hello: &hello}); err != nil {
		return 0, err
	}
	in, err = s.Recv(ctx)
	if err != nil {
		return 0, fmt.Errorf("await admission: %w", err)
	}
	switch in.Kind {
	case KindAdmitted:
		every := a.cfg.Heartbeat
		if in.Admitted != nil && in.Admitted.HeartbeatSeconds > 0 {
			every = time.Duration(in.Admitted.HeartbeatSeconds) * time.Second
		}
		return every, nil
	case KindRefused:
		r := Refusal{Code: "refused"}
		if in.Refused != nil {
			r = *in.Refused
		}
		return 0, &RefusedError{Code: r.Code, Reason: r.Reason}
	default:
		return 0, fmt.Errorf("runneragent: expected admission, got %q", in.Kind)
	}
}

func (a *Agent) attach(s Session) {
	a.mu.Lock()
	defer a.mu.Unlock()
	a.sess = s
	a.sent = map[JobRef]uint64{}
	a.resultSent = map[JobRef]bool{}
}

func (a *Agent) detach(s Session) {
	a.mu.Lock()
	defer a.mu.Unlock()
	if a.sess == s {
		a.sess = nil
	}
}

func (a *Agent) send(ctx context.Context, s Session, m Outbound) error {
	a.sendMu.Lock()
	defer a.sendMu.Unlock()
	return s.Send(ctx, m)
}

// held lists jobs the runner still owns, for hello and heartbeats.
func (a *Agent) held(ctx context.Context) []HeldJob {
	pending, err := a.d.Spool.Pending(ctx)
	if err != nil {
		a.log.Error("read spool for held jobs", "error", err)
	}
	spooled := map[JobRef]SpooledJob{}
	for _, p := range pending {
		spooled[p.Ref] = p
	}
	a.mu.Lock()
	out := []HeldJob{}
	for _, js := range a.jobs {
		if js.fenced {
			continue
		}
		state := HeldRunning
		if js.done {
			state = HeldTerminal
		}
		out = append(out, HeldJob{JobID: js.ref.JobID, Generation: js.ref.Generation, State: state, LastOutputSeq: js.lastSeq})
		delete(spooled, js.ref)
	}
	a.mu.Unlock()
	for ref, p := range spooled {
		if p.Result == nil {
			continue
		}
		last := p.LastAcked
		if n := len(p.Frames); n > 0 {
			last = p.Frames[n-1].Seq
		}
		out = append(out, HeldJob{JobID: ref.JobID, Generation: ref.Generation, State: HeldTerminal, LastOutputSeq: last})
	}
	sort.Slice(out, func(i, k int) bool {
		if out[i].JobID != out[k].JobID {
			return out[i].JobID < out[k].JobID
		}
		return out[i].Generation < out[k].Generation
	})
	return out
}

func (a *Agent) heartbeatLoop(ctx context.Context, s Session, every time.Duration) error {
	for {
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-a.d.Clock.After(every):
		}
		if err := a.sendHeartbeat(ctx, s); err != nil {
			return err
		}
	}
}

func (a *Agent) sendHeartbeat(ctx context.Context, s Session) error {
	tel, err := a.d.Telemetry.Sample(ctx)
	if err != nil {
		a.log.Warn("telemetry sample failed; reporting unavailable", "error", err)
		tel = runner.Telemetry{SampledAt: a.d.Clock.Now()}
	}
	hb := Heartbeat{Telemetry: tel, Sources: a.d.Sources.Sources(ctx), Jobs: a.held(ctx)}
	return a.send(ctx, s, Outbound{Kind: KindHeartbeat, Heartbeat: &hb})
}

func (a *Agent) recvLoop(ctx context.Context, s Session) error {
	for {
		in, err := s.Recv(ctx)
		if err != nil {
			return err
		}
		ref := JobRef{JobID: in.JobID, Generation: in.Generation}
		switch in.Kind {
		case KindDispatch:
			a.onDispatch(ctx, s, in)
		case KindCancel:
			a.onCancel(ref)
		case KindAck:
			a.onAck(ctx, ref, in)
		case KindFenced:
			a.fence(ctx, ref)
		case KindProbe:
			go a.onProbe(ctx, s, in.Probe)
		default:
			a.log.Warn("ignoring unexpected server message", "kind", in.Kind)
		}
	}
}

func (a *Agent) onDispatch(ctx context.Context, s Session, in Inbound) {
	if in.Dispatch == nil || in.JobID == "" {
		a.log.Warn("ignoring dispatch without payload", "job", in.JobID)
		return
	}
	d := *in.Dispatch
	d.JobID, d.Generation = in.JobID, in.Generation
	ref := JobRef{JobID: d.JobID, Generation: d.Generation}
	accept := Outbound{Kind: KindAccept, JobID: d.JobID, Generation: d.Generation}

	a.mu.Lock()
	if a.seen[ref] {
		js := a.jobs[d.JobID]
		live := js != nil && js.ref == ref && !js.fenced
		a.mu.Unlock()
		if live {
			_ = a.send(ctx, s, accept) // idempotent re-acknowledgment; no second execution
		}
		return
	}
	a.seen[ref] = true
	a.mu.Unlock()

	if ok, reason := a.d.Exec.Supports(d); !ok {
		_ = a.send(ctx, s, Outbound{Kind: KindReject, JobID: d.JobID, Generation: d.Generation, RejectReason: reason})
		return
	}
	a.mu.Lock()
	old := a.jobs[d.JobID]
	a.mu.Unlock()
	if old != nil && old.ref.Generation < d.Generation {
		a.fence(ctx, old.ref)
	}
	if err := a.send(ctx, s, accept); err != nil {
		a.log.Warn("accept not delivered; job still starts and replays", "job", d.JobID, "error", err)
	}
	a.startJob(d)
}

func (a *Agent) startJob(d Dispatch) {
	ref := JobRef{JobID: d.JobID, Generation: d.Generation}
	a.mu.Lock()
	jctx, cancel := context.WithCancel(a.runCtx)
	js := &jobState{ref: ref, cancel: cancel}
	a.jobs[d.JobID] = js
	a.mu.Unlock()
	go func() {
		defer cancel()
		res := a.d.Exec.Execute(jctx, d, func(ctx context.Context, token string) error {
			return a.emit(ctx, js, token)
		})
		a.mu.Lock()
		fenced := js.fenced
		a.mu.Unlock()
		if fenced {
			return
		}
		a.mu.Lock()
		js.done = true
		a.mu.Unlock()
		bg := context.WithoutCancel(jctx)
		if err := a.d.Spool.PutResult(bg, ref, res); err != nil {
			a.log.Error("spool result", "job", ref.JobID, "error", err)
			return
		}
		a.flush(bg, ref)
	}()
}

// emit appends one frame (blocking while the spool is full) then flushes.
func (a *Agent) emit(ctx context.Context, js *jobState, token string) error {
	a.mu.Lock()
	seq := js.lastSeq + 1
	a.mu.Unlock()
	for {
		wake := a.ackSignal()
		err := a.d.Spool.AppendOutput(ctx, js.ref, OutputFrame{Seq: seq, Token: token})
		if err == nil {
			break
		}
		if !errors.Is(err, ErrSpoolFull) {
			return err
		}
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-wake:
		}
	}
	a.mu.Lock()
	js.lastSeq = seq
	fenced := js.fenced
	a.mu.Unlock()
	if fenced {
		_ = a.d.Spool.Drop(context.WithoutCancel(ctx), js.ref)
		return context.Canceled
	}
	a.flush(ctx, js.ref)
	return nil
}

func (a *Agent) ackSignal() <-chan struct{} {
	a.mu.Lock()
	defer a.mu.Unlock()
	return a.ackWake
}

func (a *Agent) wake() {
	a.mu.Lock()
	close(a.ackWake)
	a.ackWake = make(chan struct{})
	a.mu.Unlock()
}

// flush sends every spooled frame of ref beyond what this session already
// sent (or the server acknowledged), then the terminal result once.
func (a *Agent) flush(ctx context.Context, ref JobRef) {
	a.flushMu.Lock()
	defer a.flushMu.Unlock()
	a.mu.Lock()
	s := a.sess
	from := a.sent[ref]
	resultSent := a.resultSent[ref]
	a.mu.Unlock()
	if s == nil {
		return
	}
	sj, ok, err := a.d.Spool.Get(ctx, ref)
	if err != nil || !ok {
		return
	}
	if sj.LastAcked > from {
		from = sj.LastAcked
	}
	for _, f := range sj.Frames {
		if f.Seq <= from {
			continue
		}
		frame := f
		if err := a.send(ctx, s, Outbound{Kind: KindOutput, JobID: ref.JobID, Generation: ref.Generation, Seq: f.Seq, Output: &frame}); err != nil {
			return
		}
		a.mu.Lock()
		if a.sess == s {
			a.sent[ref] = f.Seq
		}
		a.mu.Unlock()
	}
	if sj.Result != nil && !resultSent {
		if err := a.send(ctx, s, Outbound{Kind: KindResult, JobID: ref.JobID, Generation: ref.Generation, Result: sj.Result}); err != nil {
			return
		}
		a.mu.Lock()
		if a.sess == s {
			a.resultSent[ref] = true
		}
		a.mu.Unlock()
	}
}

// replay resends everything the server has not acknowledged.
func (a *Agent) replay(ctx context.Context) {
	pending, err := a.d.Spool.Pending(ctx)
	if err != nil {
		a.log.Error("read spool for replay", "error", err)
		return
	}
	for _, p := range pending {
		a.flush(ctx, p.Ref)
	}
}

func (a *Agent) onAck(ctx context.Context, ref JobRef, in Inbound) {
	if in.AckResult {
		if err := a.d.Spool.AckResult(ctx, ref); err != nil {
			a.log.Error("ack result", "job", ref.JobID, "error", err)
		}
		a.mu.Lock()
		if js := a.jobs[ref.JobID]; js != nil && js.ref == ref {
			delete(a.jobs, ref.JobID)
		}
		a.mu.Unlock()
	} else if err := a.d.Spool.Ack(ctx, ref, in.Seq); err != nil {
		a.log.Error("ack output", "job", ref.JobID, "error", err)
	}
	a.wake()
}

// onProbe refreshes local sources and answers with the probed source; a
// source this runner does not have is reported with unknown auth, never ok.
func (a *Agent) onProbe(ctx context.Context, s Session, req *ProbeRequest) {
	sources := a.d.Sources.Probe(ctx)
	if req == nil {
		if err := a.sendHeartbeat(ctx, s); err != nil {
			a.log.Warn("probe heartbeat failed", "error", err)
		}
		return
	}
	src := Source{Kind: req.SourceKind, CredentialRef: req.CredentialRef,
		Health: runner.SourceHealth{Auth: "unknown", Detail: "source not present on this runner"}}
	for _, c := range sources {
		if c.Kind == req.SourceKind && c.CredentialRef == req.CredentialRef {
			src = c
			break
		}
	}
	if err := a.send(ctx, s, Outbound{Kind: KindProbeResult, ProbeResult: &src}); err != nil {
		a.log.Warn("probe result not delivered", "error", err)
	}
}

func (a *Agent) onCancel(ref JobRef) {
	a.mu.Lock()
	defer a.mu.Unlock()
	if js := a.jobs[ref.JobID]; js != nil && js.ref == ref {
		js.cancel()
	}
}

// fence kills the local execution of ref and forgets it; any later result
// of that generation is discarded.
func (a *Agent) fence(ctx context.Context, ref JobRef) {
	a.mu.Lock()
	a.seen[ref] = true
	if js := a.jobs[ref.JobID]; js != nil && js.ref == ref {
		js.fenced = true
		js.cancel()
		delete(a.jobs, ref.JobID)
	}
	a.mu.Unlock()
	if err := a.d.Spool.Drop(context.WithoutCancel(ctx), ref); err != nil {
		a.log.Error("drop fenced job", "job", ref.JobID, "error", err)
	}
	a.wake()
}
