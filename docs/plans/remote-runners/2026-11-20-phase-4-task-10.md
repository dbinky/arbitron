**Goal:** Move a pinned session when its source becomes exhausted or logged out, or its runner is lost or draining, even while the run is idle and waiting for human input. After a restart, every pending turn intent is either adopted from a still-live remote turn or fenced and resent with its exact durable message. Drain moves pinned sessions at their turn boundaries and reports sessions that cannot move, naming the missing group.

**Architecture:** `runner.SessionRecoveryController` listens to recorded pool events through the new `(*EventRecorder).AddListener`. For each matching event it enqueues an idempotent move for the affected pinned sessions; the `PinnedSessionIndex` lists them from `node_executions`. A worker flushes the queue on pool-change wakes and on a retry tick. Each move goes through `engine.SessionRecoverer`, implemented by `conversationHandler`. That path acquires the session's own turn lease, so a busy session defers to its turn boundary, and then calls the P4 T8 mover. When the mover finds no target, it publishes `execution.status=waiting_capacity` and the move stays queued until the pool changes. `StartupScan` replays `TurnIntentStore.Pending()`. For each intent it adopts a live remote turn when one exists, otherwise it fences that turn's ledger jobs and resends the exact intent through the normal turn path; the existing intent is reused, never duplicated. It then reconciles every pin against the registry. `(*Service).Drain` merges the controller's per-runner session progress and blockers into `DrainProgress`.

**Tech Stack:** Go 1.25.5, testify, modernc SQLite, google/uuid.

// adapter/sqlite/session_pin_index.go
package sqlite

import (
	"context"
	"database/sql"
	"fmt"

	"github.com/arbitron/arbitron/domain/runner"
)

var _ runner.PinnedSessionIndex = (*SessionPinIndex)(nil)

// SessionPinIndex implements runner.PinnedSessionIndex over node_executions.
type SessionPinIndex struct{ db *sql.DB }

// NewSessionPinIndex creates a SessionPinIndex.
func NewSessionPinIndex(db *sql.DB) *SessionPinIndex { return &SessionPinIndex{db: db} }

func (x *SessionPinIndex) ListPinned(ctx context.Context) ([]runner.PinnedSession, error) {
	rows, err := x.db.QueryContext(ctx, `
SELECT ne.id, ne.instance_id, ne.node_id, ne.session_pin_runner, COALESCE(ne.source_id, '')
FROM node_executions ne
JOIN instances i ON i.id = ne.instance_id
WHERE i.state = 'waiting'
  AND i.current_node_id = ne.node_id
  AND ne.session_pin_runner IS NOT NULL AND ne.session_pin_runner != ''
  AND ne.attempt = (SELECT MAX(x.attempt) FROM node_executions x WHERE x.instance_id = ne.instance_id AND x.node_id = ne.node_id)
ORDER BY ne.instance_id`)
	if err != nil {
		return nil, fmt.Errorf("listing pinned sessions: %w", err)
	}
	defer rows.Close()
	var out []runner.PinnedSession
	for rows.Next() {
		var id, run, node, pin, src string
		if err := rows.Scan(&id, &run, &node, &pin, &src); err != nil {
			return nil, fmt.Errorf("scanning pinned session: %w", err)
		}
		execID, err := parseUUID(id)
		if err != nil {
			return nil, err
		}
		runID, err := parseUUID(run)
		if err != nil {
			return nil, err
		}
		out = append(out, runner.PinnedSession{RunID: runID, NodeID: node, ExecutionID: execID,
			RunnerID: runner.RunnerID(pin), SourceID: runner.SourceID(src)})
	}
	return out, rows.Err()
}

// domain/engine/session_recovery.go
package engine

import (
	"context"

	"github.com/google/uuid"
)

// AdoptedTurn is the output of a remote turn that was still live (or had
// completed) when the server restarted, adopted instead of resent.
type AdoptedTurn struct {
	Response  string
	SessionID string

    Completions []StepCompletion
    Lease TurnLease
    Identity InteractiveTurnIdentity
}

// SessionRecoverer runs session recovery on behalf of the system (no user).
type SessionRecoverer interface {
	// RecoverSession moves the waiting session of (runID,nodeID) off `from`
	// under its own turn lease. A busy session returns an error wrapping
	// ErrInteractiveTurnInProgress (retry at the turn boundary); a session no
	// longer waiting or no longer pinned to `from` returns nil (stale trigger).
	RecoverSession(ctx context.Context, runID uuid.UUID, nodeID, reason, from string) error
	// ResumePendingTurn replays a durable intent (message/end/abandon) through
	// the normal turn path, reusing the intent. adopted != nil commits that
	// output instead of dispatching.
	ResumePendingTurn(ctx context.Context, in TurnIntent, adopted *AdoptedTurn) error
}

type resumedTurnKey struct{}

type resumedTurn struct {
	intent  TurnIntent
	adopted *AdoptedTurn
}

func contextWithResumedTurn(ctx context.Context, r resumedTurn) context.Context {
	return context.WithValue(ctx, resumedTurnKey{}, r)
}

func resumedTurnFrom(ctx context.Context) (resumedTurn, bool) {
	r, ok := ctx.Value(resumedTurnKey{}).(resumedTurn)
	return r, ok
}

// TurnAdopter returns the output of a turn whose remote job is still live

type TurnAdopter interface {
	AdoptTurn(ctx context.Context, job Job) (engine.AdoptedTurn, bool, error)

// SessionDrainer moves the sessions pinned to a draining runner at their turn
// boundaries and reports what is left.
type SessionDrainer interface {
	DrainSessions(ctx context.Context, id RunnerID) (pending int, blocked []DrainBlocker, err error)
}

// SessionRecoveryDeps are the controller's collaborators. Adopter is optional.
type SessionRecoveryDeps struct {
	Registry      Registry
	Ledger        Ledger
	Pins          PinnedSessionIndex
	Intents       engine.TurnIntentStore
	Recoverer     engine.SessionRecoverer
	Invocations   engine.InvocationStore
	Adopter       TurnAdopter
	Notifier      *PoolNotifier
	Clock         func() time.Time
	RetryInterval time.Duration
}

type sessionKey struct {
	run  uuid.UUID
	node string
}

type queuedMove struct {
	session PinnedSession
	reason  string
}

// SessionRecoveryController turns pool events into idempotent session moves,
// even while runs wait for human input, and resumes durable intents at startup.
type SessionRecoveryController struct {
	d       SessionRecoveryDeps
	mu      sync.Mutex
	queue   map[sessionKey]queuedMove
	blocked map[sessionKey]string // group with no eligible target
	retry   map[uuid.UUID]engine.TurnIntent
	kick    chan struct{}
	flushMu sync.Mutex
}

// NewSessionRecoveryController validates deps.
func NewSessionRecoveryController(d SessionRecoveryDeps) *SessionRecoveryController {
	if d.Registry == nil || d.Ledger == nil || d.Pins == nil || d.Intents == nil || d.Recoverer == nil || d.Invocations == nil || d.Notifier == nil {
		panic("runner: NewSessionRecoveryController requires Registry, Ledger, Pins, Intents, Recoverer, Invocations and Notifier")
	}
	if d.Clock == nil {
		d.Clock = time.Now
	}
	if d.RetryInterval <= 0 {
		d.RetryInterval = 5 * time.Second
	}
	return &SessionRecoveryController{d: d, queue: map[sessionKey]queuedMove{}, blocked: map[sessionKey]string{},
		retry: map[uuid.UUID]engine.TurnIntent{}, kick: make(chan struct{}, 1)}
}

// HandlePoolEvent enqueues moves for the sessions an event affects. It never
// blocks (it is an EventRecorder listener).
func (c *SessionRecoveryController) HandlePoolEvent(ctx context.Context, ev PoolEvent) {
	var reason string
	var match func(PinnedSession) bool
	switch ev.Kind {
	case "runner.lost":
		reason, match = engine.MoveReasonRunnerLost, func(s PinnedSession) bool { return s.RunnerID == ev.RunnerID }
	case "runner.draining":
		reason, match = engine.MoveReasonDrain, func(s PinnedSession) bool { return s.RunnerID == ev.RunnerID }
	case "source.logged_out":
		reason, match = engine.MoveReasonLoggedOut, func(s PinnedSession) bool { return s.SourceID == ev.SourceID }
	case "source.health_changed":
		src, err := c.d.Registry.GetSource(ctx, ev.SourceID)
		if err != nil {
			slog.Warn("session recovery: read source", "source", ev.SourceID, "error", err)
			return
		}
		r, ok := sourceMoveReason(src.Health)
		if !ok {
			return
		}
		reason, match = r, func(s PinnedSession) bool { return s.SourceID == ev.SourceID }
	default:
		return // cordon, suspect, load: never move pins
	}
	pins, err := c.d.Pins.ListPinned(ctx)
	if err != nil {
		slog.Warn("session recovery: list pinned sessions", "error", err)
		return
	}
	for _, s := range pins {
		if match(s) {
			c.enqueue(s, reason)
		}
	}
	c.wake()
}

func sourceMoveReason(h SourceHealth) (string, bool) {
	if h.Auth == AuthState("logged_out") {
		return engine.MoveReasonLoggedOut, true
	}
	if h.Capacity == CapacityState("exhausted") {
		return engine.MoveReasonExhausted, true
	}
	return "", false
}

func (c *SessionRecoveryController) enqueue(s PinnedSession, reason string) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.queue[sessionKey{s.RunID, s.NodeID}] = queuedMove{session: s, reason: reason}
}

func (c *SessionRecoveryController) wake() {
	select {
	case c.kick <- struct{}{}:
	default:
	}
}

// Flush attempts every queued move and retryable intent once. Busy sessions
// (turn in flight) and sessions without a target stay queued.
func (c *SessionRecoveryController) Flush(ctx context.Context) {
	c.flushMu.Lock()
	defer c.flushMu.Unlock()
	c.mu.Lock()
	moves := make([]queuedMove, 0, len(c.queue))
	for _, m := range c.queue {
		moves = append(moves, m)
	}
	c.mu.Unlock()
	for _, m := range moves {
		c.attemptMove(ctx, m)
	}
}

func (c *SessionRecoveryController) attemptMove(ctx context.Context, m queuedMove) {
	key := sessionKey{m.session.RunID, m.session.NodeID}
	inv, err := c.d.Invocations.Begin(ctx, m.session.RunID, m.session.NodeID)
	if err != nil {
		slog.Warn("session recovery: invocation", "run", m.session.RunID, "error", err)
		return
	}
	mctx := engine.ContextWithStepIdentity(ctx, engine.StepIdentity{RunID: m.session.RunID, NodeID: m.session.NodeID,
		InvocationID: inv.ID, OperationID: "session-move", Kind: engine.OperationTurn, Attempt: 1})
	err = c.d.Recoverer.RecoverSession(mctx, m.session.RunID, m.session.NodeID, m.reason, string(m.session.RunnerID))
	c.mu.Lock()
	defer c.mu.Unlock()
	var wait *engine.SessionWaitError
	switch {
	case err == nil:
		delete(c.queue, key)
		delete(c.blocked, key)
	case errors.Is(err, instance.ErrInteractiveTurnInProgress):
		// Turn in flight: retried at the boundary (next wake or tick).
	case errors.As(err, &wait):
		c.blocked[key] = wait.Group
	default:
		slog.Warn("session recovery: move failed", "run", m.session.RunID, "reason", m.reason, "error", err)
	}
}

// DrainSessions implements SessionDrainer.
func (c *SessionRecoveryController) DrainSessions(ctx context.Context, id RunnerID) (int, []DrainBlocker, error) {
	pins, err := c.d.Pins.ListPinned(ctx)
	if err != nil {
		return 0, nil, fmt.Errorf("session drain: list pinned: %w", err)
	}
	for _, s := range pins {
		if s.RunnerID == id {
			c.enqueue(s, engine.MoveReasonDrain)
		}
	}
	c.Flush(ctx)
	pins, err = c.d.Pins.ListPinned(ctx)
	if err != nil {
		return 0, nil, fmt.Errorf("session drain: list pinned: %w", err)
	}
	pending := 0
	var blocked []DrainBlocker
	c.mu.Lock()
	defer c.mu.Unlock()
	for _, s := range pins {
		if s.RunnerID != id {
			continue
		}
		pending++
		if g, ok := c.blocked[sessionKey{s.RunID, s.NodeID}]; ok {
			run := s.RunID
			blocked = append(blocked, DrainBlocker{Kind: "session", RunID: &run, Group: g, Reason: "no eligible runner serves the session's model group"})
		}
	}
	return pending, blocked, nil
}

// Start runs the worker: it flushes on enqueue, on every pool change and
// every RetryInterval, until ctx ends.
func (c *SessionRecoveryController) Start(ctx context.Context) {
	wake, cancel := c.d.Notifier.Subscribe()
	go func() {
		defer cancel()
		t := time.NewTicker(c.d.RetryInterval)
		defer t.Stop()
		for {
			select {
			case <-ctx.Done():
				return
			case <-c.kick:
			case <-wake:
			case <-t.C:
			}
			c.Flush(ctx)
		}
	}()
}
