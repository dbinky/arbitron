**Goal:** Move a pinned session when its source becomes exhausted or logged out, or its runner is lost or draining, even while the run is idle and waiting for human input. After a restart, every pending turn intent is either adopted from a still-live remote turn or fenced and resent with its exact durable message. Drain moves pinned sessions at their turn boundaries and reports sessions that cannot move, naming the missing group.

**Architecture:** `runner.SessionRecoveryController` listens to recorded pool events through the new `(*EventRecorder).AddListener`. For each matching event it enqueues an idempotent move for the affected pinned sessions; the `PinnedSessionIndex` lists them from `node_executions`. A worker flushes the queue on pool-change wakes and on a retry tick. Each move goes through `engine.SessionRecoverer`, implemented by `conversationHandler`. That path acquires the session's own turn lease, so a busy session defers to its turn boundary, and then calls the P4 T8 mover. When the mover finds no target, it publishes `execution.status=waiting_capacity` and the move stays queued until the pool changes. `StartupScan` replays `TurnIntentStore.Pending()`. For each intent it adopts a live remote turn when one exists, otherwise it fences that turn's ledger jobs and resends the exact intent through the normal turn path; the existing intent is reused, never duplicated. It then reconciles every pin against the registry. `(*Service).Drain` merges the controller's per-runner session progress and blockers into `DrainProgress`.

// seedPinned (Authorized fixture change, unstick 2026-11-23; signature kept for
// P4 T14): a run in `state` at `node` with ONE open node visit holding
// `attempt` executions. Attempts before the last are pinned to "older-runner";
// the last is pinned to `pin` with source `source` (pin "" = never pinned).
// Attempt a gets session generation a, all stamped with the open invocation.
func seedPinned(t *testing.T, db testDB, state, node string, attempt int, pin, source string) (uuid.UUID, uuid.UUID) {
	t.Helper()
	runID := seedInstance(t, db, localUserID(t), state)
	_, err := db.DB.Exec(`UPDATE instances SET current_node_id = ? WHERE id = ?`, node, uuidToString(runID))
	require.NoError(t, err)
	inv, err := NewInvocationRepo(db.DB).Begin(context.Background(), runID, node)
	require.NoError(t, err)
	var last uuid.UUID
	for a := 1; a <= attempt; a++ {
		p, src := pin, source
		if a < attempt {
			p, src = "older-runner", "older-runner/anthropic_subscription/-"
		}
		last = seedSessionExec(t, db, runID, node, inv.ID, a, int64(a), p, src)
	}
	return runID, last
}

// seedPinnedVisit adds one execution in the CURRENT open visit of (runID,
// node), beginning a new visit when commitPrevious closes the open one.
func seedPinnedVisit(t *testing.T, db testDB, runID uuid.UUID, node string, commitPrevious bool, attempt int, gen int64, pin, source string) (uuid.UUID, uuid.UUID) {
	t.Helper()
	ctx := context.Background()
	repo := NewInvocationRepo(db.DB)
	inv, err := repo.Begin(ctx, runID, node)
	require.NoError(t, err)
	if commitPrevious {
		require.NoError(t, repo.Commit(ctx, inv.ID))
		inv, err = repo.Begin(ctx, runID, node)
		require.NoError(t, err)
	}
	return inv.ID, seedSessionExec(t, db, runID, node, inv.ID, attempt, gen, pin, source)
}

func seedSessionExec(t *testing.T, db testDB, runID uuid.UUID, node string, invocation uuid.UUID, attempt int, gen int64, pin, source string) uuid.UUID {
	t.Helper()
	ctx := context.Background()

		Status: instance.NodeStatusSucceeded, LLMConversation: map[string]any{"messages": []any{}}, StartedAt: time.Now().UTC()}

	return exec.ID
}

package sqlite

import (
	"context"
	"database/sql"
	"fmt"

	"github.com/arbitron/arbitron/domain/runner"
)

var _ runner.PinnedSessionIndex = (*SessionPinIndex)(nil)

// activeSessionExecSQL selects the id of the ACTIVE session record of the
// outer row's (instance, node): the record stamped with the node's OPEN visit
// (node_invocations.committed_at IS NULL), highest session generation first.
// Retry attempts reset on a back-edge, so attempt order is never consulted.
// A zero/NULL session_invocation_id joins no visit and is never active.
// Shared with ListByRunner (P4 T14). The outer alias must be `ne`.
const activeSessionExecSQL = `SELECT x.id FROM node_executions x
JOIN node_invocations ni ON ni.id = x.session_invocation_id
WHERE x.instance_id = ne.instance_id AND x.node_id = ne.node_id
  AND ni.run_id = x.instance_id AND ni.node_id = x.node_id AND ni.committed_at IS NULL
ORDER BY x.session_generation DESC, x.started_at DESC, x.rowid DESC LIMIT 1`

// SessionPinIndex implements runner.PinnedSessionIndex over node_executions.
type SessionPinIndex struct{ db *sql.DB }

// NewSessionPinIndex creates a SessionPinIndex.
func NewSessionPinIndex(db *sql.DB) *SessionPinIndex { return &SessionPinIndex{db: db} }

// ListPinned lists waiting runs whose active session record at the current
// node is pinned. The source is reported only when it belongs to the pinned
// runner; a source left over from before a move is unknown ("").
func (x *SessionPinIndex) ListPinned(ctx context.Context) ([]runner.PinnedSession, error) {
	rows, err := x.db.QueryContext(ctx, `
SELECT ne.id, ne.instance_id, ne.node_id, ne.session_pin_runner,
       CASE WHEN substr(COALESCE(ne.source_id, ''), 1, length(ne.session_pin_runner) + 1) = ne.session_pin_runner || '/'
            THEN ne.source_id ELSE '' END
FROM node_executions ne
JOIN instances i ON i.id = ne.instance_id
WHERE i.state = 'waiting'
  AND i.current_node_id = ne.node_id
  AND ne.session_pin_runner IS NOT NULL AND ne.session_pin_runner != ''
  AND ne.id = (`+activeSessionExecSQL+`)
ORDER BY ne.instance_id`)
	if err != nil {
		return nil, fmt.Errorf("listing pinned sessions: %w", err)
	}
	defer rows.Close()
	var out []runner.PinnedSession
	for rows.Next() {
		var id, run, node, pin, src string
		if err := rows.Scan(&id, &run, &node, &pin, &src); err != nil {
			return nil, fmt.Errorf("scanning pinned session: %w", err)
		}
		execID, err := parseUUID(id)
		if err != nil {
			return nil, err
		}
		runID, err := parseUUID(run)
		if err != nil {
			return nil, err
		}
		out = append(out, runner.PinnedSession{RunID: runID, NodeID: node, ExecutionID: execID,
			RunnerID: runner.RunnerID(pin), SourceID: runner.SourceID(src)})
	}
	return out, rows.Err()
}

// RecoverSession moves the waiting session of (runID,nodeID) off `from` with
// no user message, under the session's own turn lease. It acts as the run's
// owner (system recovery), so no intervention is attributed.
func (h *conversationHandler) RecoverSession(ctx context.Context, runID uuid.UUID, nodeID, reason, from string) error {
	if !AllowedMoveReason(reason) {
		return fmt.Errorf("recover session %s: reason %q: %w", runID, reason, ErrSessionMoveNotAllowed)
	}
	if h.mover == nil {
		return errors.New("conversation: RecoverSession requires SetSessionMover")
	}
	inst, err := h.instances.Get(ctx, runID)
	if errors.Is(err, instance.ErrNotFound) {
		return nil // stale trigger
	}
	if err != nil {
		return fmt.Errorf("recover session %s: %w", runID, err)
	}
	if inst.State != instance.StateWaiting || inst.CurrentNodeID == nil || *inst.CurrentNodeID != nodeID {
		return nil // no longer waiting at that node: nothing is pinned
	}
	ctx = identity.WithIdentity(ctx, identity.Identity{UserID: inst.OwnerID, Role: identity.RoleUser})
	_, _, _, group, err := h.loadAndValidate(ctx, runID)
	if errors.Is(err, instance.ErrNotInteractiveWait) {
		return nil
	}
	if err != nil {
		return err
	}
	ctx, hold, err := h.acquireTurn(ctx, runID, group)
	if err != nil {
		return err // busy => wraps ErrInteractiveTurnInProgress: retried at the turn boundary
	}
	defer hold.Release()
	exec, err := h.sessionExecution(ctx, runID, nodeID, group)
	if err != nil {
		return err
	}
	if exec.SessionPinRunner == nil || *exec.SessionPinRunner != from {
		return nil // already moved (idempotent)
	}
	accepted, ok := exec.LLMConversation[ConversationKeyAcceptedTurnBinding]
	if !ok {
		return fmt.Errorf("recover session %s: %w: no committed accepted binding", runID, ErrSessionRestoreFailed)
	}
	var snap AcceptedTurnBinding
	if err := decodeConversationValue(accepted, &snap); err != nil {
		return fmt.Errorf("recover session %s: accepted binding: %w", runID, err)
	}
	binding, err := h.bindings.RestoreTurnBinding(ctx, snap) // P4 T3: never re-resolves the current profile
	if err != nil {
		return fmt.Errorf("recover session %s: %w", runID, err)
	}
	sid, _ := StepIdentityFromContext(ctx)
	res, err := h.mover.MoveSession(ctx, hold.Lease(), SessionMoveRequest{
		RunID: runID, InvocationID: sid.InvocationID, ExecutionID: exec.ID, NodeID: nodeID,
		TurnIndex: userTurnCount(parseConversation(exec.LLMConversation)), Reason: reason, From: from,
		Binding: binding, WorkDir: groupWorkDir(group, inst), ResourceID: groupResourceID(group),
		RestoreWorkspace: true,
	})
	if err != nil {
		return err // SessionWaitError surfaces to the controller (stays queued)
	}
	hold.Adopt(TurnLease{Key: hold.Lease().Key, Token: hold.Lease().Token, Generation: res.Generation, ExpiresAt: hold.Lease().ExpiresAt})
	return nil
}

// ResumePendingTurn replays a durable message/end/abandon intent through the
// normal turn path, reusing the intent (resumeOrPersist) and the recovered
// lease carried in ctx (ContextWithRecoveredTurnLease). adopted != nil commits
// that live output with its exact receipts instead of dispatching.
func (h *conversationHandler) ResumePendingTurn(ctx context.Context, in TurnIntent, adopted *AdoptedTurn) error {
	switch in.Kind {
	case TurnIntentMessage, TurnIntentEnd, TurnIntentAbandon:
	default:
		return fmt.Errorf("resume intent %s (kind %q): %w", in.ID, in.Kind, ErrNotSessionTurnIntent)
	}
	if adopted != nil && len(adopted.Completions) == 0 {
		return fmt.Errorf("resume intent %s: %w", in.ID, ErrAdoptedTurnIncomplete)
	}
	inst, err := h.instances.Get(ctx, in.RunID)
	if err != nil {
		return fmt.Errorf("resume intent %s: %w", in.ID, err)
	}
	ctx = identity.WithIdentity(ctx, identity.Identity{UserID: inst.OwnerID, Role: identity.RoleUser})
	ctx = contextWithResumedTurn(ctx, resumedTurn{intent: in, adopted: adopted})
	switch in.Kind {
	case TurnIntentMessage:
		return h.HandleMessage(ctx, in.RunID, in.Message)
	case TurnIntentEnd:
		_, err := h.EndSession(ctx, in.RunID)
		return err
	default:
		return h.AbandonSession(ctx, in.RunID)
	}
}

type TurnAdopter interface {
	AdoptTurn(ctx context.Context, job Job) (engine.AdoptedTurn, bool, error)

// SessionDrainer moves the sessions pinned to a draining runner at their turn
// boundaries and reports what is left.
type SessionDrainer interface {
	DrainSessions(ctx context.Context, id RunnerID) (pending int, blocked []DrainBlocker, err error)
}

// SessionRecoveryDeps are the controller's collaborators. Adopter is optional.
type SessionRecoveryDeps struct {
	Registry      Registry
	Ledger        Ledger
	Pins          PinnedSessionIndex
	Intents       engine.TurnIntentStore
	Recoverer     engine.SessionRecoverer
	Invocations   engine.InvocationStore

type sessionKey struct {

	node string

type queuedMove struct {
	session PinnedSession
	reason  string
}

// SessionRecoveryController turns pool events into idempotent session moves,
// even while runs wait for human input, and resumes durable intents at startup.
type SessionRecoveryController struct {
	d       SessionRecoveryDeps
	mu      sync.Mutex
	queue   map[sessionKey]queuedMove
	blocked map[sessionKey]string // group with no eligible target
	retry   map[uuid.UUID]engine.TurnIntent
	kick    chan struct{}
	flushMu sync.Mutex
}

// NewSessionRecoveryController validates deps.
func NewSessionRecoveryController(d SessionRecoveryDeps) *SessionRecoveryController {
	if d.Registry == nil || d.Ledger == nil || d.Pins == nil || d.Intents == nil || d.Recoverer == nil || d.Invocations == nil || d.Notifier == nil {
		panic("runner: NewSessionRecoveryController requires Registry, Ledger, Pins, Intents, Recoverer, Invocations and Notifier")
	}
	if d.Clock == nil {
		d.Clock = time.Now
	}
	if d.RetryInterval <= 0 {
		d.RetryInterval = 5 * time.Second
	}
	return &SessionRecoveryController{d: d, queue: map[sessionKey]queuedMove{}, blocked: map[sessionKey]string{},
		retry: map[uuid.UUID]engine.TurnIntent{}, kick: make(chan struct{}, 1)}
}

// HandlePoolEvent enqueues moves for the sessions an event affects. It never
// blocks (it is an EventRecorder listener).
func (c *SessionRecoveryController) HandlePoolEvent(ctx context.Context, ev PoolEvent) {
	var reason string
	var match func(PinnedSession) bool
	switch ev.Kind {
	case "runner.lost":
		reason, match = engine.MoveReasonRunnerLost, func(s PinnedSession) bool { return s.RunnerID == ev.RunnerID }
	case "runner.draining":
		reason, match = engine.MoveReasonDrain, func(s PinnedSession) bool { return s.RunnerID == ev.RunnerID }
	case "source.logged_out":
		reason, match = engine.MoveReasonLoggedOut, func(s PinnedSession) bool { return s.SourceID == ev.SourceID }
	case "source.health_changed":
		src, err := c.d.Registry.GetSource(ctx, ev.SourceID)
		if err != nil {
			slog.Warn("session recovery: read source", "source", ev.SourceID, "error", err)
			return
		}
		r, ok := sourceMoveReason(src.Health)
		if !ok {
			return
		}
		reason, match = r, func(s PinnedSession) bool { return s.SourceID == ev.SourceID }
	default:
		return // cordon, suspect, load: never move pins
	}
	pins, err := c.d.Pins.ListPinned(ctx)
	if err != nil {
		slog.Warn("session recovery: list pinned sessions", "error", err)
		return
	}
	for _, s := range pins {
		if match(s) {
			c.enqueue(s, reason)
		}
	}
	c.wake()
}

func sourceMoveReason(h SourceHealth) (string, bool) {
	if h.Auth == AuthState("logged_out") {
		return engine.MoveReasonLoggedOut, true
	}
	if h.Capacity == CapacityState("exhausted") {
		return engine.MoveReasonExhausted, true
	}
	return "", false
}

func (c *SessionRecoveryController) enqueue(s PinnedSession, reason string) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.queue[sessionKey{s.RunID, s.NodeID}] = queuedMove{session: s, reason: reason}
}

func (c *SessionRecoveryController) wake() {
	select {
	case c.kick <- struct{}{}:
	default:
	}
}

// Flush attempts every queued move and retryable intent once. Busy sessions
// (turn in flight) and sessions without a target stay queued.
func (c *SessionRecoveryController) Flush(ctx context.Context) {
	c.flushMu.Lock()
	defer c.flushMu.Unlock()
	c.mu.Lock()
	moves := make([]queuedMove, 0, len(c.queue))
	for _, m := range c.queue {
		moves = append(moves, m)
	}
	c.mu.Unlock()
	for _, m := range moves {
		c.attemptMove(ctx, m)
	}
}

func (c *SessionRecoveryController) attemptMove(ctx context.Context, m queuedMove) {
	key := sessionKey{m.session.RunID, m.session.NodeID}
	inv, err := c.d.Invocations.Begin(ctx, m.session.RunID, m.session.NodeID)
	if err != nil {
		slog.Warn("session recovery: invocation", "run", m.session.RunID, "error", err)
		return
	}
	mctx := engine.ContextWithStepIdentity(ctx, engine.StepIdentity{RunID: m.session.RunID, NodeID: m.session.NodeID,
		InvocationID: inv.ID, OperationID: "session-move", Kind: engine.OperationTurn, Attempt: 1})
	err = c.d.Recoverer.RecoverSession(mctx, m.session.RunID, m.session.NodeID, m.reason, string(m.session.RunnerID))
	c.mu.Lock()
	defer c.mu.Unlock()
	var wait *engine.SessionWaitError
	switch {
	case err == nil:
		delete(c.queue, key)
		delete(c.blocked, key)
	case errors.Is(err, instance.ErrInteractiveTurnInProgress):
		// Turn in flight: retried at the boundary (next wake or tick).
	case errors.As(err, &wait):
		c.blocked[key] = wait.Group
	default:
		slog.Warn("session recovery: move failed", "run", m.session.RunID, "reason", m.reason, "error", err)
	}
}

// DrainSessions implements SessionDrainer.
func (c *SessionRecoveryController) DrainSessions(ctx context.Context, id RunnerID) (int, []DrainBlocker, error) {
	pins, err := c.d.Pins.ListPinned(ctx)
	if err != nil {
		return 0, nil, fmt.Errorf("session drain: list pinned: %w", err)
	}
	for _, s := range pins {
		if s.RunnerID == id {
			c.enqueue(s, engine.MoveReasonDrain)
		}
	}
	c.Flush(ctx)
	pins, err = c.d.Pins.ListPinned(ctx)
	if err != nil {
		return 0, nil, fmt.Errorf("session drain: list pinned: %w", err)
	}
	pending := 0
	var blocked []DrainBlocker
	c.mu.Lock()
	defer c.mu.Unlock()
	for _, s := range pins {
		if s.RunnerID != id {
			continue
		}
		pending++
		if g, ok := c.blocked[sessionKey{s.RunID, s.NodeID}]; ok {
			run := s.RunID
			blocked = append(blocked, DrainBlocker{Kind: "session", RunID: &run, Group: g, Reason: "no eligible runner serves the session's model group"})
		}
	}
	return pending, blocked, nil
}

// Start runs the worker: it flushes on enqueue, on every pool change and
// every RetryInterval, until ctx ends.
func (c *SessionRecoveryController) Start(ctx context.Context) {
	wake, cancel := c.d.Notifier.Subscribe()
	go func() {
		defer cancel()
		t := time.NewTicker(c.d.RetryInterval)
		defer t.Stop()
		for {
			select {
			case <-ctx.Done():
				return
			case <-c.kick:
			case <-wake:
			case <-t.C:
			}
			c.Flush(ctx)
		}
	}()
}

// domain/runner/turn_recovery.go
package runner

import (
	"context"
	"errors"
	"fmt"
	"strings"
	"sync"
	"time"

	"github.com/arbitron/arbitron/domain/engine"
	"github.com/arbitron/arbitron/domain/instance"
	"github.com/google/uuid"
)

var (
	// ErrTurnRecoveryNotReady: the previous owner may still be running and is not
	// (yet) adoptable; retry after reconnect/loss evidence. Never a TTL decision.
	ErrTurnRecoveryNotReady = errors.New("runner: turn owner not adoptable and not proven stopped; retry later")
	// ErrTurnRecoveryUnproven: no durable job/resource fact exists to adopt or
	// fence (crash before dispatch while the lease is still held).
	ErrTurnRecoveryUnproven = errors.New("runner: pending turn has no dispatched job to adopt or fence")
)

// TurnStopEvidence turns real liveness/stop facts into a replay proof (Cycle 5).
type TurnStopEvidence interface {
	CanReplayTurn(ctx context.Context, in engine.TurnIntent, jobs []Job) (engine.TurnRecoveryProof, error)
}

// TurnRecoveryDeps are the coordinator's collaborators; all are required.
type TurnRecoveryDeps struct {
	Ledger         Ledger
	Leases         engine.TurnLeaseStore
	RecoveryLeases engine.TurnLeaseRecoveryStore
	Adopter        TurnAdopter
	StopEvidence   TurnStopEvidence
	TTL            time.Duration
	Clock          func() time.Time
}

// TurnRecoveryCoordinator is the single owner of pending-turn recovery: it
// adopts the exact live job of an intent or proves its owners stopped, and
// only then takes the session lease through RecoverTurnLease.
type TurnRecoveryCoordinator struct {
	d        TurnRecoveryDeps
	mu       sync.Mutex
	inflight map[string]bool // lease keys being recovered in this process
}

// NewTurnRecoveryCoordinator validates deps (P4 T13 fails startup on a nil).
func NewTurnRecoveryCoordinator(d TurnRecoveryDeps) *TurnRecoveryCoordinator {
	if d.Ledger == nil || d.Leases == nil || d.RecoveryLeases == nil || d.Adopter == nil || d.StopEvidence == nil || d.TTL <= 0 {
		panic("runner: NewTurnRecoveryCoordinator requires Ledger, Leases, RecoveryLeases, Adopter, StopEvidence and a positive TTL")
	}
	if d.Clock == nil {
		d.Clock = time.Now
	}
	return &TurnRecoveryCoordinator{d: d, inflight: map[string]bool{}}
}

// operationMatches: equality or a sub-operation ("turn/1/stream"); "turn/1"
// never matches "turn/10".
func operationMatches(jobOp, intentOp string) bool {
	return jobOp == intentOp || strings.HasPrefix(jobOp, intentOp+"/")
}

func (c *TurnRecoveryCoordinator) turnJobs(ctx context.Context, in engine.TurnIntent) ([]Job, error) {
	all, err := c.d.Ledger.ListByRun(ctx, in.RunID)
	if err != nil {
		return nil, fmt.Errorf("turn recovery %s: jobs: %w", in.ID, err)
	}
	var out []Job
	for _, j := range all {
		if j.Key.InvocationID == in.InvocationID && operationMatches(j.Key.OperationID, in.OperationID) {
			out = append(out, j)
		}
	}
	return out, nil
}

func jobFenceRef(j Job) engine.JobFenceRef {
	return engine.JobFenceRef{JobID: j.ID, RunnerID: string(j.RunnerID), Generation: j.Generation}
}

// PrepareTurn returns the lease the resumed operation must run under and, when
// the previous owner's job is still live, its adopted output (no re-dispatch).
func (c *TurnRecoveryCoordinator) PrepareTurn(ctx context.Context, in engine.TurnIntent) (engine.TurnLease, *engine.AdoptedTurn, error) {
	if in.ID == uuid.Nil || in.LeaseKey == "" || in.InvocationID == uuid.Nil || in.OperationID == "" {
		return engine.TurnLease{}, nil, fmt.Errorf("turn recovery: %w: incomplete intent %s", engine.ErrTurnRecoveryProofInvalid, in.ID)
	}
	c.mu.Lock()
	if c.inflight[in.LeaseKey] {
		c.mu.Unlock()
		return engine.TurnLease{}, nil, fmt.Errorf("turn recovery %s: %w", in.LeaseKey, instance.ErrInteractiveTurnInProgress)
	}
	c.inflight[in.LeaseKey] = true
	c.mu.Unlock()
	defer func() { c.mu.Lock(); delete(c.inflight, in.LeaseKey); c.mu.Unlock() }()

	jobs, err := c.turnJobs(ctx, in)
	if err != nil {
		return engine.TurnLease{}, nil, err
	}
	var live []Job
	for _, j := range jobs {
		if j.State.HoldsSlot() {
			live = append(live, j)
		}
	}
	cur, held, err := c.d.Leases.Current(ctx, in.LeaseKey)
	if err != nil {
		return engine.TurnLease{}, nil, fmt.Errorf("turn recovery %s: lease: %w", in.ID, err)
	}
	if !held {
		// Released by an earlier recoverable failure: an ordinary acquire.
		l, err := c.d.Leases.Acquire(ctx, in.LeaseKey, c.d.TTL)
		if err != nil {
			return engine.TurnLease{}, nil, err
		}
		if len(live) == 0 {
			return l, nil, nil
		}
		// A live job without a held lease: fence it before anything resends.
		if _, err := c.d.StopEvidence.CanReplayTurn(ctx, in, jobs); err != nil {
			_ = c.d.Leases.Release(context.WithoutCancel(ctx), l)
			return engine.TurnLease{}, nil, err
		}
		l, err = c.d.Leases.BumpGeneration(ctx, l)
		return l, nil, err
	}
	if c.d.Clock().Before(cur.ExpiresAt) {
		// A holder renewed recently (another resumer, or a live process): busy.
		return engine.TurnLease{}, nil, fmt.Errorf("turn recovery %s: %w", in.LeaseKey, instance.ErrInteractiveTurnInProgress)
	}
	if len(live) == 1 {
		adopted, ok, err := c.d.Adopter.AdoptTurn(ctx, live[0])
		if err != nil {
			return engine.TurnLease{}, nil, fmt.Errorf("turn recovery %s: adopt: %w", in.ID, err)
		}
		if ok {
			if len(adopted.Completions) == 0 || adopted.Completions[0].JobFence != jobFenceRef(live[0]) {
				return engine.TurnLease{}, nil, fmt.Errorf("turn recovery %s: %w", in.ID, engine.ErrAdoptedTurnIncomplete)
			}
			proof := engine.TurnRecoveryProof{IntentID: in.ID, Adopted: true, Jobs: []engine.JobFenceRef{jobFenceRef(live[0])}}
			l, err := c.d.RecoveryLeases.RecoverTurnLease(ctx, cur, proof, c.d.TTL)
			if err != nil {
				return engine.TurnLease{}, nil, fmt.Errorf("turn recovery %s: %w", in.ID, err)
			}
			adopted.Lease = l
			adopted.Identity = engine.InteractiveTurnIdentity{RunID: in.RunID, NodeID: in.NodeID, InvocationID: in.InvocationID,
				RequestID: in.RequestID, TurnIndex: in.TurnIndex, Generation: l.Generation}
			return l, &adopted, nil
		}
	}
	proof, err := c.d.StopEvidence.CanReplayTurn(ctx, in, jobs)
	if err != nil {
		return engine.TurnLease{}, nil, err
	}
	if len(proof.Jobs) == 0 && proof.Resource == nil {
		return engine.TurnLease{}, nil, fmt.Errorf("turn recovery %s: %w", in.ID, ErrTurnRecoveryUnproven)
	}
	proof.IntentID, proof.Adopted = in.ID, false
	l, err := c.d.RecoveryLeases.RecoverTurnLease(ctx, cur, proof, c.d.TTL)
	if err != nil {
		return engine.TurnLease{}, nil, fmt.Errorf("turn recovery %s: %w", in.ID, err)
	}
	return l, nil, nil
}

package runner

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/arbitron/arbitron/domain/engine"
)

// RunnerPresence reads the hub's live connection/revocation facts. *Hub implements it.
type RunnerPresence interface {
	Connected(id RunnerID) bool
	Revoked(id RunnerID) bool
}

// TurnStopEvidenceDeps: Stopper is P3 T10's supervised built-in stopper;
// Started is server start; Grace is cfg.Runners.RestartGraceSeconds.
type TurnStopEvidenceDeps struct {
	Registry Registry
	Presence RunnerPresence
	Ledger   Ledger
	Stopper  BuiltInStopper
	Grace    time.Duration
	Started  time.Time
	Clock    func() time.Time
}

type turnStopEvidence struct{ d TurnStopEvidenceDeps }

// NewTurnStopEvidence composes real liveness, revocation and stop facts.
func NewTurnStopEvidence(d TurnStopEvidenceDeps) (TurnStopEvidence, error) {
	if d.Registry == nil || d.Presence == nil || d.Ledger == nil || d.Stopper == nil || d.Started.IsZero() || d.Grace <= 0 {
		return nil, errors.New("runner: NewTurnStopEvidence requires Registry, Presence, Ledger, Stopper, Started and a positive Grace")
	}
	if d.Clock == nil {
		d.Clock = time.Now
	}
	return &turnStopEvidence{d: d}, nil
}

// CanReplayTurn fences every live owner whose stop is confirmed and returns
// the fences as a replay proof. Any owner that may still run => NotReady.
func (e *turnStopEvidence) CanReplayTurn(ctx context.Context, in engine.TurnIntent, jobs []Job) (engine.TurnRecoveryProof, error) {
	proof := engine.TurnRecoveryProof{IntentID: in.ID}
	for _, j := range jobs {
		if j.State.HoldsSlot() {
			if err := e.stopped(ctx, j); err != nil {
				return engine.TurnRecoveryProof{}, err
			}
			if err := e.d.Ledger.Fail(ctx, j.Fence(), OutcomeRunnerLost); err != nil && !errors.Is(err, ErrFenced) {
				return engine.TurnRecoveryProof{}, fmt.Errorf("turn stop evidence: fence %s: %w", j.ID, err)
			}
		}
		proof.Jobs = append(proof.Jobs, engine.JobFenceRef{JobID: j.ID, RunnerID: string(j.RunnerID), Generation: j.Generation})
	}
	return proof, nil
}

func (e *turnStopEvidence) stopped(ctx context.Context, j Job) error {
	if j.RunnerID == BuiltInRunnerID {
		if err := e.d.Stopper.StopAndWait(ctx, j.Fence()); err != nil {
			return fmt.Errorf("%w: built-in job %s not stopped: %v", ErrTurnRecoveryNotReady, j.ID, err)
		}
		return nil
	}
	if e.d.Presence.Revoked(j.RunnerID) {
		return nil // key revoked: the runner can no longer report or write
	}
	r, _, err := e.d.Registry.GetRunner(ctx, j.RunnerID)
	if errors.Is(err, ErrRunnerNotFound) {
		return nil // removed runner: confirmed loss
	}
	if err != nil {
		return fmt.Errorf("turn stop evidence: runner %s: %w", j.RunnerID, err)
	}
	if e.d.Presence.Connected(j.RunnerID) || r.Liveness != LivenessLost {
		return fmt.Errorf("%w: runner %s is %s", ErrTurnRecoveryNotReady, j.RunnerID, r.Liveness)
	}
	if e.d.Clock().Before(e.d.Started.Add(e.d.Grace)) {
		return fmt.Errorf("%w: runner %s may reconnect within the restart grace", ErrTurnRecoveryNotReady, j.RunnerID)
	}
	return nil
}

// PendingFirstTurnResumer resumes a durable FIRST-turn intent: Engine for node
// visits, GroupResolutionBridge (P4 T11) for resolution intents.
type PendingFirstTurnResumer interface {
	ResumeFirstTurn(ctx context.Context, in TurnIntent) error
}

var _ PendingFirstTurnResumer = (*Engine)(nil)

// AdoptedTurns hands an adopted live first turn from session recovery to the
// re-entered InteractiveExecutor (which consumes it instead of dispatching).
type AdoptedTurns struct {
	mu  sync.Mutex
	set map[uuid.UUID]AdoptedTurn
}

func NewAdoptedTurns() *AdoptedTurns { return &AdoptedTurns{set: map[uuid.UUID]AdoptedTurn{}} }

func (a *AdoptedTurns) Put(intentID uuid.UUID, t AdoptedTurn) {
	a.mu.Lock()
	defer a.mu.Unlock()
	a.set[intentID] = t
}

func (a *AdoptedTurns) Take(intentID uuid.UUID) (AdoptedTurn, bool) {
	a.mu.Lock()
	defer a.mu.Unlock()
	t, ok := a.set[intentID]
	delete(a.set, intentID)
	return t, ok
}

type adoptedTurnKey struct{}

// ContextWithAdoptedTurn carries an adopted first turn into ResumeFirstTurn.
func ContextWithAdoptedTurn(ctx context.Context, t AdoptedTurn) context.Context {
	return context.WithValue(ctx, adoptedTurnKey{}, t)
}

// SetAdoptedTurns shares the hand-off registry with the InteractiveExecutor
// (P4 T13 passes the same instance to both).
func (e *Engine) SetAdoptedTurns(a *AdoptedTurns) { e.adoptedTurns = a }

// ResumeFirstTurn re-enters the run at the intent's node visit. The exact open
// invocation must still be current; the re-executed InteractiveExecutor reuses
// the pending first intent's prompt and nonce (P4 T3 Cycle 3). The run is
// claimed so generic restart recovery (P3 T10) does not fail it.
func (e *Engine) ResumeFirstTurn(ctx context.Context, in TurnIntent) error {
	if in.Kind != TurnIntentFirst {
		return fmt.Errorf("resume first turn %s: kind %q: %w", in.ID, in.Kind, instance.ErrNotInteractiveWait)
	}
	inst, err := e.instances.Get(ctx, in.RunID)
	if err != nil {
		return fmt.Errorf("resume first turn %s: %w", in.ID, err)
	}
	if inst.State != instance.StateRunning || inst.CurrentNodeID == nil || *inst.CurrentNodeID != in.NodeID {
		return fmt.Errorf("resume first turn %s: run %s not at %s: %w", in.ID, in.RunID, in.NodeID, instance.ErrNotInteractiveWait)
	}
	inv, err := e.invocationStore().Get(ctx, in.InvocationID)
	if err != nil || inv.CommittedAt != nil || inv.RunID != in.RunID || inv.NodeID != in.NodeID {
		return fmt.Errorf("resume first turn %s: invocation %s is not the open visit: %w", in.ID, in.InvocationID, instance.ErrNotInteractiveWait)
	}
	if t, ok := ctx.Value(adoptedTurnKey{}).(AdoptedTurn); ok {
		if e.adoptedTurns == nil || len(t.Completions) == 0 {
			return fmt.Errorf("resume first turn %s: %w", in.ID, ErrAdoptedTurnIncomplete)
		}
		e.adoptedTurns.Put(in.ID, t)
	}
	e.sessionClaims.Store(in.RunID, struct{}{})
	return e.StartInstance(context.WithoutCancel(ctx), in.RunID)
}

func (e *Engine) claimedBySessionRecovery(runID uuid.UUID) bool {
	_, ok := e.sessionClaims.Load(runID)
	return ok
}

// TurnPreparer is the coordinator port (*TurnRecoveryCoordinator).
type TurnPreparer interface {
	PrepareTurn(ctx context.Context, in engine.TurnIntent) (engine.TurnLease, *engine.AdoptedTurn, error)
}

// StartupScan resumes every pending durable intent through the coordinator
// (any kind), then reconciles pins against the registry. P4 T13 runs it after
// reconnect evidence is available and BEFORE generic RecoverRunningInstances.
func (c *SessionRecoveryController) StartupScan(ctx context.Context) error {
	pending, err := c.d.Intents.Pending(ctx)
	if err != nil {
		return fmt.Errorf("session recovery: pending intents: %w", err)
	}
	for _, in := range pending {
		c.resumeIntent(ctx, in)
	}
	return c.Reconcile(ctx)
}

func (c *SessionRecoveryController) resumeIntent(ctx context.Context, in engine.TurnIntent) {
	lease, adopted, err := c.d.Coordinator.PrepareTurn(ctx, in)
	if err == nil {
		switch in.Kind {
		case engine.TurnIntentFirst:
			// The re-entered executor/bridge acquires the session lease itself.
			_ = c.d.Leases.Release(context.WithoutCancel(ctx), lease)
			fctx := ctx
			if adopted != nil {
				fctx = engine.ContextWithAdoptedTurn(ctx, *adopted)
			}
			err = c.d.FirstTurns.ResumeFirstTurn(fctx, in)
		case engine.TurnIntentMove:
			err = c.d.Moves.ResumeMove(engine.ContextWithRecoveredTurnLease(ctx, lease), in)
			_ = c.d.Leases.Release(context.WithoutCancel(ctx), lease) // idempotent; a no-op once handed over
		default:
			err = c.d.Recoverer.ResumePendingTurn(engine.ContextWithRecoveredTurnLease(ctx, lease), in, adopted)
			_ = c.d.Leases.Release(context.WithoutCancel(ctx), lease)
		}
	}
	c.mu.Lock()
	defer c.mu.Unlock()
	switch {
	case err == nil:
		delete(c.retry, in.ID)
	case errors.Is(err, instance.ErrNotInteractiveWait) || errors.Is(err, instance.ErrNotFound):
		// Explicit terminal outcome: the run left the session; record and close.
		delete(c.retry, in.ID)
		slog.Info("session recovery: intent obsolete", "intent", in.ID, "run", in.RunID, "error", err)
		if cerr := c.d.Intents.Complete(context.WithoutCancel(ctx), in.ID); cerr != nil {
			c.retry[in.ID] = in
		}
	default:
		// Busy, not-ready, unproven, transport or DB: the intent stays durable.
		c.retry[in.ID] = in
		slog.Warn("session recovery: intent retained", "intent", in.ID, "kind", in.Kind, "error", err)
	}
}

// Reconcile enqueues moves for pins whose runner is gone, lost or draining,
// or whose source is exhausted or logged out, while the server was down.
func (c *SessionRecoveryController) Reconcile(ctx context.Context) error {
	snap, err := c.d.Registry.Snapshot(ctx)
	if err != nil {
		return fmt.Errorf("session recovery: snapshot: %w", err)
	}
	runners := map[RunnerID]Runner{}
	for _, r := range snap.Runners {
		runners[r.ID] = r
	}
	sources := map[SourceID]Source{}
	for _, s := range snap.Sources {
		sources[s.ID] = s
	}
	pins, err := c.d.Pins.ListPinned(ctx)
	if err != nil {
		return fmt.Errorf("session recovery: list pinned: %w", err)
	}
	for _, s := range pins {
		r, ok := runners[s.RunnerID]
		switch {
		case !ok || r.Liveness == LivenessLost:
			c.enqueue(s, engine.MoveReasonRunnerLost)
		case r.Draining:
			c.enqueue(s, engine.MoveReasonDrain)
		default:
			if src, ok := sources[s.SourceID]; ok {
				if reason, move := sourceMoveReason(src.Health); move {
					c.enqueue(s, reason)
				}
			}
		}
	}
	c.wake()
	return nil
}
