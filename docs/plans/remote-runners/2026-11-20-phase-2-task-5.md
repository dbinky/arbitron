// adapter/http/runner_translate.go
package http

import (
	"fmt"
	"time"

	"github.com/google/uuid"

	"github.com/arbitron/arbitron/adapter/runnerproto"
	"github.com/arbitron/arbitron/domain/modelidentity"
	"github.com/arbitron/arbitron/domain/runner"
)

func malformedf(format string, args ...any) error {
	return fmt.Errorf("%w: "+format, append([]any{runner.ErrMalformedMessage}, args...)...)
}

func parseJob(e runnerproto.Envelope) (uuid.UUID, error) {
	id, err := uuid.Parse(e.JobID)
	if err != nil || id == uuid.Nil || e.Generation < 1 {
		return uuid.Nil, malformedf("%s needs a job id and generation", e.Kind)
	}
	return id, nil
}

// toInbound maps one post-admission frame to the hub's domain message.
func toInbound(e runnerproto.Envelope) (runner.Inbound, error) {
	switch e.Kind {
	case runnerproto.KindHeartbeat:
		var hb runnerproto.Heartbeat
		if err := runnerproto.DecodePayload(e, &hb); err != nil {
			return runner.Inbound{}, malformedf("heartbeat: %v", err)
		}
		info, err := heartbeatInfo(hb)
		if err != nil {
			return runner.Inbound{}, err
		}
		return runner.Inbound{Kind: runner.MsgHeartbeat, Heartbeat: &info}, nil
	case runnerproto.KindAccept:
		id, err := parseJob(e)
		if err != nil {
			return runner.Inbound{}, err
		}
		return runner.Inbound{Kind: runner.MsgAccept, JobID: id, Generation: e.Generation, Accept: true}, nil
	case runnerproto.KindReject:
		id, err := parseJob(e)
		if err != nil {
			return runner.Inbound{}, err
		}
		var rj runnerproto.Reject
		if len(e.Payload) > 0 {
			if err := runnerproto.DecodePayload(e, &rj); err != nil {
				return runner.Inbound{}, malformedf("reject: %v", err)
			}
		}
		return runner.Inbound{Kind: runner.MsgReject, JobID: id, Generation: e.Generation, RejectReason: rj.Reason}, nil
	case runnerproto.KindOutput:
		id, err := parseJob(e)
		if err != nil {
			return runner.Inbound{}, err
		}
		var o runnerproto.Output
		if err := runnerproto.DecodePayload(e, &o); err != nil {
			return runner.Inbound{}, malformedf("output: %v", err)
		}
		if o.Seq != e.Seq {
			return runner.Inbound{}, malformedf("output envelope seq %d != payload seq %d", e.Seq, o.Seq)
		}
		return runner.Inbound{Kind: runner.MsgOutput, JobID: id, Generation: e.Generation, Seq: o.Seq,
			Output: &runner.OutputFrame{Seq: o.Seq, Token: o.Token}}, nil
	case runnerproto.KindResult:
		id, err := parseJob(e)
		if err != nil {
			return runner.Inbound{}, err
		}
		var res runnerproto.Result
		if err := runnerproto.DecodePayload(e, &res); err != nil {
			return runner.Inbound{}, malformedf("result: %v", err)
		}
		rr := resultToDomain(res)
		return runner.Inbound{Kind: runner.MsgResult, JobID: id, Generation: e.Generation, Result: &rr}, nil
	default:
		return runner.Inbound{}, malformedf("unexpected %q after admission", e.Kind)
	}
}

func heartbeatInfo(hb runnerproto.Heartbeat) (runner.HeartbeatInfo, error) {
	jobs, err := heldJobs(hb.Jobs)
	if err != nil {
		return runner.HeartbeatInfo{}, err
	}
	info := runner.HeartbeatInfo{Telemetry: telemetryToDomain(hb.Telemetry), Jobs: jobs}
	if hb.Sources != nil {
		info.Sources = sourcesToDomain(hb.Sources)
	}
	return info, nil
}

// helloInfo maps the admission hello.
func helloInfo(h runnerproto.Hello) (runner.HelloInfo, error) {
	held, err := heldJobs(h.Held)
	if err != nil {

// fromOutbound maps a hub message to a frame. id/heartbeatSeconds fill admitted.
func fromOutbound(m runner.Outbound, id runner.RunnerID, heartbeatSeconds int) (runnerproto.Envelope, error) {
	jobID := m.JobID.String()
	switch m.Kind {
	case runner.MsgAdmitted:
		return runnerproto.NewEnvelope(runnerproto.KindAdmitted, &runnerproto.Admitted{RunnerID: string(id),
			HeartbeatSeconds: heartbeatSeconds, ProtocolVersion: runnerproto.ProtocolVersion})
	case runner.MsgDispatch:
		if m.Dispatch == nil {
			return runnerproto.Envelope{}, fmt.Errorf("runner connect: dispatch without payload")
		}
		return runnerproto.NewJobEnvelope(runnerproto.KindDispatch, jobID, m.Generation, 0, dispatchToWire(*m.Dispatch))
	case runner.MsgAck:
		if m.Ack == nil {
			return runnerproto.Envelope{}, fmt.Errorf("runner connect: ack without payload")
		}
		return runnerproto.NewJobEnvelope(runnerproto.KindAck, jobID, m.Generation, 0, &runnerproto.Ack{Seq: m.Ack.Seq, Result: m.Ack.Result})
	case runner.MsgFenced:
		return runnerproto.NewJobEnvelope(runnerproto.KindFenced, jobID, m.Generation, 0, &runnerproto.Fenced{Reason: "stale generation"})
	case runner.MsgCancel:
		return runnerproto.NewJobEnvelope(runnerproto.KindCancel, jobID, m.Generation, 0, &runnerproto.Cancel{Reason: "cancelled by server"})
	default:
		return runnerproto.Envelope{}, fmt.Errorf("runner connect: cannot send %q", m.Kind)
	}
}

// dispatchToWire copies exactly the hermetic chat fields; the domain payload

			Backend: c.Backend, ContextWindow: c.ContextWindow, Effort: c.Effort, Tools: c.Tools}

package http

import (
	"context"

// RunnerKeyAuthenticator is the runner-channel key port (*runner.KeyService).
type RunnerKeyAuthenticator interface {
	Authenticate(ctx context.Context, plaintext string) (runner.RunnerKey, error)
	ProveMachine(ctx context.Context, k runner.RunnerKey, nonce []byte, pub ed25519.PublicKey, sig []byte) (runner.RunnerKey, error)
}

type runnerKeyCtxKey struct{}

// WithRunnerKey stores the authenticated runner key on the request context.
func WithRunnerKey(ctx context.Context, k runner.RunnerKey) context.Context {
	return context.WithValue(ctx, runnerKeyCtxKey{}, k)
}

// RunnerKeyFromContext returns the key set by RunnerKeyMiddleware.
func RunnerKeyFromContext(ctx context.Context) (runner.RunnerKey, bool) {
	k, ok := ctx.Value(runnerKeyCtxKey{}).(runner.RunnerKey)
	return k, ok
}

// runnerKeyQueryParams are parameter names refused outright: a key in a URL
// leaks into proxy and access logs.
var runnerKeyQueryParams = []string{"key", "runner_key", "runnerkey", "token", "access_token", "api_key", "authorization"}

// RunnerKeyMiddleware authenticates runner-channel requests from the
// Authorization header ONLY. It never reads or alters other headers, so
// Cloudflare Access headers reach the edge-validated request unchanged. It
// logs refusal codes, never the key or its hash.
type RunnerKeyMiddleware struct {
	keys RunnerKeyAuthenticator
	log  *slog.Logger
}

// NewRunnerKeyMiddleware builds the middleware; a nil log uses slog.Default().
func NewRunnerKeyMiddleware(keys RunnerKeyAuthenticator, log *slog.Logger) *RunnerKeyMiddleware {
	if keys == nil {
		panic("http: NewRunnerKeyMiddleware requires a key authenticator")
	}
	if log == nil {
		log = slog.Default()
	}
	return &RunnerKeyMiddleware{keys: keys, log: log}
}

func keyInQuery(q url.Values) bool {
	for name, vals := range q {
		for _, p := range runnerKeyQueryParams {
			if strings.EqualFold(name, p) {
				return true
			}
		}
		for _, v := range vals {
			if strings.HasPrefix(v, runner.RunnerKeyPrefix) {
				return true
			}
		}
	}
	return false
}

func (m *RunnerKeyMiddleware) refuse(w http.ResponseWriter, r *http.Request, code, msg string) {
	m.log.Warn("runner request refused", "code", code, "path", r.URL.Path, "remote", r.RemoteAddr)
	w.Header().Set("WWW-Authenticate", RunnerAuthScheme)
	WriteError(w, http.StatusUnauthorized, code, msg)
}

// Handler authenticates and stores the key on the context.
func (m *RunnerKeyMiddleware) Handler(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if keyInQuery(r.URL.Query()) {
			m.refuse(w, r, "RUNNER_KEY_IN_QUERY", "runner keys are accepted only in the Authorization header")
			return
		}
		plaintext, ok := runnerKeyFromHeader(r.Header)
		if !ok {
			m.refuse(w, r, "RUNNER_KEY_REQUIRED", "Authorization: Runner <key> is required")
			return
		}
		k, err := m.keys.Authenticate(r.Context(), plaintext)
		switch {
		case err == nil:
		case errors.Is(err, runner.ErrKeyRevoked):
			m.refuse(w, r, "RUNNER_KEY_REVOKED", "runner key revoked")
			return
		case errors.Is(err, runner.ErrKeyNotFound):
			m.refuse(w, r, "RUNNER_KEY_INVALID", "runner key not recognised")
			return
		default:
			m.log.Error("runner key check failed", "error", err)
			WriteError(w, http.StatusInternalServerError, "INTERNAL_ERROR", "runner key check failed")
			return
		}
		next.ServeHTTP(w, r.WithContext(WithRunnerKey(r.Context(), k)))
	})
}

// wsRunnerConn implements runner.RunnerConn over one WebSocket. Writes are
// serialised and bounded by WriteTimeout (never by the caller's step context,
// which would tear the socket mid-frame).
type wsRunnerConn struct {
	c                *websocket.Conn
	runnerID         runner.RunnerID
	heartbeatSeconds int
	writeTimeout     time.Duration

	mu     sync.Mutex
	closed bool
	done   chan struct{}
}

func newWSRunnerConn(c *websocket.Conn, id runner.RunnerID, hb int, wt time.Duration) *wsRunnerConn {
	return &wsRunnerConn{c: c, runnerID: id, heartbeatSeconds: hb, writeTimeout: wt, done: make(chan struct{})}
}

func (w *wsRunnerConn) write(ctx context.Context, e runnerproto.Envelope) error {
	b, err := runnerproto.Encode(e)
	if err != nil {
		return err
	}
	w.mu.Lock()
	defer w.mu.Unlock()
	if w.closed {
		return errRunnerConnClosed
	}
	wctx, cancel := context.WithTimeout(context.WithoutCancel(ctx), w.writeTimeout)
	defer cancel()
	return w.c.Write(wctx, websocket.MessageText, b)
}

// Send implements runner.RunnerConn.
func (w *wsRunnerConn) Send(ctx context.Context, m runner.Outbound) error {
	e, err := fromOutbound(m, w.runnerID, w.heartbeatSeconds)
	if err != nil {
		return err
	}
	return w.write(ctx, e)
}

func clampReason(s string) string {
	const max = 120 // WebSocket close reasons are limited to 123 bytes
	if len(s) > max {
		return s[:max]
	}
	return s
}

func (w *wsRunnerConn) markClosed() bool {
	w.mu.Lock()
	defer w.mu.Unlock()
	if w.closed {
		return false
	}
	w.closed = true
	return true
}

// Close implements runner.RunnerConn for hub-initiated closes (lost,
// superseded, revoked). It returns immediately; the close handshake runs in
// the background so the hub never blocks on a slow peer.
func (w *wsRunnerConn) Close(code int, reason string) error {
	if !w.markClosed() {
		return nil
	}
	go func() {
		_ = w.c.Close(websocket.StatusCode(code), clampReason(reason))
		close(w.done)
	}()
	return nil
}

// closeSync closes on the handler goroutine (handshake/read-loop failures).
func (w *wsRunnerConn) closeSync(code int, reason string) {
	if !w.markClosed() {
		return
	}
	_ = w.c.Close(websocket.StatusCode(code), clampReason(reason))
	close(w.done)
}

// waitClosed waits (bounded) for a background close handshake to finish.
func (w *wsRunnerConn) waitClosed(d time.Duration) {
	w.mu.Lock()
	closed := w.closed
	w.mu.Unlock()
	if !closed {
		return
	}
	select {
	case <-w.done:
	case <-time.After(d):
	}
}

}

// handshake: challenge -> challenge_response (machine proof) -> hello -> Admit.
func (h *RunnerConnectHandler) handshake(ctx context.Context, conn *wsRunnerConn, key runner.RunnerKey) (*runner.Session, bool) {
	nonce, err := runner.NewChallengeNonce()
	if err != nil {
		h.log.Error("runner challenge unavailable", "runner", string(key.RunnerID), "error", err)
		conn.closeSync(int(websocket.StatusInternalError), "challenge unavailable")
		return nil, false
	}
	ch, err := runnerproto.NewEnvelope(runnerproto.KindChallenge, &runnerproto.Challenge{KeyID: key.ID.String(), Nonce: base64.StdEncoding.EncodeToString(nonce)})
	if err != nil || conn.write(ctx, ch) != nil {
		return nil, false
	}

	env, err := h.readEnvelope(ctx, conn, h.cfg.HandshakeTimeout)
	if err != nil || env.Kind != runnerproto.KindChallengeResponse {
		h.closeMalformed(conn, key.RunnerID, "challenge response")
		return nil, false
	}
	var resp runnerproto.ChallengeResponse
	if err := runnerproto.DecodePayload(env, &resp); err != nil {
		h.closeMalformed(conn, key.RunnerID, "challenge response")
		return nil, false
	}
	pub, sig, err := resp.Decoded()
	if err != nil {
		h.closeMalformed(conn, key.RunnerID, "challenge response")
		return nil, false
	}
	proven, err := h.keys.ProveMachine(ctx, key, nonce, pub, sig)
	switch {
	case err == nil:
	case errors.Is(err, runner.ErrWrongMachine), errors.Is(err, runner.ErrKeyAlreadyClaimed):
		h.refuse(ctx, conn, key.RunnerID, &runnerproto.Refused{Code: "wrong_machine", Reason: "runner key is bound to a different machine"}, runner.CloseWrongMachine)
		return nil, false
	case errors.Is(err, runner.ErrKeyRevoked):
		h.refuse(ctx, conn, key.RunnerID, &runnerproto.Refused{Code: "revoked", Reason: "runner key revoked"}, runner.CloseRevoked)
		return nil, false
	default:
		h.log.Error("runner machine check failed", "runner", string(key.RunnerID), "error", err)
		conn.closeSync(int(websocket.StatusInternalError), "machine check failed")
		return nil, false
	}

	env, err = h.readEnvelope(ctx, conn, h.cfg.HandshakeTimeout)
	if err != nil || env.Kind != runnerproto.KindHello {
		h.closeMalformed(conn, key.RunnerID, "hello")
		return nil, false
	}
	var hello runnerproto.Hello
	if err := runnerproto.DecodePayload(env, &hello); err != nil {
		h.closeMalformed(conn, key.RunnerID, "hello")
		return nil, false
	}
	info, err := helloInfo(hello)
	if err != nil {
		h.closeMalformed(conn, key.RunnerID, "hello")
		return nil, false
	}
	s, err := h.hub.Admit(ctx, proven, conn, info)
	var inc *runner.IncompatibleError
	switch {
	case err == nil:
		return s, true
	case errors.As(err, &inc):
		h.refuse(ctx, conn, key.RunnerID, &runnerproto.Refused{Code: "incompatible", Reason: inc.Compat.Reason,
			ProtocolMin: runnerproto.ProtocolVersion, ProtocolMax: runnerproto.ProtocolVersion}, runner.CloseIncompatible)
	case errors.Is(err, runner.ErrRunnerRevoked):
		h.refuse(ctx, conn, key.RunnerID, &runnerproto.Refused{Code: "revoked", Reason: "runner key revoked"}, runner.CloseRevoked)
	default:
		h.log.Error("runner admission failed", "runner", string(key.RunnerID), "error", err)
		conn.closeSync(int(websocket.StatusInternalError), "admission failed")
	}
	return nil, false
}

func closeFor(err error) (int, string) {
	switch {
	case errors.Is(err, runner.ErrRunnerRevoked):
		return runner.CloseRevoked, "runner key revoked"
	case errors.Is(err, runner.ErrSessionClosed):
		return runner.CloseSuperseded, "session superseded"
	case errors.Is(err, runner.ErrMalformedMessage):
		return runner.CloseMalformed, "malformed message"
	default:
		return int(websocket.StatusInternalError), "server error"
	}
}

// readLoop feeds frames to the hub until the socket ends. A closed or idle
// socket is NOT runner loss: the hub's heartbeat-gap liveness decides that.
func (h *RunnerConnectHandler) readLoop(ctx context.Context, conn *wsRunnerConn, s *runner.Session) {
	for {
		env, err := h.readEnvelope(ctx, conn, h.cfg.IdleTimeout)
		if err != nil {
			if isProtocolError(err) {
				h.closeMalformed(conn, s.RunnerID, "frame")
			} else {
				h.log.Debug("runner socket ended", "runner", string(s.RunnerID), "status", int(websocket.CloseStatus(err)))
			}
			return
		}
		in, err := toInbound(env)
		if err != nil {
			h.closeMalformed(conn, s.RunnerID, env.Kind)
			return
		}
		if err := h.hub.Handle(ctx, s, in); err != nil {
			code, reason := closeFor(err)
			if code == int(websocket.StatusInternalError) {
				h.log.Error("runner message failed", "runner", string(s.RunnerID), "kind", env.Kind, "error", err)
			} else {
				h.log.Warn("runner connection closed", "runner", string(s.RunnerID), "reason", reason)
			}
			conn.closeSync(code, reason)
			return
		}
	}
}

// ServeHTTP upgrades an authenticated runner request and runs the session.
func (h *RunnerConnectHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
	key, ok := RunnerKeyFromContext(r.Context())
	if !ok {
		WriteError(w, http.StatusUnauthorized, "RUNNER_KEY_REQUIRED", "Authorization: Runner <key> is required")
		return
	}
	c, err := websocket.Accept(w, r, nil)
	if err != nil {
		return // Accept wrote the HTTP error
	}
	defer func() { _ = c.CloseNow() }()
	c.SetReadLimit(runnerproto.MaxFrameBytes)
	conn := newWSRunnerConn(c, key.RunnerID, h.cfg.HeartbeatSeconds, h.cfg.WriteTimeout)
	defer conn.waitClosed(h.cfg.WriteTimeout)
	ctx := r.Context()
	s, ok := h.handshake(ctx, conn, key)
	if !ok {
		return
	}
	defer h.hub.Disconnect(s)
	h.log.Info("runner connected", "runner", string(s.RunnerID), "key_id", s.KeyID.String())
	h.readLoop(ctx, conn, s)
	h.log.Info("runner disconnected", "runner", string(s.RunnerID))
}
