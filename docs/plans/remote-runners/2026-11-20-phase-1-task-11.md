# Remote Runners — Phase 1 Task 11: Claude outcome classification (RR-03)

**Goal:** Classify every failed `claude` turn from structured stream-json fields only, as exhausted (with a normalized reset and window), logged out, transient, or other. Attach the result with `runner.WithOutcome` before sanitizing. The sanitized error must still `errors.As` to `*runner.OutcomeError`, and success-path headroom hints must reach the health monitor.

**Architecture:**
- `parseEvent` learns `rate_limit_event`, the assistant `error` kind, `api_error_status` / `apiErrorStatus`, and `quotaLimits` / `rate_limit_info` / `apiErrorParams.rate_limit_info`. It decodes these leniently into a prose-free `outcomeSignal`, so a wrong JSON type is ignored rather than failing the line.
- An `outcomeTracker` folds the signals across the turn. Every terminal-error path in `collectResultWithPolicy` and `streamState` calls `tracker.attach(err)` before `SanitizeProcessError`.
- `sanitizedProcessError` gains an `As` method that exposes only a redacted `*runner.OutcomeError` copy. It never exposes the raw cause.
- Success-path hints such as `allowed_warning` go through a new context observer, `runner.ObserveOutcome`. The P1 T13 health-reporting runner set installs it.

**Tech Stack:** Go 1.25, `encoding/json`, testify; JSONL fixtures under `adapter/claudecli/testdata/outcomes/`.

```go
package runner

import "context"

type outcomeObserverKey struct{}

// ContextWithOutcomeObserver installs fn to receive success-path observations
// (headroom hints, window snapshots) that adapters cannot return through a
// provider response. A nil fn installs nothing.
func ContextWithOutcomeObserver(ctx context.Context, fn func(SourceOutcome)) context.Context {
	if fn == nil {
		return ctx
	}
	return context.WithValue(ctx, outcomeObserverKey{}, fn)
}

// ObserveOutcome reports a success-path observation to the installed observer.
// Without an observer (probes, legacy tests) it does nothing.
func ObserveOutcome(ctx context.Context, o SourceOutcome) {
	if fn, ok := ctx.Value(outcomeObserverKey{}).(func(SourceOutcome)); ok && fn != nil {
		fn(o)
	}
}
```

```go
package claudecli

import (
	"context"
	"encoding/json"
	"strings"
	"time"

	pool "github.com/arbitron/arbitron/domain/runner"
)

// rateLimitInfo is a decoded rate_limit_info / quotaLimits object: status,
// reset (epoch seconds; milliseconds tolerated), window type, utilization 0..1.
type rateLimitInfo struct {
	Status        string
	ResetsAt      *float64
	RateLimitType string
	Utilization   *float64
}

type rawRateLimitInfo struct {
	Status        string   `json:"status"`
	ResetsAt      *float64 `json:"resetsAt"`
	RateLimitType string   `json:"rateLimitType"`
	Utilization   *float64 `json:"utilization"`
}

// outcomeSignal is the structured, prose-free failure evidence on one event.
type outcomeSignal struct {
	ErrorKind      string
	APIErrorStatus int
	Limit          *rateLimitInfo
}

// decodeSignal leniently extracts the outcome fields from an envelope. A field
// of an unexpected JSON type is ignored (never fails the line), so unknown or
// changed shapes fail closed to "no signal" => class other.
func decodeSignal(env rawEnvelope) *outcomeSignal {
	var s outcomeSignal
	var kind string
	if len(env.Error) > 0 && json.Unmarshal(env.Error, &kind) == nil {
		s.ErrorKind = strings.TrimSpace(kind)
	}
	for _, raw := range []json.RawMessage{env.APIErrorStatus, env.APIErrorStatusCamel} {
		var n int
		if len(raw) > 0 && json.Unmarshal(raw, &n) == nil && n != 0 {
			s.APIErrorStatus = n
		}
	}
	candidates := []json.RawMessage{env.RateLimitInfo, env.QuotaLimits}
	if len(env.APIErrorParams) > 0 {
		var p struct {
			RateLimitInfo json.RawMessage `json:"rate_limit_info"`
		}
		if json.Unmarshal(env.APIErrorParams, &p) == nil {
			candidates = append(candidates, p.RateLimitInfo)
		}
	}
	for _, raw := range candidates {
		if len(raw) == 0 {
			continue
		}
		var r rawRateLimitInfo
		if json.Unmarshal(raw, &r) != nil || r.Status == "" {
			continue
		}
		s.Limit = &rateLimitInfo{Status: r.Status, ResetsAt: r.ResetsAt, RateLimitType: r.RateLimitType, Utilization: r.Utilization}
		if r.Status == "rejected" {
			break // the rejecting window is the authoritative one
		}
	}
	if s.ErrorKind == "" && s.APIErrorStatus == 0 && s.Limit == nil {
		return nil
	}
	return &s
}

var loggedOutKinds = map[string]bool{
	"authentication_failed": true, "oauth_org_not_allowed": true, "account_on_hold": true,
	"verification_required": true, "billing_error": true,
}

var transientKinds = map[string]bool{"rate_limit": true, "overloaded": true, "server_error": true}

// outcomeTracker folds the structured signals seen across one claude turn.
type outcomeTracker struct {
	errorKind string
	status    int
	limit     *rateLimitInfo
	rejected  *rateLimitInfo // latched: a rejected window is never overwritten by a later allowed one
}

func (t *outcomeTracker) observe(ev streamEvent) {
	s := ev.Signal
	if s == nil {
		return
	}
	if s.ErrorKind != "" {
		t.errorKind = s.ErrorKind
	}
	if s.APIErrorStatus != 0 {
		t.status = s.APIErrorStatus
	}
	if s.Limit != nil {
		t.limit = s.Limit
		if s.Limit.Status == "rejected" {
			t.rejected = s.Limit
		}
	}
}

// classify returns the outcome, and ok=false when no structured evidence was seen.
// Order: logged-out (needs a human) > exhausted (rejected quota) > transient > other.
func (t *outcomeTracker) classify() (pool.SourceOutcome, bool) {
	switch {
	case loggedOutKinds[t.errorKind] || t.status == 401:
		return pool.SourceOutcome{Class: pool.OutcomeLoggedOut}, true
	case t.rejected != nil:
		o := pool.SourceOutcome{Class: pool.OutcomeExhausted, ResetAt: normalizeEpoch(t.rejected.ResetsAt)}
		if w, ok := windowFor(*t.rejected, o.ResetAt); ok {
			o.Windows = []pool.LimitWindow{w}
		}
		return o, true
	case transientKinds[t.errorKind] || t.status == 429 || (t.status >= 500 && t.status <= 599):
		return pool.SourceOutcome{Class: pool.OutcomeTransient}, true
	case t.errorKind != "" || t.status != 0:
		return pool.SourceOutcome{Class: pool.OutcomeOther}, true
	}

		return nil
	}

	}

// reportSuccess publishes an allowed_warning headroom hint for a turn that
// completed. 1-utilization when reported; absent utilization stays unknown.
func (t *outcomeTracker) reportSuccess(ctx context.Context) {
	if t.limit == nil || t.limit.Status != "allowed_warning" {
		return
	}
	o := pool.SourceOutcome{Class: pool.OutcomeNone}
	if u := t.limit.Utilization; u != nil {
		h := clamp01(1 - *u)
		o.HeadroomHint = &h
	}
	if w, ok := windowFor(*t.limit, normalizeEpoch(t.limit.ResetsAt)); ok {
		o.Windows = []pool.LimitWindow{w}
	}
	pool.ObserveOutcome(ctx, o)
}

// windowFor keys the window by its rateLimitType (identity), model scope and
// duration — never by primary/secondary position.
func windowFor(l rateLimitInfo, reset *time.Time) (pool.LimitWindow, bool) {
	if l.RateLimitType == "" {
		return pool.LimitWindow{}, false
	}
	w := pool.LimitWindow{LimitID: l.RateLimitType, ResetAt: reset}
	switch {
	case l.RateLimitType == "five_hour":
		w.DurationMins = 300
	case strings.HasPrefix(l.RateLimitType, "seven_day"):
		w.DurationMins = 10080
		switch l.RateLimitType {
		case "seven_day_opus":
			w.ModelScope = "opus"
		case "seven_day_sonnet":
			w.ModelScope = "sonnet"
		}
	}
	if l.Utilization != nil {
		p := clamp01(*l.Utilization) * 100
		w.UsedPercent = &p
	}
	return w, true
}

// normalizeEpoch converts an epoch in seconds (or milliseconds, when > 1e12) to
// UTC. Non-positive or missing values are unknown (nil).
func normalizeEpoch(v *float64) *time.Time {
	if v == nil || *v <= 0 {
		return nil
	}
	sec := *v
	if sec > 1e12 {
		sec /= 1000
	}
	t := time.Unix(int64(sec), 0).UTC()
	return &t
}

func clamp01(v float64) float64 {
	if v < 0 {
		return 0
	}
	if v > 1 {
		return 1
	}
	return v
}
```

```go
// parseEvent parses a single stream-json line. ok is false for blank lines and
// unrecognised event types (both are skipped without error). A non-nil error

func parseEvent(line []byte) (streamEvent, bool, error) {
	trimmed := bytes.TrimSpace(line)
	if len(trimmed) == 0 {
		return streamEvent{}, false, nil
	}

	var env rawEnvelope
	if err := json.Unmarshal(trimmed, &env); err != nil {
		return streamEvent{}, false, fmt.Errorf("claudecli: parse stream-json line: %w", err)
	}

	switch eventKind(env.Type) {
	case eventSystem:
		return streamEvent{Kind: eventSystem, SessionID: env.SessionID}, true, nil
	case eventUser:
		return streamEvent{Kind: eventUser, SessionID: env.SessionID}, true, nil
	case eventRateLimit:
		return streamEvent{Kind: eventRateLimit, SessionID: env.SessionID, Signal: decodeSignal(env)}, true, nil
	case eventAssistant:
		// An assistant message that is not an object fails closed: its text is
		// the agent's output, and silently dropping it would lose a turn.
		var msg *rawMessage
		if len(env.Message) > 0 {
			if err := json.Unmarshal(env.Message, &msg); err != nil {
				return streamEvent{}, false, fmt.Errorf("claudecli: parse stream-json assistant message: %w", err)
			}
		}
		return streamEvent{
			Kind:      eventAssistant,
			SessionID: env.SessionID,
			Text:      assistantText(msg),
			Signal:    decodeSignal(env),
		}, true, nil
	case eventResult:
		ev := streamEvent{
			Kind:      eventResult,
			SessionID: env.SessionID,
			Text:      env.Result,
			IsError:   env.IsError,
			Signal:    decodeSignal(env),
		}
		if env.Usage != nil {
			ev.TokensIn = env.Usage.totalInput()
			ev.TokensOut = env.Usage.OutputTokens
		}
		for m := range env.ModelUsage {
			ev.ModelUsage = append(ev.ModelUsage, m)
		}
		sort.Strings(ev.ModelUsage)
		return ev, true, nil
	default:
		return streamEvent{}, false, nil
	}
}
```

```go
package claudecli

import (
	"errors"

	pool "github.com/arbitron/arbitron/domain/runner"

// SanitizeProcessError hides untrusted subprocess diagnostics before they reach
// verified-call errors or logs. It preserves errors.Is classification (including
// cancellation and response limits) and the classified *runner.OutcomeError (as
// a redacted copy), but deliberately offers no Unwrap: callers must not recover
// and log a credential-bearing StderrError tail.
func SanitizeProcessError(err error) error {
	if err == nil {
		return nil
	}
	return &sanitizedProcessError{cause: err}
}

type sanitizedProcessError struct{ cause error }

func (*sanitizedProcessError) Error() string { return "harness process failed" }

func (e *sanitizedProcessError) Is(target error) bool { return errors.Is(e.cause, target) }

// As exposes ONLY a classified outcome. The returned *OutcomeError is a fresh
// copy whose Err is this sanitized error, so the class and reset are readable
// while the raw cause (and any StderrError tail) stays unreachable.
func (e *sanitizedProcessError) As(target any) bool {
	t, ok := target.(**pool.OutcomeError)
	if !ok {
		return false
	}
	var oe *pool.OutcomeError
	if !errors.As(e.cause, &oe) {
		return false
	}
	*t = &pool.OutcomeError{Outcome: oe.Outcome, Err: e}
	return true
}
```

```go
// sendChunk. sendChunk returns false if the consumer is gone (ctx cancelled);
// Handle then sets failed and the caller should stop forwarding.

// rawLine is the on-the-wire line; parseEvent's result is consumed here.
// A parse error becomes a chunk-with-error; the caller MUST keep draining
// the runner's lines channel so the subprocess can be reaped.
func (s *streamState) Handle(rawLine string, sendChunk func(engine.ChatChunk) bool) {
	if s.failed {
		return
	}
	ev, ok, parseErr := parseEvent([]byte(rawLine))
	if parseErr != nil {
		if s.policy.sanitize {
			parseErr = SanitizeProcessError(parseErr)
		}

			// consumer gone; still mark failed so Finalize knows
		}

		return

		return

		s.sessionID = ev.SessionID

	case eventAssistant:
		if ev.Text != "" {
			if !sendChunk(engine.ChatChunk{Token: ev.Text}) {
				s.failed = true
			}
		}
	case eventResult:
		// One turn can carry several results: Claude Code re-wakes a print
		// session each time a background subagent finishes. A pinned turn needs
		// EVERY result to prove the model, so one unproven or failed wake fails
		// the whole turn (latched before Done; the caller still drains and reaps).

		if s.wantModel != "" && s.sawResult && s.policy.singleResult() {
			sendChunk(engine.ChatChunk{Error: errModelExecutionEvidence})
			s.failed = true
			return

		s.tokensOut += ev.TokensOut

			return

			err := errModelExecutionEvidence
			if s.wantModel == "" && !s.policy.sanitize {
				err = fmt.Errorf("claudecli: claude reported an error: %s", ev.Text)
			}
			sendChunk(engine.ChatChunk{Error: s.tracker.attach(err)})
			s.failed = true
		}

// been fully drained and wait() has returned. waitErr is whatever wait()
// returned (nil on clean exit). No chunk is emitted when Handle already

		return

		// Attach the classified outcome BEFORE sanitizing: the sanitized error
		// exposes only the outcome copy via As.

		return

	if s.ctx != nil {
		s.tracker.reportSuccess(s.ctx)
	}
	sendChunk(engine.ChatChunk{
		Done:      true,
		SessionID: s.sessionID,
		TokensIn:  s.tokensIn,
		TokensOut: s.tokensOut,
	})
}
```

```go
		if parseErr != nil {
			continue // keep draining; don't process further

		ev, ok, perr := parseEvent([]byte(line))
		if perr != nil {

			}

		}
```

```go
package claudecli

import (
	"context"
	"errors"
```

```go
	"os"
	"os/exec"

	"regexp"
	"strconv"
	"strings"

// HarnessFloor names a harness feature arbitron depends on and the minimum
// version that provides it (evidence: docs/evidence/remote-runners/research-exhaustion.md).
type HarnessFloor struct {
	Feature string
	Min     string
}

// SupportedVersions is the claude capability table. The effective floor is the
// highest Min. Floors are inferred from session-log and binary-schema evidence
// and are not certified by real captures (RR-03 decision 2026-11-22).
var SupportedVersions = []HarnessFloor{
	{Feature: "rate_limit_event", Min: "2.1.209"},
	{Feature: "result usage cache buckets", Min: "2.1.278"},
	{Feature: "assistant error kind and quotaLimits", Min: "2.1.284"},
}

// ErrVersionUnparseable reports `claude --version` output without a semver.
var ErrVersionUnparseable = errors.New("claudecli: unparseable claude --version output")

var semverRE = regexp.MustCompile(`\d+\.\d+\.\d+`)

// CheckVersion reports whether v meets every SupportedVersions floor. reason is
// "" when ok, else names the first unmet feature.
func CheckVersion(v string) (bool, string) {
	if strings.TrimSpace(v) == "" {
		return false, "claude version unknown"
	}
	have, ok := parseSemver(v)
	if !ok {
		return false, fmt.Sprintf("claude version %q is not parseable", v)
	}
	floor := HarnessFloor{}
	var floorV [3]int
	for _, f := range SupportedVersions {
		fv, _ := parseSemver(f.Min)
		if cmpSemver(fv, floorV) > 0 {
			floor, floorV = f, fv
		}
	}
	if cmpSemver(have, floorV) < 0 {
		return false, fmt.Sprintf("claude %s is below %s (%s)", v, floor.Min, floor.Feature)
	}
	return true, ""
}

func parseSemver(s string) ([3]int, bool) {
	m := semverRE.FindString(s)
	if m == "" {
		return [3]int{}, false
	}
	var out [3]int
	for i, p := range strings.SplitN(m, ".", 3) {
		n, err := strconv.Atoi(p)
		if err != nil {
			return [3]int{}, false
		}
		out[i] = n
	}
	return out, true
}

func cmpSemver(a, b [3]int) int {
	for i := 0; i < 3; i++ {
		if a[i] != b[i] {
			if a[i] < b[i] {
				return -1
			}
			return 1
		}
	}
	return 0
}

// probeVersionTimeout bounds `claude --version`.
const probeVersionTimeout = 10 * time.Second

// ProbeVersion runs `<bin> --version` and returns the first semver it prints.
// Process failures are sanitized (no stderr reaches the error).
func ProbeVersion(ctx context.Context, r CommandRunner, bin string) (string, error) {
	ctx, cancel := context.WithTimeout(ctx, probeVersionTimeout)
	defer cancel()
	lines, wait, err := r.Run(ctx, RunSpec{Bin: bin, Args: []string{"--version"}, Env: procenv.BaseEnv(), MaxResponseBytes: 64 << 10})
	if err != nil {
		return "", fmt.Errorf("claudecli: version probe: %w", SanitizeProcessError(err))
	}
	var first string
	for l := range lines {
		if first == "" && strings.TrimSpace(l) != "" {
			first = l
		}
	}
	if werr := wait(); werr != nil {
		return "", fmt.Errorf("claudecli: version probe: %w", SanitizeProcessError(werr))
	}
	v := semverRE.FindString(first)
	if v == "" {
		return "", ErrVersionUnparseable
	}
	return v, nil
}

// ResolveBinary returns the claude binary path and whether it was actually
// found. An explicit path counts as found; the bare "claude" fallback does not.
func ResolveBinary(explicit string) (string, bool) {
	return resolveBinaryFound(explicit, exec.LookPath, func(p string) (bool, error) {
		_, err := os.Stat(p)
		return err == nil, nil
	})
}

func resolveBinaryFound(explicit string, lookPath func(string) (string, error), exists func(string) (bool, error)) (string, bool) {
	if explicit != "" {
		return resolveBinary(explicit), true
	}
	if p, err := lookPath("claude"); err == nil {
		return p, true
	}
	for _, dir := range procenv.StandardBinDirs() {
		c := filepath.Join(dir, "claude")
		if ok, _ := exists(c); ok {
			return c, true
		}
	}
	return "claude", false
}
```
