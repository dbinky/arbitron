# Remote Runners — Phase 1 Task 10: Built-in StepRunner and routing selector decorators

**Goal:** Every provider and process operation that a workflow run executes is routed through `StepDispatcher.Dispatch`. The decorators keep the existing factory validation unchanged, and the built-in runner executes the placed job with the undecorated inner selectors.

**Architecture:**
- `domain/runner/selectors.go` adds four decorators:
  - `ChatProviderSelector`
  - `AgentProviderSelector`
  - `InteractiveStreamSelector`
  - `ProcessRunner`
- Each decorated selector calls `inner.XFor(b)` first, so validation errors and refusals surface exactly as today. It then returns a routed provider. On each call, that provider builds `ExecRequirements` from `GroupFor` and the binding (exact model, effective backend, tools, `RequireExecutedModel`) and calls `Dispatch`.
- `adapter/runner/builtin` is the `StepRunner` for `BuiltInRunnerID`. It re-selects through the inner selectors and returns adapter errors unchanged, so `runner.WithOutcome` classification reaches the dispatcher.

**Tech Stack:** Go 1.25, testify.

// recordingDispatcher records every (req, job) and answers from a per-op table,
// so the routed provider must hand over the right payload to get the right answer.
type recordingDispatcher struct {
	mu   sync.Mutex
	reqs []ExecRequirements
	jobs []StepJob
	out  map[OpType]StepOutput
	err  error
}

func (d *recordingDispatcher) Dispatch(_ context.Context, req ExecRequirements, job StepJob) (StepOutput, error) {
	d.mu.Lock()
	defer d.mu.Unlock()
	d.reqs = append(d.reqs, req)
	d.jobs = append(d.jobs, job)
	return d.out[req.Op], d.err
}

// innerChats is a validating inner selector: it refuses bindings whose Model is
// "refused" and records every binding it saw. Its providers must NEVER be called
// by the routed provider (execution belongs to the runner).
type innerChats struct {
	seen  []provider.ResolvedBinding
	calls int
}

func (s *innerChats) ChatProviderFor(b provider.ResolvedBinding) (engine.ChatProvider, error) {
	s.seen = append(s.seen, b)
	if b.Model == "refused" {
		return nil, errInnerRefused
	}
	return countingChat{&s.calls}, nil
}

func (s *innerChats) AgentProviderFor(b provider.ResolvedBinding) (engine.AgentProvider, error) {

	if b.Model == "refused" {
		return nil, errInnerRefused

import (
	"context"

// dispatcher is the slice of *StepDispatcher the routed providers need. The
// exported constructors take the concrete *StepDispatcher (contracts §11); the
// interface lets unit tests record dispatches.
type dispatcher interface {
	Dispatch(ctx context.Context, req ExecRequirements, job StepJob) (StepOutput, error)
}

// errMissingStepOutput reports a dispatcher success that carried no payload for
// the requested op. It is a programming error surfaced as a step failure.
var errMissingStepOutput = errors.New("runner: dispatch returned no output for the operation")

// BuiltInAffinity is the Phase 1 affinity: every operation stays on the
// built-in runner.
func BuiltInAffinity(provider.ResolvedBinding) Affinity { return AffinityBuiltIn }

// NewRoutingChatSelector wraps a chat selector so every Chat/StreamChat call is
// placed through the dispatcher. inner is consulted first, so validation errors
// surface unchanged. A nil affinity uses BuiltInAffinity.
func NewRoutingChatSelector(inner engine.ChatProviderSelector, d *StepDispatcher, affinity func(provider.ResolvedBinding) Affinity) engine.ChatProviderSelector {
	if d == nil {
		panic("runner: NewRoutingChatSelector requires a dispatcher")
	}
	return newRoutingChatSelector(inner, d, affinity)
}

// NewRoutingAgentSelector wraps an agent selector so RunAgent is placed.
func NewRoutingAgentSelector(inner engine.AgentProviderSelector, d *StepDispatcher) engine.AgentProviderSelector {
	if d == nil {
		panic("runner: NewRoutingAgentSelector requires a dispatcher")
	}

// requirementsFor builds the placement requirements for a model operation.
// The binding's own tool allowlist wins over the request scope (same precedence
// the adapters apply); pinned bindings always require executed-model evidence.
func requirementsFor(ctx context.Context, b provider.ResolvedBinding, op OpType, aff Affinity, requireModel bool, scope []string) (ExecRequirements, error) {
	g, err := GroupFor(b, op)
	if err != nil {
		return ExecRequirements{}, fmt.Errorf("route %s for model %q: %w", op, b.Model, err)
	}
	fingerprint, err := EndpointFingerprint(b)
	if err != nil { return ExecRequirements{}, err }
	tools := scope
	if len(b.AllowedTools) > 0 {
		tools = b.AllowedTools
	}
	return ExecRequirements{
		Provider: b.Provider,
		EndpointFingerprint: fingerprint,
		ContextWindow: b.ContextWindow,
		Effort: b.Effort,
		Group:                g,
		Model:                b.Model,
		Backend:              b.Backend,
		Op:                   op,
		Affinity:             aff,
		RunID:                runIDFrom(ctx),
		RequireExecutedModel: requireModel || b.RequiresModelEvidence(),
		Tools:                append([]string(nil), tools...),
	}, nil
}

// runIDFrom reads the run from the step identity. A missing identity yields
// uuid.Nil here; the dispatcher refuses it with engine.ErrMissingStepIdentity.
func runIDFrom(ctx context.Context) uuid.UUID {
	if si, ok := engine.StepIdentityFromContext(ctx); ok {
		return si.RunID
	}
	return uuid.Nil
}

// Package builtin is the in-process StepRunner for runner.BuiltInRunnerID. It
// executes placed jobs with the UNDECORATED selectors (so it never re-enters
// the dispatcher) and the local process runner.
package builtin

import (
	"context"
	"errors"
	"fmt"

	"github.com/arbitron/arbitron/domain/engine"
	"github.com/arbitron/arbitron/domain/provider"
	"github.com/arbitron/arbitron/domain/runner"
)

// ErrMalformedJob is returned for a StepJob whose payload does not match its op.
var ErrMalformedJob = errors.New("builtin runner: malformed step job")

const (
	opChat    runner.OpType = "chat"
	opStream  runner.OpType = "stream"
	opAgent   runner.OpType = "agent"
	opProcess runner.OpType = "process"
)

// Deps are the inner (undecorated) execution paths.
type Deps struct {
	Chats   engine.ChatProviderSelector
	Agents  engine.AgentProviderSelector
	Streams engine.InteractiveStreamSelector
	Process engine.ProcessRunner
}

// Runner executes jobs placed on the built-in runner.
type Runner struct{ d Deps }

var _ runner.StepRunner = (*Runner)(nil)

// New builds the built-in runner. Any nil dependency panics: a half-wired
// built-in runner would silently fail every placed step.
func New(d Deps) *Runner {
	if d.Chats == nil || d.Agents == nil || d.Streams == nil || d.Process == nil {
		panic("builtin: New requires Chats, Agents, Streams and Process")
	}
	return &Runner{d: d}
}

// Run executes one placed job. Adapter errors are returned unchanged so the
// outcome attached with runner.WithOutcome reaches the dispatcher's classifier.
func (r *Runner) Run(ctx context.Context, _ runner.Assignment, job runner.StepJob) (runner.StepOutput, error) {
	switch job.Op {
	case opChat:
		if job.Chat == nil {
			return runner.StepOutput{}, fmt.Errorf("%w: chat op without request", ErrMalformedJob)
		}
		p, err := r.d.Chats.ChatProviderFor(job.Binding)
		if err != nil {
			return runner.StepOutput{}, err
		}
		resp, err := p.Chat(ctx, *job.Chat)
		return runner.StepOutput{Chat: resp}, err
	case opStream:
		if job.Chat == nil {
			return runner.StepOutput{}, fmt.Errorf("%w: stream op without request", ErrMalformedJob)
		}
		p, err := r.streamProvider(job.Binding)
		if err != nil {
			return runner.StepOutput{}, err
		}
		ch, err := p.StreamChat(ctx, *job.Chat)
		return runner.StepOutput{Stream: ch}, err
	case opAgent:
		if job.Agent == nil {
			return runner.StepOutput{}, fmt.Errorf("%w: agent op without mission", ErrMalformedJob)
		}
		p, err := r.d.Agents.AgentProviderFor(job.Binding)
		if err != nil {
			return runner.StepOutput{}, err
		}
		res, err := p.RunAgent(ctx, *job.Agent)
		return runner.StepOutput{Agent: res}, err
	case opProcess:
		if job.Process == nil {
			return runner.StepOutput{}, fmt.Errorf("%w: process op without spec", ErrMalformedJob)
		}
		res, err := r.d.Process.RunProcess(ctx, *job.Process)
		return runner.StepOutput{Process: &res}, err
	default:
		return runner.StepOutput{}, fmt.Errorf("%w: unknown op %q", ErrMalformedJob, job.Op)
	}
}

// streamProvider mirrors the interactive selector's split: agent-kind bindings
// use the tool-enabled interactive stream; every other kind uses the chat
// selector's tool-suppressed stream.
func (r *Runner) streamProvider(b provider.ResolvedBinding) (engine.ChatProvider, error) {
	if b.Kind == provider.ProfileKindAgent {
		return r.d.Streams.StreamProviderFor(b)
	}
	return r.d.Chats.ChatProviderFor(b)
}

|---|---|---|
| empty scope | unchanged argv (today's posture) | P3 T15 resolves empty scopes before remote dispatch; no new restriction here. |

| `WebSearch` absent | `-c web_search="disabled"` | |

**Context-window contract:**
- `binding.ContextWindow` empty or `default` emits nothing.
- Otherwise the tier is validated and floored through the same `CapabilityRules.Validate(backend, model, tier, EffortDefault)` the profile write uses (skipped only for an `Empty()` ruleset), then mapped:
  - `200k` → `-c model_context_window=200000`;
  - `1m` → `-c model_context_window=1000000`.
- Any other tier, or a Validate error, refuses the launch: `ErrContextWindowUnsupported`, or the Validate error. Default `capabilities.default.yaml` keeps codex at `context_tiers: [default]`, so production emits a context override only when an operator configures codex tiers.

package codex

import (
	"errors"
	"fmt"
	"strings"

	"github.com/arbitron/arbitron/domain/provider"
)

// ErrToolScopeUnsupported refuses a launch whose tool scope codex-cli cannot
// enforce (see the P1 T10 Cycle 6 evidence: apply_patch and view_image cannot
// be unregistered, reads go through the shell tool).
var ErrToolScopeUnsupported = errors.New("codex: tool scope cannot be enforced by codex-cli")

// ErrContextWindowUnsupported refuses a context tier with no codex emission.
var ErrContextWindowUnsupported = errors.New("codex: context window has no codex-cli emission")

// codexContextConfigKey is codex's ConfigToml context-window field.
const codexContextConfigKey = "model_context_window"

// toolPolicy is the codex launch posture derived from a non-empty tool scope.
type toolPolicy struct {
	Restricted bool     // --strict-config + per-server MCP disable
	ReadOnly   bool     // no write tool granted: read-only sandbox
	Overrides  []string // fixed-order "-c" pairs
}

var (
	codexWriteTools = map[string]bool{"Write": true, "Edit": true, "MultiEdit": true, "NotebookEdit": true}
	codexReadTools  = map[string]bool{"Read": true, "Grep": true, "Glob": true, "LS": true}
	// No separate codex surface: Bash already governs this reach.
	codexInertTools = map[string]bool{"WebFetch": true, "Task": true, "TodoWrite": true}
)

func codexToolPolicy(scope []string) (toolPolicy, error) {
	if len(scope) == 0 {
		return toolPolicy{}, nil
	}
	var bash, write, search bool
	for _, raw := range scope {
		name := strings.TrimSpace(raw)
		switch {
		case name == "Bash":
			bash = true
		case name == "WebSearch":
			search = true
		case codexWriteTools[name]:
			write = true
		case codexReadTools[name], codexInertTools[name]:
		default:
			// Quote the exact profile entry (e.g. "Bash(gh:*)") so the operator
			// knows which allowed-tools item to change.
			return toolPolicy{}, fmt.Errorf("%w: tool %q has no codex-cli equivalent", ErrToolScopeUnsupported, name)
		}
	}
	if !bash {
		return toolPolicy{}, fmt.Errorf("%w: scope lacks %q: codex reads and searches only through its shell tool and cannot unregister view_image/apply_patch", ErrToolScopeUnsupported, "Bash")
	}
	p := toolPolicy{Restricted: true, ReadOnly: !write,
		Overrides: []string{"-c", "features.plugins=false", "-c", "features.apps=false"}}
	if !search {
		p.Overrides = append(p.Overrides, "-c", `web_search="disabled"`)
	}
	return p, nil
}

func codexContextArgs(b provider.ResolvedBinding, rules provider.CapabilityRules) ([]string, error) {
	tier := b.ContextWindow
	if tier == "" || tier == provider.ContextDefault {
		return nil, nil
	}
	if !rules.Empty() {
		backend := b.Backend
		if backend == "" {
			backend = codexBackend
		}
		effective, _, err := rules.Validate(backend, b.Model, tier, provider.EffortDefault)
		if err != nil {
			return nil, err
		}
		tier = effective
	}
	switch tier {
	case provider.ContextDefault:
		return nil, nil
	case provider.Context200K:
		return []string{"-c", codexContextConfigKey + "=200000"}, nil
	case provider.Context1M:
		return []string{"-c", codexContextConfigKey + "=1000000"}, nil
	}
	return nil, fmt.Errorf("%w: %q", ErrContextWindowUnsupported, tier)
}
