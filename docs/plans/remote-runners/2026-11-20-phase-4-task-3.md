## Alignment correction — durable accepted route and idempotency

The persisted resend source includes the entire accepted execution constraint, not only Model/Tools. Add `RequestID uuid.UUID` and `Binding AcceptedTurnBinding` to `TurnIntent`; migration 045 columns `request_id`, `accepted_binding`. The accepted first turn also writes the snapshot into `LLMConversation[ConversationKeyAcceptedTurnBinding]` in P4 T6. Idle recovery and later turns use that committed snapshot, so editing/removing a profile does not change an open session's accepted backend/model/tool policy.

Capture after effective backend/context/effort/tool scope are resolved, before dispatch. `RouteConfigHash` hashes canonical provider/endpoint/Extras metadata; never persist RevealKey, credential values, raw arbitrary Extras or endpoint auth query values. `RequireExecutedModel` uses accepted `RequiresModelEvidence`. The adapter restores local subscription bindings directly, or loads the original ConnectionRef from `ConnectionStore.Get`, validates route hash and current authorization/existence, and attaches a fresh lazy key via the existing provider key-resolution path. It restores all accepted model/backend/tool/effort/pin fields; it never resolves a changed alias or expands tool scope. Missing/revoked/changed connection produces a durable actionable restoration error; no credential/family substitution.

One pending operation per lease/session has a unique request ID (generated before persistence, reused by retries). Persist must compare immutable payload on duplicate RequestID: identical is idempotent, different message/prompt/binding is an error. SQL unique partial index prevents multiple pending intents for the same lease identity. First-turn replay checks invocation, not just lease key/node. Operation matching uses equality or `prefix + "/"` (never matches `turn/1` with `turn/10`). Pending intent remains pending on recoverable capacity/congestion/transport/commit failure or exhausted bounded recovery budget; only durable success or a recorded definitive terminal operator outcome completes it. End/Abandon state transition and intent completion share an idempotent transaction, with no gap where a crash repeats an applied completion.

**Wiring:** P4 T13 creates `providersadapter.NewTurnBindingRestorer(connStore)` once, injects it into all three consumers before FreezeExecutors; P4 T9/T10 consume accepted snapshot. No unknown/empty snapshot defaults to today's alias. Legacy sessions are explicitly adopted once from immutable run pins plus historical execution evidence, or stop with visible restoration error; legacy tests supply a snapshot explicitly.

**Goal:** Persist every accepted interactive operation (first turn, later message, End, Abandon) as a durable turn intent BEFORE dispatch/acknowledgement, and complete it only when the turn finishes, so a restarted server can resend the exact message and a re-executed first turn reuses the original rendered prompt and nonce.

**Architecture:** `TurnIntent` / `TurnIntentStore` (contracts §17) are appended to `domain/engine/turn_lease.go`; `adapter/sqlite/turn_intent_repo.go` implements them over `turn_intents` (migration 045). `conversationHandler.HandleMessage` resolves the binding first, then persists the intent (exact user message, resolved model, tool scope, prior checkpoint ref, lease generation, operation ID `turn/<n>`), then acknowledges (publishes the user message) and dispatches. `InteractiveExecutor.Execute` persists the rendered prompt, system prompt, nonce, tool scope and exact model before the first dispatch and, when a pending first-turn intent already exists for the run, reuses it verbatim instead of re-rendering. An intent stays pending only when the operation was interrupted (its context was cancelled: crash, shutdown, lost lease); success and definitive failures complete it. P4 T6 later moves the success-path completion into the atomic turn commit.

package sqlite

import (
	"context"
	"database/sql"
	"encoding/json"
	"errors"
	"fmt"
	"time"

	"github.com/arbitron/arbitron/domain/engine"
	"github.com/arbitron/arbitron/domain/instance"
	"github.com/google/uuid"
)

var _ engine.TurnIntentStore = (*TurnIntentRepo)(nil)

// TurnIntentRepo implements engine.TurnIntentStore over turn_intents
// (migration 045). Times are unix milliseconds.
type TurnIntentRepo struct {
	db  *sql.DB
	now func() time.Time
}

// NewTurnIntentRepo creates the repo.
func NewTurnIntentRepo(db *sql.DB) *TurnIntentRepo { return &TurnIntentRepo{db: db, now: time.Now} }

// SetClock replaces the clock (tests).
func (r *TurnIntentRepo) SetClock(now func() time.Time) { r.now = now }

const turnIntentColumns = `id, lease_key, run_id, invocation_id, node_id, turn_index, kind, prompt, system_prompt, nonce, message,
model, tools, prior_checkpoint_ref, session_generation, operation_id, created_at, completed_at`

func (r *TurnIntentRepo) Persist(ctx context.Context, in engine.TurnIntent) error {
	if in.ID == uuid.Nil {
		return errors.New("turn intent: id is required")
	}
	created := in.CreatedAt
	if created.IsZero() {
		created = r.now()
	}
	tools := in.Tools
	if tools == nil {
		tools = []string{}
	}
	toolsJSON, err := json.Marshal(tools)
	if err != nil {
		return fmt.Errorf("marshalling intent tools: %w", err)
	}
	_, err = r.db.ExecContext(ctx, `INSERT INTO turn_intents (`+turnIntentColumns+`)
VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, NULL)`,
		uuidToString(in.ID), in.LeaseKey, uuidToString(in.RunID), uuidToString(in.InvocationID), in.NodeID, in.TurnIndex,
		in.Kind, in.Prompt, in.SystemPrompt, in.Nonce, in.Message, in.Model, string(toolsJSON), in.PriorCheckpointRef,
		in.SessionGeneration, in.OperationID, created.UTC().UnixMilli())
	if err != nil {
		return fmt.Errorf("persisting turn intent %s: %w", in.ID, err)
	}
	return nil
}

// completeIntentTx marks an intent completed; idempotent. Shared with the
// turn-commit transaction (P4 T6).
func completeIntentTx(ctx context.Context, q sqlExecQuerier, id uuid.UUID, at time.Time) error {
	res, err := q.ExecContext(ctx, `UPDATE turn_intents SET completed_at = ? WHERE id = ? AND completed_at IS NULL`,
		at.UTC().UnixMilli(), uuidToString(id))
	if err != nil {
		return fmt.Errorf("completing turn intent %s: %w", id, err)
	}
	if n, err := res.RowsAffected(); err != nil {
		return fmt.Errorf("completing turn intent %s: %w", id, err)
	} else if n == 1 {
		return nil
	}
	var count int
	if err := q.QueryRowContext(ctx, `SELECT COUNT(*) FROM turn_intents WHERE id = ?`, uuidToString(id)).Scan(&count); err != nil {
		return fmt.Errorf("checking turn intent %s: %w", id, err)
	}
	if count == 0 {
		return fmt.Errorf("turn intent %s: %w", id, instance.ErrNotFound)
	}
	return nil
}

func (r *TurnIntentRepo) Complete(ctx context.Context, id uuid.UUID) error {
	return completeIntentTx(ctx, r.db, id, r.now())
}

func (r *TurnIntentRepo) Pending(ctx context.Context) ([]engine.TurnIntent, error) {
	rows, err := r.db.QueryContext(ctx, `SELECT `+turnIntentColumns+` FROM turn_intents WHERE completed_at IS NULL ORDER BY created_at, id`)
	if err != nil {
		return nil, fmt.Errorf("listing pending turn intents: %w", err)
	}
	defer rows.Close()
	var out []engine.TurnIntent
	for rows.Next() {
		in, err := scanTurnIntent(rows)
		if err != nil {
			return nil, err
		}
		out = append(out, in)
	}
	return out, rows.Err()
}

func (r *TurnIntentRepo) PendingFor(ctx context.Context, leaseKey string) (*engine.TurnIntent, error) {
	row := r.db.QueryRowContext(ctx, `SELECT `+turnIntentColumns+` FROM turn_intents
WHERE completed_at IS NULL AND lease_key = ? ORDER BY created_at DESC, id DESC LIMIT 1`, leaseKey)
	in, err := scanTurnIntent(row)
	if errors.Is(err, sql.ErrNoRows) {
		return nil, nil
	}
	if err != nil {
		return nil, err
	}
	return &in, nil
}

type rowScanner interface{ Scan(dest ...any) error }

func scanTurnIntent(s rowScanner) (engine.TurnIntent, error) {
	var in engine.TurnIntent
	var id, run, inv, tools string
	var created int64
	var completed sql.NullInt64
	if err := s.Scan(&id, &in.LeaseKey, &run, &inv, &in.NodeID, &in.TurnIndex, &in.Kind, &in.Prompt, &in.SystemPrompt,
		&in.Nonce, &in.Message, &in.Model, &tools, &in.PriorCheckpointRef, &in.SessionGeneration, &in.OperationID,
		&created, &completed); err != nil {
		if errors.Is(err, sql.ErrNoRows) {
			return in, err
		}
		return in, fmt.Errorf("scanning turn intent: %w", err)
	}
	var err error
	if in.ID, err = parseUUID(id); err != nil {
		return in, err
	}
	if in.RunID, err = parseUUID(run); err != nil {
		return in, err
	}
	if in.InvocationID, err = parseUUID(inv); err != nil {
		return in, err
	}
	if err := json.Unmarshal([]byte(tools), &in.Tools); err != nil {
		return in, fmt.Errorf("decoding intent tools: %w", err)
	}
	if len(in.Tools) == 0 {
		in.Tools = nil
	}
	in.CreatedAt = time.UnixMilli(created).UTC()
	if completed.Valid {
		t := time.UnixMilli(completed.Int64).UTC()
		in.CompletedAt = &t
	}
	return in, nil
}
