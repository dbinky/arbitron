# Remote Runners — Phase 1 Task 5: Runner pool schema, registry and pool event log

**Goal:** Persist the runner pool: migration `041_runner_pool` (all six pool tables), the `runner.Registry` port with its SQLite adapter `RunnerRegistryRepo`, and the `runner.EventLog` port with `PoolEventRepo` (append, newest-first list, prune).

**Architecture:** The ports live in `domain/runner/registry.go` and `domain/runner/eventlog.go`; the adapters are hand-written SQL repos in `adapter/sqlite` modelled on `audit_repo.go`. Source health is stored as columns on `runner_sources` keyed by the source identity `(runner, kind, credential_ref)`, so two connections can never share health. `Snapshot` derives slot and in-flight counts only from `offered`/`running` `runner_jobs` rows (the ledger, P1 T6, owns those rows). `UpsertRunner` refreshes machine facts but never overwrites operator-owned config (step limit, labels, cordon, drain).

**Tech Stack:** Go 1.25.5, modernc SQLite (`database/sql`, single connection), `encoding/json` columns, testify.

**Spec / Design:** spec "Runner pool and registration" (step limit, labels, cordon), "Capacity exhaustion and source health" (per-source health), "Operator visibility" (pool events); design phase 1 §3.2 (registry: persist runners, config, sources with health; built-in registered at boot), §3.3 (slot accounting from offered/running only; workspace homes reserved), §3.9 (bounded pool event log with cause), §8 cross-phase "two connections never share health", "queued waits use no slots". Contracts: `docs/plans/remote-runners/2026-11-20-contracts.md` §1 (041 table list), §8 (Registry, PoolEvent, Cause, EventLog, repo names/constructors).

Schema decisions (migration 041, consumed by P1 T6 ledger, P1 T14, P2 T2 `043` column additions, P3 T1 homes):
- `runners.step_limit` `CHECK (step_limit BETWEEN 1 AND 256)`; `labels`/`harness_versions` JSON text; `compat_state` default `'compatible'`; `liveness` default `'offline'`.
- `runner_sources` stores health as columns (`auth`, `capacity`, `reset_at`, `headroom REAL NULL` — NULL = unknown, never 0, `windows` JSON, `congested`, `congested_until`, `last_probe_at`, `last_outcome`, `detail`), `UNIQUE(runner_id, kind, credential_ref)`, `FOREIGN KEY runner_id → runners ON DELETE CASCADE`.
- `runner_jobs` `UNIQUE(run_id, invocation_id, operation_id, attempt)` (the idempotency key); `runner_id`/`source_id` NULL while queued; `wait` JSON; `state CHECK IN (queued, offered, running, completed, failed, fenced)`. No FK on `run_id`: confirmation probes (P1 T13) use synthetic run IDs. Indexes `(runner_id, state)`, `(source_id, state)`, `(state)`, `(run_id)`.
- `runner_job_attempts` `PRIMARY KEY (job_id, generation)`, FK → `runner_jobs ON DELETE CASCADE`.
- `workspace_homes` keyed by `run_id` (Phase 3 fills; Phase 1 never writes).
- `pool_events.seq INTEGER PRIMARY KEY AUTOINCREMENT` gives a stable newest-first order independent of second-precision timestamps.

-- Remote Runners pool (Phase 1). Runners, their capacity sources with health,
-- the job ledger (intent -> offered -> running -> terminal), attempt history,
-- workspace homes (populated from Phase 3) and the bounded pool event log.

CREATE TABLE IF NOT EXISTS runners (
    id               TEXT PRIMARY KEY,
    name             TEXT NOT NULL,
    built_in         INTEGER NOT NULL DEFAULT 0,
    step_limit       INTEGER NOT NULL DEFAULT 8 CHECK (step_limit BETWEEN 1 AND 256),
    labels           TEXT NOT NULL DEFAULT '{}',
    cordoned         INTEGER NOT NULL DEFAULT 0,
    draining         INTEGER NOT NULL DEFAULT 0,
    compat_state     TEXT NOT NULL DEFAULT 'compatible',
    compat_reason    TEXT NOT NULL DEFAULT '',
    liveness         TEXT NOT NULL DEFAULT 'offline',
    last_contact     TEXT,
    os               TEXT NOT NULL DEFAULT '',
    runner_version   TEXT NOT NULL DEFAULT '',
    harness_versions TEXT NOT NULL DEFAULT '{}',
    workspace_root   TEXT NOT NULL DEFAULT '',
    created_at       TEXT NOT NULL,
    updated_at       TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS runner_sources (
    id              TEXT PRIMARY KEY,
    runner_id       TEXT NOT NULL REFERENCES runners(id) ON DELETE CASCADE,
    kind            TEXT NOT NULL,
    credential_ref  TEXT NOT NULL DEFAULT '',
    families        TEXT NOT NULL DEFAULT '[]',
    backends        TEXT NOT NULL DEFAULT '[]',
    models          TEXT NOT NULL DEFAULT '[]',
    auth            TEXT NOT NULL DEFAULT 'unknown',
    capacity        TEXT NOT NULL DEFAULT 'available',
    reset_at        TEXT,
    headroom        REAL,
    windows         TEXT NOT NULL DEFAULT '[]',
    congested       INTEGER NOT NULL DEFAULT 0,
    congested_until TEXT,
    last_probe_at   TEXT,
    last_outcome    TEXT NOT NULL DEFAULT '',
    detail          TEXT NOT NULL DEFAULT '',
    updated_at      TEXT NOT NULL,
    UNIQUE (runner_id, kind, credential_ref)
);
CREATE INDEX IF NOT EXISTS idx_runner_sources_runner ON runner_sources (runner_id);

CREATE TABLE IF NOT EXISTS runner_jobs (
    id             TEXT PRIMARY KEY,
    run_id         TEXT NOT NULL,
    invocation_id  TEXT NOT NULL,
    operation_id   TEXT NOT NULL,
    attempt        INTEGER NOT NULL CHECK (attempt >= 1),
    node_id        TEXT NOT NULL,
    op             TEXT NOT NULL,
    group_kind     TEXT NOT NULL,
    group_ref      TEXT NOT NULL DEFAULT '',
    group_family   TEXT NOT NULL DEFAULT '',
    model          TEXT NOT NULL DEFAULT '',
    runner_id      TEXT,
    source_id      TEXT,
    generation     INTEGER NOT NULL DEFAULT 0,
    state          TEXT NOT NULL CHECK (state IN ('queued', 'offered', 'running', 'completed', 'failed', 'fenced')),
    lease_until    TEXT,
    reason         TEXT NOT NULL DEFAULT '',
    wait           TEXT,
    result_ref     TEXT NOT NULL DEFAULT '',
    checkpoint_ref TEXT NOT NULL DEFAULT '',
    created_at     TEXT NOT NULL,
    updated_at     TEXT NOT NULL,
    UNIQUE (run_id, invocation_id, operation_id, attempt)
);
CREATE INDEX IF NOT EXISTS idx_runner_jobs_runner_state ON runner_jobs (runner_id, state);
CREATE INDEX IF NOT EXISTS idx_runner_jobs_source_state ON runner_jobs (source_id, state);
CREATE INDEX IF NOT EXISTS idx_runner_jobs_state ON runner_jobs (state);
CREATE INDEX IF NOT EXISTS idx_runner_jobs_run ON runner_jobs (run_id);

CREATE TABLE IF NOT EXISTS runner_job_attempts (
    job_id     TEXT NOT NULL REFERENCES runner_jobs(id) ON DELETE CASCADE,
    generation INTEGER NOT NULL,
    runner_id  TEXT NOT NULL,
    source_id  TEXT NOT NULL DEFAULT '',
    reason     TEXT NOT NULL DEFAULT '',
    outcome    TEXT NOT NULL DEFAULT '',
    started_at TEXT NOT NULL,
    ended_at   TEXT,
    PRIMARY KEY (job_id, generation)
);

CREATE TABLE IF NOT EXISTS workspace_homes (
    run_id             TEXT PRIMARY KEY,
    runner_id          TEXT NOT NULL DEFAULT '',

    checkpoint_version INTEGER NOT NULL DEFAULT 0,
    dirty              INTEGER NOT NULL DEFAULT 0,

CREATE TABLE IF NOT EXISTS pool_events (

    kind          TEXT NOT NULL,

    run_id        TEXT,
    cause_kind    TEXT NOT NULL DEFAULT '',
    cause_actor   TEXT NOT NULL DEFAULT '',

CREATE INDEX IF NOT EXISTS idx_pool_events_run ON pool_events (run_id);

import (

// MaxStepLimit bounds a runner's concurrent step limit.
const MaxStepLimit = 256

// RunnerConfigPatch is an operator change to a runner. nil = unchanged;
// a non-nil Labels map replaces the labels.
type RunnerConfigPatch struct {
	StepLimit *int
	Labels    map[string]string
}

// Registry persists runners, their operator config and their sources' health.
type Registry interface {
	Snapshot(ctx context.Context) (PoolSnapshot, error) // runners+sources+slots+in-flight (+homes from P3); Telemetry filled by caller
	GetRunner(ctx context.Context, id RunnerID) (Runner, []Source, error)
	UpsertRunner(ctx context.Context, r Runner) error
	UpdateConfig(ctx context.Context, id RunnerID, p RunnerConfigPatch) (Runner, error)

	SetDraining(ctx context.Context, id RunnerID, draining bool) error
	SetLiveness(ctx context.Context, id RunnerID, l Liveness, at time.Time) error
	SetCompat(ctx context.Context, id RunnerID, c Compatibility) error
	ReplaceSources(ctx context.Context, id RunnerID, sources []Source) error // keeps existing Health for unchanged IDs
	RecordHealth(ctx context.Context, id SourceID, h SourceHealth) error
	GetSource(ctx context.Context, id SourceID) (Source, error)
	ListRunners(ctx context.Context) ([]Runner, error)
}

// decodeList decodes a JSON array column; "[]" and "" decode to a nil slice.
func decodeList[T any](raw string, dst *[]T) error {
	if raw == "" || raw == "[]" {
		*dst = nil
		return nil
	}
	if err := json.Unmarshal([]byte(raw), dst); err != nil {
		return fmt.Errorf("decoding JSON list: %w", err)
	}
	return nil
}

func encodeList[T any](v []T) (string, error) {
	if len(v) == 0 {
		return "[]", nil
	}
	b, err := json.Marshal(v)
	if err != nil {
		return "", fmt.Errorf("encoding JSON list: %w", err)
	}
	return string(b), nil
}

func headroomArg(h *float64) sql.NullFloat64 {
	if h == nil {
		return sql.NullFloat64{}
	}
	return sql.NullFloat64{Float64: *h, Valid: true}
}

// Snapshot reads runners, sources and live slot counts in one transaction.
func (r *RunnerRegistryRepo) Snapshot(ctx context.Context) (runner.PoolSnapshot, error) {
	tx, err := r.db.BeginTx(ctx, nil)
	if err != nil {
		return runner.PoolSnapshot{}, fmt.Errorf("beginning snapshot: %w", err)
	}
	defer func() { _ = tx.Rollback() }()

	runners, err := listRunners(ctx, tx)
	if err != nil {
		return runner.PoolSnapshot{}, err
	}
	sources, err := listSources(ctx, tx, `SELECT `+sourceColumns+` FROM runner_sources ORDER BY runner_id, id`)
	if err != nil {
		return runner.PoolSnapshot{}, err
	}
	snap := runner.PoolSnapshot{
		Runners: runners, Sources: sources,
		SlotsInUse: map[runner.RunnerID]int{}, SourceInFlight: map[runner.SourceID]int{},
		Telemetry: map[runner.RunnerID]runner.Telemetry{}, Homes: map[uuid.UUID]runner.Home{},
	}
	rows, err := tx.QueryContext(ctx, `SELECT runner_id, COALESCE(source_id, ''), COUNT(*) FROM runner_jobs
		WHERE state IN ('offered', 'running') AND runner_id IS NOT NULL GROUP BY runner_id, source_id`)
	if err != nil {
		return runner.PoolSnapshot{}, fmt.Errorf("counting slots: %w", err)
	}
	defer rows.Close()
	for rows.Next() {
		var rid, sid string
		var n int
		if err := rows.Scan(&rid, &sid, &n); err != nil {
			return runner.PoolSnapshot{}, fmt.Errorf("scanning slot count: %w", err)
		}
		snap.SlotsInUse[runner.RunnerID(rid)] += n
		if sid != "" {
			snap.SourceInFlight[runner.SourceID(sid)] += n
		}
	}
	if err := rows.Err(); err != nil {
		return runner.PoolSnapshot{}, err
	}
	return snap, nil
}

// UpsertRunner registers a runner or refreshes its machine facts. Operator-owned
// configuration (step limit, labels, cordon, drain) is written only on insert.
func (r *RunnerRegistryRepo) UpsertRunner(ctx context.Context, rn runner.Runner) error {
	if rn.ID == "" {
		return errors.New("runner registry: runner id is required")
	}
	if !runner.ValidStepLimit(rn.StepLimit) {
		return fmt.Errorf("%w: %d", runner.ErrInvalidStepLimit, rn.StepLimit)
	}
	labels, err := extrasToJSON(rn.Labels)
	if err != nil {
		return err
	}
	harness, err := extrasToJSON(rn.HarnessVersions)
	if err != nil {
		return err
	}
	compat := rn.Compat.State
	if compat == "" {
		compat = runner.CompatCompatible
	}
	liveness := rn.Liveness
	if liveness == "" {
		liveness = runner.LivenessOffline
	}
	now := r.now().UTC()
	created := rn.CreatedAt
	if created.IsZero() {
		created = now
	}
	name := rn.Name
	if name == "" {
		name = string(rn.ID)
	}
	_, err = r.db.ExecContext(ctx, `INSERT INTO runners (`+runnerColumns+`, updated_at)
		VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
		ON CONFLICT(id) DO UPDATE SET
			name = excluded.name, built_in = excluded.built_in,
			compat_state = excluded.compat_state, compat_reason = excluded.compat_reason,
			liveness = excluded.liveness, last_contact = excluded.last_contact,
			os = excluded.os, runner_version = excluded.runner_version,
			harness_versions = excluded.harness_versions, workspace_root = excluded.workspace_root,
			updated_at = excluded.updated_at`,
		string(rn.ID), name, boolToInt(rn.BuiltIn), rn.StepLimit, labels, boolToInt(rn.Cordoned), boolToInt(rn.Draining),
		string(compat), rn.Compat.Reason, string(liveness), timeToNullString(rn.LastContact), rn.OS, rn.RunnerVersion,
		harness, rn.WorkspaceRoot, timeToString(created), timeToString(now))
	if err != nil {
		return fmt.Errorf("upserting runner: %w", err)
	}
	return nil
}

// UpdateConfig applies an operator patch atomically and returns the runner.
func (r *RunnerRegistryRepo) UpdateConfig(ctx context.Context, id runner.RunnerID, p runner.RunnerConfigPatch) (runner.Runner, error) {
	if p.StepLimit != nil && !runner.ValidStepLimit(*p.StepLimit) {
		return runner.Runner{}, fmt.Errorf("%w: %d", runner.ErrInvalidStepLimit, *p.StepLimit)
	}
	tx, err := r.db.BeginTx(ctx, nil)
	if err != nil {
		return runner.Runner{}, fmt.Errorf("beginning config update: %w", err)
	}
	defer func() { _ = tx.Rollback() }()
	now := timeToString(r.now())
	if err := requireRunnerRow(ctx, tx, id); err != nil {
		return runner.Runner{}, err
	}
	if p.StepLimit != nil {
		if _, err := tx.ExecContext(ctx, `UPDATE runners SET step_limit = ?, updated_at = ? WHERE id = ?`, *p.StepLimit, now, string(id)); err != nil {
			return runner.Runner{}, fmt.Errorf("updating step limit: %w", err)
		}
	}
	if p.Labels != nil {
		labels, err := extrasToJSON(p.Labels)
		if err != nil {
			return runner.Runner{}, err
		}
		if _, err := tx.ExecContext(ctx, `UPDATE runners SET labels = ?, updated_at = ? WHERE id = ?`, labels, now, string(id)); err != nil {
			return runner.Runner{}, fmt.Errorf("updating labels: %w", err)
		}
	}
	rn, err := scanRunner(tx.QueryRowContext(ctx, `SELECT `+runnerColumns+` FROM runners WHERE id = ?`, string(id)))
	if err != nil {
		return runner.Runner{}, fmt.Errorf("reading runner: %w", err)
	}
	if err := tx.Commit(); err != nil {
		return runner.Runner{}, fmt.Errorf("committing config update: %w", err)
	}
	return rn, nil
}

func requireRunnerRow(ctx context.Context, q groupQueryer, id runner.RunnerID) error {

		return fmt.Errorf("%w: %s", runner.ErrRunnerNotFound, id)
	}
	if err != nil {
		return fmt.Errorf("reading runner: %w", err)
	}
	return nil
}

// ReplaceSources sets the runner's advertised sources. Existing IDs keep their
// health (only advertised facts are refreshed); new IDs start unknown/available
// unless the caller supplies health; missing IDs are deleted.
func (r *RunnerRegistryRepo) ReplaceSources(ctx context.Context, id runner.RunnerID, sources []runner.Source) error {
	tx, err := r.db.BeginTx(ctx, nil)
	if err != nil {
		return fmt.Errorf("beginning source replace: %w", err)
	}
	defer func() { _ = tx.Rollback() }()
	if err := requireRunnerRow(ctx, tx, id); err != nil {
		return err
	}
	now := timeToString(r.now())
	keep := map[runner.SourceID]bool{}
	for _, s := range sources {
		want := runner.MakeSourceID(id, s.Kind, s.CredentialRef)
		if s.ID == "" {
			s.ID = want
		}
		if s.ID != want {
			return fmt.Errorf("runner registry: source id %q is not its identity %q", s.ID, want)
		}
		h := s.Health
		if h.Auth == "" {
			h.Auth = runner.AuthUnknown
		}
		if h.Capacity == "" {
			h.Capacity = runner.CapacityAvailable
		}
		families, err := encodeList(s.Families)
		if err != nil {
			return err
		}
		backends, err := encodeList(s.Backends)
		if err != nil {
			return err
		}
		models, err := encodeList(s.Models)
		if err != nil {
			return err
		}
		windows, err := encodeList(h.Windows)
		if err != nil {
			return err
		}
		if _, err := tx.ExecContext(ctx, `INSERT INTO runner_sources (`+sourceColumns+`, updated_at)
			VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
			ON CONFLICT(id) DO UPDATE SET families = excluded.families, backends = excluded.backends,
				models = excluded.models, updated_at = excluded.updated_at`,
			string(s.ID), string(id), string(s.Kind), s.CredentialRef, families, backends, models,
			string(h.Auth), string(h.Capacity), timeToNullString(h.ResetAt), headroomArg(h.Headroom), windows,
			boolToInt(h.Congested), timeToNullString(h.CongestedUntil), timeToNullString(h.LastProbeAt),
			string(h.LastOutcome), h.Detail, now); err != nil {
			return fmt.Errorf("upserting source %s: %w", s.ID, err)
		}
		keep[s.ID] = true
	}
	existing, err := listSources(ctx, tx, `SELECT `+sourceColumns+` FROM runner_sources WHERE runner_id = ?`, string(id))
	if err != nil {
		return err
	}
	for _, s := range existing {
		if keep[s.ID] {
			continue
		}
		if _, err := tx.ExecContext(ctx, `DELETE FROM runner_sources WHERE id = ?`, string(s.ID)); err != nil {
			return fmt.Errorf("deleting source %s: %w", s.ID, err)
		}
	}
	if err := tx.Commit(); err != nil {
		return fmt.Errorf("committing source replace: %w", err)
	}
	return nil
}

// RecordHealth overwrites one source's health.
func (r *RunnerRegistryRepo) RecordHealth(ctx context.Context, id runner.SourceID, h runner.SourceHealth) error {
	windows, err := encodeList(h.Windows)
	if err != nil {
		return err
	}
	res, err := r.db.ExecContext(ctx, `UPDATE runner_sources SET auth = ?, capacity = ?, reset_at = ?, headroom = ?, windows = ?,
		congested = ?, congested_until = ?, last_probe_at = ?, last_outcome = ?, detail = ?, updated_at = ? WHERE id = ?`,
		string(h.Auth), string(h.Capacity), timeToNullString(h.ResetAt), headroomArg(h.Headroom), windows,
		boolToInt(h.Congested), timeToNullString(h.CongestedUntil), timeToNullString(h.LastProbeAt),
		string(h.LastOutcome), h.Detail, timeToString(r.now()), string(id))
	if err != nil {
		return fmt.Errorf("recording source health: %w", err)
	}
	if n, err := res.RowsAffected(); err != nil || n == 0 {
		return fmt.Errorf("%w: %s", runner.ErrSourceNotFound, id)
	}
	return nil
}

// GetSource reads one source with its health.
func (r *RunnerRegistryRepo) GetSource(ctx context.Context, id runner.SourceID) (runner.Source, error) {
	s, err := scanSource(r.db.QueryRowContext(ctx, `SELECT `+sourceColumns+` FROM runner_sources WHERE id = ?`, string(id)))
	if errors.Is(err, sql.ErrNoRows) {
		return runner.Source{}, fmt.Errorf("%w: %s", runner.ErrSourceNotFound, id)
	}
	if err != nil {
		return runner.Source{}, fmt.Errorf("reading source: %w", err)
	}
	return s, nil
}

package sqlite

import (
	"context"
	"database/sql"
	"errors"
	"fmt"

	"github.com/arbitron/arbitron/domain/runner"
	"github.com/google/uuid"
)

// Compile-time: satisfies the pool event log port.
var _ runner.EventLog = (*PoolEventRepo)(nil)

// defaultPoolEventLimit applies when List is called with limit <= 0.
const defaultPoolEventLimit = 100

// PoolEventRepo implements runner.EventLog over SQLite (migration 041).
type PoolEventRepo struct {
	db *sql.DB
}

// NewPoolEventRepo creates a PoolEventRepo.
func NewPoolEventRepo(db *sql.DB) *PoolEventRepo { return &PoolEventRepo{db: db} }

func (r *PoolEventRepo) Append(ctx context.Context, ev runner.PoolEvent) error {
	if ev.ID == uuid.Nil || ev.Kind == "" {
		return errors.New("pool events: id and kind are required")
	}
	detail, err := toJSONString(ev.Detail)
	if err != nil {
		return fmt.Errorf("encoding pool event detail: %w", err)
	}
	var run sql.NullString
	if ev.RunID != nil {
		run = sql.NullString{String: uuidToString(*ev.RunID), Valid: true}
	}
	_, err = r.db.ExecContext(ctx, `INSERT INTO pool_events (id, at, kind, runner_id, source_id, run_id, cause_kind, cause_actor, cause_outcome, detail)
		VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
		uuidToString(ev.ID), timeToString(ev.At), ev.Kind, string(ev.RunnerID), string(ev.SourceID), run,
		ev.Cause.Kind, ev.Cause.Actor, string(ev.Cause.Outcome), detail)
	if err != nil {
		return fmt.Errorf("appending pool event: %w", err)
	}
	return nil
}

func (r *PoolEventRepo) List(ctx context.Context, limit int) ([]runner.PoolEvent, error) {
	if limit <= 0 {
		limit = defaultPoolEventLimit
	}
	rows, err := r.db.QueryContext(ctx, `SELECT id, at, kind, runner_id, source_id, run_id, cause_kind, cause_actor, cause_outcome, detail
		FROM pool_events ORDER BY seq DESC LIMIT ?`, limit)
	if err != nil {
		return nil, fmt.Errorf("listing pool events: %w", err)
	}
	defer rows.Close()
	var out []runner.PoolEvent
	for rows.Next() {
		var ev runner.PoolEvent
		var id, at, rid, sid, outcome, detail string
		var run sql.NullString
		if err := rows.Scan(&id, &at, &ev.Kind, &rid, &sid, &run, &ev.Cause.Kind, &ev.Cause.Actor, &outcome, &detail); err != nil {
			return nil, fmt.Errorf("scanning pool event: %w", err)
		}
		if ev.ID, err = parseUUID(id); err != nil {
			return nil, err
		}
		if ev.At, err = parseTime(at); err != nil {
			return nil, err
		}
		ev.RunnerID, ev.SourceID, ev.Cause.Outcome = runner.RunnerID(rid), runner.SourceID(sid), runner.OutcomeClass(outcome)
		if run.Valid {
			u, err := parseUUID(run.String)
			if err != nil {
				return nil, err
			}
			ev.RunID = &u
		}
		if ev.Detail, err = fromJSONString(detail); err != nil {
			return nil, fmt.Errorf("decoding pool event detail: %w", err)
		}
		out = append(out, ev)
	}
	return out, rows.Err()
}

// Prune keeps only the newest keep events.
func (r *PoolEventRepo) Prune(ctx context.Context, keep int) error {
	if keep < 0 {
		return fmt.Errorf("pool events: keep must be >= 0, got %d", keep)
	}
	if _, err := r.db.ExecContext(ctx,
		`DELETE FROM pool_events WHERE seq NOT IN (SELECT seq FROM pool_events ORDER BY seq DESC LIMIT ?)`, keep); err != nil {
		return fmt.Errorf("pruning pool events: %w", err)
	}
	return nil
}
