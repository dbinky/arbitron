# Remote Runners — Phase 3 Task 5: Durable step commit and committed-result replay (RR-10)

**Goal:** Make a node's successful completion one durable fact — execution record, variables, engine continuation, invocation commit, job completion under the ledger fence and the linked workspace checkpoint, written in a single SQLite transaction — and make the engine replay a committed result instead of executing the step again, while every persistence failure fails the step visibly (no more ignored `NodeExecution.Create` errors).

**Architecture:** `domain/engine/step_commit.go` defines the contract types (`StepCommit`, `JobFenceRef`, `StepCommitter`), `Engine.SetStepCommitter`, `ErrStepPersistence` and the two engine helpers (`commitNodeSuccess`, `committedReplay`). When a committer is installed, the run loop checks `CommittedResult` for the node's open invocation before executing it and, after a successful non-wait execution, hands the whole step to `CommitStep` instead of `nodeExecutions.Create` + `UpdateVariables` + `UpdateCurrentNode` + invocation commit. `adapter/sqlite/step_commit_repo.go` implements the port in one transaction, checking the job fence with the same `checkJobFenceTx` the checkpoint catalog uses.

**Tech Stack:** Go 1.25, `database/sql` + modernc SQLite (hand-written SQL, pattern `adapter/sqlite/audit_repo.go`), testify.

**Spec / Design:** spec §"Recovery", §"Workspace mobility"; design phase 3 §3.3 (completion boundary), §3.5 (server restart: "replay any committed result/continuation before creating an invocation"), §5 (row "Durable invocation/retry identity, terminal result/checkpoint and engine continuation"), §6 (engine recovery/commit seam), §8 checkpoint/ownership regression "crash after result/checkpoint commit but before engine advancement replays without repeating the completed step" and Failure "a stale result after failover is rejected". Contracts: `2026-11-20-contracts.md` §3 (InvocationStore), §4 (PlacementSink / PlacementRecord), §8 (Fence, ErrFenced), §16 (Commit protocol RR-10).

Semantics fixed by this task (consumed by P3 T6, T10, T13):
- `CommitStep` is all-or-nothing. On a stale or terminal-for-another-generation job fence it returns an error wrapping `runner.ErrFenced` and writes nothing.
- A non-terminal commit (`NextNodeID != ""`) marks the invocation committed and moves `instances.current_node_id` in the same transaction. A terminal commit (`NextNodeID == ""`) leaves the invocation **open** and `current_node_id` on the node, so a restart re-enters the node, `InvocationStore.Begin` returns that open invocation and `CommittedResult` replays it; the engine commits the terminal invocation only after `SetCompleted` succeeded.
- `CommittedResult` returns `Execution` as an identity stub (`ID`, `InstanceID`, `NodeID`); the full record is read through `instance.NodeExecutionRepository`.
- When `StepCommit.CheckpointRef == ""` and `JobFence != nil`, the adapter stores the newest `workspace_checkpoints` row published by that job generation (`runner.CheckpointRef.String()` format), which is how P3 T6's `CommitBuiltIn` / `AcceptPush` results are linked to the step.

```go
package engine

import (
	"context"
	"errors"

	"github.com/arbitron/arbitron/domain/instance"
	"github.com/google/uuid"
)

// ErrStepPersistence marks a step whose durable record could not be written.
// The run loop halts the run as failed: a step that is not durably recorded
// must never be treated as done (RR-10).
var ErrStepPersistence = errors.New("engine: step persistence failed")

// StepCommit is the single durable fact that a node finished: its execution
// record, the run's variables after the node's output was applied, the engine
// continuation and the job/checkpoint it ran under (RR-10).
type StepCommit struct {
	RunID, InvocationID uuid.UUID
	NodeID              string
	Execution           *instance.NodeExecution
	Variables           map[string]any
	// NextNodeID is the continuation; "" means the node was the run's last step.
	NextNodeID string
	// CheckpointRef is the workspace checkpoint the result is linked to. When
	// empty and JobFence is set, the committer links the checkpoint published
	// under that job generation.
	CheckpointRef string
	// JobFence is the ledger job the step ran under; nil for steps that were
	// never dispatched (transform, conditional, waits).
	JobFence *JobFenceRef
}

// JobFenceRef names the ledger job generation a result belongs to.
type JobFenceRef struct {
	JobID      uuid.UUID
	RunnerID   string
	Generation int64
}

// StepCommitter persists StepCommits atomically and answers whether an
// invocation already committed (restart replay).
type StepCommitter interface {
	// CommitStep writes the whole commit or nothing. A stale job fence is an
	// error and writes nothing.
	CommitStep(ctx context.Context, c StepCommit) error
	// CommittedResult returns the commit recorded for invocationID, if any.
	CommittedResult(ctx context.Context, invocationID uuid.UUID) (*StepCommit, bool, error)
}

// SetStepCommitter installs the durable step committer. nil restores the
// per-write persistence path (whose errors also fail the step).
func (e *Engine) SetStepCommitter(c StepCommitter) {
	e.mu.Lock()
	defer e.mu.Unlock()
	e.stepCommitter = c
}

func (e *Engine) committer() StepCommitter {
	e.mu.Lock()
	defer e.mu.Unlock()
	return e.stepCommitter
}
```

```go
func (m *memCommitter) CommitStep(ctx context.Context, c engine.StepCommit) error {
	m.mu.Lock()
	if m.failErr != nil && c.NodeID == m.failNode {
		m.mu.Unlock()
		return m.failErr
	}
	if c.Execution == nil || c.InvocationID == uuid.Nil || c.RunID == uuid.Nil {
		m.mu.Unlock()
		return errors.New("mem committer: incomplete commit")
	}
	if c.JobFence != nil {
		if gen, ok := m.liveGen[c.JobFence.JobID]; !ok || gen != c.JobFence.Generation {
			m.mu.Unlock()
			return fmt.Errorf("%w: job %s gen %d", errMemFenced, c.JobFence.JobID, c.JobFence.Generation)
		}
	}
	m.mu.Unlock()

	if err := m.execs.Create(ctx, c.Execution); err != nil {
		return err
	}
	if err := m.inst.UpdateVariables(ctx, c.RunID, c.Variables); err != nil {
		return err
	}
	if c.NextNodeID != "" {
		if err := m.inst.UpdateCurrentNode(ctx, c.RunID, c.NextNodeID); err != nil {
			return err
		}
		if err := m.invs.Commit(ctx, c.InvocationID); err != nil {
			return err
		}
	}
	m.mu.Lock()
	m.commits[c.InvocationID] = c
	m.order = append(m.order, c)
	after := m.after
	m.mu.Unlock()
	if after != nil {
		return after(c)
	}
	return nil
}

func (m *memCommitter) CommittedResult(_ context.Context, invocationID uuid.UUID) (*engine.StepCommit, bool, error) {
	m.mu.Lock()
	defer m.mu.Unlock()
	c, ok := m.commits[invocationID]
	if !ok {
		return nil, false, nil
	}
	cp := c
	return &cp, true, nil
}

func (m *memCommitter) committed() []engine.StepCommit {
	m.mu.Lock()
	defer m.mu.Unlock()
	return append([]engine.StepCommit(nil), m.order...)
}
```

```go
// commitNodeSuccess applies a successful, non-wait node result through the
// step committer: output into vars, continuation, then one atomic CommitStep.
// The commit is detached from cancellation — a cancel landing after the node
// finished must not lose its result. Any commit error fails the run.
func (e *Engine) commitNodeSuccess(
	ctx context.Context,
	committer StepCommitter,
	instanceID uuid.UUID,
	nodeID string,
	node workflow.Node,
	result *NodeResult,
	nodeExec *instance.NodeExecution,
	invocationID uuid.UUID,
	vars *variable.VariableStore,
	adj map[string][]edgeTarget,
) (nextNodeID string, halt bool) {
	if result != nil && result.Output != nil {
		vars.SetNodeOutput(nodeID, result.Output)
	}
	outputPort := ""
	if result != nil {
		outputPort = result.OutputPort
	}
	next := e.findNextNode(nodeID, outputPort, adj)

	fence, err := jobFenceFromExecution(nodeExec)
	if err == nil && nodeExec == nil {
		err = fmt.Errorf("%w: node %q finished without an execution record", ErrStepPersistence, nodeID)
	}
	if err == nil {
		err = committer.CommitStep(context.WithoutCancel(ctx), StepCommit{
			RunID:        instanceID,
			InvocationID: invocationID,
			NodeID:       nodeID,
			Execution:    nodeExec,
			Variables:    vars.Snapshot(),
			NextNodeID:   next,
			JobFence:     fence,
		})
	}
	if err != nil {
		slog.Error("step commit failed; failing run", "instanceID", instanceID, "nodeID", nodeID, "error", err)
		e.eventBus.Publish(instanceID, Event{
			Type:      EventTypeNodeFailed,
			Timestamp: time.Now(),
			Data:      map[string]any{"nodeId": nodeID, "nodeType": string(node.Type), "error": err.Error()},
		})
		e.failInstance(ctx, instanceID, nodeID)
		return "", true
	}
	e.eventBus.Publish(instanceID, Event{
		Type:      EventTypeNodeCompleted,
		Timestamp: time.Now(),
		Data:      map[string]any{"nodeId": nodeID, "nodeType": string(node.Type)},
	})
	return next, false
}

// jobFenceFromExecution derives the job fence from the attempt's placement
// history: the last record is the generation that produced the result. No
// history means the step was not dispatched. A record that does not name a
// complete fence fails closed.
func jobFenceFromExecution(exec *instance.NodeExecution) (*JobFenceRef, error) {
	if exec == nil || len(exec.PlacementHistory) == 0 {
		return nil, nil
	}
	last := exec.PlacementHistory[len(exec.PlacementHistory)-1]
	jobID, err := uuid.Parse(last.JobID)
	if err != nil || jobID == uuid.Nil || last.RunnerID == "" || last.Generation < 1 {
		return nil, fmt.Errorf("%w: placement record (job %q, runner %q, generation %d) does not name a job fence",
			ErrStepPersistence, last.JobID, last.RunnerID, last.Generation)
	}
	return &JobFenceRef{JobID: jobID, RunnerID: last.RunnerID, Generation: last.Generation}, nil
}
```

```go
// committedReplay looks up a committed result for the node's open visit. It
// returns nil when no committer is installed or nothing was committed. Begin
// returns the same open invocation executeWithRetry will reuse, so asking
// never creates a second visit.
func (e *Engine) committedReplay(ctx context.Context, instanceID uuid.UUID, nodeID string) (*StepCommit, error) {
	committer := e.committer()
	if committer == nil {
		return nil, nil
	}
	inv, err := e.invocationStore().Begin(ctx, instanceID, nodeID)
	if err != nil {
		return nil, fmt.Errorf("%w: opening invocation for node %q: %v", ErrStepPersistence, nodeID, err)

	if err != nil {
		return nil, fmt.Errorf("%w: reading committed result for node %q: %v", ErrStepPersistence, nodeID, err)
	}
	if !ok {
		return nil, nil
	}
	if c.InvocationID == uuid.Nil {
		c.InvocationID = inv.ID
	}
	return c, nil
}
```

```go
import (
```

```go
// StepCommitRepo implements engine.StepCommitter (RR-10): a node's execution
// record, the run's variables and continuation, the invocation commit, the
// job's completion under its fence and the linked checkpoint are written in a
// single transaction. Hand-written SQL, modeled on AuditRepo.
type StepCommitRepo struct {
	db  *sql.DB
	now func() time.Time
}

var _ engine.StepCommitter = (*StepCommitRepo)(nil)

// NewStepCommitRepo creates a StepCommitRepo over db.
func NewStepCommitRepo(db *sql.DB) *StepCommitRepo {
	return &StepCommitRepo{db: db, now: time.Now}
}

// CommitStep writes c atomically. A stale job fence returns an error wrapping
// runner.ErrFenced and writes nothing.
func (r *StepCommitRepo) CommitStep(ctx context.Context, c engine.StepCommit) error {
	if c.RunID == uuid.Nil || c.InvocationID == uuid.Nil || c.NodeID == "" || c.Execution == nil {
		return errors.New("step commit: run, invocation, node and execution are required")
	}
	if c.Execution.InstanceID == uuid.Nil || c.Execution.NodeID != c.NodeID {
		return fmt.Errorf("step commit: execution record does not belong to node %q", c.NodeID)
	}
	varsJSON, err := toJSONString(c.Variables)
	if err != nil {
		return fmt.Errorf("step commit: marshalling variables: %w", err)
	}

	tx, err := r.db.BeginTx(ctx, nil)
	if err != nil {
		return fmt.Errorf("step commit: begin: %w", err)
	}
	defer func() { _ = tx.Rollback() }()
	now := r.now().UTC()

	var fence *runner.Fence
	if c.JobFence != nil {
		f := runner.Fence{JobID: c.JobFence.JobID, RunnerID: runner.RunnerID(c.JobFence.RunnerID), Generation: c.JobFence.Generation}
		if err := checkJobFenceTx(ctx, tx, f, "completed"); err != nil {
			return fmt.Errorf("step commit: %w", err)
		}
		// The dispatcher normally completed the job already; this closes the
		// window where the result is committed before Complete landed.
		if _, err := tx.ExecContext(ctx,
			`UPDATE runner_jobs SET state = 'completed' WHERE id = ? AND state IN ('offered', 'running')`,
			uuidToString(f.JobID)); err != nil {
			return fmt.Errorf("step commit: completing job: %w", err)
		}
		fence = &f
	}

	var committedAt sql.NullString
	err = tx.QueryRowContext(ctx,
		`SELECT committed_at FROM node_invocations WHERE id = ? AND run_id = ? AND node_id = ?`,
		uuidToString(c.InvocationID), uuidToString(c.RunID), c.NodeID).Scan(&committedAt)
	if errors.Is(err, sql.ErrNoRows) {
		return fmt.Errorf("step commit: %w: %s for node %q", engine.ErrInvocationNotFound, c.InvocationID, c.NodeID)
	}
	if err != nil {
		return fmt.Errorf("step commit: reading invocation: %w", err)
	}
	if committedAt.Valid {
		return fmt.Errorf("step commit: invocation %s is already committed", c.InvocationID)
	}

	if err := insertNodeExecutionTx(ctx, tx, c.Execution); err != nil {
		return fmt.Errorf("step commit: %w", err)
	}

	res, err := tx.ExecContext(ctx,
		`UPDATE instances SET variables = ?, updated_at = strftime('%Y-%m-%dT%H:%M:%SZ') WHERE id = ?`,
		varsJSON, uuidToString(c.RunID))
	if err != nil {
		return fmt.Errorf("step commit: updating variables: %w", err)
	}
	if n, err := res.RowsAffected(); err != nil || n != 1 {
		return fmt.Errorf("step commit: run %s not found", c.RunID)
	}
	if c.NextNodeID != "" {
		if _, err := tx.ExecContext(ctx,
			`UPDATE instances SET current_node_id = ?, updated_at = strftime('%Y-%m-%dT%H:%M:%SZ') WHERE id = ?`,
			c.NextNodeID, uuidToString(c.RunID)); err != nil {
			return fmt.Errorf("step commit: advancing run: %w", err)
		}
		if _, err := tx.ExecContext(ctx,
			`UPDATE node_invocations SET committed_at = ? WHERE id = ? AND committed_at IS NULL`,
			timeToString(now), uuidToString(c.InvocationID)); err != nil {
			return fmt.Errorf("step commit: committing invocation: %w", err)
		}
	}

	checkpointRef := c.CheckpointRef
	if checkpointRef == "" && fence != nil {
		var version int64
		var hash string
		err := tx.QueryRowContext(ctx,
			`SELECT version, manifest_hash FROM workspace_checkpoints
			 WHERE run_id = ? AND job_id = ? AND job_generation = ? ORDER BY version DESC LIMIT 1`,
			uuidToString(c.RunID), uuidToString(fence.JobID), fence.Generation).Scan(&version, &hash)
		switch {
		case err == nil:
			checkpointRef = runner.CheckpointRef{RunID: c.RunID, Version: version, ManifestHash: hash}.String()
		case !errors.Is(err, sql.ErrNoRows):
			return fmt.Errorf("step commit: reading checkpoint: %w", err)
		}
	}

	var jobID, jobRunner sql.NullString
	var jobGen sql.NullInt64
	if c.JobFence != nil {
		jobID = sql.NullString{String: uuidToString(c.JobFence.JobID), Valid: true}
		jobRunner = sql.NullString{String: c.JobFence.RunnerID, Valid: true}
		jobGen = sql.NullInt64{Int64: c.JobFence.Generation, Valid: true}
	}
	if _, err := tx.ExecContext(ctx,
		`INSERT INTO step_commits (invocation_id, run_id, node_id, execution_id, next_node_id, variables, checkpoint_ref, job_id, job_runner_id, job_generation, committed_at)
		 VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
		uuidToString(c.InvocationID), uuidToString(c.RunID), c.NodeID, uuidToString(c.Execution.ID),
		c.NextNodeID, varsJSON, checkpointRef, jobID, jobRunner, jobGen, timeToString(now)); err != nil {
		return fmt.Errorf("step commit: recording commit: %w", err)
	}
	if err := tx.Commit(); err != nil {
		return fmt.Errorf("step commit: commit: %w", err)
	}
	return nil
}

// CommittedResult returns the commit recorded for invocationID. Execution is
// an identity stub (ID, InstanceID, NodeID).
func (r *StepCommitRepo) CommittedResult(ctx context.Context, invocationID uuid.UUID) (*engine.StepCommit, bool, error) {
	var runStr, nodeID, execStr, next, varsJSON, ref string
	var jobID, jobRunner sql.NullString
	var jobGen sql.NullInt64
	err := r.db.QueryRowContext(ctx,
		`SELECT run_id, node_id, execution_id, next_node_id, variables, checkpoint_ref, job_id, job_runner_id, job_generation
		 FROM step_commits WHERE invocation_id = ?`, uuidToString(invocationID)).
		Scan(&runStr, &nodeID, &execStr, &next, &varsJSON, &ref, &jobID, &jobRunner, &jobGen)
	if errors.Is(err, sql.ErrNoRows) {
		return nil, false, nil
	}
	if err != nil {
		return nil, false, fmt.Errorf("reading step commit: %w", err)
	}
	runID, err := parseUUID(runStr)
	if err != nil {
		return nil, false, fmt.Errorf("step commit run id: %w", err)
	}
	execID, err := parseUUID(execStr)
	if err != nil {
		return nil, false, fmt.Errorf("step commit execution id: %w", err)
	}
	vars := map[string]any{}
	if err := json.Unmarshal([]byte(varsJSON), &vars); err != nil {
		return nil, false, fmt.Errorf("step commit variables: %w", err)
	}
	c := &engine.StepCommit{
		RunID: runID, InvocationID: invocationID, NodeID: nodeID,
		Execution:  &instance.NodeExecution{ID: execID, InstanceID: runID, NodeID: nodeID},
		Variables:  vars,
		NextNodeID: next, CheckpointRef: ref,
	}
	if jobID.Valid {
		jid, err := parseUUID(jobID.String)
		if err != nil {
			return nil, false, fmt.Errorf("step commit job id: %w", err)
		}
		c.JobFence = &engine.JobFenceRef{JobID: jid, RunnerID: jobRunner.String, Generation: jobGen.Int64}
	}
	return c, true, nil
}

// insertNodeExecutionTx writes one node_executions row, including the P1 T14
// placement columns (migration 042), inside tx.
func insertNodeExecutionTx(ctx context.Context, tx *sql.Tx, ex *instance.NodeExecution) error {
	resolved, err := toNullJSONString(ex.ResolvedInputs)
	if err != nil {
		return fmt.Errorf("marshalling resolved inputs: %w", err)
	}
	output, err := toNullJSONString(ex.Output)
	if err != nil {
		return fmt.Errorf("marshalling output: %w", err)
	}
	errJSON, err := toNullJSONString(ex.Error)
	if err != nil {
		return fmt.Errorf("marshalling error: %w", err)
	}
	conv, err := toNullJSONString(ex.LLMConversation)
	if err != nil {
		return fmt.Errorf("marshalling llm conversation: %w", err)
	}
	history := ex.PlacementHistory
	if history == nil {
		history = []instance.PlacementRecord{}
	}
	historyJSON, err := json.Marshal(history)
	if err != nil {
		return fmt.Errorf("marshalling placement history: %w", err)
	}
	if _, err := tx.ExecContext(ctx,
		`INSERT INTO node_executions (id, instance_id, node_id, attempt, status, resolved_inputs, output, error,
		   llm_conversation, llm_model, llm_tokens_in, llm_tokens_out, script_stdout, script_stderr, script_exit_code,
		   started_at, completed_at, runner_id, source_id, placement_reason, placement_history)
		 VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
		uuidToString(ex.ID), uuidToString(ex.InstanceID), ex.NodeID, int64(ex.Attempt), string(ex.Status),
		resolved, output, errJSON, conv, ptrToNullString(ex.LLMModel),
		intPtrToNullInt64(ex.LLMTokensIn), intPtrToNullInt64(ex.LLMTokensOut),
		ptrToNullString(ex.ScriptStdout), ptrToNullString(ex.ScriptStderr), intPtrToNullInt64(ex.ScriptExitCode),
		timeToString(ex.StartedAt), timeToNullString(ex.CompletedAt),
		ptrToNullString(ex.RunnerID), ptrToNullString(ex.SourceID), ptrToNullString(ex.PlacementReason),
		string(historyJSON)); err != nil {
		return fmt.Errorf("inserting node execution: %w", err)
	}
	return nil
}
```
