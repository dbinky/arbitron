---

**Goal:** One in-process `runner.Hub` owns every connected runner session: it admits runners against a compatibility table, records their sources, source health and telemetry from heartbeats, turns heartbeat gaps (never socket drops) into `suspect`/`lost`, offers jobs and waits for accept/output/result with per-job generation fencing, expires unaccepted offers, replays held jobs on reconnect without output gaps or duplicates, and disconnects revoked runners immediately.

**Architecture:** Pure domain code in `domain/runner` with no transport: the WebSocket endpoint (P2 T5) translates `runnerproto` envelopes to `runner.Inbound` and calls `Admit`/`Handle`/`Disconnect`; the remote `StepRunner` (P2 T6) calls `Run`. Persistent state lives in the Phase 1 `Registry` and `Ledger`; only sessions, contact times, job waiters and the bounded telemetry window are in memory. `Tick` is the single liveness/expiry pass (driven by a fake clock in tests, by `Start` in production). Test fixtures (real SQLite pool, fake clock, recording conn, recording health transitioner, in-memory runner) live in `adapter/runner/hubtest`, reused by P2 T5/T6/T10.

**Tech Stack:** Go 1.25.5, `sync`, `encoding/json`, testify; real SQLite (`adapter/sqlite` P1 T5/T6 repos) in tests.

**Spec / Design:** Spec "Execution and placement", "Recovery and session continuity" (brief interruptions do not move work), "Compatibility", "Administration and visibility" (incompatible runners visible, no work); design phase 2 §3.1 (invariants: liveness from heartbeats only; ~30 s suspect, ~120 s lost; generation fencing; unaccepted offer TTL), §3.2 (admission and compatibility table, chat_only on root mismatch, darwin only, scratch root outside workspace root), §3.4 (bounded telemetry window), §3.5 (offer/accept/output/result/ack/fenced), §3.6 (held-job reconciliation on reconnect, resend pending offers), §7 (error table), §8 (testing bullets owned below). Contracts: `docs/plans/remote-runners/2026-11-20-contracts.md` §6 (vocabulary), §6 Outcomes, §7 (`Telemetry`, `Place`), §8 (`Registry`, `Ledger`, `PoolEvent`), §9 (event kinds), §11 (`StepJob`, `Assignment`, `StepOutput`, `HealthTransitioner`, `PoolNotifier`), §15 (Hub, `RunnerConn`, `Outbound`, `Inbound`, `HubDeps`, `HubConfig`, `CompatTable`).

// domain/runner/hub_compat.go
package runner

import (
	"fmt"
	"path/filepath"
	"sort"
	"strconv"
	"strings"
	"time"

	"github.com/arbitron/arbitron/domain/config"
)

// CompatTable is the server's admission table (design phase 2 §3.2).
// Harness maps a harness name ("claude", "codex") to its minimum version; a
// harness the runner does not report is not checked (the runner simply has no
// such source). RunnerMin "unreleased" disables remote execution while preserving built-in operation.
type CompatTable struct {
	ProtocolMin, ProtocolMax int
	RunnerMin                string
	RunnerMaxExclusive       string
	Harness                  map[string]string
}

// Check classifies a hello. Incompatible runners stay visible but receive no
// work; chat_only runners receive only hermetic (AffinityNone) work because
// their execution root differs from the server's.
func (c CompatTable) Check(h HelloInfo, serverRoot string) Compatibility {
	if h.OS != "darwin" {
		return Compatibility{State: CompatIncompatible, Reason: fmt.Sprintf("unsupported OS %q: remote runners require macOS (darwin)", h.OS)}
	}
	if h.ProtocolVersion < c.ProtocolMin || h.ProtocolVersion > c.ProtocolMax {
		return Compatibility{State: CompatIncompatible, Reason: fmt.Sprintf("protocol version %d outside supported range %d-%d", h.ProtocolVersion, c.ProtocolMin, c.ProtocolMax)}
	}
	if c.RunnerMin == "" || c.RunnerMaxExclusive == "" || !versionAtLeast(h.RunnerVersion, c.RunnerMin) || versionAtLeast(h.RunnerVersion, c.RunnerMaxExclusive) {
		return Compatibility{State: CompatIncompatible, Reason: fmt.Sprintf("runner version %q is older than the minimum %q", h.RunnerVersion, c.RunnerMin)}
	}
	names := make([]string, 0, len(c.Harness))
	for name := range c.Harness {
		names = append(names, name)
	}
	sort.Strings(names)
	for _, name := range names {
		have, ok := h.HarnessVersions[name]
		if !ok || strings.TrimSpace(have) == "" {
			continue
		}
		if !versionAtLeast(have, c.Harness[name]) {
			return Compatibility{State: CompatIncompatible, Reason: fmt.Sprintf("harness %s %q is older than the minimum %q", name, have, c.Harness[name])}
		}
	}
	if h.ScratchRoot == "" || !filepath.IsAbs(h.ScratchRoot) {
		return Compatibility{State: CompatIncompatible, Reason: fmt.Sprintf("scratch root %q must be an absolute path", h.ScratchRoot)}
	}
	if h.WorkspaceRoot != "" && filepath.IsAbs(h.WorkspaceRoot) && within(filepath.Clean(h.ScratchRoot), filepath.Clean(h.WorkspaceRoot)) {
		return Compatibility{State: CompatIncompatible, Reason: fmt.Sprintf("scratch root %q must be outside the workspace root %q", h.ScratchRoot, h.WorkspaceRoot)}
	}
	if serverRoot == "" || h.WorkspaceRoot == "" || !filepath.IsAbs(h.WorkspaceRoot) ||
		filepath.Clean(h.WorkspaceRoot) != filepath.Clean(serverRoot) {
		return Compatibility{State: CompatChatOnly, Reason: fmt.Sprintf("workspace root %q does not match the server root %q: hermetic chat only", h.WorkspaceRoot, serverRoot)}
	}
	return Compatibility{State: CompatCompatible}
}

// within reports whether child is parent or below it.
func within(child, parent string) bool {
	rel, err := filepath.Rel(parent, child)
	if err != nil {
		return false
	}
	return rel == "." || (rel != ".." && !strings.HasPrefix(rel, ".."+string(filepath.Separator)))
}

// parseVersion reads the leading dotted-numeric version of strings such as
// "2.1.30 (Claude Code)", "v0.3.0" or "0.50.0-beta".
func parseVersion(v string) ([]int, bool) {
	v = strings.TrimSpace(v)
	if i := strings.IndexAny(v, " \t("); i >= 0 {
		v = v[:i]
	}
	v = strings.TrimPrefix(v, "v")
	if i := strings.IndexAny(v, "-+"); i >= 0 {
		v = v[:i]
	}
	if v == "" {
		return nil, false
	}
	parts := strings.Split(v, ".")
	out := make([]int, 0, len(parts))
	for _, p := range parts {
		n, err := strconv.Atoi(p)
		if err != nil || n < 0 {
			return nil, false
		}
		out = append(out, n)
	}
	return out, true
}

// versionAtLeast is false when either side is unparseable (fail closed).
func versionAtLeast(have, min string) bool {
	h, ok1 := parseVersion(have)
	m, ok2 := parseVersion(min)
	if !ok1 || !ok2 {
		return false
	}
	at := func(v []int, i int) int {
		if i < len(v) {
			return v[i]
		}
		return 0
	}
	for i := 0; i < len(h) || i < len(m); i++ {
		if a, b := at(h, i), at(m, i); a != b {
			return a > b
		}
	}
	return true
}

// HubConfigFromRunners converts the validated runners config (P2 T2) into the
// hub's durations. serverRoot is made absolute so the root comparison in Check
// is meaningful for the default relative "./workspaces".
func HubConfigFromRunners(rc config.RunnersConfig, serverRoot string, compat CompatTable) HubConfig {
	root := serverRoot
	if serverRoot != "" {
		if abs, err := filepath.Abs(serverRoot); err == nil {
			root = abs
		}
	}
	return HubConfig{
		Heartbeat:           time.Duration(rc.HeartbeatSeconds) * time.Second,
		SuspectAfter:        time.Duration(rc.SuspectAfterSeconds) * time.Second,
		LostAfter:           time.Duration(rc.LostAfterSeconds) * time.Second,
		OfferTTL:            time.Duration(rc.OfferTTLSeconds) * time.Second,
		TelemetryWindow:     rc.TelemetryWindow,
		ServerWorkspaceRoot: root,
		Compat:              compat,
	}
}

func TestHubHeartbeat_SourceHealthTransitions(t *testing.T) {

	sid := runner.MakeSourceID("mac-a", runner.CredAnthropicSubscription, "")

	beat := func(hl runner.SourceHealth) {
		t.Helper()
		src := hubtest.AnthropicSource()
		src.Health = hl
		require.NoError(t, h.Handle(ctx, s, heartbeat(src)))
	}

	beat(runner.SourceHealth{Auth: runner.AuthLoggedOut, Capacity: runner.CapacityAvailable})
	beat(runner.SourceHealth{Auth: runner.AuthLoggedOut, Capacity: runner.CapacityAvailable})
	calls := p.Health.Calls()
	require.Len(t, calls, 1, "a persisting logged_out is applied once")
	assert.Equal(t, sid, calls[0].Source)
	assert.Equal(t, runner.OutcomeLoggedOut, calls[0].Outcome.Class)
	assert.False(t, place(t, p, anthropicChat(runner.AffinityNone)).Placed, "logged-out source gets no work")

	beat(runner.SourceHealth{Auth: runner.AuthOK, Capacity: runner.CapacityAvailable})
	src, err := p.Registry.GetSource(ctx, sid)
	require.NoError(t, err)
	assert.Equal(t, runner.AuthOK, src.Health.Auth)
	assert.Len(t, p.EventsOf(t, "source.recovered"), 1)

	reset := p.Clock.Now().Add(3 * time.Hour)
	beat(runner.SourceHealth{Auth: runner.AuthOK, Capacity: runner.CapacityExhausted, ResetAt: &reset})
	calls = p.Health.Calls()
	require.Len(t, calls, 2)
	assert.Equal(t, runner.OutcomeExhausted, calls[1].Outcome.Class)
	require.NotNil(t, calls[1].Outcome.ResetAt)
	assert.True(t, calls[1].Outcome.ResetAt.Equal(reset))

	beat(runner.SourceHealth{Auth: runner.AuthOK, Capacity: runner.CapacityAvailable, Headroom: f64(0.4)})
	src, err = p.Registry.GetSource(ctx, sid)
	require.NoError(t, err)
	assert.Equal(t, runner.CapacityProbation, src.Health.Capacity, "recovery from exhaustion goes through probation")
	assert.Nil(t, src.Health.ResetAt)
	require.NotNil(t, src.Health.Headroom)
	assert.Equal(t, 0.4, *src.Health.Headroom)
}

	Inbox RemoteInbox
	Registry Registry
	Ledger   Ledger
	Events   *EventRecorder
	Health   HealthTransitioner
	Notifier *PoolNotifier
	Clock    func() time.Time

// Hub owns every remote runner session in this server process.
type Hub struct {
	reg      Registry
	ledger   Ledger
	events   *EventRecorder
	health   HealthTransitioner
	notifier *PoolNotifier
	now      func() time.Time
	cfg      HubConfig

	// admitMu serialises admission with loss/revocation so a revoked or lost
	// runner can never be re-admitted halfway through being torn down.
	admitMu sync.Mutex

	mu        sync.Mutex
	sessions  map[RunnerID]*Session
	contact   map[RunnerID]time.Time
	revoked   map[RunnerID]bool
	telemetry map[RunnerID][]Telemetry

// NewHub validates deps and config and builds the hub.
func NewHub(d HubDeps) *Hub {
	if d.Registry == nil || d.Ledger == nil || d.Events == nil || d.Health == nil || d.Notifier == nil {
		panic("runner: NewHub requires Registry, Ledger, Events, Health and Notifier")
	}
	c := d.Config
	if c.Heartbeat <= 0 || c.SuspectAfter <= 0 || c.LostAfter <= c.SuspectAfter || c.OfferTTL <= 0 || c.TelemetryWindow < 1 {
		panic(fmt.Sprintf("runner: invalid hub config %+v", c))
	}
	if d.Clock == nil {
		d.Clock = time.Now
	}
	return &Hub{
		reg: d.Registry, ledger: d.Ledger, events: d.Events, health: d.Health, notifier: d.Notifier, now: d.Clock, cfg: c,
		sessions: map[RunnerID]*Session{}, contact: map[RunnerID]time.Time{}, revoked: map[RunnerID]bool{},
		telemetry: map[RunnerID][]Telemetry{}, jobs: map[uuid.UUID]*remoteJob{},
	}

func remoteKind(k CredentialKind) bool {
	return k == CredAnthropicSubscription || k == CredOpenAISubscription || k == CredLunaRoute
}

// reportedSources converts reports to registry sources: remote kinds only,
// de-duplicated by source id, with neutral health (applyReportedHealth then
// records the reported state through the proper transitions).
func reportedSources(id RunnerID, reps []SourceReport) []Source {
	out := make([]Source, 0, len(reps))
	seen := map[SourceID]bool{}
	for _, r := range reps {
		if !remoteKind(r.Kind) {
			continue
		}
		sid := MakeSourceID(id, r.Kind, r.CredentialRef)
		if seen[sid] {
			continue
		}
		seen[sid] = true
		out = append(out, Source{ID: sid, RunnerID: id, Kind: r.Kind, CredentialRef: r.CredentialRef,
			Families: r.Families, Backends: r.Backends, Models: r.Models,
			Health: SourceHealth{Auth: AuthUnknown, Capacity: CapacityAvailable}})
	}
	return out
}

func sourceSignature(s Source) string {
	fam := make([]string, 0, len(s.Families))
	for _, f := range s.Families {
		fam = append(fam, string(f))
	}
	sort.Strings(fam)
	be := append([]string(nil), s.Backends...)
	sort.Strings(be)
	md := append([]string(nil), s.Models...)
	sort.Strings(md)
	return string(s.ID) + "|" + strings.Join(fam, ",") + "|" + strings.Join(be, ",") + "|" + strings.Join(md, ",")
}

func sameSourceSet(a, b []Source) bool {
	if len(a) != len(b) {
		return false
	}
	sa := make([]string, 0, len(a))
	for _, s := range a {
		sa = append(sa, sourceSignature(s))
	}
	sb := make([]string, 0, len(b))
	for _, s := range b {
		sb = append(sb, sourceSignature(s))
	}
	sort.Strings(sa)
	sort.Strings(sb)
	return reflect.DeepEqual(sa, sb)
}

// applyReportedHealth maps runner-reported health onto the registry. New
// logged_out / exhausted states go through the HealthTransitioner (which owns
// alerts and events); other observations are merged directly. Recovery from
// exhaustion passes through probation; clearing logged_out emits source.recovered.
func (h *Hub) applyReportedHealth(ctx context.Context, id RunnerID, reps []SourceReport, now time.Time) error {
	for _, rep := range reps {
		if !remoteKind(rep.Kind) {
			continue
		}
		sid := MakeSourceID(id, rep.Kind, rep.CredentialRef)
		cur, err := h.reg.GetSource(ctx, sid)
		if err != nil {
			return fmt.Errorf("hub: read source %s: %w", sid, err)
		}
		rh := rep.Health
		switch {
		case rh.Auth == AuthLoggedOut && cur.Health.Auth != AuthLoggedOut:
			if err := h.health.ApplyOutcome(ctx, sid, SourceOutcome{Class: OutcomeLoggedOut}); err != nil {
				return err
			}
		case rh.Capacity == CapacityExhausted && cur.Health.Capacity != CapacityExhausted:
			if err := h.health.ApplyOutcome(ctx, sid, SourceOutcome{Class: OutcomeExhausted, ResetAt: rh.ResetAt, Windows: rh.Windows, HeadroomHint: rh.Headroom}); err != nil {
				return err
			}
		default:
			next := cur.Health
			recovered := false
			if rh.Auth == AuthOK && cur.Health.Auth != AuthOK {
				recovered = cur.Health.Auth == AuthLoggedOut
				next.Auth = AuthOK
			}
			if rh.Capacity == CapacityAvailable && cur.Health.Capacity == CapacityExhausted {
				next.Capacity = CapacityProbation
				next.ResetAt = nil
			}
			if rh.Headroom != nil {
				next.Headroom = rh.Headroom
			}
			if rh.Windows != nil {
				next.Windows = rh.Windows
			}
			probed := now
			if rh.LastProbeAt != nil {
				probed = *rh.LastProbeAt
			}
			changed := !reflect.DeepEqual(next, cur.Health)
			next.LastProbeAt = &probed
			if changed {
				if err := h.reg.RecordHealth(ctx, sid, next); err != nil {
					return err
				}
				h.notifier.Notify()
			}
			if recovered {
				h.record(ctx, evtSourceRecovered, id, sid, Cause{Kind: "outcome", Actor: string(id)}, map[string]any{"kind": string(rep.Kind)})
			}
		}
	}
	return nil
}

// Admit admits an authenticated, machine-proven runner (the endpoint has
// already verified key and machine binding). On success the runner is
// online, its sources are recorded, and MsgAdmitted has been sent first.
func (h *Hub) Admit(ctx context.Context, key RunnerKey, conn RunnerConn, hello HelloInfo) (*Session, error) {
	if conn == nil {
		return nil, fmt.Errorf("runner: Admit requires a connection")
	}
	id := key.RunnerID
	h.admitMu.Lock()
	defer h.admitMu.Unlock()
	if key.RevokedAt != nil || h.isRevoked(id) {
		return nil, ErrRunnerRevoked
	}
	r, _, err := h.reg.GetRunner(ctx, id)
	if err != nil {
		return nil, fmt.Errorf("admit %s: %w", id, err)
	}
	if r.BuiltIn {
		return nil, fmt.Errorf("admit %s: the built-in runner cannot connect remotely", id)
	}
	now := h.now()
	prevLiveness := r.Liveness
	r.OS, r.RunnerVersion, r.HarnessVersions, r.WorkspaceRoot = hello.OS, hello.RunnerVersion, hello.HarnessVersions, hello.WorkspaceRoot
	if err := h.reg.UpsertRunner(ctx, r); err != nil {
		return nil, fmt.Errorf("admit %s: %w", id, err)
	}
	compat := h.cfg.Compat.Check(hello, h.cfg.ServerWorkspaceRoot)
	if err := h.reg.SetCompat(ctx, id, compat); err != nil {
		return nil, fmt.Errorf("admit %s: %w", id, err)
	}
	if compat.State == CompatIncompatible {
		if err := h.reg.SetLiveness(ctx, id, LivenessOffline, now); err != nil {
			return nil, fmt.Errorf("admit %s: %w", id, err)
		}
		h.record(ctx, evtRunnerIncompatible, id, "", Cause{Kind: "system", Actor: string(id)}, map[string]any{
			"reason": compat.Reason, "os": hello.OS, "protocolVersion": hello.ProtocolVersion, "runnerVersion": hello.RunnerVersion})
		h.notifier.Notify()
		return nil, &IncompatibleError{Compat: compat}
	}
	if err := h.reg.ReplaceSources(ctx, id, reportedSources(id, hello.Sources)); err != nil {
		return nil, fmt.Errorf("admit %s: %w", id, err)
	}
	if err := h.applyReportedHealth(ctx, id, hello.Sources, now); err != nil {
		return nil, fmt.Errorf("admit %s: %w", id, err)
	}
	if err := h.reg.SetLiveness(ctx, id, LivenessOnline, now); err != nil {
		return nil, fmt.Errorf("admit %s: %w", id, err)
	}

	h.mu.Lock()
	old := h.sessions[id]
	_, seen := h.contact[id]
	h.serial++
	s := &Session{RunnerID: id, KeyID: key.ID, conn: conn, serial: h.serial}
	h.sessions[id] = s
	h.contact[id] = now
	h.mu.Unlock()
	if old != nil {
		_ = old.conn.Close(CloseSuperseded, "superseded by a newer connection from this runner")
	}
	if err := conn.Send(ctx, Outbound{Kind: MsgAdmitted}); err != nil {
		h.Disconnect(s)
		return nil, fmt.Errorf("admit %s: send admitted: %w", id, err)
	}
	kind := evtRunnerConnected
	if seen || prevLiveness == LivenessSuspect || prevLiveness == LivenessLost {
		kind = evtRunnerReconnected
	}
	h.record(ctx, kind, id, "", Cause{Kind: "liveness", Actor: string(id)}, map[string]any{
		"runnerVersion": hello.RunnerVersion, "compat": string(compat.State)})
	h.notifier.Notify()
	return s, nil
}

// Disconnect forgets a closed socket. It never changes liveness: only
// heartbeat gaps do (Tick).
func (h *Hub) Disconnect(s *Session) {
	if s == nil {
		return
	}
	h.mu.Lock()
	defer h.mu.Unlock()
	if h.sessions[s.RunnerID] == s {
		delete(h.sessions, s.RunnerID)
	}
}

// Handle processes one message from an admitted session. A non-nil error
// tells the endpoint to close the connection.
func (h *Hub) Handle(ctx context.Context, s *Session, in Inbound) error {
	if s == nil {
		return ErrSessionClosed
	}
	if h.isRevoked(s.RunnerID) {
		return ErrRunnerRevoked
	}
	if !h.current(s) {
		return ErrSessionClosed
	}
	now := h.now()
	h.mu.Lock()
	h.contact[s.RunnerID] = now
	h.mu.Unlock()
	switch in.Kind {
	case MsgHeartbeat:
		if in.Heartbeat == nil {
			return fmt.Errorf("%w: heartbeat without payload", ErrMalformedMessage)
		}
		return h.handleHeartbeat(ctx, s, *in.Heartbeat, now)
	default:
		return fmt.Errorf("%w: unexpected kind %q", ErrMalformedMessage, in.Kind)
	}
}

func (h *Hub) handleHeartbeat(ctx context.Context, s *Session, hb HeartbeatInfo, now time.Time) error {
	t := hb.Telemetry
	if t.SampledAt.IsZero() {
		t.SampledAt = now
	}
	h.RecordTelemetry(s.RunnerID, t)
	r, cur, err := h.reg.GetRunner(ctx, s.RunnerID)
	if err != nil {
		return err
	}
	if err := h.reg.SetLiveness(ctx, s.RunnerID, LivenessOnline, now); err != nil {
		return err
	}
	if r.Liveness == LivenessSuspect {
		h.record(ctx, evtRunnerReconnected, s.RunnerID, "", Cause{Kind: "liveness", Actor: string(s.RunnerID)}, map[string]any{"from": string(r.Liveness)})
		h.notifier.Notify()
	}
	if hb.Sources != nil {
		next := reportedSources(s.RunnerID, hb.Sources)
		if !sameSourceSet(cur, next) {
			if err := h.reg.ReplaceSources(ctx, s.RunnerID, next); err != nil {
				return err
			}
			h.notifier.Notify()
		}
		if err := h.applyReportedHealth(ctx, s.RunnerID, hb.Sources, now); err != nil {
			return err
		}
	}
	return nil
}

// RecordTelemetry appends t to id's bounded in-memory window (heartbeats and
// the built-in loop of P2 T7).
func (h *Hub) RecordTelemetry(id RunnerID, t Telemetry) {
	h.mu.Lock()
	defer h.mu.Unlock()
	w := append(h.telemetry[id], t)
	if len(w) > h.cfg.TelemetryWindow {
		w = append([]Telemetry(nil), w[len(w)-h.cfg.TelemetryWindow:]...)
	}
	h.telemetry[id] = w
}

// Latest returns the newest sample per runner (TelemetrySource).
func (h *Hub) Latest() map[RunnerID]Telemetry {
	h.mu.Lock()
	defer h.mu.Unlock()
	out := make(map[RunnerID]Telemetry, len(h.telemetry))
	for id, w := range h.telemetry {
		if len(w) > 0 {
			out[id] = w[len(w)-1]
		}
	}
	return out
}

// Window returns a copy of id's recent samples, oldest first.
func (h *Hub) Window(id RunnerID) []Telemetry {
	h.mu.Lock()
	defer h.mu.Unlock()
	return append([]Telemetry(nil), h.telemetry[id]...)
}

// domain/runner/hub_run.go
package runner

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"time"

	"github.com/google/uuid"

	"github.com/arbitron/arbitron/domain/engine"

const streamBuffer = 64

// Run offers one reserved job generation to its remote runner and waits for
// the outcome. Chat returns after the terminal result; stream returns the
// token channel once the runner accepts. Errors carry outcomes (WithOutcome):
// runner loss and offer expiry are OutcomeRunnerLost (the dispatcher re-places
// without spending a retry); unsupported work and rejections are OutcomeOther.
func (h *Hub) Run(ctx context.Context, a Assignment, job StepJob) (StepOutput, error) {
	id := a.Fence.RunnerID
	if id == "" || id == BuiltInRunnerID {
		return StepOutput{}, fmt.Errorf("runner: hub cannot execute on runner %q", id)
	}
	now := h.now()
	d, err := buildDispatch(a, job, now.Add(h.cfg.LostAfter))
	if err != nil {
		return StepOutput{}, WithOutcome(err, SourceOutcome{Class: OutcomeOther})
	}
	j := &remoteJob{ctx: ctx, a: a, dispatch: d, offeredAt: now, acceptedCh: make(chan struct{}), done: make(chan struct{})}
	if d.Op == OpStream {
		j.stream = make(chan engine.ChatChunk, streamBuffer)
	}
	h.mu.Lock()
	if h.revoked[id] {
		h.mu.Unlock()
		return StepOutput{}, WithOutcome(fmt.Errorf("%w: %s", ErrRunnerRevoked, id), SourceOutcome{Class: OutcomeRunnerLost})
	}
	h.jobs[a.Fence.JobID] = j
	s := h.sessions[id]
	h.mu.Unlock()
	h.send(ctx, s, dispatchMsg(j))

	select {
	case <-j.acceptedCh:
	case <-j.done:
	case <-ctx.Done():
		h.abandon(ctx, j, ctx.Err())
		return StepOutput{}, ctx.Err()
	}
	if j.stream != nil {
		h.mu.Lock()
		accepted := j.accepted
		h.mu.Unlock()
		if accepted {
			return StepOutput{Stream: j.stream}, nil
		}
		return h.jobOutput(j)
	}
	select {
	case <-j.done:
	case <-ctx.Done():
		h.abandon(ctx, j, ctx.Err())
		return StepOutput{}, ctx.Err()
	}
	return h.jobOutput(j)
}

func dispatchMsg(j *remoteJob) Outbound {
	d := j.dispatch
	return Outbound{Kind: MsgDispatch, JobID: j.a.Fence.JobID, Generation: j.a.Fence.Generation, Dispatch: &d}
}

// buildDispatch builds the hermetic wire request. It never calls
// Binding.RevealKey and never copies BaseURL, ConnectionRef or WorkDir.
func buildDispatch(a Assignment, job StepJob, leaseUntil time.Time) (DispatchPayload, error) {
	if job.Op != OpChat && job.Op != OpStream {
		return DispatchPayload{}, fmt.Errorf("%w: %s", ErrRemoteOpUnsupported, job.Op)
	}
	if job.Chat == nil {
		return DispatchPayload{}, fmt.Errorf("%w: %s job without a chat request", ErrRemoteOpUnsupported, job.Op)
	}
	kind := a.Job.Group.Kind
	if !remoteKind(kind) {
		return DispatchPayload{}, fmt.Errorf("%w: %s", ErrRemoteCredential, kind)
	}
	c := job.Chat
	var schema json.RawMessage
	if c.OutputSchema != nil {
		b, err := json.Marshal(c.OutputSchema)
		if err != nil {
			return DispatchPayload{}, fmt.Errorf("runner: encode output schema: %w", err)
		}
		schema = b
	}
	tools := c.ToolScope
	if len(job.Binding.AllowedTools) > 0 {
		tools = job.Binding.AllowedTools
	}
	if c.SuppressTools {
		tools = nil
	}
	model := c.Model
	if model == "" {
		model = job.Binding.Model
	}
	return DispatchPayload{Op: job.Op, LeaseUntil: leaseUntil, Chat: &ChatDispatch{
		Model: model, SystemPrompt: c.SystemPrompt, Prompt: c.Prompt, MaxTokens: c.MaxTokens, Temperature: c.Temperature,
		OutputSchema: schema, SessionID: c.SessionID, RequireExecutedModel: c.RequireExecutedModel || job.Binding.RequiresModelEvidence(), Stream: job.Op == OpStream,
		CredentialKind: kind, Backend: job.Binding.Backend, ContextWindow: string(job.Binding.ContextWindow),
		Effort: string(job.Binding.Effort), Tools: append([]string(nil), tools...),
	}}, nil
}

func (h *Hub) jobOutput(j *remoteJob) (StepOutput, error) {
	h.mu.Lock()
	res, err := j.result, j.err
	h.mu.Unlock()
	if err != nil {
		return StepOutput{}, err
	}
	return remoteOutput(res)
}

func remoteOutput(res *RemoteResult) (StepOutput, error) {
	if res == nil {
		return StepOutput{}, WithOutcome(fmt.Errorf("%w: empty result", ErrRemoteStepFailed), SourceOutcome{Class: OutcomeOther})
	}
	if !res.OK {
		return StepOutput{}, remoteError(res)
	}
	if res.Chat == nil {
		return StepOutput{}, WithOutcome(fmt.Errorf("%w: result carries no chat payload", ErrRemoteStepFailed), SourceOutcome{Class: OutcomeOther})
	}
	resp := &engine.ChatResponse{Output: res.Chat.Output, Model: res.Chat.Model, TokensIn: res.Chat.TokensIn,
		TokensOut: res.Chat.TokensOut, SessionID: res.Chat.SessionID}
	if len(res.Chat.Conversation) > 0 {
		var conv any
		if json.Unmarshal(res.Chat.Conversation, &conv) == nil {
			resp.Conversation = conv
		}
	}
	return StepOutput{Chat: resp}, nil
}

// remoteError maps a failed result. A runner may report only source outcomes
// (exhausted, logged_out, transient, other); anything else is other.
func remoteError(res *RemoteResult) error {
	o := res.Outcome
	switch o.Class {
	case OutcomeExhausted, OutcomeLoggedOut, OutcomeTransient, OutcomeOther:
	default:
		o.Class = OutcomeOther
	}
	msg := res.ErrorText
	if msg == "" {
		msg = "no detail"
	}
	return WithOutcome(fmt.Errorf("%w: %s", ErrRemoteStepFailed, msg), o)
}

func terminalChunk(res *RemoteResult, err error) engine.ChatChunk {
	if err != nil {
		return engine.ChatChunk{Done: true, Error: err}
	}
	if res == nil || !res.OK {
		if res == nil {
			return engine.ChatChunk{Done: true, Error: WithOutcome(fmt.Errorf("%w: empty result", ErrRemoteStepFailed), SourceOutcome{Class: OutcomeOther})}
		}
		return engine.ChatChunk{Done: true, Error: remoteError(res)}
	}
	c := engine.ChatChunk{Done: true}
	if res.Chat != nil {
		c.TokensIn, c.TokensOut, c.SessionID = res.Chat.TokensIn, res.Chat.TokensOut, res.Chat.SessionID
	}
	return c
}

func (h *Hub) markAccepted(j *remoteJob) {
	h.mu.Lock()
	defer h.mu.Unlock()
	if !j.accepted {
		j.accepted = true
		close(j.acceptedCh)
	}
}

// finish resolves a waiter exactly once. For an accepted stream it emits the
// terminal chunk and closes the channel under emitMu.
func (h *Hub) finish(j *remoteJob, res *RemoteResult, err error) bool {
	h.mu.Lock()
	if j.finished {
		h.mu.Unlock()
		return false
	}
	j.finished, j.result, j.err = true, res, err
	if h.jobs[j.a.Fence.JobID] == j {
		delete(h.jobs, j.a.Fence.JobID)
	}
	accepted := j.accepted
	h.mu.Unlock()
	if j.stream != nil && accepted {
		j.emitMu.Lock()
		select {
		case j.stream <- terminalChunk(res, err):
		case <-j.ctx.Done():
		}
		close(j.stream)
		j.emitMu.Unlock()
	}
	close(j.done)
	return true
}

func (h *Hub) abandon(ctx context.Context, j *remoteJob, cause error) {
	if h.finish(j, nil, cause) {
		h.send(context.WithoutCancel(ctx), h.session(j.a.Fence.RunnerID),
			Outbound{Kind: MsgCancel, JobID: j.a.Fence.JobID, Generation: j.a.Fence.Generation})
	}
}

// liveJob returns the unfinished waiter matching f exactly, or nil.
func (h *Hub) liveJob(f Fence) *remoteJob {
	h.mu.Lock()
	defer h.mu.Unlock()
	j := h.jobs[f.JobID]
	if j == nil || j.finished || j.a.Fence != f {
		return nil
	}
	return j
}

func (h *Hub) handleJob(ctx context.Context, s *Session, in Inbound, now time.Time) error {
	if in.JobID == uuid.Nil || in.Generation < 1 {
		return fmt.Errorf("%w: %s without job id or generation", ErrMalformedMessage, in.Kind)
	}
	f := Fence{JobID: in.JobID, RunnerID: s.RunnerID, Generation: in.Generation}
	j := h.liveJob(f)
	if j == nil {
		return h.handleOrphan(ctx, s, in, f)
	}
	switch in.Kind {
	case MsgAccept:
		h.markAccepted(j)
		if err := h.ledger.ExtendLease(ctx, f, now.Add(h.cfg.LostAfter)); err != nil {
			return h.fenceLive(ctx, s, j, err)
		}
		return nil
	case MsgReject:
		h.finish(j, nil, WithOutcome(fmt.Errorf("%w: %s", ErrJobRejected, in.RejectReason), SourceOutcome{Class: OutcomeOther}))
		return nil
	case MsgOutput:
		if in.Output == nil {
			return fmt.Errorf("%w: output without payload", ErrMalformedMessage)
		}
		return h.handleOutput(ctx, s, j, *in.Output)
	default: // MsgResult
		if in.Result == nil {
			return fmt.Errorf("%w: result without payload", ErrMalformedMessage)
		}
		if err := h.ledger.FenceCheck(ctx, f); err != nil {
			return h.fenceLive(ctx, s, j, err)
		}
		h.markAccepted(j)
		h.finish(j, in.Result, nil)
		h.send(ctx, s, Outbound{Kind: MsgAck, JobID: f.JobID, Generation: f.Generation, Ack: &AckPayload{Result: true}})
		return nil
	}
}

// fenceLive handles a ledger fence failure for a live waiter: the runner is
// told to stop and the step fails (its generation is no longer current).
func (h *Hub) fenceLive(ctx context.Context, s *Session, j *remoteJob, err error) error {
	if !errors.Is(err, ErrFenced) {
		return err
	}
	h.send(ctx, s, Outbound{Kind: MsgFenced, JobID: j.a.Fence.JobID, Generation: j.a.Fence.Generation})
	h.finish(j, nil, WithOutcome(err, SourceOutcome{Class: OutcomeOther}))
	return nil
}

// handleOutput delivers in-order frames exactly once; duplicates and gaps are
// answered with the last contiguous sequence so the runner resends from there.
func (h *Hub) handleOutput(ctx context.Context, s *Session, j *remoteJob, f OutputFrame) error {
	h.markAccepted(j)
	j.emitMu.Lock()
	h.mu.Lock()
	deliver := f.Seq == j.lastSeq+1 && !j.finished
	if deliver {
		j.lastSeq = f.Seq
	}
	ack := j.lastSeq
	h.mu.Unlock()
	if deliver && j.stream != nil {
		select {
		case j.stream <- engine.ChatChunk{Token: f.Token}:
		case <-j.ctx.Done():
		}
	}
	j.emitMu.Unlock()
	h.send(ctx, s, Outbound{Kind: MsgAck, JobID: j.a.Fence.JobID, Generation: j.a.Fence.Generation, Ack: &AckPayload{Seq: ack}})
	return nil
}

// alreadyFinished reports whether the ledger shows f's generation terminal on
// this runner (a result the server already consumed).
func (h *Hub) alreadyFinished(ctx context.Context, f Fence) bool {
	job, err := h.ledger.Get(ctx, f.JobID)
	if err != nil {
		return false
	}
	return job.RunnerID == f.RunnerID && job.Generation == f.Generation && (job.State == stJobCompleted || job.State == stJobFailed)
}

// handleOrphan answers a message for a job generation this process is not
// waiting on: a replayed result of an already-consumed generation is
// acknowledged; everything else is fenced so the runner stops spending tokens.
func (h *Hub) handleOrphan(ctx context.Context, s *Session, in Inbound, f Fence) error {
	if in.Kind == MsgResult && h.alreadyFinished(ctx, f) {
		h.send(ctx, s, Outbound{Kind: MsgAck, JobID: f.JobID, Generation: f.Generation, Ack: &AckPayload{Result: true}})
		return nil
	}
	h.send(ctx, s, Outbound{Kind: MsgFenced, JobID: f.JobID, Generation: f.Generation})
	return nil
}

// refreshHeld extends leases of live held jobs and fences unknown ones.
func (h *Hub) refreshHeld(ctx context.Context, s *Session, held []HeldJobInfo, now time.Time) error {
	for _, hj := range held {
		f := Fence{JobID: hj.JobID, RunnerID: s.RunnerID, Generation: hj.Generation}
		j := h.liveJob(f)
		if j == nil {
			if hj.State == HeldTerminal && h.alreadyFinished(ctx, f) {
				continue
			}
			h.send(ctx, s, Outbound{Kind: MsgFenced, JobID: f.JobID, Generation: f.Generation})
			continue
		}
		if err := h.ledger.ExtendLease(ctx, f, now.Add(h.cfg.LostAfter)); err != nil {
			if ferr := h.fenceLive(ctx, s, j, err); ferr != nil {
				return ferr
			}
		}
	}
	return nil
}

// domain/runner/hub_liveness.go
package runner

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/google/uuid"
)

// Start runs Tick every Heartbeat until ctx ends.
func (h *Hub) Start(ctx context.Context) {
	go func() {
		t := time.NewTicker(h.cfg.Heartbeat)
		defer t.Stop()
		for {
			select {
			case <-ctx.Done():
				return
			case <-t.C:
				_ = h.Tick(ctx)
			}
		}
	}()
}

// Tick is one liveness pass: expire unaccepted offers, then classify every
// online/suspect remote runner by the age of its last contact. Socket state
// is irrelevant: only heartbeat/message gaps count.
func (h *Hub) Tick(ctx context.Context) error {
	now := h.now()
	h.expireOffers(ctx, now)
	runners, err := h.reg.ListRunners(ctx)
	if err != nil {
		return fmt.Errorf("hub tick: %w", err)
	}
	var errs []error
	for _, r := range runners {
		if r.BuiltIn || (r.Liveness != LivenessOnline && r.Liveness != LivenessSuspect) {
			continue
		}
		last, ok := h.lastContact(r)
		if !ok {
			continue
		}
		gap := now.Sub(last)
		switch {
		case gap >= h.cfg.LostAfter:
			errs = append(errs, h.markLost(ctx, r.ID, last, CloseLost, fmt.Sprintf("no contact for %s", gap.Round(time.Second)),
				Cause{Kind: "liveness", Actor: "hub", Outcome: OutcomeRunnerLost}))
		case gap >= h.cfg.SuspectAfter && r.Liveness == LivenessOnline:
			if err := h.reg.SetLiveness(ctx, r.ID, LivenessSuspect, last); err != nil {
				errs = append(errs, err)
				continue
			}
			h.record(ctx, evtRunnerSuspect, r.ID, "", Cause{Kind: "liveness", Actor: "hub"}, map[string]any{"gapSeconds": int(gap.Seconds())})
			h.notifier.Notify()
		}
	}
	return errors.Join(errs...)
}

func (h *Hub) lastContact(r Runner) (time.Time, bool) {
	h.mu.Lock()
	t, ok := h.contact[r.ID]
	h.mu.Unlock()
	if ok {
		return t, true
	}
	if r.LastContact != nil {
		return *r.LastContact, true
	}
	return time.Time{}, false
}

// markLost tears a runner down: close its socket, mark it lost (before any
// waiter learns, so re-placement can never pick it), fail its waiters with
// OutcomeRunnerLost (the dispatcher requeues them with a new generation),
// release slots held by jobs no waiter owns, record runner.lost and wake.
func (h *Hub) markLost(ctx context.Context, id RunnerID, at time.Time, code int, reason string, cause Cause) error {
	h.admitMu.Lock()
	defer h.admitMu.Unlock()
	h.mu.Lock()
	s := h.sessions[id]
	delete(h.sessions, id)
	var waiters []*remoteJob
	for _, j := range h.jobs {
		if j.a.Fence.RunnerID == id && !j.finished {
			waiters = append(waiters, j)
		}
	}
	h.mu.Unlock()
	if s != nil {
		_ = s.conn.Close(code, reason)
	}
	setErr := h.reg.SetLiveness(ctx, id, LivenessLost, at)
	lostErr := WithOutcome(fmt.Errorf("%w: %s: %s", ErrRunnerLost, id, reason), SourceOutcome{Class: OutcomeRunnerLost})
	owned := map[uuid.UUID]bool{}
	for _, j := range waiters {
		owned[j.a.Fence.JobID] = true
		h.finish(j, nil, lostErr)
	}
	active, listErr := h.ledger.ListActiveByRunner(ctx, id)
	var failErrs []error
	for _, job := range active {
		if owned[job.ID] || job.State == stJobQueued {
			continue
		}
		f := Fence{JobID: job.ID, RunnerID: id, Generation: job.Generation}
		if err := h.ledger.Fail(ctx, f, OutcomeRunnerLost); err != nil && !errors.Is(err, ErrFenced) {
			failErrs = append(failErrs, err)
		}
	}
	h.record(ctx, evtRunnerLost, id, "", cause, map[string]any{"reason": reason, "interruptedSteps": len(waiters)})
	h.notifier.Notify()
	return errors.Join(append([]error{setErr, listErr}, failErrs...)...)
}

// expireOffers fails offers not accepted within OfferTTL with OutcomeRunnerLost
// so the dispatcher re-places them (new generation); the runner is told to cancel.
func (h *Hub) expireOffers(ctx context.Context, now time.Time) {
	h.mu.Lock()
	var expired []*remoteJob
	for _, j := range h.jobs {
		if !j.finished && !j.accepted && now.Sub(j.offeredAt) >= h.cfg.OfferTTL {
			expired = append(expired, j)
		}
	}
	h.mu.Unlock()
	for _, j := range expired {
		err := WithOutcome(fmt.Errorf("%w: not accepted within %s", ErrOfferExpired, h.cfg.OfferTTL), SourceOutcome{Class: OutcomeRunnerLost})
		if h.finish(j, nil, err) {
			h.send(ctx, h.session(j.a.Fence.RunnerID), Outbound{Kind: MsgCancel, JobID: j.a.Fence.JobID, Generation: j.a.Fence.Generation})
		}
	}
}

// RevokeRunner implements SessionRevoker: the runner is refused from now on,
// its socket is closed with CloseRevoked and it is treated as lost.
func (h *Hub) RevokeRunner(ctx context.Context, id RunnerID) error {
	h.mu.Lock()
	h.revoked[id] = true
	h.mu.Unlock()
	h.record(ctx, evtRunnerKeyRevoked, id, "", Cause{Kind: "operator", Outcome: OutcomeRunnerLost}, nil)
	return h.markLost(ctx, id, h.now(), CloseRevoked, "runner key revoked", Cause{Kind: "operator", Outcome: OutcomeRunnerLost})
}

// domain/runner/hub_reconcile.go
package runner

import (
	"context"
	"errors"
	"time"

	"github.com/google/uuid"
)

// reconcileHeld answers each job the runner reports holding at admission:
// a live generation is (re)accepted, its lease extended, and the last
// contiguous output sequence acked so the runner resends only what is
// missing; an already-consumed terminal result is acked; anything else is
// fenced. It returns the set of job ids the runner claimed.
func (h *Hub) reconcileHeld(ctx context.Context, s *Session, held []HeldJobInfo, now time.Time) map[uuid.UUID]bool {
	claimed := make(map[uuid.UUID]bool, len(held))
	for _, hj := range held {
		claimed[hj.JobID] = true
		f := Fence{JobID: hj.JobID, RunnerID: s.RunnerID, Generation: hj.Generation}
		j := h.liveJob(f)
		if j == nil {
			if hj.State == HeldTerminal && h.alreadyFinished(ctx, f) {
				h.send(ctx, s, Outbound{Kind: MsgAck, JobID: f.JobID, Generation: f.Generation, Ack: &AckPayload{Result: true}})
				continue
			}
			h.send(ctx, s, Outbound{Kind: MsgFenced, JobID: f.JobID, Generation: f.Generation})
			continue
		}
		h.markAccepted(j)
		if err := h.ledger.ExtendLease(ctx, f, now.Add(h.cfg.LostAfter)); err != nil {
			if errors.Is(err, ErrFenced) {
				_ = h.fenceLive(ctx, s, j, err)
			}
			continue
		}
		h.mu.Lock()
		seq := j.lastSeq
		h.mu.Unlock()
		h.send(ctx, s, Outbound{Kind: MsgAck, JobID: f.JobID, Generation: f.Generation, Ack: &AckPayload{Seq: seq}})
	}
	return claimed
}

// resendOffers re-sends every unaccepted offer for this runner that the
// runner did not claim. The runner dedupes by (job, generation).
func (h *Hub) resendOffers(ctx context.Context, s *Session, claimed map[uuid.UUID]bool) {
	h.mu.Lock()
	var pending []*remoteJob
	for _, j := range h.jobs {
		if j.a.Fence.RunnerID == s.RunnerID && !j.finished && !j.accepted && !claimed[j.a.Fence.JobID] {
			pending = append(pending, j)
		}
	}
	h.mu.Unlock()
	for _, j := range pending {
		h.send(ctx, s, dispatchMsg(j))
	}
}
