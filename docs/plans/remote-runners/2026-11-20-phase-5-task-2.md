## Hook algorithm

1. Serialize loads per subject epoch: one active REST load at a time, plus one dirty flag for coalesced invalidations. An event during a load marks dirty and schedules one fresh load AFTER the active load finishes. Never replay buffered events as data patches atop the result. If data arrived before an event, the next load captures its authoritative state; if event races a read, dirty followup closes the race.
2. Increment epoch on subject/auth/enabled change or unmount. Ignore old epoch successes AND errors and cancel scheduled work. A load started for runner A must never populate runner B. No stale request can overwrite a more recently requested subject or clear that subject's error/loading state. (The dashboard has no in-page identity switch today; a caller whose identity can change folds it into `subject`.)
3. Initial load and manual retry call load; error preserves last successfully read data with `error` and `stale=true`. A REST failure also marks data stale while WebSocket remains connected. Coalesce events/manual retries/poll/reconnect; ensure failed loads don't starve dirty followup, bounded by debounce/backoff.
4. Disconnect sets stale immediately and starts 5s polling; reconnect triggers an authoritative refresh. Connected 15s refresh updates telemetry because heartbeats need not be bus events. A failed freshness refresh keeps stale until a successful load. Socket connected alone does not prove data freshness. (Implementation: EVERY transition to connected, including the first open, marks not-fresh and requests a load, because events published before the socket opened are never replayed.)
5. `loadPool` may issue independent runners/events/running/waiting GETs concurrently, then deduplicates execution rows by durable job identity. It MUST NOT label this a transactional snapshot or derive revision from event timestamps. Read states holding pending/recovery work as the upstream projector supports, including waiting interactive runs. Runner detail returns `{detail,pool}`: detail is the authoritative detail response, pool is snapshotState([detail], filteredEvents, authorized assignment execution data); applyRunnerEvent invalidates this subject without patching detail/pool from events. None of these read hooks loads/stores runner-key plaintext.
6. Countdown uses absolute reset timestamp, validates finite dates, clamps display at zero/awaiting confirmation, and never mutates source health after reset. `useNow` drives rendering; unavailable timestamps remain unknown.

```ts
import { formatDuration } from './time'

/**
 * Countdown to a source's next capacity reset, computed client-side from the
 * server's normalized ABSOLUTE timestamp (never from elapsed time alone). A
 * passed reset does not mean the source is back: the server confirms recovery
 * with a probe, so the label clamps at "awaiting confirmation" and this
 * helper never reads or changes source health.
 */
export function formatCountdown(resetAt: string | null | undefined, now: number): string {
  if (!resetAt || !Number.isFinite(now)) return 'reset time unknown'
  const at = Date.parse(resetAt)
  if (!Number.isFinite(at)) return 'reset time unknown'
  const left = at - now
  if (left <= 0) return 'reset due · awaiting confirmation'
  return `resets in ${formatDuration(left)}`
}
```

```ts
import type { LiveEvent } from '../api/runnerDecode'
import type { ExecutionStatus, PoolEvent, RunnerView } from '../api/types'
import type { Applied } from './useLiveSnapshot'

/** Recent pool events kept in view; also the REST `limit` the hooks request. */
export const POOL_EVENT_LIMIT = 50

/**
 * One authoritative read of the pool. The runners, event log and execution
 * rows come from independent REST reads with no shared revision, so nothing
 * here claims they are a transactional snapshot.
 */
export interface PoolState {
  runners: RunnerView[]
  /** Display log only, newest first. Never proof that `runners` reflects an event. */
  events: PoolEvent[]
  executions: ExecutionStatus[]
}

function dedupeEvents(events: PoolEvent[]): PoolEvent[] {
  const seen = new Set<string>()
  const out: PoolEvent[] = []
  for (const e of events) {
    if (seen.has(e.id)) continue
    seen.add(e.id)
    out.push(e)
    if (out.length === POOL_EVENT_LIMIT) break
  }
  return out
}

export function snapshotState(runners: RunnerView[], events: PoolEvent[], executions: ExecutionStatus[]): PoolState {
  return { runners, events: dedupeEvents(events), executions }
}

/** Pure helper for already-authoritative rows. MUST NOT be fed unversioned system events. */
export function upsertExecution(list: ExecutionStatus[], status: ExecutionStatus): ExecutionStatus[] {
  const idx = list.findIndex((s) => s.jobId === status.jobId)
  if (idx === -1) return [...list, status]
  const next = [...list]
  next[idx] = status
  return next
}

/** Pure helper for already-authoritative rows. MUST NOT be fed unversioned system events. */
export function pruneExecutions(list: ExecutionStatus[], runId: string, nodeId: string): ExecutionStatus[] {
  const next = list.filter((s) => !(s.runId === runId && s.nodeId === nodeId))
  return next.length === list.length ? list : next
}

/** Adds a live pool event to the display log (deduplicated by id); data rows are untouched. */
export function withLoggedEvent(state: PoolState, ev: PoolEvent): PoolState {
  if (state.events.some((e) => e.id === ev.id)) return state
  return { ...state, events: [ev, ...state.events].slice(0, POOL_EVENT_LIMIT) }
}

/**
 * Events are invalidations. Every pool/source/job/workspace/session event and
 * every execution.status / node completion asks for an authoritative REST
 * reload; none patches health, liveness, counts, telemetry or executions.
 * Event `at` is never used as a cursor or watermark.
 */
export function applyLiveEvent(state: PoolState, live: LiveEvent): Applied<PoolState> {
  switch (live.kind) {
    case 'pool':
      return { state: withLoggedEvent(state, live.event), refetch: true }
    case 'execution':
    case 'node_done':
      return { state, refetch: true }
    default:
      return { state, refetch: false }
  }
}
```

```ts
import { useCallback, useEffect, useRef, useState } from 'react'
import { decodeLiveEvent, type LiveEvent } from '../api/runnerDecode'
import type { WebSocketEvent } from '../api/types'
import { useSystemWebSocket } from './useSystemWebSocket'

/** Poll interval while the system socket is down (precedent: GRAPH_STATE_POLL_MS). */
export const POOL_POLL_MS = 5000
/**
 * Refresh interval while connected. Heartbeat telemetry is deliberately not a
 * bus event, so live CPU, memory and heartbeat age are refreshed from REST.
 */
export const POOL_REFRESH_MS = 15_000
/** Coalesces bursts of invalidations (and the dirty follow-up) into one REST read. */
export const POOL_EVENT_DEBOUNCE_MS = 300

/** Result of applying one live event: the next (display-only) state and whether an authoritative reload is needed. */
export interface Applied<S> {
  state: S
  refetch: boolean
}

export interface LiveSnapshotOptions<S> {
  /** False: nothing is fetched and no state is kept. */
  enabled: boolean
  /** Changing the subject starts a new epoch: state resets and reloads. Fold identity into it when it can change. */
  subject: string
  load: () => Promise<S>
  apply: (state: S, live: LiveEvent) => Applied<S>
}

export interface LiveSnapshot<S> {
  /** Last successfully loaded snapshot (plus display-only log entries); null until the first success. */
  state: S | null
  loading: boolean
  error: string | null
  connected: boolean
  /** True while disconnected, before the first success, after a failed load, or until a (re)connect refresh succeeds. */
  stale: boolean
  refetch: () => Promise<void>
}

/**
 * Authoritative REST snapshot with event invalidation.
 *
 * - One active load per subject epoch plus one dirty flag: an invalidation
 *   during a load schedules exactly one fresh load after it settles. Events
 *   are never replayed as data patches on top of a result.
 * - A subject/enabled change or unmount bumps the epoch: older successes and
 *   errors are ignored and scheduled work is cancelled.
 * - A failed load keeps the last good state, sets `error` and `stale`.
 * - Disconnected: stale immediately, poll every POOL_POLL_MS. Every
 *   (re)connect triggers an authoritative refresh; stale clears only after a
 *   load succeeds while connected. Connected: refresh every POOL_REFRESH_MS.
 */
export function useLiveSnapshot<S>({ enabled, subject, load, apply }: LiveSnapshotOptions<S>): LiveSnapshot<S> {
  const [state, setState] = useState<S | null>(null)
  const [loading, setLoading] = useState(enabled)
  const [error, setError] = useState<string | null>(null)
  const [fresh, setFresh] = useState(false)
  const stateRef = useRef<S | null>(null)
  const loadRef = useRef(load)
  loadRef.current = load
  const applyRef = useRef(apply)
  applyRef.current = apply
  const enabledRef = useRef(enabled)
  enabledRef.current = enabled
  const epoch = useRef(0)
  const active = useRef<Promise<void> | null>(null)
  const dirty = useRef(false)
  const timer = useRef<ReturnType<typeof setTimeout> | null>(null)

  const commit = useCallback((next: S | null) => {
    stateRef.current = next
    setState(next)
  }, [])

  const startLoadRef = useRef<() => Promise<void>>(() => Promise.resolve())

  const schedule = useCallback(() => {
    if (timer.current) return // one pending reload already covers this invalidation
    timer.current = setTimeout(() => {
      timer.current = null
      void startLoadRef.current()
    }, POOL_EVENT_DEBOUNCE_MS)
  }, [])

  const startLoad = useCallback((): Promise<void> => {
    if (!enabledRef.current) return Promise.resolve()
    if (active.current) {
      dirty.current = true
      return active.current
    }
    const mine = epoch.current
    dirty.current = false
    const run = (async () => {
      try {
        const next = await loadRef.current()
        if (mine !== epoch.current) return
        commit(next)
        setError(null)
        setFresh(true)
      } catch (err) {
        if (mine !== epoch.current) return
        setError(err instanceof Error ? err.message : 'Failed to load')
        setFresh(false)
      } finally {
        if (mine === epoch.current) {
          active.current = null
          setLoading(false)
          if (dirty.current) {
            dirty.current = false
            schedule()
          }
        }
      }
    })()
    active.current = run
    return run
  }, [commit, schedule])
  startLoadRef.current = startLoad

  useEffect(() => {
    epoch.current += 1
    active.current = null
    dirty.current = false
    commit(null)
    setError(null)
    setFresh(false)
    setLoading(enabled)
    if (enabled) void startLoadRef.current()
    return () => {
      epoch.current += 1
      active.current = null
      dirty.current = false
      if (timer.current) clearTimeout(timer.current)
      timer.current = null
    }
  }, [subject, enabled, commit])

  const onEvent = useCallback(
    (event: WebSocketEvent) => {
      if (!enabledRef.current) return
      const live = decodeLiveEvent(event)
      if (live.kind === 'ignored') return
      const current = stateRef.current
      let wanted = true
      if (current !== null) {
        const out = applyRef.current(current, live)
        if (out.state !== current) commit(out.state)
        wanted = out.refetch
      }
      if (!wanted) return
      if (active.current) dirty.current = true
      else schedule()
    },
    [commit, schedule],
  )

  const { connected } = useSystemWebSocket(onEvent)

  const wasConnected = useRef(false)
  useEffect(() => {
    if (connected === wasConnected.current) return
    wasConnected.current = connected
    setFresh(false)
    // Events published while the socket was closed (or before it first opened) are never replayed.
    if (connected) void startLoadRef.current()
  }, [connected])

  const interval = connected ? POOL_REFRESH_MS : POOL_POLL_MS
  useEffect(() => {
    if (!enabled) return
    const id = setInterval(() => {
      void startLoadRef.current()
    }, interval)
    return () => clearInterval(id)
  }, [enabled, interval, subject])

  return { state, loading, error, connected, stale: enabled && (!connected || !fresh || error !== null), refetch: startLoad }
}
```

```ts
import { api } from '../api/client'
import { decodeExecutionStatus } from '../api/runnerDecode'
import { listPoolEvents, listRunners } from '../api/runners'
import type { EnrichedInstanceSummary, ExecutionStatus } from '../api/types'
import { applyLiveEvent, POOL_EVENT_LIMIT, snapshotState, type PoolState } from './poolState'
import { useLiveSnapshot, type LiveSnapshot } from './useLiveSnapshot'

/**
 * Execution rows of the runs the caller may see (the server already filters
 * by visibility), decoded tolerantly, tagged with the owning run when the entry
 * omits it, and deduplicated by durable job id (a run can appear in both the
 * running and waiting reads because they are independent).
 */
export function executionsOf(rows: EnrichedInstanceSummary[]): ExecutionStatus[] {
  const out: ExecutionStatus[] = []
  const seen = new Set<string>()
  for (const row of rows) {
    if (!Array.isArray(row.execution)) continue
    for (const raw of row.execution as unknown[]) {
      const obj = typeof raw === 'object' && raw !== null ? (raw as Record<string, unknown>) : {}
      const runId = typeof obj.runId === 'string' && obj.runId !== '' ? obj.runId : row.id
      const status = decodeExecutionStatus({ ...obj, runId })
      if (!status || seen.has(status.jobId)) continue
      seen.add(status.jobId)
      out.push(status)
    }
  }
  return out
}

/**
 * Four independent authorized reads issued concurrently. This is NOT a
 * transactional snapshot and carries no revision. Running AND waiting runs:
 * pending/recovery work and idle interactive sessions are waiting at the
 * instance level.
 */
export async function loadPool(): Promise<PoolState> {
  const [runners, events, running, waiting] = await Promise.all([
    listRunners(),
    listPoolEvents(POOL_EVENT_LIMIT),
    api.instances.list({ state: 'running' }),
    api.instances.list({ state: 'waiting' }),
  ])
  return snapshotState(runners, events, executionsOf([...running, ...waiting]))
}

export function usePool(): LiveSnapshot<PoolState> {
  return useLiveSnapshot({ enabled: true, subject: 'pool', load: loadPool, apply: applyLiveEvent })
}
```

```ts
import { useCallback } from 'react'
import { decodeExecutionStatus, type LiveEvent } from '../api/runnerDecode'
import { getRunner, listPoolEvents } from '../api/runners'
import type { ExecutionStatus, RunnerDetail } from '../api/types'
import { POOL_EVENT_LIMIT, snapshotState, withLoggedEvent, type PoolState } from './poolState'
import { useLiveSnapshot, type Applied, type LiveSnapshot } from './useLiveSnapshot'

export interface RunnerState {
  /** The authoritative detail response; never patched from events. */
  detail: RunnerDetail
  /** snapshotState([detail], this runner's logged events, visible assignment execution rows). */
  pool: PoolState
}

/** Visible (server-authorized) assignments as execution rows; hidden runs are already redacted to runId null. */
function assignmentExecutions(d: RunnerDetail): ExecutionStatus[] {
  const out: ExecutionStatus[] = []
  for (const a of d.assignments) {
    if (a.hidden || !a.runId) continue
    const s = decodeExecutionStatus({ runId: a.runId, nodeId: a.nodeId, jobId: a.jobId, state: a.state, runnerId: d.id, sourceId: a.sourceId })
    if (s) out.push(s)
  }
  return out
}

function visibleRuns(s: RunnerState): Set<string> {
  return new Set(s.detail.assignments.flatMap((a) => (a.runId ? [a.runId] : [])))
}

/**
 * Invalidations scoped to this runner: its own pool events, source-only
 * events that name none of the runners or one of its sources (conservative),
 * and execution/node/pool events for runs assigned to it. The detail is never
 * patched; only this runner's events join the display log.
 */
export function applyRunnerEvent(s: RunnerState, live: LiveEvent, runnerId: string): Applied<RunnerState> {
  switch (live.kind) {
    case 'pool': {
      const ev = live.event
      const mine = ev.runnerId === runnerId
      const relevant =
        mine ||
        (ev.runnerId === '' && (ev.sourceId === '' || ev.sourceId.startsWith(`${runnerId}/`))) ||
        (ev.runId !== null && visibleRuns(s).has(ev.runId))
      if (!relevant) return { state: s, refetch: false }
      const pool = mine ? withLoggedEvent(s.pool, ev) : s.pool
      return { state: pool === s.pool ? s : { ...s, pool }, refetch: true }
    }
    case 'execution':
      return { state: s, refetch: live.status.runnerId === runnerId || visibleRuns(s).has(live.status.runId) }
    case 'node_done':
      return { state: s, refetch: visibleRuns(s).has(live.instanceId) }
    default:
      return { state: s, refetch: false }
  }
}

export function useRunner(id: string | undefined): LiveSnapshot<RunnerState> {
  const load = useCallback(async (): Promise<RunnerState> => {
    if (!id) throw new Error('no runner id')
    const [detail, events] = await Promise.all([getRunner(id), listPoolEvents(POOL_EVENT_LIMIT)])
    return { detail, pool: snapshotState([detail], events.filter((e) => e.runnerId === id), assignmentExecutions(detail)) }
  }, [id])
  const apply = useCallback((s: RunnerState, live: LiveEvent) => applyRunnerEvent(s, live, id ?? ''), [id])
  return useLiveSnapshot({ enabled: Boolean(id), subject: id ?? '', load, apply })
}
```
