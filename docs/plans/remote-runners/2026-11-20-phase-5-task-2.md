# Remote Runners — Phase 5 Task 2: Pool data hooks (usePool / useRunner) with live merge, ordering and polling fallback

**Goal:** Load pool/runner REST snapshots, merge live system-stream events in order, and fall back to a stale banner plus polling when the socket drops.
**Architecture:** A pure reducer (`hooks/poolState.ts`) applies decoded `LiveEvent`s to a `PoolState` with two ordering guards: pool-event ids already in the REST log are never re-applied, and an event older than the newest event already reflected for that runner (or job) never patches state. A generic `useLiveSnapshot` owns load/apply/buffer-during-reload/debounced-reload/polling; `usePool` and `useRunner` are thin wrappers. A countdown helper renders reset times from server absolute timestamps.
**Tech Stack:** React 19 hooks, TypeScript, Vitest 4 fake timers, jsdom.
**Spec / Design:** spec "Administration and visibility" (live views); design phase 5 §3.5 (pool data hooks), §4 steps 1–3, §5 row "Event delivery and polling fallback" (RR-16 version/cursor handling), §7 (WebSocket disconnect → stale banner + polling; unknown event kinds ignored), §8. Contracts: `docs/plans/remote-runners/2026-11-20-contracts.md` §§9, 10, 14, 18.

```ts
import { formatDuration } from './time'

/**
 * Countdown to a source's next capacity reset, computed client-side from the
 * server's normalized ABSOLUTE timestamp (never from elapsed time alone). A
 * passed reset does not mean the source is back: the server confirms recovery
 * with a probe, so the label says it is awaiting confirmation.
 */
export function formatCountdown(resetAt: string | null | undefined, now: number): string {
  if (!resetAt) return 'reset time unknown'
  const at = Date.parse(resetAt)
  if (!Number.isFinite(at)) return 'reset time unknown'
  const left = at - now
  if (left <= 0) return 'reset due · awaiting confirmation'
  return `resets in ${formatDuration(left)}`
}
```

```ts
import type { LiveEvent } from '../api/runnerDecode'
import type {
  DrainProgress,
  ExecutionStatus,
  PoolEvent,
  RunnerSource,
  RunnerView,
  SourceAuthState,
  SourceCapacityState,
} from '../api/types'
import type { Applied } from './useLiveSnapshot'

/** Recent pool events kept in view; also the REST `limit` the hooks request. */
export const POOL_EVENT_LIMIT = 50

export interface PoolState {
  runners: RunnerView[]
  events: PoolEvent[]
  executions: ExecutionStatus[]
  /** Pool event ids already reflected (snapshot log + applied live events). */
  seenEvents: ReadonlySet<string>
  /** Newest server event time (ms) already reflected per runner; older events no longer patch it. */
  runnerAsOf: Readonly<Record<string, number>>
  /** Newest applied execution.status time (ms) per job. */
  execAsOf: Readonly<Record<string, number>>
}

const EMPTY_DRAIN: DrainProgress = { activeSteps: 0, homedWorkspaces: 0, sessions: 0, blocked: [], complete: false }
const CAPACITIES: readonly SourceCapacityState[] = ['available', 'exhausted', 'probation']
const AUTHS: readonly SourceAuthState[] = ['ok', 'logged_out', 'unknown']

function ms(isoTime: string): number {
  const t = Date.parse(isoTime)
  return Number.isFinite(t) ? t : 0
}

/**
 * Builds state from a REST snapshot. The event log was read together with the
 * runners, so every logged event is already reflected in them: their ids are
 * "seen" and their newest time per runner is the ordering watermark.
 */
export function snapshotState(runners: RunnerView[], events: PoolEvent[], executions: ExecutionStatus[]): PoolState {
  const runnerAsOf: Record<string, number> = {}
  for (const e of events) {
    if (!e.runnerId) continue
    runnerAsOf[e.runnerId] = Math.max(runnerAsOf[e.runnerId] ?? 0, ms(e.at))
  }
  return {
    runners,
    events: events.slice(0, POOL_EVENT_LIMIT),
    executions,
    seenEvents: new Set(events.map((e) => e.id)),
    runnerAsOf,
    execAsOf: {},
  }
}

export function upsertExecution(list: ExecutionStatus[], status: ExecutionStatus): ExecutionStatus[] {
  const idx = list.findIndex((s) => s.jobId === status.jobId)
  if (idx === -1) return [...list, status]
  const next = [...list]
  next[idx] = status
  return next
}

export function pruneExecutions(list: ExecutionStatus[], runId: string, nodeId: string): ExecutionStatus[] {
  const next = list.filter((s) => !(s.runId === runId && s.nodeId === nodeId))
  return next.length === list.length ? list : next
}

function withSource(r: RunnerView, updated: RunnerSource): RunnerView {
  return { ...r, sources: r.sources.map((s) => (s.id === updated.id ? updated : s)) }
}

function healthFromDetail(s: RunnerSource, d: Record<string, unknown>): RunnerSource | null {
  const capacity = typeof d.capacity === 'string' && (CAPACITIES as readonly string[]).includes(d.capacity) ? (d.capacity as SourceCapacityState) : null
  const auth = typeof d.auth === 'string' && (AUTHS as readonly string[]).includes(d.auth) ? (d.auth as SourceAuthState) : null
  if (!capacity && !auth) return null
  return {
    ...s,
    health: {
      ...s.health,
      ...(auth ? { auth } : {}),
      ...(capacity ? { capacity, resetAt: typeof d.resetAt === 'string' && d.resetAt !== '' ? d.resetAt : null } : {}),
      ...(typeof d.congested === 'boolean' ? { congested: d.congested } : {}),
    },
  }
}

interface Patched {
  runner: RunnerView
  refetch: boolean
}

/** Applies what an event kind states for certain; anything else asks for an authoritative reload. */
function patchRunner(r: RunnerView, ev: PoolEvent): Patched {
  const src = r.sources.find((s) => s.id === ev.sourceId)
  switch (ev.kind) {
    case 'runner.cordoned':
      return { runner: { ...r, cordoned: true }, refetch: false }
    case 'runner.uncordoned':
      return { runner: { ...r, cordoned: false }, refetch: false }
    case 'runner.connected':
    case 'runner.reconnected':
      return { runner: { ...r, liveness: 'online', lastContact: ev.at }, refetch: true }
    case 'runner.suspect':
      return { runner: { ...r, liveness: 'suspect' }, refetch: false }
    case 'runner.lost':
      return { runner: { ...r, liveness: 'lost' }, refetch: true }
    case 'runner.key_revoked':
      return { runner: { ...r, liveness: 'offline' }, refetch: true }
    case 'runner.incompatible': {
      const state = ev.detail.state === 'chat_only' ? 'chat_only' : 'incompatible'
      const reason = typeof ev.detail.reason === 'string' && ev.detail.reason !== '' ? ev.detail.reason : r.compat.reason
      return { runner: { ...r, compat: { state, reason } }, refetch: false }
    }
    case 'runner.draining':
      return { runner: { ...r, draining: true }, refetch: true }
    case 'runner.drained':
      return { runner: { ...r, draining: true, drain: { ...(r.drain ?? EMPTY_DRAIN), blocked: [], complete: true } }, refetch: true }
    case 'source.logged_out':
      return src
        ? { runner: withSource(r, { ...src, health: { ...src.health, auth: 'logged_out' } }), refetch: false }
        : { runner: r, refetch: true }
    case 'source.recovered':
      // A restored login says nothing about capacity: an exhausted source stays
      // exhausted until the server reports otherwise (reload confirms).
      return { runner: src ? withSource(r, { ...src, health: { ...src.health, auth: 'ok' } }) : r, refetch: true }
    case 'source.health_changed': {
      const updated = src ? healthFromDetail(src, ev.detail) : null
      return updated ? { runner: withSource(r, updated), refetch: false } : { runner: r, refetch: true }
    }
    default:
      // runner.updated, job.*, workspace.moved, session.moved, pool.*: counts/config changed.
      return { runner: r, refetch: true }
  }
}

function applyPoolEvent(state: PoolState, ev: PoolEvent): Applied<PoolState> {
  if (state.seenEvents.has(ev.id)) return { state, refetch: false }
  const seenEvents = new Set(state.seenEvents)
  seenEvents.add(ev.id)
  const at = ms(ev.at)
  if (ev.runnerId && at < (state.runnerAsOf[ev.runnerId] ?? 0)) {
    // Already superseded by what the snapshot (or a newer event) reflects.
    return { state: { ...state, seenEvents }, refetch: false }
  }
  const events = [ev, ...state.events].slice(0, POOL_EVENT_LIMIT)
  if (!ev.runnerId) return { state: { ...state, seenEvents, events }, refetch: true }
  const runnerAsOf = { ...state.runnerAsOf, [ev.runnerId]: at }
  const idx = state.runners.findIndex((r) => r.id === ev.runnerId)
  if (idx === -1) return { state: { ...state, seenEvents, events, runnerAsOf }, refetch: true }
  const patched = patchRunner(state.runners[idx], ev)
  const runners = patched.runner === state.runners[idx] ? state.runners : state.runners.map((r, i) => (i === idx ? patched.runner : r))
  return { state: { ...state, seenEvents, events, runnerAsOf, runners }, refetch: patched.refetch }
}

function applyExecution(state: PoolState, status: ExecutionStatus, atIso: string): Applied<PoolState> {
  const at = ms(atIso)
  if (at < (state.execAsOf[status.jobId] ?? 0)) return { state, refetch: false }
  return {
    state: { ...state, executions: upsertExecution(state.executions, status), execAsOf: { ...state.execAsOf, [status.jobId]: at } },
    refetch: false,
  }
}

export function applyLiveEvent(state: PoolState, live: LiveEvent): Applied<PoolState> {
  switch (live.kind) {
    case 'pool':
      return applyPoolEvent(state, live.event)
    case 'execution':
      return applyExecution(state, live.status, live.at)
    case 'node_done': {
      const executions = pruneExecutions(state.executions, live.instanceId, live.nodeId)
      return { state: executions === state.executions ? state : { ...state, executions }, refetch: false }
    }
    default:
      return { state, refetch: false }
  }
}
```

```ts
import { useCallback, useEffect, useRef, useState } from 'react'
import { decodeLiveEvent, type LiveEvent } from '../api/runnerDecode'
import type { WebSocketEvent } from '../api/types'
import { useRefetchOnReconnect } from './useRefetchOnReconnect'
import { useSystemWebSocket } from './useSystemWebSocket'

/** Poll interval while the system socket is down (precedent: GRAPH_STATE_POLL_MS). */
export const POOL_POLL_MS = 5000
/**
 * Refresh interval while connected. Heartbeat telemetry is deliberately not a
 * bus event (the Phase 2 hub keeps a sample window instead), so live CPU,
 * memory and heartbeat age are refreshed from REST at this slower cadence.
 */
export const POOL_REFRESH_MS = 15_000
/** Coalesces bursts of reload-worthy events into one REST read. */
export const POOL_EVENT_DEBOUNCE_MS = 300

/** Result of applying one live event: the next state and whether a REST reload is needed. */
export interface Applied<S> {
  state: S
  refetch: boolean
}

export interface LiveSnapshotOptions<S> {
  /** False: nothing is fetched and no state is kept. */
  enabled: boolean
  /** Changing the subject resets state and reloads. */
  subject: string
  load: () => Promise<S>
  apply: (state: S, live: LiveEvent) => Applied<S>
}

export interface LiveSnapshot<S> {
  /** Latest applied snapshot + live events; null until the first successful load. */
  state: S | null
  loading: boolean
  error: string | null
  connected: boolean
  /** True while the system socket is down: views show a stale banner while this hook polls. */
  stale: boolean
  refetch: () => Promise<void>
}

/**
 * REST snapshot + live system-stream merge. Events that arrive while a reload
 * is in flight are buffered and applied on top of the new snapshot (anything
 * already in it is skipped by the reducer's id/At ordering); events processed
 * before a reload began are reflected by the snapshot itself and never
 * replayed. Reload-worthy events are debounced. Disconnected: polls every
 * POOL_POLL_MS and resyncs on reconnect; connected: refreshes every
 * POOL_REFRESH_MS for telemetry.
 */
export function useLiveSnapshot<S>({ enabled, subject, load, apply }: LiveSnapshotOptions<S>): LiveSnapshot<S> {
  const [state, setState] = useState<S | null>(null)
  const [loading, setLoading] = useState(enabled)
  const [error, setError] = useState<string | null>(null)
  const stateRef = useRef<S | null>(null)
  const loadRef = useRef(load)
  loadRef.current = load
  const applyRef = useRef(apply)
  applyRef.current = apply
  const issued = useRef(0)
  const inFlight = useRef(0)
  const pending = useRef<LiveEvent[]>([])
  const debounce = useRef<ReturnType<typeof setTimeout> | null>(null)

  const commit = useCallback((next: S | null) => {
    stateRef.current = next
    setState(next)
  }, [])

  const refetch = useCallback(async () => {
    if (!enabled) return
    const seq = ++issued.current
    inFlight.current += 1
    try {
      const snapshot = await loadRef.current()
      if (seq !== issued.current) return
      let next = snapshot
      for (const live of pending.current) next = applyRef.current(next, live).state
      pending.current = []
      commit(next)
      setError(null)
    } catch (err) {
      if (seq !== issued.current) return
      setError(err instanceof Error ? err.message : 'Failed to load')
      const current = stateRef.current
      if (current !== null) {
        let next = current
        for (const live of pending.current) next = applyRef.current(next, live).state
        if (next !== current) commit(next)
      }
      pending.current = []
    } finally {
      inFlight.current -= 1
      if (seq === issued.current) setLoading(false)
    }
  }, [enabled, commit])

  const refetchRef = useRef(refetch)
  refetchRef.current = refetch

  useEffect(() => {
    issued.current += 1 // fence results for the previous subject
    pending.current = []
    commit(null)
    setError(null)
    setLoading(enabled)
    if (enabled) void refetchRef.current()
    return () => {
      issued.current += 1
      if (debounce.current) clearTimeout(debounce.current)
      debounce.current = null
    }
  }, [subject, enabled, commit])

  const scheduleRefetch = useCallback(() => {
    if (debounce.current) clearTimeout(debounce.current)
    debounce.current = setTimeout(() => {
      debounce.current = null
      void refetchRef.current()
    }, POOL_EVENT_DEBOUNCE_MS)
  }, [])

  const onEvent = useCallback(
    (event: WebSocketEvent) => {
      const live = decodeLiveEvent(event)
      if (live.kind === 'ignored') return
      if (inFlight.current > 0) {
        pending.current.push(live)
        return
      }
      const current = stateRef.current
      if (current === null) return
      const { state: next, refetch: wanted } = applyRef.current(current, live)
      if (next !== current) commit(next)
      if (wanted) scheduleRefetch()
    },
    [commit, scheduleRefetch],
  )

  const { connected } = useSystemWebSocket(onEvent)
  useRefetchOnReconnect(connected, () => {
    void refetchRef.current()
  })

  const interval = connected ? POOL_REFRESH_MS : POOL_POLL_MS
  useEffect(() => {
    if (!enabled) return
    const timer = setInterval(() => {
      if (inFlight.current === 0) void refetchRef.current()
    }, interval)
    return () => clearInterval(timer)
  }, [enabled, interval])

  return { state, loading, error, connected, stale: enabled && !connected, refetch }
}
```

```ts
/**
 * Execution statuses of the runs the caller may see (the server already
 * filters the list by visibility), decoded tolerantly and tagged with the
 * owning run when the entry omits it.
 */
export function executionsOf(rows: EnrichedInstanceSummary[]): ExecutionStatus[] {
  const out: ExecutionStatus[] = []
  for (const row of rows) {
    if (!Array.isArray(row.execution)) continue
    for (const raw of row.execution as unknown[]) {
      const obj = typeof raw === 'object' && raw !== null ? (raw as Record<string, unknown>) : {}
      const runId = typeof obj.runId === 'string' && obj.runId !== '' ? obj.runId : row.id
      const status = decodeExecutionStatus({ ...obj, runId })
      if (status) out.push(status)
    }
  }
  return out
}

/**
 * One pool snapshot. Running AND waiting runs: an idle interactive session
 * recovering capacity is waiting at the instance level (design P5 §5).
 */
export async function loadPool(): Promise<PoolState> {
  const [runners, events, running, waiting] = await Promise.all([
    listRunners(),
    listPoolEvents(POOL_EVENT_LIMIT),
    api.instances.list({ state: 'running' }),
    api.instances.list({ state: 'waiting' }),
  ])
  return snapshotState(runners, events, executionsOf([...running, ...waiting]))
}

  return useLiveSnapshot({ enabled: true, subject: 'pool', load: loadPool, apply: applyLiveEvent })
}
```

```ts
import { useCallback } from 'react'
import type { LiveEvent } from '../api/runnerDecode'
import { getRunner, listPoolEvents } from '../api/runners'
import type { RunnerDetail } from '../api/types'
import { applyLiveEvent, POOL_EVENT_LIMIT, snapshotState, type PoolState } from './poolState'
import { useLiveSnapshot, type Applied, type LiveSnapshot } from './useLiveSnapshot'

export interface RunnerState {
  detail: RunnerDetail
  /** One-runner pool state: this runner's events and ordering watermarks. */
  pool: PoolState
}

/** Only this runner's pool events patch the detail; everything else is ignored. */
export function applyRunnerEvent(s: RunnerState, live: LiveEvent, runnerId: string): Applied<RunnerState> {
  if (live.kind !== 'pool' || live.event.runnerId !== runnerId) return { state: s, refetch: false }
  const { state: pool, refetch } = applyLiveEvent(s.pool, live)
  if (pool === s.pool) return { state: s, refetch }
  const patched = pool.runners.find((r) => r.id === runnerId)
  return { state: { pool, detail: patched ? { ...s.detail, ...patched } : s.detail }, refetch }
}

export function useRunner(id: string | undefined): LiveSnapshot<RunnerState> {
  const load = useCallback(async (): Promise<RunnerState> => {
    if (!id) throw new Error('no runner id')
    const [detail, events] = await Promise.all([getRunner(id), listPoolEvents(POOL_EVENT_LIMIT)])
    return { detail, pool: snapshotState([detail], events.filter((e) => e.runnerId === id), []) }
  }, [id])
  const apply = useCallback((s: RunnerState, live: LiveEvent) => applyRunnerEvent(s, live, id ?? ''), [id])
  return useLiveSnapshot({ enabled: Boolean(id), subject: id ?? '', load, apply })
}
```
