**Goal:** Build every driven adapter behind the `domain/runneragent` ports, then compose them in `cmd/arbitron-runner/main.go`. A Mac can then run `arbitron-runner run`, dial the server with its runner key, prove its machine key, advertise its own sources, execute hermetic chat steps in throw-away scratch directories, and spool output until the server acknowledges it.

**Architecture:** The adapters under `adapter/runneragent/*` are:
- `config`: reads `~/.arbitron-runner/config.yaml` and `runner.env`;
- `identity`: the ed25519 machine key;
- `spool`: a durable append-log in front of `runneragent.MemorySpool`;
- `wsclient`: nhooyr WebSocket, translating `runneragent` mirrors to and from `runnerproto`;
- `executor`: chat through the existing `adapter/provider.Factory` and `adapter/claudecli`;
- `sources`: local discovery and read-only probes.

`cmd/arbitron-runner` is the runner's composition root. Its `buildAgent` is the only place these adapters are constructed. A test drives `buildAgent` end to end against a WebSocket stub that speaks `runnerproto`.

**Tech Stack:** Go 1.25.5, `nhooyr.io/websocket` v1.8.17, koanf (yaml parser + file provider), `crypto/ed25519` + `crypto/x509` (PKCS#8 PEM), cobra, testify.

**Decisions:**
1. **`openai_subscription` chat is rejected** with an explicit reason. `adapter/provider.Factory.ChatProviderFor` (factory.go:192) has no codex chat arm because codex is agent-only. The runner therefore does not advertise an `openai_subscription` source in Phase 2, and the hub never places such a step on it.
2. **Probers.** The runner implements its own `runner.SourceProber`s:
   - `sources.ClaudeAuth` runs `claude auth status` with `procenv.BaseEnv()`, so a stray `ANTHROPIC_API_KEY` is never forwarded.
   - `sources.LunaRouteAuth` wraps the existing `adapter/probe` LunaRoute probe with the runner's own key.

   P1 T13's server probers have no constructor in the contracts and read server connections, so the runner implements its own against the same contract port.
3. **Spool.** The runner's bounded, unacked in-memory window is `runneragent.MemorySpool` (P2 T8), which already satisfies the backpressure semantics under the shared `spooltest` suite. `FileSpool` adds a durable per-job append log in front of it. `runnerproto.OutputRing` is not used on the runner side; its sequence numbering is server-assigned, while the runner's `Spool` contract validates agent-assigned sequence numbers.

// Load reads configPath and the runner.env beside it.
func Load(configPath string) (Config, error) {
	abs, err := filepath.Abs(configPath)
	if err != nil {
		return Config{}, fmt.Errorf("runner config: %w", err)
	}
	k := koanf.New(".")
	if err := k.Load(koanffile.Provider(abs), yaml.Parser()); err != nil {
		return Config{}, fmt.Errorf("runner config %s: %w", abs, err)
	}
	for _, key := range k.Keys() {
		if !allowedKeys[key] {
			return Config{}, fmt.Errorf("runner config %s: unknown key %q", abs, key)
		}
	}
	var fc fileConfig
	if err := k.Unmarshal("", &fc); err != nil {
		return Config{}, fmt.Errorf("runner config %s: %w", abs, err)
	}

	dir := filepath.Dir(abs)
	cfg := Config{ConfigDir: dir, ClaudeBinary: strings.TrimSpace(fc.ClaudeBinary), CodexBinary: strings.TrimSpace(fc.CodexBinary)}
	if cfg.WorkspaceRoot, err = absPath("workspace_root", fc.WorkspaceRoot, ""); err != nil {
		return Config{}, err
	}
	if cfg.ScratchRoot, err = absPath("scratch_root", fc.ScratchRoot, filepath.Join(dir, "scratch")); err != nil {
		return Config{}, err
	}
	if cfg.SpoolDir, err = absPath("spool_dir", fc.SpoolDir, filepath.Join(dir, "spool")); err != nil {
		return Config{}, err
	}
	if cfg.Heartbeat, err = seconds(k, "heartbeat_seconds", fc.HeartbeatSeconds, defaultHeartbeat, 1, 60); err != nil {
		return Config{}, err
	}
	if cfg.ProbeInterval, err = seconds(k, "probe_interval_seconds", fc.ProbeIntervalSeconds, defaultProbeInterval, 5, 3600); err != nil {
		return Config{}, err
	}
	cfg.OutputBufferFrames = defaultOutputBufferFrames
	if k.Exists("output_buffer_frames") {
		if fc.OutputBufferFrames < 16 || fc.OutputBufferFrames > 1_000_000 {
			return Config{}, fmt.Errorf("runner config: output_buffer_frames must be 16..1000000, got %d", fc.OutputBufferFrames)
		}
		cfg.OutputBufferFrames = fc.OutputBufferFrames
	}

	env, err := readEnvFile(filepath.Join(dir, EnvFileName))
	if err != nil {
		return Config{}, err
	}
	if cfg.ServerURL, err = validateServerURL(env["ARBITRON_SERVER_URL"]); err != nil {
		return Config{}, err
	}
	key := env["ARBITRON_RUNNER_KEY"]
	if !strings.HasPrefix(key, "ark_") || len(key) <= len("ark_") {
		return Config{}, errors.New("runner config: ARBITRON_RUNNER_KEY in runner.env must be an ark_ runner key")
	}
	cfg.RunnerKey = key
	if cfg.ExtraHeaders, err = ParseExtraHeaders(env["ARBITRON_EXTRA_HEADERS"]); err != nil {
		return Config{}, fmt.Errorf("runner config: %w", err)
	}
	if cfg.LunaRoute, err = lunaRoute(fc, env["LUNAROUTE_API_KEY"]); err != nil {
		return Config{}, err
	}
	return cfg, nil
}

func absPath(name, val, def string) (string, error) {
	v := strings.TrimSpace(val)
	if v == "" {
		v = def
	}
	if v == "" {
		return "", fmt.Errorf("runner config: %s is required", name)
	}
	if !filepath.IsAbs(v) {
		return "", fmt.Errorf("runner config: %s must be an absolute path, got %q", name, v)
	}
	return filepath.Clean(v), nil
}

func seconds(k *koanf.Koanf, key string, v int, def time.Duration, lo, hi int) (time.Duration, error) {
	if !k.Exists(key) {
		return def, nil
	}
	if v < lo || v > hi {
		return 0, fmt.Errorf("runner config: %s must be %d..%d, got %d", key, lo, hi, v)
	}
	return time.Duration(v) * time.Second, nil
}

func lunaRoute(fc fileConfig, apiKey string) (LunaRoute, error) {
	lr := LunaRoute{ConnectionRef: strings.TrimSpace(fc.LunaRoute.ConnectionRef), Models: fc.LunaRoute.Models, APIKey: strings.TrimSpace(apiKey)}
	for _, f := range fc.LunaRoute.Families {
		f = strings.TrimSpace(f)
		if !familyRE.MatchString(f) || modelidentity.Family(f) == modelidentity.FamilyUnresolved {
			return LunaRoute{}, fmt.Errorf("runner config: lunaroute.families entry %q is not a model family", f)
		}
		lr.Families = append(lr.Families, modelidentity.Family(f))
	}
	if lr.APIKey != "" && !provider.ValidLunaRouteKey(lr.APIKey) {
		return LunaRoute{}, errors.New("runner config: LUNAROUTE_API_KEY in runner.env is not a LunaRoute key (lr_...)")
	}
	configured := lr.ConnectionRef != "" || len(lr.Families) > 0
	if configured && (lr.ConnectionRef == "" || len(lr.Families) == 0) {
		return LunaRoute{}, errors.New("runner config: lunaroute needs both connection_ref and families")
	}
	if configured && lr.APIKey == "" {
		return LunaRoute{}, errors.New("runner config: lunaroute is configured but LUNAROUTE_API_KEY is missing from runner.env")
	}
	return lr, nil
}

// readEnvFile parses KEY=VALUE lines. The file holds the runner key, so it
// must not be readable by group or others.
func readEnvFile(path string) (map[string]string, error) {
	info, err := os.Stat(path)
	if err != nil {
		return nil, fmt.Errorf("runner config: %s: %w", path, err)
	}
	if perm := info.Mode().Perm(); perm&0o077 != 0 {
		return nil, fmt.Errorf("runner config: %s has mode %04o; it holds the runner key and must be 0600 (chmod 600 %s)", path, perm, path)
	}
	f, err := os.Open(path)
	if err != nil {
		return nil, fmt.Errorf("runner config: %s: %w", path, err)
	}
	defer func() { _ = f.Close() }()
	out := map[string]string{}
	sc := bufio.NewScanner(f)
	for n := 1; sc.Scan(); n++ {
		line := strings.TrimSpace(sc.Text())
		if line == "" || strings.HasPrefix(line, "#") {
			continue
		}
		line = strings.TrimPrefix(line, "export ")
		key, val, ok := strings.Cut(line, "=")
		if !ok || strings.TrimSpace(key) == "" {
			return nil, fmt.Errorf("runner config: %s line %d is not KEY=VALUE", path, n)
		}
		out[strings.TrimSpace(key)] = unquote(strings.TrimSpace(val))
	}
	if err := sc.Err(); err != nil {
		return nil, fmt.Errorf("runner config: %s: %w", path, err)
	}
	return out, nil
}

func unquote(v string) string {
	if len(v) >= 2 && (v[0] == '"' && v[len(v)-1] == '"' || v[0] == '\'' && v[len(v)-1] == '\'') {
		return v[1 : len(v)-1]
	}
	return v
}

func validateServerURL(raw string) (string, error) {
	raw = strings.TrimSpace(raw)
	if raw == "" {
		return "", errors.New("runner config: ARBITRON_SERVER_URL is missing from runner.env")
	}
	u, err := url.Parse(raw)
	if err != nil || u.Host == "" || u.User != nil || u.RawQuery != "" || u.Fragment != "" || u.Opaque != "" {
		return "", errors.New("runner config: ARBITRON_SERVER_URL must be an absolute URL without credentials, query or fragment")
	}
	switch u.Scheme {
	case "https":
	case "http":
		if !isLoopback(u.Hostname()) {
			return "", errors.New("runner config: ARBITRON_SERVER_URL must use https (plain http is only allowed for a loopback server)")
		}
	default:
		return "", errors.New("runner config: ARBITRON_SERVER_URL must use https")
	}
	return strings.TrimRight(u.String(), "/"), nil
}

func isLoopback(host string) bool {
	if host == "localhost" {
		return true
	}
	ip := net.ParseIP(host)
	return ip != nil && ip.IsLoopback()
}

// ParseExtraHeaders parses ARBITRON_EXTRA_HEADERS ("Name: value;Name2: value2").
// Headers that would override authentication or the WebSocket handshake are refused.
func ParseExtraHeaders(s string) (http.Header, error) {
	h := http.Header{}
	for i, part := range strings.Split(s, ";") {
		part = strings.TrimSpace(part)
		if part == "" {
			continue
		}
		name, value, ok := strings.Cut(part, ":")
		name, value = strings.TrimSpace(name), strings.TrimSpace(value)
		if !ok || name == "" || value == "" || !headerNameRE.MatchString(name) || strings.ContainsAny(value, "\r\n") {
			return nil, fmt.Errorf("ARBITRON_EXTRA_HEADERS entry %d must be \"Name: value\"", i+1)
		}
		canon := http.CanonicalHeaderKey(name)
		if forbiddenHdr[canon] {
			return nil, fmt.Errorf("ARBITRON_EXTRA_HEADERS may not set %s", canon)
		}
		h.Add(canon, value)
	}
	return h, nil
}

// Package identity stores the runner's ed25519 machine key (PKCS#8 PEM, 0600).
package identity

import (
	"context"
	"crypto/ed25519"
	"crypto/rand"
	"crypto/x509"
	"encoding/pem"
	"errors"
	"fmt"
	"io/fs"
	"os"
	"path/filepath"
	"sync"
)

// FileStore implements runneragent.IdentityStore.
type FileStore struct {
	path string
	mu   sync.Mutex
	key  ed25519.PrivateKey
}

// New returns a store for the key at path.
func New(path string) *FileStore { return &FileStore{path: path} }

// Key loads the machine key, creating it on first use.
func (s *FileStore) Key(ctx context.Context) (ed25519.PrivateKey, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	if s.key != nil {
		return s.key, nil
	}
	k, err := s.load()
	if errors.Is(err, fs.ErrNotExist) {
		k, err = s.create()
		if errors.Is(err, fs.ErrExist) { // another process created it first
			k, err = s.load()
		}
	}
	if err != nil {
		return nil, err
	}
	s.key = k
	return k, nil
}

func (s *FileStore) load() (ed25519.PrivateKey, error) {
	info, err := os.Stat(s.path)
	if err != nil {
		return nil, err
	}
	if perm := info.Mode().Perm(); perm&0o077 != 0 {
		return nil, fmt.Errorf("identity: %s has mode %04o; the machine key must be 0600", s.path, perm)
	}
	b, err := os.ReadFile(s.path)
	if err != nil {
		return nil, err
	}
	blk, _ := pem.Decode(b)
	if blk == nil || blk.Type != "PRIVATE KEY" {
		if len(b) == 0 { // a concurrent creator has not written yet
			return nil, fmt.Errorf("identity: %s is empty", s.path)
		}
		return nil, fmt.Errorf("identity: %s is not a PEM private key", s.path)
	}
	parsed, err := x509.ParsePKCS8PrivateKey(blk.Bytes)
	if err != nil {
		return nil, fmt.Errorf("identity: %s: %w", s.path, err)
	}
	k, ok := parsed.(ed25519.PrivateKey)
	if !ok {
		return nil, fmt.Errorf("identity: %s is not an ed25519 key", s.path)
	}
	return k, nil
}

func (s *FileStore) create() (ed25519.PrivateKey, error) {
	if err := os.MkdirAll(filepath.Dir(s.path), 0o700); err != nil {
		return nil, fmt.Errorf("identity: %w", err)
	}
	_, priv, err := ed25519.GenerateKey(rand.Reader)
	if err != nil {
		return nil, fmt.Errorf("identity: generate: %w", err)
	}
	der, err := x509.MarshalPKCS8PrivateKey(priv)
	if err != nil {
		return nil, fmt.Errorf("identity: encode: %w", err)
	}
	tmp, err := os.CreateTemp(filepath.Dir(s.path), ".machine_ed25519-*")
	if err != nil {
		return nil, fmt.Errorf("identity: %w", err)
	}
	defer func() { _ = os.Remove(tmp.Name()) }()
	if err := tmp.Chmod(0o600); err != nil {
		_ = tmp.Close()
		return nil, err
	}
	if err := pem.Encode(tmp, &pem.Block{Type: "PRIVATE KEY", Bytes: der}); err != nil {
		_ = tmp.Close()
		return nil, err
	}
	if err := tmp.Sync(); err != nil {
		_ = tmp.Close()
		return nil, err
	}
	if err := tmp.Close(); err != nil {
		return nil, err
	}
	// os.Link fails with ErrExist when another process won the race, so the
	// published key is complete and never overwritten.
	if err := os.Link(tmp.Name(), s.path); err != nil {
		return nil, err
	}
	return priv, nil
}

// Package spool is the runner's durable output spool: a per-job append log in
// front of the bounded runneragent.MemorySpool.
package spool

import (
	"bufio"
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"io/fs"
	"os"
	"path/filepath"
	"sort"
	"strconv"
	"strings"
	"sync"

	"github.com/google/uuid"

	"github.com/arbitron/arbitron/domain/runneragent"
)

type record struct {
	Op     string              `json:"op"` // "out" | "ack" | "res"
	Seq    uint64              `json:"seq,omitempty"`
	Token  string              `json:"tok,omitempty"`
	Result *runneragent.Result `json:"res,omitempty"`
}

// FileSpool implements runneragent.Spool with crash-durable terminal results.
type FileSpool struct {
	dir string
	mem *runneragent.MemorySpool
	mu  sync.Mutex
}

var _ runneragent.Spool = (*FileSpool)(nil)

// Open replays every job log under dir into a spool bounded by capacity.
func Open(dir string, capacity int) (*FileSpool, error) {
	if err := os.MkdirAll(dir, 0o700); err != nil {
		return nil, fmt.Errorf("spool: %w", err)
	}
	s := &FileSpool{dir: dir, mem: runneragent.NewMemorySpool(capacity)}
	logs, err := filepath.Glob(filepath.Join(dir, "*.log"))
	if err != nil {
		return nil, fmt.Errorf("spool: %w", err)
	}
	sort.Strings(logs)
	for _, p := range logs {
		if err := s.replay(p); err != nil {
			return nil, err
		}
	}
	return s, nil
}

func checkRef(ref runneragent.JobRef) error {
	if _, err := uuid.Parse(ref.JobID); err != nil || ref.Generation < 1 {
		return fmt.Errorf("spool: invalid job reference %q/%d", ref.JobID, ref.Generation)
	}
	return nil
}

func (s *FileSpool) path(ref runneragent.JobRef) string {
	return filepath.Join(s.dir, fmt.Sprintf("%s.%d.log", ref.JobID, ref.Generation))
}

func parseName(p string) (runneragent.JobRef, bool) {
	base := strings.TrimSuffix(filepath.Base(p), ".log")
	i := strings.LastIndex(base, ".")
	if i < 0 {
		return runneragent.JobRef{}, false
	}
	gen, err := strconv.ParseInt(base[i+1:], 10, 64)
	ref := runneragent.JobRef{JobID: base[:i], Generation: gen}
	return ref, err == nil && checkRef(ref) == nil
}

func (s *FileSpool) replay(p string) error {
	ref, ok := parseName(p)
	if !ok {
		return nil // not ours
	}
	f, err := os.Open(p)
	if err != nil {
		return fmt.Errorf("spool: %w", err)
	}
	defer func() { _ = f.Close() }()
	ctx := context.Background()
	sc := bufio.NewScanner(f)
	sc.Buffer(make([]byte, 64<<10), 4<<20)
	for sc.Scan() {
		var r record
		if err := json.Unmarshal(sc.Bytes(), &r); err != nil {
			break // truncated tail from a crash mid-write
		}
		switch r.Op {
		case "out":
			if err := s.mem.AppendOutput(ctx, ref, runneragent.OutputFrame{Seq: r.Seq, Token: r.Token}); err != nil {
				return fmt.Errorf("spool: replay %s: %w (raise output_buffer_frames or clear the spool)", filepath.Base(p), err)
			}
		case "ack":
			_ = s.mem.Ack(ctx, ref, r.Seq)
		case "res":
			if r.Result != nil {
				_ = s.mem.PutResult(ctx, ref, *r.Result)
			}
		}
	}
	return nil
}

func (s *FileSpool) appendRecord(ref runneragent.JobRef, r record, durable bool) error {
	b, err := json.Marshal(r)
	if err != nil {
		return fmt.Errorf("spool: %w", err)
	}
	f, err := os.OpenFile(s.path(ref), os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o600)
	if err != nil {
		return fmt.Errorf("spool: %w", err)
	}
	if _, err := f.Write(append(b, '\n')); err != nil {
		_ = f.Close()
		return fmt.Errorf("spool: %w", err)
	}
	if durable {
		if err := f.Sync(); err != nil {
			_ = f.Close()
			return fmt.Errorf("spool: %w", err)
		}
	}
	return f.Close()
}

// AppendOutput spools f; ErrSpoolFull applies backpressure, nothing is dropped.
func (s *FileSpool) AppendOutput(ctx context.Context, ref runneragent.JobRef, f runneragent.OutputFrame) error {
	if err := checkRef(ref); err != nil {
		return err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if err := s.mem.AppendOutput(ctx, ref, f); err != nil {
		return err
	}
	return s.appendRecord(ref, record{Op: "out", Seq: f.Seq, Token: f.Token}, false)
}

// PutResult durably retains the terminal result (fsync).
func (s *FileSpool) PutResult(ctx context.Context, ref runneragent.JobRef, r runneragent.Result) error {
	if err := checkRef(ref); err != nil {
		return err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if err := s.mem.PutResult(ctx, ref, r); err != nil {
		return err
	}
	res := r
	return s.appendRecord(ref, record{Op: "res", Result: &res}, true)
}

// Ack drops frames up to seq.
func (s *FileSpool) Ack(ctx context.Context, ref runneragent.JobRef, seq uint64) error {
	if err := checkRef(ref); err != nil {
		return err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if _, ok, _ := s.mem.Get(ctx, ref); !ok {
		return nil
	}
	if err := s.mem.Ack(ctx, ref, seq); err != nil {
		return err
	}
	return s.appendRecord(ref, record{Op: "ack", Seq: seq}, false)
}

// AckResult forgets a job whose result the server durably accepted.
func (s *FileSpool) AckResult(ctx context.Context, ref runneragent.JobRef) error {
	return s.Drop(ctx, ref)
}

// Drop forgets the job and deletes its log.
func (s *FileSpool) Drop(ctx context.Context, ref runneragent.JobRef) error {
	if err := checkRef(ref); err != nil {
		return err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if err := s.mem.Drop(ctx, ref); err != nil {
		return err
	}
	if err := os.Remove(s.path(ref)); err != nil && !errors.Is(err, fs.ErrNotExist) {
		return fmt.Errorf("spool: %w", err)
	}
	return nil
}

// Get returns one job's spooled state.
func (s *FileSpool) Get(ctx context.Context, ref runneragent.JobRef) (runneragent.SpooledJob, bool, error) {
	return s.mem.Get(ctx, ref)
}

// Pending returns every spooled job.
func (s *FileSpool) Pending(ctx context.Context) ([]runneragent.SpooledJob, error) {
	return s.mem.Pending(ctx)
}

// toEnvelope translates a runner domain message into a protocol frame.
func toEnvelope(m runneragent.Outbound) (runnerproto.Envelope, error) {
	switch m.Kind {
	case runneragent.KindChallengeResponse:
		if m.ChallengeResponse == nil {
			break
		}
		return runnerproto.NewEnvelope(runnerproto.KindChallengeResponse, runnerproto.ChallengeResponse{
			PublicKey: base64.StdEncoding.EncodeToString(m.ChallengeResponse.PublicKey),
			Signature: base64.StdEncoding.EncodeToString(m.ChallengeResponse.Signature),
		})
	case runneragent.KindHello:
		if m.Hello == nil {
			break
		}
		h := m.Hello
		return runnerproto.NewEnvelope(runnerproto.KindHello, runnerproto.Hello{
			OS: h.OS, Arch: h.Arch, ProtocolVersion: h.ProtocolVersion, RunnerVersion: h.RunnerVersion,
			HarnessVersions: h.HarnessVersions, WorkspaceRoot: h.WorkspaceRoot, ScratchRoot: h.ScratchRoot,
			Capabilities: h.Capabilities, Sources: wireSources(h.Sources), Held: wireHeld(h.Held),
		})
	case runneragent.KindHeartbeat:
		if m.Heartbeat == nil {
			break
		}
		return runnerproto.NewEnvelope(runnerproto.KindHeartbeat, runnerproto.Heartbeat{
			Telemetry: wireTelemetry(m.Heartbeat.Telemetry), Sources: wireSources(m.Heartbeat.Sources), Jobs: wireHeld(m.Heartbeat.Jobs),
		})
	case runneragent.KindAccept:
		return runnerproto.NewJobEnvelope(runnerproto.KindAccept, m.JobID, m.Generation, 0, nil)
	case runneragent.KindReject:
		return runnerproto.NewJobEnvelope(runnerproto.KindReject, m.JobID, m.Generation, 0, runnerproto.Reject{Reason: m.RejectReason})
	case runneragent.KindOutput:
		if m.Output == nil {
			break
		}
		return runnerproto.NewJobEnvelope(runnerproto.KindOutput, m.JobID, m.Generation, m.Output.Seq, runnerproto.Output{Seq: m.Output.Seq, Token: m.Output.Token})
	case runneragent.KindResult:
		if m.Result == nil {
			break
		}
		r := runnerproto.Result{OK: m.Result.OK, Outcome: wireOutcome(m.Result.Outcome), ErrorText: m.Result.ErrorText}
		if m.Result.OK && m.Result.Chat != nil {
			c := m.Result.Chat
			r.Chat = &runnerproto.WireChatResult{Output: c.Output, Model: c.Model, TokensIn: c.TokensIn, TokensOut: c.TokensOut, SessionID: c.SessionID}
		}
		return runnerproto.NewJobEnvelope(runnerproto.KindResult, m.JobID, m.Generation, 0, r)
	case runneragent.KindProbeResult:
		if m.ProbeResult == nil {
			break
		}
		return runnerproto.NewEnvelope(runnerproto.KindProbeResult, runnerproto.ProbeResult{Source: wireSource(*m.ProbeResult)})
	}
	return runnerproto.Envelope{}, fmt.Errorf("wsclient: cannot encode %q message", m.Kind)
}

// fromEnvelope translates a server frame into the runner domain mirror.
// Kinds the runner does not handle pass through with only their envelope
// fields so the agent can ignore them.
func fromEnvelope(e runnerproto.Envelope) (runneragent.Inbound, error) {
	in := runneragent.Inbound{Kind: e.Kind, JobID: e.JobID, Generation: e.Generation}
	switch e.Kind {
	case runnerproto.KindChallenge:
		var c runnerproto.Challenge
		if err := runnerproto.DecodePayload(e, &c); err != nil {
			return runneragent.Inbound{}, err
		}
		nonce, err := base64.StdEncoding.DecodeString(c.Nonce)
		if err != nil {
			return runneragent.Inbound{}, fmt.Errorf("%w: challenge nonce", runnerproto.ErrMalformed)
		}
		in.Challenge = &runneragent.Challenge{KeyID: c.KeyID, Nonce: nonce}
	case runnerproto.KindAdmitted:
		var a runnerproto.Admitted
		if err := runnerproto.DecodePayload(e, &a); err != nil {
			return runneragent.Inbound{}, err
		}
		in.Admitted = &runneragent.Admission{RunnerID: a.RunnerID, HeartbeatSeconds: a.HeartbeatSeconds}
	case runnerproto.KindRefused:
		var r runnerproto.Refused
		if err := runnerproto.DecodePayload(e, &r); err != nil {
			return runneragent.Inbound{}, err
		}
		in.Refused = &runneragent.Refusal{Code: r.Code, Reason: r.Reason}
	case runnerproto.KindDispatch:
		var d runnerproto.Dispatch
		if err := runnerproto.DecodePayload(e, &d); err != nil {
			return runneragent.Inbound{}, err
		}
		c := d.Chat
		in.Dispatch = &runneragent.Dispatch{JobID: e.JobID, Generation: e.Generation, Op: d.Op, LeaseUntil: d.LeaseUntil, Chat: &runneragent.ChatJob{
			Model: c.Model, SystemPrompt: c.SystemPrompt, Prompt: c.Prompt, MaxTokens: c.MaxTokens, Temperature: c.Temperature,
			OutputSchema: c.OutputSchema, SessionID: c.SessionID, RequireExecutedModel: c.RequireExecutedModel, Stream: c.Stream,
			CredentialKind: c.CredentialKind, Backend: c.Backend, ContextWindow: c.ContextWindow, Effort: c.Effort, Tools: c.Tools,
		}}
	case runnerproto.KindAck:
		var a runnerproto.Ack
		if err := runnerproto.DecodePayload(e, &a); err != nil {
			return runneragent.Inbound{}, err
		}
		in.Seq, in.AckResult = a.Seq, a.Result
	case runnerproto.KindProbe:
		var p runnerproto.Probe
		if err := runnerproto.DecodePayload(e, &p); err != nil {
			return runneragent.Inbound{}, err
		}
		in.Probe = &runneragent.ProbeRequest{SourceKind: runner.CredentialKind(p.SourceKind), CredentialRef: p.CredentialRef}
	}
	return in, nil
}

package wsclient

import (
	"context"
	"errors"
	"fmt"
	"net/http"
	"strings"
	"sync"

	"nhooyr.io/websocket"

	"github.com/arbitron/arbitron/adapter/runnerproto"
	"github.com/arbitron/arbitron/domain/runneragent"
)

// ConnectPath is the server's runner WebSocket endpoint.
const ConnectPath = "/api/v1/runners/connect"

// Options configures the client. RunnerKey travels only in the
// Authorization header.
type Options struct {
	ServerURL  string
	RunnerKey  string
	Headers    http.Header
	HTTPClient *http.Client
}

// Client implements runneragent.Connection.
type Client struct {
	endpoint   string
	key        string
	headers    http.Header
	httpClient *http.Client
}

var _ runneragent.Connection = (*Client)(nil)

// New validates o and returns a client.
func New(o Options) (*Client, error) {
	base := strings.TrimRight(strings.TrimSpace(o.ServerURL), "/")
	if base == "" {
		return nil, errors.New("wsclient: server URL is required")
	}
	if !strings.HasPrefix(o.RunnerKey, "ark_") {
		return nil, errors.New("wsclient: runner key must be an ark_ key")
	}
	for name := range o.Headers {
		if c := http.CanonicalHeaderKey(name); c == "Authorization" || c == "Host" {
			return nil, fmt.Errorf("wsclient: extra headers may not set %s", c)
		}
	}
	return &Client{endpoint: base + ConnectPath, key: o.RunnerKey, headers: o.Headers.Clone(), httpClient: o.HTTPClient}, nil
}

// Dial opens one session.
func (c *Client) Dial(ctx context.Context) (runneragent.Session, error) {
	h := c.headers.Clone()
	if h == nil {
		h = http.Header{}
	}
	h.Set("Authorization", "Runner "+c.key)
	conn, resp, err := websocket.Dial(ctx, c.endpoint, &websocket.DialOptions{HTTPHeader: h, HTTPClient: c.httpClient})
	if err != nil {
		if resp != nil && (resp.StatusCode == http.StatusUnauthorized || resp.StatusCode == http.StatusForbidden) {
			return nil, &runneragent.RefusedError{Code: "unauthorized", Reason: fmt.Sprintf("server refused the runner key (HTTP %d)", resp.StatusCode)}
		}
		return nil, fmt.Errorf("wsclient: dial %s: %w", c.endpoint, err)
	}
	conn.SetReadLimit(runnerproto.MaxFrameBytes)
	return &session{conn: conn}, nil
}

type session struct {
	conn *websocket.Conn
	wmu  sync.Mutex
}

func (s *session) Send(ctx context.Context, m runneragent.Outbound) error {
	env, err := toEnvelope(m)
	if err != nil {
		return err
	}
	b, err := runnerproto.Encode(env)
	if err != nil {
		return err
	}
	s.wmu.Lock()
	defer s.wmu.Unlock()
	return s.conn.Write(ctx, websocket.MessageText, b)
}

func (s *session) Recv(ctx context.Context) (runneragent.Inbound, error) {
	typ, b, err := s.conn.Read(ctx)
	if err != nil {
		return runneragent.Inbound{}, err
	}
	if typ != websocket.MessageText {
		return runneragent.Inbound{}, fmt.Errorf("%w: binary frame", runnerproto.ErrMalformed)
	}
	env, err := runnerproto.Decode(b)
	if err != nil {
		return runneragent.Inbound{}, err
	}
	return fromEnvelope(env)
}

func (s *session) Close() error {
	return s.conn.Close(websocket.StatusNormalClosure, "runner closing")
}

func New(sel engine.ChatProviderSelector, cfg Config) (*Executor, error) {

	}
	if !filepath.IsAbs(cfg.ScratchRoot) {

// Supports reports whether this runner can run d.
func (e *Executor) Supports(d runneragent.Dispatch) (bool, string) {
	if d.Chat == nil || (d.Op != runneragent.OpChat && d.Op != runneragent.OpStream) {
		return false, fmt.Sprintf("operation %q is not supported by this runner", d.Op)
	}
	switch runner.CredentialKind(d.Chat.CredentialKind) {
	case runner.CredAnthropicSubscription:
		if b := d.Chat.Backend; b != "" && b != "claude-code" {
			return false, fmt.Sprintf("anthropic_subscription chat runs only on the claude harness, not %q", b)
		}
		return true, ""
	case runner.CredLunaRoute:
		if e.cfg.LunaRoute == nil {
			return false, "no LunaRoute key is configured on this runner"
		}
		return true, ""
	case runner.CredOpenAISubscription:
		return false, "openai_subscription chat is not supported on remote runners in this release"
	default:
		return false, fmt.Sprintf("credential kind %q cannot run on this runner", d.Chat.CredentialKind)
	}
}

// Execute runs d in a fresh scratch directory that is removed afterwards.
func (e *Executor) Execute(ctx context.Context, d runneragent.Dispatch, emit runneragent.EmitFunc) runneragent.Result {
	if ok, reason := e.Supports(d); !ok {
		return failure(runner.SourceOutcome{Class: runner.OutcomeOther}, reason)
	}
	dir, err := os.MkdirTemp(e.cfg.ScratchRoot, fmt.Sprintf("job-%s-g%d-", d.JobID, d.Generation))
	if err != nil {
		return failure(runner.SourceOutcome{Class: runner.OutcomeOther}, "runner scratch directory unavailable")
	}
	defer func() { _ = os.RemoveAll(dir) }()

	p, err := e.sel.ChatProviderFor(e.binding(d.Chat))
	if err != nil {
		return e.errResult(err)
	}
	c := d.Chat
	req := engine.ChatRequest{
		Model: c.Model, SystemPrompt: c.SystemPrompt, Prompt: c.Prompt, MaxTokens: c.MaxTokens,
		Temperature: c.Temperature, WorkDir: dir, SessionID: c.SessionID, RequireExecutedModel: c.RequireExecutedModel,
	}
	if len(c.OutputSchema) > 0 {
		var schema any
		if err := json.Unmarshal(c.OutputSchema, &schema); err != nil {
			return failure(runner.SourceOutcome{Class: runner.OutcomeOther}, "invalid output schema")
		}
		req.OutputSchema = schema
	}
	if d.Op == runneragent.OpStream || c.Stream {
		return e.stream(ctx, p, req, emit)
	}
	resp, err := p.Chat(ctx, req)
	if err != nil {
		return e.errResult(err)
	}
	out, err := outputText(resp.Output)
	if err != nil {
		return failure(runner.SourceOutcome{Class: runner.OutcomeOther}, "chat output is not serializable")
	}
	model := resp.Model
	if model == "" {
		model = c.Model
	}
	return runneragent.Result{OK: true, Chat: &runneragent.ChatResult{Output: out, Model: model, SessionID: resp.SessionID, TokensIn: resp.TokensIn, TokensOut: resp.TokensOut}}
}

func (e *Executor) stream(ctx context.Context, p engine.ChatProvider, req engine.ChatRequest, emit runneragent.EmitFunc) runneragent.Result {
	sctx, cancel := context.WithCancel(ctx)
	defer cancel()
	ch, err := p.StreamChat(sctx, req)
	if err != nil {
		return e.errResult(err)
	}
	drain := func() {
		cancel()
		for range ch {
		}
	}
	var sb strings.Builder
	var final engine.ChatChunk
	for c := range ch {
		if c.Error != nil {
			drain()
			return e.errResult(c.Error)
		}
		if c.Token != "" {
			sb.WriteString(c.Token)
			if err := emit(sctx, c.Token); err != nil {
				drain()
				return failure(runner.SourceOutcome{Class: runner.OutcomeOther}, "output delivery stopped")
			}
		}
		if c.Done {
			final = c

  // ClaudeModelVerifier natively confirms the exact Claude models the server
  // requests at admission (contracts D1). Only verified models are advertised.
  type ClaudeModelVerifier struct {
  	confirm runner.ConfirmationExecutor
  	workDir string
  	log     *slog.Logger

  	mu        sync.Mutex
  	requested []string
  	verified  map[string]bool
  	refused   map[string]bool

  func NewClaudeModelVerifier(c runner.ConfirmationExecutor, workDir string, log *slog.Logger) *ClaudeModelVerifier {
  	if c == nil {
  		panic("sources: NewClaudeModelVerifier requires a confirmation executor")
  	}
  	if log == nil {
  		log = slog.New(slog.DiscardHandler)
  	}
  	return &ClaudeModelVerifier{confirm: c, workDir: workDir, log: log, verified: map[string]bool{}, refused: map[string]bool{}}
  }

  // Request replaces the requested set and confirms every uncached model, one
  // at a time. It returns when all are checked or ctx ends.
  func (v *ClaudeModelVerifier) Request(ctx context.Context, models []string) {
  	v.mu.Lock()
  	v.requested = append([]string(nil), models...)
  	v.mu.Unlock()
  	for _, m := range models {
  		if ctx.Err() != nil {
  			return
  		}
  		v.mu.Lock()
  		cached := v.verified[m] || v.refused[m]
  		v.mu.Unlock()
  		if cached {
  			continue
  		}
  		res, err := v.confirm.Confirm(ctx, provider.ResolvedBinding{Kind: provider.ProfileKindChat, Provider: claudeChatProvider, Model: m}, v.workDir)
  		if err == nil {
  			err = runner.VerifyConfirmation(m, &res)
  		}
  		v.mu.Lock()
  		switch {
  		case err == nil:
  			v.verified[m] = true
  		case errors.Is(err, runner.ErrConfirmationModelMismatch):
  			v.refused[m] = true
  			v.log.Warn("claude model not advertised: native confirmation did not prove the exact model", "model", m)
  		default:
  			v.log.Warn("claude model not advertised: native confirmation failed; retried at next admission", "model", m, "error_class", "confirm_failed")
  		}
  		v.mu.Unlock()
  	}
  }

  // Verified returns the sorted requested models that verified.
  func (v *ClaudeModelVerifier) Verified() []string {
  	v.mu.Lock()
  	defer v.mu.Unlock()
  	var out []string
  	for _, m := range v.requested {
  		if v.verified[m] && !slices.Contains(out, m) {
  			out = append(out, m)
  		}
  	}
  	slices.Sort(out)
  	return out
  }
