# Remote Runners — Phase 1 Task 9: ProcessRunner port and LocalProcessRunner

**Goal:** Script and tool_call nodes stop calling `adapter/process/runner.go:run` directly and instead build a fully-resolved `engine.ProcessSpec` and execute it through the new domain `engine.ProcessRunner` port, whose local adapter is `process.LocalProcessRunner`.

**Architecture:** A new port `domain/engine/process_runner.go` describes "run this rendered command with this env, stdin, cwd and timeout". The executors resolve node-env templates on the server (`NodeEnv`), carry passthrough variable NAMES only (values are read on the executing machine), validate env-name collisions before dispatch (so a doomed step never consumes a slot), and call the port. `LocalProcessRunner` is today's `run()` plus `buildEnvFromResolved` (same collision policy as the old `buildEnv`). Task 16 swaps the composition-root runner for the routed one (`runner.NewRoutedProcessRunner`, Task 10).

**Tech Stack:** Go 1.25, `os/exec`, testify.

**Spec / Design:** `docs/specs/2026-11-19-remote-runners-spec.md` (Execution and placement: "Scripts and workspace-based tools run with that workspace"); `docs/designs/2026-11-19-remote-runners-design-phase-1.md` §3.7 (ProcessRunner port), §3.8 (built-in runner working directories), §5 row "Workspace affinity; commands/env/stdin/timeouts" (RR-09). Contracts: `docs/plans/remote-runners/2026-11-20-contracts.md` §5 (Process port), §0 (Process facts), §11 (StepJob.Process).

package engine

import "context"

// ProcessKind names the node family a ProcessSpec was built for.
type ProcessKind string

const (
	ProcessScript   ProcessKind = "script"
	ProcessToolCall ProcessKind = "tool_call"
)

// ProcessSpec is a fully rendered subprocess request. It is built on the server
// by the script/tool_call executors and executed by whichever runner placement
// chose. NodeEnv values are already template-resolved; Passthrough carries env
// variable NAMES only, whose values are read on the executing machine.
type ProcessSpec struct {
	Kind        ProcessKind
	Command     string            // interpreter (script) or command (tool_call)
	Args        []string          // rendered; for script Args[0] is the script path (Phase 1)
	Dir         string            // cwd; script = script dir; tool_call = "" (inherit) until Phase 3
	Stdin       []byte            // nil = none
	NodeEnv     map[string]string // resolved node env
	Passthrough []string          // passthrough NAMES
	TimeoutSecs int               // 0 = none
	WorkDir     string            // canonical {{workspace}} for affinity; "" = no workspace
	Bundle      *ScriptBundleRef  // nil in Phase 1
}

// ScriptBundleRef names a content-addressed script bundle (Phase 3).
type ScriptBundleRef struct {
	Hash       string
	Entrypoint string // bundle-relative
}

// ProcessResult is the outcome of a completed (or attempted) subprocess.
type ProcessResult struct {
	Stdout   string
	Stderr   string
	ExitCode int
}

// ProcessRunner executes a ProcessSpec. A completed non-zero exit is reported in
// ProcessResult.ExitCode with a nil error; a non-nil error means an execution
// fault (start failure, timeout/cancellation, wait failure).
type ProcessRunner interface {
	RunProcess(ctx context.Context, spec ProcessSpec) (ProcessResult, error)
}

package process

import (
	"fmt"
	"os"
	"sort"
	"strings"

	"github.com/arbitron/arbitron/adapter/procenv"
	"github.com/arbitron/arbitron/domain/variable"
)

// passthroughSet returns the effective passthrough names: trimmed, non-empty,
// and never a base-allowlist name (a base name cannot be shadowed).
func passthroughSet(passthrough []string) map[string]bool {
	set := make(map[string]bool, len(passthrough))
	for _, name := range passthrough {
		name = strings.TrimSpace(name)
		if name == "" || procenv.IsBaseName(name) {
			continue
		}
		set[name] = true
	}
	return set
}

// validateNodeEnvNames rejects a node-env key colliding with a base or
// passthrough name, preventing secret/base spoofing. It is pure on names, so
// executors run it on the server before dispatch and every runner runs it
// again in buildEnvFromResolved.
func validateNodeEnvNames(passthrough []string, nodeEnv map[string]string) error {
	passSet := passthroughSet(passthrough)
	keys := make([]string, 0, len(nodeEnv))
	for k := range nodeEnv {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	for _, k := range keys {
		if procenv.IsBaseName(k) || passSet[k] {
			return fmt.Errorf("node env %q collides with a base or passthrough variable name", k)
		}
	}
	return nil
}

// resolveNodeEnv substitutes {{var}} templates in node env values. It runs on
// the server, where the variable store lives.
func resolveNodeEnv(nodeEnv map[string]string, vars *variable.VariableStore) map[string]string {
	if len(nodeEnv) == 0 {
		return nil
	}
	out := make(map[string]string, len(nodeEnv))
	for k, v := range nodeEnv {
		out[k] = variable.Resolve(v, vars)
	}
	return out
}

// buildEnvFromResolved constructs the environment on the EXECUTING machine:
// the base allowlist, plus passthrough values read from this process's env, plus
// the already-resolved node env. Collisions are rejected exactly as before.
func buildEnvFromResolved(passthrough []string, nodeEnv map[string]string) ([]string, error) {
	if err := validateNodeEnvNames(passthrough, nodeEnv); err != nil {
		return nil, err
	}
	env := procenv.BaseEnv()
	names := make([]string, 0)
	for name := range passthroughSet(passthrough) {
		names = append(names, name)
	}
	sort.Strings(names)
	for _, name := range names {
		if val, ok := os.LookupEnv(name); ok {
			env = append(env, name+"="+val)
		}
	}
	keys := make([]string, 0, len(nodeEnv))
	for k := range nodeEnv {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	for _, k := range keys {
		env = append(env, k+"="+nodeEnv[k])
	}
	return env, nil
}

package process

import (
	"context"

	"github.com/arbitron/arbitron/domain/engine"
)

// LocalProcessRunner executes a ProcessSpec on this machine with os/exec. It is
// the built-in runner's process path (adapter/runner/builtin) and, until Task 16
// wires the routed runner, the executors' direct runner.
type LocalProcessRunner struct{}

var _ engine.ProcessRunner = (*LocalProcessRunner)(nil)

// NewLocalProcessRunner returns the os/exec-backed ProcessRunner.
func NewLocalProcessRunner() *LocalProcessRunner { return &LocalProcessRunner{} }

// RunProcess builds the controlled environment on this machine (passthrough
// values read here) and runs the command with the spec's cwd, stdin and timeout.
func (r *LocalProcessRunner) RunProcess(ctx context.Context, spec engine.ProcessSpec) (engine.ProcessResult, error) {
	env, err := buildEnvFromResolved(spec.Passthrough, spec.NodeEnv)
	if err != nil {
		return engine.ProcessResult{}, err
	}
	res, runErr := run(ctx, commandSpec{
		Name:        spec.Command,
		Args:        spec.Args,
		Dir:         spec.Dir,
		Stdin:       spec.Stdin,
		Env:         env,
		TimeoutSecs: spec.TimeoutSecs,
	})
	return engine.ProcessResult{Stdout: res.Stdout, Stderr: res.Stderr, ExitCode: res.ExitCode}, runErr
}

// ScriptExecutor runs external scripts through the engine.ProcessRunner port,
// piping the variable store as JSON to stdin and parsing stdout as JSON.
type ScriptExecutor struct {
	fileReader  engine.FileReader
	passthrough []string
	runner      engine.ProcessRunner
}

// NewScriptExecutor creates a ScriptExecutor. envPassthrough names the engine
// env vars (e.g. LLM_API_KEY) script nodes may inherit beyond the base
// allowlist; values are read on the executing machine. pr is the process port
// (local or routed); nil panics so a script node can never run unplaced.
func NewScriptExecutor(fr engine.FileReader, envPassthrough []string, pr engine.ProcessRunner) *ScriptExecutor {
	if pr == nil {
		panic("process: NewScriptExecutor requires a non-nil engine.ProcessRunner")
	}
	return &ScriptExecutor{fileReader: fr, passthrough: envPassthrough, runner: pr}
}

// Execute resolves the script path, renders args and node env on the server,
// and runs the script through the ProcessRunner port.
func (e *ScriptExecutor) Execute(ctx context.Context, node workflow.Node, vars *variable.VariableStore, workflowName string) (*engine.NodeResult, error) {
	scriptPath, err := e.fileReader.ResolvePath(engine.OwnerFromContext(ctx), workflowName, node.Config.ScriptFile)
	if err != nil {
		return nil, fmt.Errorf("script %q: resolve path: %w", node.ID, err)
	}

	args := make([]string, len(node.Config.Args))
	for i, arg := range node.Config.Args {
		args[i] = variable.Resolve(arg, vars)
	}

	interpreter := node.Config.Interpreter
	if interpreter == "" {
		interpreter = "/bin/sh"
	}

	if err := validateNodeEnvNames(e.passthrough, node.Config.Env); err != nil {
		return nil, fmt.Errorf("script %q: %w", node.ID, err)
	}

	stdinData, err := json.Marshal(vars.Snapshot())
	if err != nil {
		return nil, fmt.Errorf("script %q: marshal variables: %w", node.ID, err)
	}

	workDir, _ := engine.WorkDirFromVars(vars)
	res, runErr := e.runner.RunProcess(ctx, engine.ProcessSpec{
		Kind:        engine.ProcessScript,
		Command:     interpreter,
		Args:        append([]string{scriptPath}, args...),
		Dir:         filepath.Dir(scriptPath),
		Stdin:       stdinData,
		NodeEnv:     resolveNodeEnv(node.Config.Env, vars),
		Passthrough: append([]string(nil), e.passthrough...),
		TimeoutSecs: node.Config.TimeoutSeconds,
		WorkDir:     workDir,
	})

	result := &engine.NodeResult{
		ScriptStdout:   res.Stdout,
		ScriptStderr:   res.Stderr,
		ScriptExitCode: res.ExitCode,
	}
	if runErr != nil {
		return result, fmt.Errorf("script %q: %w", node.ID, runErr)
	}
	if res.ExitCode != 0 {
		return result, fmt.Errorf("script %q: exit status %d: %s", node.ID, res.ExitCode, strings.TrimSpace(res.Stderr))
	}

	var output any
	if err := json.Unmarshal([]byte(strings.TrimSpace(res.Stdout)), &output); err != nil {
		return result, fmt.Errorf("script %q: stdout is not valid JSON: %w", node.ID, err)
	}
	result.Output = output
	return result, nil
}
