**Goal:** On the run graph:
- colour each step by the runner it ran on, using a deterministic palette from the runner id; server-side pure nodes are grey;
- put workspace-move badges on the edges into the step where the workspace moved, from the run's `workspaceMoves` (home history);
- give the node panel a Placement section with the runner, source, placement reason, session pin and attempt history, the workspace's history of homes, any lossy session-move notice, and the durable recovery disclosure.

export default function RunnerStripe({ mark }: { mark: RunnerMark | undefined }) {

  const label = mark.runnerId ? `Ran on ${mark.label}` : 'Runs on the server'
  return (

      className={`${styles.stripe} ${mark.moved ? styles.moved : ''}`}

      data-color={mark.color}
      data-moved={mark.moved ? 'true' : undefined}

          )}
        </div>
      ))}
      {disclosure && (

        </p>
      )}
    </section>
