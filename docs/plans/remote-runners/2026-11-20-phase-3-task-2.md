# Remote Runners — Phase 3 Task 2: Immutable workspace checkpoint store

**Goal:** Implement `runner.CheckpointStore` as immutable, versioned tar+gzip workspace archives with verified manifests, fenced atomic publication, delta staging, safe restore, and self-contained Git metadata.
**Architecture:** Pure manifest vocabulary/hash/diff lives in `domain/runner` (`checkpoint.go`, `manifest.go`) so the server store and the runner agent (P3 T9) share one algorithm. `adapter/checkpoint.Store` owns the filesystem layout (archives, manifests, staging, read cache) and publishes through a `runner.CheckpointCatalog`; the SQLite catalog row (inserted in one transaction with the job-fence check) IS the publication, so a crash or fenced writer never changes the checkpoint of record.
**Tech Stack:** Go 1.25.5 stdlib `archive/tar`, `compress/gzip` (RR-15 decision: gzip, no new dependency), `os.Root` for traversal-safe extraction, `git` CLI only to finalize linked-worktree metadata, modernc SQLite, testify.
**Spec / Design:** spec "Recovery and session continuity" (workspace-changing step complete only when recoverable elsewhere); design phase 3 §3.3 (Manifest, Push, Built-in runner, Completion boundary), §5 rows "Committed checkpoint, manifest…" (RR-08) and "Archive compression" (RR-15), §7 (partial uploads never applied; disk full keeps previous checkpoint), §8 regressions (chmod/deletion/symlink/Git round-trip; two successful steps publish distinct manifests). Contracts: `docs/plans/remote-runners/2026-11-20-contracts.md` §16 (checkpoint names), §8 (`Fence`, `ErrFenced`), §1 (044).

// domain/runner/checkpoint.go
package runner

import (
	"context"
	"errors"
	"fmt"
	"io"
	"strconv"
	"strings"

	"github.com/google/uuid"
)

// Manifest lists one immutable checkpoint of a run's workspace. Hash covers the
// version and every entry, so two successive commits of identical content are
// still distinct manifests.
type Manifest struct {
	Version int64           `json:"version"`
	Entries []ManifestEntry `json:"entries"`
	Hash    string          `json:"hash"`
}

// ManifestEntry is one path. Mode carries fs.FileMode type bits (ModeDir,
// ModeSymlink) plus permission bits; regular files carry Size and SHA256,
// symlinks carry their link text. Deleted is used only in diffs.
type ManifestEntry struct {
	Path    string `json:"path"`
	Size    int64  `json:"size"`
	Mode    uint32 `json:"mode"`
	SHA256  string `json:"sha256,omitempty"`
	Symlink string `json:"symlink,omitempty"`
	Deleted bool   `json:"deleted,omitempty"`
}

// CheckpointRef names one published checkpoint. The zero value (Version 0)
// means "no checkpoint": an empty workspace.
type CheckpointRef struct {
	RunID        uuid.UUID `json:"runId"`
	Version      int64     `json:"version"`
	ManifestHash string    `json:"manifestHash"`
}

// IsZero reports whether r names no checkpoint.
func (r CheckpointRef) IsZero() bool { return r.Version == 0 }

// String renders "<runID>@<version>:<manifestHash>" ("" for the zero ref). The
// same format is produced in SQL by StepCommitRepo (P3 T5).
func (r CheckpointRef) String() string {
	if r.IsZero() {
		return ""
	}
	return fmt.Sprintf("%s@%d:%s", r.RunID, r.Version, r.ManifestHash)
}

// ParseCheckpointRef parses String's output.
func ParseCheckpointRef(s string) (CheckpointRef, error) {
	if s == "" {
		return CheckpointRef{}, nil
	}
	at := strings.IndexByte(s, '@')
	colon := strings.LastIndexByte(s, ':')
	if at <= 0 || colon <= at+1 || colon == len(s)-1 {
		return CheckpointRef{}, fmt.Errorf("runner: malformed checkpoint ref %q", s)
	}
	id, err := uuid.Parse(s[:at])
	if err != nil {
		return CheckpointRef{}, fmt.Errorf("runner: malformed checkpoint ref %q: %w", s, err)
	}
	v, err := strconv.ParseInt(s[at+1:colon], 10, 64)
	if err != nil || v < 1 {
		return CheckpointRef{}, fmt.Errorf("runner: malformed checkpoint ref version %q", s)
	}
	return CheckpointRef{RunID: id, Version: v, ManifestHash: s[colon+1:]}, nil
}

// StagedCheckpoint is a fully built, hash-verified next checkpoint that is not
// yet published. Dir is owned by the store; Publish or Discard removes it.
type StagedCheckpoint struct {
	RunID       uuid.UUID
	Base        CheckpointRef
	Version     int64
	Manifest    Manifest
	Dir         string
	ArchivePath string
	SizeBytes   int64
}

// DeltaControlPath is the first entry of every pushed delta archive; its JSON
// body is a DeltaControl.
const DeltaControlPath = ".arbitron-delta/control.json"

// DeltaControl lists the deletions a delta applies and the FULL manifest
// entries the runner's copy has after the step; the server verifies the
// staged tree against Expected before anything is published.
type DeltaControl struct {
	Deleted  []string        `json:"deleted"`
	Expected []ManifestEntry `json:"expected"`
}

// CheckpointStore keeps the immutable checkpoint of record for every run.
type CheckpointStore interface {
	Commit(ctx context.Context, runID uuid.UUID, srcDir string) (CheckpointRef, Manifest, error)
	Restore(ctx context.Context, ref CheckpointRef, dstDir string) error
	Manifest(ctx context.Context, ref CheckpointRef) (Manifest, error)
	Latest(ctx context.Context, runID uuid.UUID) (CheckpointRef, bool, error)
	StageDelta(ctx context.Context, runID uuid.UUID, base CheckpointRef, delta io.Reader) (StagedCheckpoint, error)
	Publish(ctx context.Context, staged StagedCheckpoint, f Fence) (CheckpointRef, error)
	Discard(ctx context.Context, staged StagedCheckpoint) error
	ReadFile(ctx context.Context, ref CheckpointRef, path string) (io.ReadCloser, error)
	Delete(ctx context.Context, runID uuid.UUID) error
}

// CheckpointCatalog is the durable index of published checkpoints. Publish is
// the single atomic publication point: it inserts the row only when
// ref.Version == latest+1 (else ErrCheckpointConflict) and, when f is non-nil,
// only when job f is still offered/running at f.Generation on f.RunnerID (else
// ErrFenced), in one transaction.
type CheckpointCatalog interface {
	Publish(ctx context.Context, ref CheckpointRef, archivePath string, sizeBytes int64, f *Fence) error
	Latest(ctx context.Context, runID uuid.UUID) (CheckpointRef, string, bool, error)
	Get(ctx context.Context, runID uuid.UUID, version int64) (CheckpointRef, string, error)
	DeleteRun(ctx context.Context, runID uuid.UUID) error
}

var (
	ErrCheckpointNotFound  = errors.New("runner: checkpoint not found")
	ErrCheckpointConflict  = errors.New("runner: checkpoint version conflict")
	ErrManifestMismatch    = errors.New("runner: staged workspace does not match the pushed manifest")
	ErrUnsupportedFileType = errors.New("runner: unsupported file type in workspace")
	ErrUnsupportedGit      = errors.New("runner: unsupported git metadata in workspace")
	ErrCorruptCheckpoint   = errors.New("runner: checkpoint content does not match its manifest")
	ErrNoCheckpointCatalog = errors.New("runner: checkpoint store has no catalog; refusing unfenced publish")
)

// domain/runner/manifest.go
package runner

import (
	"crypto/sha256"
	"encoding/hex"
	"fmt"
	"io/fs"
	"path"
	"sort"
	"strings"
)

// CleanRelPath validates a slash-separated workspace-relative path and returns
// it unchanged. Absolute paths, "..", empty/"." paths, redundant separators,
// backslashes and NUL bytes are rejected.
func CleanRelPath(p string) (string, error) {
	if p == "" || strings.HasPrefix(p, "/") || strings.ContainsAny(p, "\\\x00") {
		return "", fmt.Errorf("runner: invalid workspace path %q", p)
	}
	c := path.Clean(p)
	if c != p || c == "." || c == ".." || strings.HasPrefix(c, "../") {
		return "", fmt.Errorf("runner: invalid workspace path %q", p)
	}
	return c, nil
}

// EntryFromInfo maps a Lstat result to a manifest entry. Only directories,
// regular files and symlinks are portable; devices, sockets, named pipes and
// irregular files are rejected explicitly.
func EntryFromInfo(rel string, info fs.FileInfo, sha256Hex, link string) (ManifestEntry, error) {
	m := info.Mode()
	switch {
	case m.IsDir():
		return ManifestEntry{Path: rel, Mode: uint32(fs.ModeDir | m.Perm())}, nil
	case m.IsRegular():
		return ManifestEntry{Path: rel, Size: info.Size(), Mode: uint32(m.Perm()), SHA256: sha256Hex}, nil
	case m&fs.ModeSymlink != 0:
		return ManifestEntry{Path: rel, Mode: uint32(fs.ModeSymlink | 0o777), Symlink: link}, nil
	default:
		return ManifestEntry{}, fmt.Errorf("%w: %s (%s)", ErrUnsupportedFileType, rel, m.Type())
	}
}

func sortedEntries(es []ManifestEntry) []ManifestEntry {
	out := append([]ManifestEntry(nil), es...)
	sort.Slice(out, func(i, j int) bool { return out[i].Path < out[j].Path })
	return out
}

// ManifestHash is the SHA-256 over the version and the sorted entries.
func ManifestHash(version int64, entries []ManifestEntry) string {
	h := sha256.New()
	fmt.Fprintf(h, "arbitron-manifest-v1\nversion %d\n", version)
	for _, e := range sortedEntries(entries) {
		fmt.Fprintf(h, "%q %d %o %s %q %t\n", e.Path, e.Size, e.Mode, e.SHA256, e.Symlink, e.Deleted)
	}
	return hex.EncodeToString(h.Sum(nil))
}

// NewManifest sorts entries and computes the hash.
func NewManifest(version int64, entries []ManifestEntry) Manifest {
	s := sortedEntries(entries)
	return Manifest{Version: version, Entries: s, Hash: ManifestHash(version, s)}
}

// Verify recomputes the hash.
func (m Manifest) Verify() error {
	if got := ManifestHash(m.Version, m.Entries); got != m.Hash {
		return fmt.Errorf("%w: manifest v%d hash %s != recorded %s", ErrCorruptCheckpoint, m.Version, got, m.Hash)
	}
	return nil
}

// TotalBytes sums regular-file sizes (the move-size input for remote homes).
func (m Manifest) TotalBytes() int64 {
	var n int64
	for _, e := range m.Entries {
		n += e.Size
	}
	return n
}

// SameEntry compares two entries ignoring Deleted.
func SameEntry(a, b ManifestEntry) bool {
	return a.Path == b.Path && a.Size == b.Size && a.Mode == b.Mode && a.SHA256 == b.SHA256 && a.Symlink == b.Symlink
}

// SameEntries compares two entry sets regardless of order.
func SameEntries(a, b []ManifestEntry) bool {
	if len(a) != len(b) {
		return false
	}
	sa, sb := sortedEntries(a), sortedEntries(b)
	for i := range sa {
		if !SameEntry(sa[i], sb[i]) {
			return false
		}
	}
	return true
}

// DiffManifests returns the entries of `to` that are absent from or differ in
// `from` (to fetch on pull / to send on push), and the sorted paths of `from`
// absent from `to` (to delete).
func DiffManifests(from, to []ManifestEntry) (changed []ManifestEntry, deleted []string) {
	old := make(map[string]ManifestEntry, len(from))
	for _, e := range from {
		old[e.Path] = e
	}
	seen := make(map[string]bool, len(to))
	for _, e := range sortedEntries(to) {
		seen[e.Path] = true
		if o, ok := old[e.Path]; !ok || !SameEntry(o, e) {
			changed = append(changed, e)
		}
	}
	for _, e := range sortedEntries(from) {
		if !seen[e.Path] {
			deleted = append(deleted, e.Path)
		}
	}
	return changed, deleted
}

// memCatalog honours the load-bearing arguments: version CAS per run and the
// fence (keyed by job) it is given.
type memCatalog struct {
	mu     sync.Mutex
	rows   map[uuid.UUID][]catRow
	fences map[uuid.UUID]runner.Fence
}

type catRow struct {
	ref  runner.CheckpointRef
	path string
}

func newMemCatalog() *memCatalog {
	return &memCatalog{rows: map[uuid.UUID][]catRow{}, fences: map[uuid.UUID]runner.Fence{}}
}

func (c *memCatalog) setFence(f runner.Fence) { c.mu.Lock(); c.fences[f.JobID] = f; c.mu.Unlock() }

func (c *memCatalog) Publish(_ context.Context, ref runner.CheckpointRef, p string, _ int64, f *runner.Fence) error {
	c.mu.Lock()
	defer c.mu.Unlock()
	rows := c.rows[ref.RunID]
	if int64(len(rows))+1 != ref.Version {
		return fmt.Errorf("%w: want v%d", runner.ErrCheckpointConflict, len(rows)+1)
	}
	if f != nil {
		if cur, ok := c.fences[f.JobID]; !ok || cur != *f {
			return fmt.Errorf("%w: stale fence", runner.ErrFenced)
		}
	}
	c.rows[ref.RunID] = append(rows, catRow{ref: ref, path: p})
	return nil
}

func (c *memCatalog) Latest(_ context.Context, runID uuid.UUID) (runner.CheckpointRef, string, bool, error) {
	c.mu.Lock()
	defer c.mu.Unlock()
	rows := c.rows[runID]
	if len(rows) == 0 {
		return runner.CheckpointRef{}, "", false, nil
	}
	r := rows[len(rows)-1]
	return r.ref, r.path, true, nil
}

func (c *memCatalog) Get(_ context.Context, runID uuid.UUID, v int64) (runner.CheckpointRef, string, error) {
	c.mu.Lock()
	defer c.mu.Unlock()
	rows := c.rows[runID]
	if v < 1 || v > int64(len(rows)) {
		return runner.CheckpointRef{}, "", runner.ErrCheckpointNotFound
	}
	return rows[v-1].ref, rows[v-1].path, nil
}

func (c *memCatalog) DeleteRun(_ context.Context, runID uuid.UUID) error {
	c.mu.Lock()
	delete(c.rows, runID)
	c.mu.Unlock()
	return nil
}

func write(t *testing.T, root, rel, body string, perm os.FileMode) {
	t.Helper()
	p := filepath.Join(root, filepath.FromSlash(rel))
	require.NoError(t, os.MkdirAll(filepath.Dir(p), 0o755))
	require.NoError(t, os.WriteFile(p, []byte(body), perm))
	require.NoError(t, os.Chmod(p, perm))
}

// tree renders a directory as sorted "path mode [content|->link]" lines.
func tree(t *testing.T, root string) []string {
	t.Helper()
	var out []string
	require.NoError(t, filepath.Walk(root, func(p string, info os.FileInfo, err error) error {
		require.NoError(t, err)
		if p == root {
			return nil
		}
		rel, _ := filepath.Rel(root, p)
		rel = filepath.ToSlash(rel)
		switch {
		case info.Mode()&os.ModeSymlink != 0:
			l, _ := os.Readlink(p)
			out = append(out, rel+" link ->"+l)
		case info.IsDir():
			out = append(out, fmt.Sprintf("%s dir %o", rel, info.Mode().Perm()))
		default:
			b, _ := os.ReadFile(p)
			out = append(out, fmt.Sprintf("%s %o %s", rel, info.Mode().Perm(), b))
		}
		return nil
	}))
	sort.Strings(out)
	return out
}

	require.NoError(t, os.Mkdir(filepath.Join(ws, "empty"), 0o750))
	require.NoError(t, os.Symlink("run.sh", filepath.Join(ws, "link")))
	v1Tree := tree(t, ws)

	require.NoError(t, os.Remove(filepath.Join(ws, "link")))
	require.NoError(t, os.Symlink("sub/keep.txt", filepath.Join(ws, "link")))
	v2Tree := tree(t, ws)
	ref2, _, err := store.Commit(ctx, run, ws)

	assert.Equal(t, int64(2), ref2.Version)

	require.NoError(t, err)
	require.True(t, ok)
	assert.Equal(t, ref2, latest)
}

// buildDelta encodes a push the way the runner agent does (checkpoint.WriteDelta).
func buildDelta(t *testing.T, root string, base []runner.ManifestEntry) ([]byte, runner.DeltaControl) {
	t.Helper()
	now, err := checkpoint.ScanEntries(context.Background(), root)
	require.NoError(t, err)
	changed, deleted := runner.DiffManifests(base, now)
	paths := make([]string, 0, len(changed))
	for _, c := range changed {
		paths = append(paths, c.Path)
	}
	ctl := runner.DeltaControl{Deleted: deleted, Expected: now}
	var buf bytes.Buffer
	require.NoError(t, checkpoint.WriteDelta(&buf, root, paths, ctl))
	return buf.Bytes(), ctl
}

	write(t, server, "old.txt", "old", 0o644)
	base, baseM, err := store.Commit(ctx, run, server)
	require.NoError(t, err)

	// The runner's copy after a step: a.txt edited, old.txt deleted, new dir+file.
	runnerCopy := t.TempDir()
	require.NoError(t, store.Restore(ctx, base, runnerCopy))
	write(t, runnerCopy, "a.txt", "two", 0o600)
	require.NoError(t, os.Remove(filepath.Join(runnerCopy, "old.txt")))
	write(t, runnerCopy, "pkg/new.go", "package pkg", 0o644)
	delta, _ := buildDelta(t, runnerCopy, baseM.Entries)

	staged, err := store.StageDelta(ctx, run, base, bytes.NewReader(delta))
	require.NoError(t, err)
	assert.Equal(t, int64(2), staged.Version)

	fence := runner.Fence{JobID: uuid.New(), RunnerID: "mac-1", Generation: 2}
	cat.setFence(fence)
	stale := fence
	stale.Generation = 1
	_, err = store.Publish(ctx, staged, stale)
	require.ErrorIs(t, err, runner.ErrFenced, "a stale generation never publishes")
	latest, _, _ := store.Latest(ctx, run)
	assert.Equal(t, int64(1), latest.Version)
	_, statErr := os.Stat(staged.Dir)
	assert.True(t, os.IsNotExist(statErr), "a refused publish discards its staging")

	staged, err = store.StageDelta(ctx, run, base, bytes.NewReader(delta))
	require.NoError(t, err)
	ref, err := store.Publish(ctx, staged, fence)
	require.NoError(t, err)
	assert.Equal(t, int64(2), ref.Version)
	got := t.TempDir()
	require.NoError(t, store.Restore(ctx, ref, got))
	assert.Equal(t, tree(t, runnerCopy), tree(t, got))
}

// enospcWriter simulates a full disk after `left` bytes.
type enospcWriter struct {
	f    *os.File
	left int
}

func (w *enospcWriter) Write(p []byte) (int, error) {
	if len(p) > w.left {
		n, _ := w.f.Write(p[:w.left])
		w.left = 0
		return n, syscall.ENOSPC
	}
	n, err := w.f.Write(p)
	w.left -= n
	return n, err
}
func (w *enospcWriter) Close() error { return w.f.Close() }

func TestStore_DiskFullDuringApplyKeepsPreviousCheckpoint(t *testing.T) {

		if err != nil {
			return nil, err
		}
		if full.Load() {

// adapter/checkpoint/store.go
// Package checkpoint stores immutable, versioned workspace checkpoints as
// tar+gzip archives with verified manifests (remote runners Phase 3, D2).
package checkpoint

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"io/fs"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
	"sync"

	"github.com/arbitron/arbitron/domain/runner"
	"github.com/google/uuid"
)

var _ runner.CheckpointStore = (*Store)(nil)

// Store lays checkpoints out as
//
//	<dir>/runs/<run>/v<NNNNNNNNNN>.tar.gz          immutable archive
//	<dir>/runs/<run>/v<NNNNNNNNNN>.manifest.json   its manifest
//	<dir>/runs/<run>/v<NNNNNNNNNN>.d/              lazily extracted read cache
//	<dir>/staging/<id>/                            StageDelta output
//	<dir>/tmp/<id>/                                Commit scratch
//
// A checkpoint is published only when its catalog row is inserted; files
// without a row are orphans and are never read.
type Store struct {
	dir      string
	catalog  runner.CheckpointCatalog
	create   func(path string, perm os.FileMode) (io.WriteCloser, error)
	git      func(ctx context.Context, args ...string) ([]byte, error)
	keep     int64
	cacheMu  sync.Mutex
	runMu    sync.Mutex
	runLocks map[uuid.UUID]*sync.Mutex
}

// Option configures a Store.
type Option func(*Store)

// WithCatalog sets the durable publication index. Without it every
// Commit/Publish/Restore fails with runner.ErrNoCheckpointCatalog.
func WithCatalog(c runner.CheckpointCatalog) Option { return func(s *Store) { s.catalog = c } }

// WithFileCreator replaces the exclusive-create used for archives and
// manifests (fault injection: disk full).
func WithFileCreator(fn func(path string, perm os.FileMode) (io.WriteCloser, error)) Option {
	return func(s *Store) { s.create = fn }
}

// WithGit replaces the git runner used to finalize linked-worktree metadata.
func WithGit(fn func(ctx context.Context, args ...string) ([]byte, error)) Option {
	return func(s *Store) { s.git = fn }
}

// NewStore returns a store rooted at dir. Production wiring MUST pass
// WithCatalog(sqlite.NewCheckpointCatalogRepo(db)).
func NewStore(dir string, opts ...Option) *Store {
	s := &Store{dir: dir, create: createExclusive, git: runGit, keep: 2, runLocks: map[uuid.UUID]*sync.Mutex{}}
	for _, o := range opts {
		o(s)
	}
	return s
}

func createExclusive(path string, perm os.FileMode) (io.WriteCloser, error) {
	return os.OpenFile(path, os.O_CREATE|os.O_EXCL|os.O_WRONLY, perm)
}

func runGit(ctx context.Context, args ...string) ([]byte, error) {
	cmd := exec.CommandContext(ctx, "git", args...)
	cmd.Env = append(os.Environ(), "GIT_CONFIG_NOSYSTEM=1", "GIT_TERMINAL_PROMPT=0")
	out, err := cmd.CombinedOutput()
	if err != nil {
		return out, fmt.Errorf("git %s: %w: %s", strings.Join(args, " "), err, strings.TrimSpace(string(out)))
	}
	return out, nil
}

func (s *Store) lockRun(id uuid.UUID) func() {
	s.runMu.Lock()
	m, ok := s.runLocks[id]
	if !ok {
		m = &sync.Mutex{}
		s.runLocks[id] = m
	}
	s.runMu.Unlock()
	m.Lock()
	return m.Unlock
}

func (s *Store) runDir(id uuid.UUID) string { return filepath.Join(s.dir, "runs", id.String()) }
func (s *Store) archivePath(id uuid.UUID, v int64) string {
	return filepath.Join(s.runDir(id), fmt.Sprintf("v%010d.tar.gz", v))
}
func (s *Store) manifestPath(id uuid.UUID, v int64) string {
	return filepath.Join(s.runDir(id), fmt.Sprintf("v%010d.manifest.json", v))
}
func (s *Store) cachePath(id uuid.UUID, v int64) string {
	return filepath.Join(s.runDir(id), fmt.Sprintf("v%010d.d", v))
}

func (s *Store) scratch(kind string) (string, error) {
	base := filepath.Join(s.dir, kind)
	if err := os.MkdirAll(base, 0o700); err != nil {
		return "", fmt.Errorf("checkpoint: %s dir: %w", kind, err)
	}
	return os.MkdirTemp(base, "op-")
}

// Commit snapshots srcDir (materializing linked-worktree .git files) as the
// run's next version. It never fences: built-in callers check their job fence
// first (P3 T6); remote pushes go through StageDelta+Publish.
func (s *Store) Commit(ctx context.Context, runID uuid.UUID, srcDir string) (runner.CheckpointRef, runner.Manifest, error) {
	if s.catalog == nil {
		return runner.CheckpointRef{}, runner.Manifest{}, runner.ErrNoCheckpointCatalog
	}
	unlock := s.lockRun(runID)
	defer unlock()
	latest, _, ok, err := s.catalog.Latest(ctx, runID)
	if err != nil {
		return runner.CheckpointRef{}, runner.Manifest{}, fmt.Errorf("checkpoint: latest: %w", err)
	}
	version := int64(1)
	if ok {
		version = latest.Version + 1
	}
	tmp, err := s.scratch("tmp")
	if err != nil {
		return runner.CheckpointRef{}, runner.Manifest{}, err
	}
	defer forceRemoveAll(tmp)
	items, err := s.collect(ctx, srcDir, true, tmp)
	if err != nil {
		return runner.CheckpointRef{}, runner.Manifest{}, err
	}
	archive := filepath.Join(tmp, "archive.tar.gz")
	entries, size, err := s.writeArchiveFile(archive, items)
	if err != nil {
		return runner.CheckpointRef{}, runner.Manifest{}, fmt.Errorf("checkpoint: write archive: %w", err)
	}
	m := runner.NewManifest(version, entries)
	ref, err := s.publishLocked(ctx, runID, m, archive, size, nil)
	if err != nil {
		return runner.CheckpointRef{}, runner.Manifest{}, err
	}
	return ref, m, nil
}

// publishLocked links the archive+manifest to their immutable names and then
// inserts the catalog row (the publication). Any failure removes the linked
// files, so the checkpoint of record is unchanged.
func (s *Store) publishLocked(ctx context.Context, runID uuid.UUID, m runner.Manifest, archive string, size int64, f *runner.Fence) (runner.CheckpointRef, error) {
	if err := os.MkdirAll(s.runDir(runID), 0o700); err != nil {
		return runner.CheckpointRef{}, fmt.Errorf("checkpoint: run dir: %w", err)
	}
	mtmp := archive + ".manifest.json"
	if err := s.writeJSON(mtmp, m); err != nil {
		return runner.CheckpointRef{}, fmt.Errorf("checkpoint: write manifest: %w", err)
	}
	if err := s.clearOrphan(ctx, runID, m.Version); err != nil {
		return runner.CheckpointRef{}, err
	}
	finalArchive, finalManifest := s.archivePath(runID, m.Version), s.manifestPath(runID, m.Version)
	if err := os.Link(archive, finalArchive); err != nil {
		return runner.CheckpointRef{}, fmt.Errorf("checkpoint: link archive: %w", err)
	}
	if err := os.Link(mtmp, finalManifest); err != nil {
		_ = os.Remove(finalArchive)
		return runner.CheckpointRef{}, fmt.Errorf("checkpoint: link manifest: %w", err)
	}
	ref := runner.CheckpointRef{RunID: runID, Version: m.Version, ManifestHash: m.Hash}
	if err := s.catalog.Publish(ctx, ref, finalArchive, size, f); err != nil {
		_ = os.Remove(finalArchive)
		_ = os.Remove(finalManifest)
		return runner.CheckpointRef{}, err
	}
	s.prune(runID, m.Version)
	return ref, nil
}

func (s *Store) clearOrphan(ctx context.Context, runID uuid.UUID, v int64) error {
	_, aErr := os.Lstat(s.archivePath(runID, v))
	_, mErr := os.Lstat(s.manifestPath(runID, v))
	if errors.Is(aErr, fs.ErrNotExist) && errors.Is(mErr, fs.ErrNotExist) {
		return nil
	}
	if _, _, err := s.catalog.Get(ctx, runID, v); err == nil {
		return fmt.Errorf("%w: version %d already published", runner.ErrCheckpointConflict, v)
	} else if !errors.Is(err, runner.ErrCheckpointNotFound) {
		return fmt.Errorf("checkpoint: catalog get: %w", err)
	}
	// Files from a publish that died before its catalog row: never referenced.
	_ = os.Remove(s.archivePath(runID, v))
	_ = os.Remove(s.manifestPath(runID, v))
	forceRemoveAll(s.cachePath(runID, v))
	return nil
}

// prune keeps the newest s.keep versions' files (the previous checkpoint is
// retained until the next one is durably committed, design §3.3).
func (s *Store) prune(runID uuid.UUID, latest int64) {
	entries, err := os.ReadDir(s.runDir(runID))
	if err != nil {
		return
	}
	for _, e := range entries {
		var v int64
		if _, err := fmt.Sscanf(e.Name(), "v%d", &v); err != nil {
			continue
		}
		if v <= latest-s.keep {
			forceRemoveAll(filepath.Join(s.runDir(runID), e.Name()))
		}
	}
}

func (s *Store) writeJSON(path string, v any) error {
	b, err := json.Marshal(v)
	if err != nil {
		return err
	}
	w, err := s.create(path, 0o600)
	if err != nil {
		return err
	}
	_, werr := w.Write(b)
	cerr := w.Close()
	if werr != nil {
		return werr
	}
	return cerr
}

// Latest returns the newest published checkpoint.
func (s *Store) Latest(ctx context.Context, runID uuid.UUID) (runner.CheckpointRef, bool, error) {
	if s.catalog == nil {
		return runner.CheckpointRef{}, false, runner.ErrNoCheckpointCatalog
	}
	ref, _, ok, err := s.catalog.Latest(ctx, runID)
	if err != nil {
		return runner.CheckpointRef{}, false, fmt.Errorf("checkpoint: latest: %w", err)
	}
	return ref, ok, nil
}

// Manifest loads and verifies ref's manifest. The zero ref is the empty manifest.
func (s *Store) Manifest(_ context.Context, ref runner.CheckpointRef) (runner.Manifest, error) {
	if ref.IsZero() {
		return runner.Manifest{}, nil
	}
	b, err := os.ReadFile(s.manifestPath(ref.RunID, ref.Version))
	if errors.Is(err, fs.ErrNotExist) {
		return runner.Manifest{}, fmt.Errorf("%w: %s", runner.ErrCheckpointNotFound, ref)
	}
	if err != nil {
		return runner.Manifest{}, fmt.Errorf("checkpoint: read manifest: %w", err)
	}
	var m runner.Manifest
	if err := json.Unmarshal(b, &m); err != nil {
		return runner.Manifest{}, fmt.Errorf("%w: %s: %v", runner.ErrCorruptCheckpoint, ref, err)
	}
	if m.Version != ref.Version || m.Hash != ref.ManifestHash {
		return runner.Manifest{}, fmt.Errorf("%w: manifest differs from ref %s", runner.ErrCorruptCheckpoint, ref)
	}
	if err := m.Verify(); err != nil {
		return runner.Manifest{}, err
	}
	return m, nil
}

func (s *Store) resolve(ctx context.Context, ref runner.CheckpointRef) (string, runner.Manifest, error) {
	if s.catalog == nil {
		return "", runner.Manifest{}, runner.ErrNoCheckpointCatalog
	}
	got, path, err := s.catalog.Get(ctx, ref.RunID, ref.Version)
	if err != nil {
		return "", runner.Manifest{}, err
	}
	if got.ManifestHash != ref.ManifestHash {
		return "", runner.Manifest{}, fmt.Errorf("%w: catalog hash differs for %s", runner.ErrCorruptCheckpoint, ref)
	}
	m, err := s.Manifest(ctx, ref)
	if err != nil {
		return "", runner.Manifest{}, err
	}
	if _, err := os.Stat(path); err != nil {
		return "", runner.Manifest{}, fmt.Errorf("%w: archive for %s is no longer retained", runner.ErrCheckpointNotFound, ref)
	}
	return path, m, nil
}

// Restore wipes dstDir and extracts ref into it, verifying every entry against
// the manifest. The zero ref restores an empty workspace.
func (s *Store) Restore(ctx context.Context, ref runner.CheckpointRef, dstDir string) error {
	if err := os.MkdirAll(dstDir, 0o755); err != nil {
		return fmt.Errorf("checkpoint: restore dir: %w", err)
	}
	if ref.IsZero() {
		return wipeDir(dstDir)
	}
	path, m, err := s.resolve(ctx, ref)
	if err != nil {
		return err
	}
	if err := wipeDir(dstDir); err != nil {
		return fmt.Errorf("checkpoint: wipe %s: %w", dstDir, err)
	}
	return extractVerified(path, dstDir, m)
}

// StageDelta rebuilds base in a private staging tree, applies the delta
// (changed entries + deletions), and verifies the result equals the runner's
// claimed manifest before writing the next archive. Nothing is published.
func (s *Store) StageDelta(ctx context.Context, runID uuid.UUID, base runner.CheckpointRef, delta io.Reader) (runner.StagedCheckpoint, error) {
	stage, err := s.scratch("staging")
	if err != nil {
		return runner.StagedCheckpoint{}, err
	}
	ok := false
	defer func() {
		if !ok {
			forceRemoveAll(stage)
		}
	}()
	treeDir := filepath.Join(stage, "tree")
	if err := os.Mkdir(treeDir, 0o700); err != nil {
		return runner.StagedCheckpoint{}, err
	}
	root, err := os.OpenRoot(treeDir)
	if err != nil {
		return runner.StagedCheckpoint{}, err
	}
	defer root.Close()

	modes := map[string]fs.FileMode{}
	if !base.IsZero() {
		path, m, err := s.resolve(ctx, base)
		if err != nil {
			return runner.StagedCheckpoint{}, err
		}
		baseModes, err := extractInto(path, root, m)
		if err != nil {
			return runner.StagedCheckpoint{}, err
		}
		modes = baseModes
	}
	ctl, deltaModes, err := applyDelta(delta, root)
	if err != nil {
		return runner.StagedCheckpoint{}, err
	}
	for k, v := range deltaModes {
		modes[k] = v
	}
	for _, d := range ctl.Deleted {
		clean, err := runner.CleanRelPath(d)
		if err != nil {
			return runner.StagedCheckpoint{}, err
		}
		if err := root.RemoveAll(clean); err != nil {
			return runner.StagedCheckpoint{}, fmt.Errorf("checkpoint: delete %s: %w", clean, err)
		}
		for k := range modes {
			if k == clean || strings.HasPrefix(k, clean+"/") {
				delete(modes, k)
			}
		}
	}
	if err := applyDirModes(root, modes); err != nil {
		return runner.StagedCheckpoint{}, err
	}
	items, err := s.collect(ctx, treeDir, false, "")
	if err != nil {
		return runner.StagedCheckpoint{}, err
	}
	archive := filepath.Join(stage, "archive.tar.gz")
	entries, size, err := s.writeArchiveFile(archive, items)
	if err != nil {
		return runner.StagedCheckpoint{}, fmt.Errorf("checkpoint: write staged archive: %w", err)
	}
	if !runner.SameEntries(entries, ctl.Expected) {
		return runner.StagedCheckpoint{}, fmt.Errorf("%w: %s", runner.ErrManifestMismatch, firstDifference(entries, ctl.Expected))
	}
	ok = true
	return runner.StagedCheckpoint{
		RunID: runID, Base: base, Version: base.Version + 1,
		Manifest: runner.NewManifest(base.Version+1, entries),
		Dir:      stage, ArchivePath: archive, SizeBytes: size,
	}, nil
}

func firstDifference(got, want []runner.ManifestEntry) string {
	changed, deleted := runner.DiffManifests(want, got)
	switch {
	case len(changed) > 0:
		return "path " + changed[0].Path + " differs"
	case len(deleted) > 0:
		return "path " + deleted[0] + " missing"
	default:
		return "entry sets differ"
	}
}

// Publish makes a staged checkpoint the run's newest version under fence f.
// The staging directory is removed whether or not publication succeeds.
func (s *Store) Publish(ctx context.Context, staged runner.StagedCheckpoint, f runner.Fence) (runner.CheckpointRef, error) {
	if s.catalog == nil {
		return runner.CheckpointRef{}, runner.ErrNoCheckpointCatalog
	}
	if err := s.ownsStaging(staged.Dir); err != nil {
		return runner.CheckpointRef{}, err
	}
	defer forceRemoveAll(staged.Dir)
	unlock := s.lockRun(staged.RunID)
	defer unlock()
	fence := f
	return s.publishLocked(ctx, staged.RunID, staged.Manifest, staged.ArchivePath, staged.SizeBytes, &fence)
}

// Discard drops a staged checkpoint.
func (s *Store) Discard(_ context.Context, staged runner.StagedCheckpoint) error {
	if err := s.ownsStaging(staged.Dir); err != nil {
		return err
	}
	forceRemoveAll(staged.Dir)
	return nil
}

func (s *Store) ownsStaging(dir string) error {
	rel, err := filepath.Rel(filepath.Join(s.dir, "staging"), dir)
	if err != nil || rel == "." || strings.HasPrefix(rel, "..") || strings.Contains(rel, string(filepath.Separator)) {
		return fmt.Errorf("checkpoint: %q is not a staging directory of this store", dir)
	}
	return nil
}

// ReadFile opens one regular file of a published checkpoint (served to
// runners during pull). Symlinks and directories are refused; the content is
// the immutable committed version, never the mutable working copy.
func (s *Store) ReadFile(ctx context.Context, ref runner.CheckpointRef, p string) (io.ReadCloser, error) {
	clean, err := runner.CleanRelPath(p)
	if err != nil {
		return nil, err
	}
	cache, err := s.ensureCache(ctx, ref)
	if err != nil {
		return nil, err
	}
	root, err := os.OpenRoot(cache)
	if err != nil {
		return nil, err
	}
	fi, err := root.Lstat(clean)
	if err != nil {
		root.Close()
		return nil, fmt.Errorf("%w: %s in %s", runner.ErrCheckpointNotFound, clean, ref)
	}
	if !fi.Mode().IsRegular() {
		root.Close()
		return nil, fmt.Errorf("checkpoint: %s is not a regular file", clean)
	}
	f, err := root.Open(clean)
	if err != nil {
		root.Close()
		return nil, err
	}
	return rootFile{File: f, root: root}, nil
}

type rootFile struct {
	*os.File
	root *os.Root
}

func (f rootFile) Close() error {
	err := f.File.Close()
	f.root.Close()
	return err
}

func (s *Store) ensureCache(ctx context.Context, ref runner.CheckpointRef) (string, error) {
	cache := s.cachePath(ref.RunID, ref.Version)
	s.cacheMu.Lock()
	defer s.cacheMu.Unlock()
	if fi, err := os.Stat(cache); err == nil && fi.IsDir() {
		return cache, nil
	}
	path, m, err := s.resolve(ctx, ref)
	if err != nil {
		return "", err
	}
	tmp, err := os.MkdirTemp(s.runDir(ref.RunID), "cache-")
	if err != nil {
		return "", err
	}
	if err := extractVerified(path, tmp, m); err != nil {
		forceRemoveAll(tmp)
		return "", err
	}
	if err := os.Rename(tmp, cache); err != nil {
		forceRemoveAll(tmp)
		return "", err
	}
	return cache, nil
}

// Delete removes every checkpoint of a run (workspace removal, P3 T11).
func (s *Store) Delete(ctx context.Context, runID uuid.UUID) error {
	if s.catalog == nil {
		return runner.ErrNoCheckpointCatalog
	}
	unlock := s.lockRun(runID)
	defer unlock()
	if err := s.catalog.DeleteRun(ctx, runID); err != nil {
		return fmt.Errorf("checkpoint: delete catalog rows: %w", err)
	}
	forceRemoveAll(s.runDir(runID))
	return nil
}

func wipeDir(dir string) error {
	es, err := os.ReadDir(dir)
	if err != nil {
		return err
	}
	for _, e := range es {
		p := filepath.Join(dir, e.Name())
		forceRemoveAll(p)
		if _, err := os.Lstat(p); err == nil {
			return fmt.Errorf("checkpoint: could not remove %s", p)
		}
	}
	return nil
}

// forceRemoveAll makes directories writable first so read-only trees (git
// objects, 0555 dirs) can be removed.
func forceRemoveAll(p string) {
	_ = filepath.WalkDir(p, func(q string, d fs.DirEntry, err error) error {
		if err == nil && d.IsDir() {
			_ = os.Chmod(q, 0o700)
		}
		return nil
	})
	_ = os.RemoveAll(p)
}

package checkpoint

import (
	"archive/tar"
	"compress/gzip"
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
	"io"

const maxDeltaControlBytes = 64 << 20

type countingWriter struct {
	w io.Writer
	n int64
}

func (c *countingWriter) Write(p []byte) (int, error) {
	n, err := c.w.Write(p)
	c.n += int64(n)
	return n, err
}

func (s *Store) writeArchiveFile(path string, items []item) ([]runner.ManifestEntry, int64, error) {
	w, err := s.create(path, 0o600)
	if err != nil {
		return nil, 0, err
	}
	cw := &countingWriter{w: w}
	entries, werr := writeArchive(cw, items)
	cerr := w.Close()
	if werr != nil {
		return nil, 0, werr
	}
	if cerr != nil {
		return nil, 0, cerr
	}
	return entries, cw.n, nil
}

func writeArchive(w io.Writer, items []item) ([]runner.ManifestEntry, error) {
	gz := gzip.NewWriter(w)
	tw := tar.NewWriter(gz)
	entries := make([]runner.ManifestEntry, 0, len(items))
	for _, it := range items {
		e, err := writeItem(tw, it)
		if err != nil {
			return nil, err
		}
		entries = append(entries, e)
	}
	if err := tw.Close(); err != nil {
		return nil, err
	}
	if err := gz.Close(); err != nil {
		return nil, err
	}
	return entries, nil
}

func writeItem(tw *tar.Writer, it item) (runner.ManifestEntry, error) {
	m := it.info.Mode()
	hdr := &tar.Header{Name: it.rel, ModTime: it.info.ModTime(), Format: tar.FormatPAX}
	switch {
	case m.IsDir():
		hdr.Typeflag, hdr.Name, hdr.Mode = tar.TypeDir, it.rel+"/", int64(m.Perm())
	case m.IsRegular():
		hdr.Typeflag, hdr.Mode, hdr.Size = tar.TypeReg, int64(m.Perm()), it.info.Size()
	case m&fs.ModeSymlink != 0:
		hdr.Typeflag, hdr.Linkname, hdr.Mode = tar.TypeSymlink, it.link, 0o777
	default:
		return runner.ManifestEntry{}, fmt.Errorf("%w: %s (%s)", runner.ErrUnsupportedFileType, it.rel, m.Type())
	}
	if err := tw.WriteHeader(hdr); err != nil {
		return runner.ManifestEntry{}, err
	}
	sum := ""
	if m.IsRegular() {
		f, err := os.Open(it.abs)
		if err != nil {
			return runner.ManifestEntry{}, err
		}
		h := sha256.New()
		n, err := io.Copy(io.MultiWriter(tw, h), f)
		f.Close()
		if err != nil {
			return runner.ManifestEntry{}, fmt.Errorf("checkpoint: archive %s: %w", it.rel, err)
		}
		if n != hdr.Size {
			return runner.ManifestEntry{}, fmt.Errorf("checkpoint: %s changed while archiving", it.rel)
		}
		sum = hex.EncodeToString(h.Sum(nil))
	}
	return runner.EntryFromInfo(it.rel, it.info, sum, it.link)
}

// extractVerified extracts an archive into dst and checks it reproduces m
// exactly (truncated or tampered archives fail with ErrCorruptCheckpoint).
func extractVerified(archive, dst string, m runner.Manifest) error {
	root, err := os.OpenRoot(dst)
	if err != nil {
		return err
	}
	defer root.Close()
	modes, err := extractInto(archive, root, m)
	if err != nil {
		return err
	}
	return applyDirModes(root, modes)
}

func extractInto(archive string, root *os.Root, m runner.Manifest) (map[string]fs.FileMode, error) {
	f, err := os.Open(archive)
	if err != nil {
		return nil, err
	}
	defer f.Close()
	gz, err := gzip.NewReader(f)
	if err != nil {
		return nil, fmt.Errorf("%w: %v", runner.ErrCorruptCheckpoint, err)
	}
	defer gz.Close()
	seen, modes, err := applyTar(tar.NewReader(gz), root, false)
	if err != nil {
		return nil, err
	}
	if len(seen) != len(m.Entries) {
		return nil, fmt.Errorf("%w: archive has %d entries, manifest %d", runner.ErrCorruptCheckpoint, len(seen), len(m.Entries))
	}
	for _, e := range m.Entries {
		got, ok := seen[e.Path]
		if !ok || !runner.SameEntry(got, e) {
			return nil, fmt.Errorf("%w: %s", runner.ErrCorruptCheckpoint, e.Path)
		}
	}
	return modes, nil
}

func applyDelta(r io.Reader, root *os.Root) (runner.DeltaControl, map[string]fs.FileMode, error) {
	gz, err := gzip.NewReader(r)
	if err != nil {
		return runner.DeltaControl{}, nil, fmt.Errorf("checkpoint: delta gzip: %w", err)
	}
	defer gz.Close()
	tr := tar.NewReader(gz)
	hdr, err := tr.Next()
	if err != nil {
		return runner.DeltaControl{}, nil, fmt.Errorf("checkpoint: delta: %w", err)
	}
	if hdr.Name != runner.DeltaControlPath || hdr.Typeflag != tar.TypeReg || hdr.Size > maxDeltaControlBytes {
		return runner.DeltaControl{}, nil, fmt.Errorf("checkpoint: delta must start with %s", runner.DeltaControlPath)
	}
	b, err := io.ReadAll(io.LimitReader(tr, hdr.Size))
	if err != nil {
		return runner.DeltaControl{}, nil, fmt.Errorf("checkpoint: delta control: %w", err)
	}
	var ctl runner.DeltaControl
	if err := json.Unmarshal(b, &ctl); err != nil {
		return runner.DeltaControl{}, nil, fmt.Errorf("checkpoint: delta control: %w", err)
	}
	for _, e := range ctl.Expected {
		if _, err := runner.CleanRelPath(e.Path); err != nil {
			return runner.DeltaControl{}, nil, err
		}
	}
	_, modes, err := applyTar(tr, root, true)
	if err != nil {
		return runner.DeltaControl{}, nil, err
	}
	return ctl, modes, nil
}

// applyTar writes entries under root (os.Root refuses escapes). Directories are
// created 0700 and their recorded modes returned for applyDirModes, so
// read-only directories never block their children.
func applyTar(tr *tar.Reader, root *os.Root, overwrite bool) (map[string]runner.ManifestEntry, map[string]fs.FileMode, error) {
	seen := map[string]runner.ManifestEntry{}
	modes := map[string]fs.FileMode{}
	for {
		hdr, err := tr.Next()
		if errors.Is(err, io.EOF) {
			return seen, modes, nil
		}
		if err != nil {
			return nil, nil, fmt.Errorf("checkpoint: read archive: %w", err)
		}
		name, err := runner.CleanRelPath(strings.TrimSuffix(hdr.Name, "/"))
		if err != nil {
			return nil, nil, err
		}
		if overwrite {
			if err := removeExisting(root, name, hdr.Typeflag == tar.TypeDir); err != nil {
				return nil, nil, err
			}
		}
		if dir := path.Dir(name); dir != "." {
			if err := root.MkdirAll(dir, 0o700); err != nil {
				return nil, nil, fmt.Errorf("checkpoint: mkdir %s: %w", dir, err)
			}
		}
		perm := fs.FileMode(hdr.Mode).Perm()
		switch hdr.Typeflag {
		case tar.TypeDir:
			if err := root.MkdirAll(name, 0o700); err != nil {
				return nil, nil, fmt.Errorf("checkpoint: mkdir %s: %w", name, err)
			}
			modes[name] = perm
			seen[name] = runner.ManifestEntry{Path: name, Mode: uint32(fs.ModeDir | perm)}
		case tar.TypeReg:
			f, err := root.OpenFile(name, os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0o600)
			if err != nil {
				return nil, nil, fmt.Errorf("checkpoint: create %s: %w", name, err)
			}
			h := sha256.New()
			n, cpErr := io.Copy(io.MultiWriter(f, h), tr)
			clErr := f.Close()
			if cpErr != nil {
				return nil, nil, fmt.Errorf("checkpoint: write %s: %w", name, cpErr)
			}
			if clErr != nil {
				return nil, nil, fmt.Errorf("checkpoint: close %s: %w", name, clErr)
			}
			if err := root.Chmod(name, perm); err != nil {
				return nil, nil, err
			}
			seen[name] = runner.ManifestEntry{Path: name, Size: n, Mode: uint32(perm), SHA256: hex.EncodeToString(h.Sum(nil))}
		case tar.TypeSymlink:
			if err := root.Symlink(hdr.Linkname, name); err != nil {
				return nil, nil, fmt.Errorf("checkpoint: symlink %s: %w", name, err)
			}
			seen[name] = runner.ManifestEntry{Path: name, Mode: uint32(fs.ModeSymlink | 0o777), Symlink: hdr.Linkname}
		default:
			return nil, nil, fmt.Errorf("%w: archive entry %s has type %q", runner.ErrUnsupportedFileType, name, hdr.Typeflag)
		}
	}
}

func removeExisting(root *os.Root, name string, wantDir bool) error {
	fi, err := root.Lstat(name)
	if errors.Is(err, fs.ErrNotExist) {
		return nil
	}
	if err != nil {
		return err
	}
	if wantDir && fi.IsDir() {
		return nil
	}
	return root.RemoveAll(name)
}

func applyDirModes(root *os.Root, modes map[string]fs.FileMode) error {
	names := make([]string, 0, len(modes))
	for n := range modes {
		names = append(names, n)
	}
	sort.Sort(sort.Reverse(sort.StringSlice(names))) // children before parents
	for _, n := range names {
		if err := root.Chmod(n, modes[n]); err != nil {
			return fmt.Errorf("checkpoint: chmod %s: %w", n, err)
		}
	}
	return nil
}

// WriteDelta encodes a push: the DeltaControl entry first, then every changed
// path (directories, regular files, symlinks) read from root.
func WriteDelta(w io.Writer, root string, changed []string, ctl runner.DeltaControl) error {
	gz := gzip.NewWriter(w)
	tw := tar.NewWriter(gz)
	b, err := json.Marshal(ctl)
	if err != nil {
		return err
	}
	if err := tw.WriteHeader(&tar.Header{Name: runner.DeltaControlPath, Typeflag: tar.TypeReg, Mode: 0o600, Size: int64(len(b)), Format: tar.FormatPAX}); err != nil {
		return err
	}
	if _, err := tw.Write(b); err != nil {
		return err
	}
	paths := append([]string(nil), changed...)
	sort.Strings(paths)
	for _, p := range paths {
		clean, err := runner.CleanRelPath(p)
		if err != nil {
			return err
		}
		abs := filepath.Join(root, filepath.FromSlash(clean))
		info, err := os.Lstat(abs)
		if err != nil {
			return err
		}
		link := ""
		if info.Mode()&fs.ModeSymlink != 0 {
			if link, err = os.Readlink(abs); err != nil {
				return err
			}
		}
		if _, err := writeItem(tw, item{rel: clean, abs: abs, info: info, link: link}); err != nil {
			return err
		}
	}
	if err := tw.Close(); err != nil {
		return err
	}
	return gz.Close()
}

// adapter/checkpoint/walk.go
package checkpoint

import (
	"context"
	"crypto/sha256"
	"encoding/hex"
	"fmt"
	"io"
	"io/fs"
	"os"
	"path/filepath"
	"sort"
	"strings"

	"github.com/arbitron/arbitron/domain/runner"
)

type item struct {
	rel  string // slash-separated, workspace-relative
	abs  string // where the bytes are read from (may be a materialized git dir)
	info fs.FileInfo
	link string
}

// collect walks root without following symlinks. A `.git` FILE whose gitdir is
// inside root is kept verbatim; one pointing outside root is materialized into
// a self-contained `.git` directory when materialize is true, and rejected
// (ErrUnsupportedGit) otherwise. Sockets/fifos inside git directories are
// runtime artefacts (fsmonitor) and are skipped; anywhere else they are
// rejected with ErrUnsupportedFileType.
func (s *Store) collect(ctx context.Context, root string, materialize bool, tmp string) ([]item, error) {
	abs, err := filepath.Abs(root)
	if err != nil {
		return nil, err
	}
	real, err := filepath.EvalSymlinks(abs)
	if err != nil {
		return nil, fmt.Errorf("checkpoint: workspace %s: %w", root, err)
	}
	if fi, err := os.Stat(real); err != nil || !fi.IsDir() {
		return nil, fmt.Errorf("checkpoint: workspace %s is not a directory", root)
	}
	var items []item
	err = filepath.WalkDir(real, func(p string, d fs.DirEntry, werr error) error {
		if werr != nil {
			return werr
		}
		if err := ctx.Err(); err != nil {
			return err
		}
		if p == real {
			return nil
		}
		relOS, err := filepath.Rel(real, p)
		if err != nil {
			return err
		}
		rel := filepath.ToSlash(relOS)
		info, err := os.Lstat(p)
		if err != nil {
			return err
		}
		mode := info.Mode()
		portable := mode.IsDir() || mode.IsRegular() || mode&fs.ModeSymlink != 0
		if !portable && insideGitDir(rel) {
			return nil
		}
		if d.Name() == ".git" && mode.IsRegular() {
			target, err := readGitFile(p)
			if err != nil {
				return fmt.Errorf("%w: %s: %v", runner.ErrUnsupportedGit, rel, err)
			}
			if within(real, target) {
				items = append(items, item{rel: rel, abs: p, info: info})
				return nil
			}
			if !materialize {
				return fmt.Errorf("%w: %s links to a git directory outside the workspace", runner.ErrUnsupportedGit, rel)
			}
			gitDir, err := s.materializeLinkedGit(ctx, target, tmp)
			if err != nil {
				return fmt.Errorf("%s: %w", rel, err)
			}
			sub, err := walkPlain(gitDir, rel)
			if err != nil {
				return err
			}
			items = append(items, sub...)
			return nil
		}
		if !portable {
			return fmt.Errorf("%w: %s (%s)", runner.ErrUnsupportedFileType, rel, mode.Type())
		}
		link := ""
		if mode&fs.ModeSymlink != 0 {
			if link, err = os.Readlink(p); err != nil {
				return err
			}
		}
		items = append(items, item{rel: rel, abs: p, info: info, link: link})
		return nil
	})
	if err != nil {
		return nil, err
	}
	sort.Slice(items, func(i, j int) bool { return items[i].rel < items[j].rel })
	return items, nil
}

// walkPlain lists a materialized git directory under the workspace path rel.
func walkPlain(dir, rel string) ([]item, error) {
	var out []item
	err := filepath.WalkDir(dir, func(p string, d fs.DirEntry, werr error) error {
		if werr != nil {
			return werr
		}
		info, err := os.Lstat(p)
		if err != nil {
			return err
		}
		sub, err := filepath.Rel(dir, p)
		if err != nil {
			return err
		}
		name := rel
		if sub != "." {
			name = rel + "/" + filepath.ToSlash(sub)
		}
		mode := info.Mode()
		switch {
		case mode.IsDir(), mode.IsRegular():
			out = append(out, item{rel: name, abs: p, info: info})
		case mode&fs.ModeSymlink != 0:
			l, err := os.Readlink(p)
			if err != nil {
				return err
			}
			out = append(out, item{rel: name, abs: p, info: info, link: l})
		}
		return nil
	})
	return out, err
}

func insideGitDir(rel string) bool {
	return strings.HasPrefix(rel, ".git/") || strings.Contains(rel, "/.git/")
}

func readGitFile(p string) (string, error) {
	b, err := os.ReadFile(p)
	if err != nil {
		return "", err
	}
	line := strings.TrimSpace(strings.SplitN(string(b), "\n", 2)[0])
	if !strings.HasPrefix(line, "gitdir:") {
		return "", fmt.Errorf("not a gitdir file")
	}
	target := strings.TrimSpace(strings.TrimPrefix(line, "gitdir:"))
	if !filepath.IsAbs(target) {
		target = filepath.Join(filepath.Dir(p), target)
	}
	return filepath.EvalSymlinks(filepath.Clean(target))
}

func within(root, target string) bool {
	rel, err := filepath.Rel(root, target)
	return err == nil && rel != ".." && !strings.HasPrefix(rel, ".."+string(filepath.Separator)) && !filepath.IsAbs(rel)
}

func hashFile(p string) (string, error) {
	f, err := os.Open(p)
	if err != nil {
		return "", err
	}
	defer f.Close()
	h := sha256.New()
	if _, err := io.Copy(h, f); err != nil {
		return "", err
	}
	return hex.EncodeToString(h.Sum(nil)), nil
}

// ScanEntries computes the manifest entries of a working copy with the same
// algorithm the store uses (strict: an external .git link is an error). The
// runner agent (P3 T9) uses it to diff and push.
func ScanEntries(ctx context.Context, root string) ([]runner.ManifestEntry, error) {
	s := &Store{}
	items, err := s.collect(ctx, root, false, "")
	if err != nil {
		return nil, err
	}
	out := make([]runner.ManifestEntry, 0, len(items))
	for _, it := range items {
		sum := ""
		if it.info.Mode().IsRegular() {
			if sum, err = hashFile(it.abs); err != nil {
				return nil, err
			}
		}
		e, err := runner.EntryFromInfo(it.rel, it.info, sum, it.link)
		if err != nil {
			return nil, err
		}
		out = append(out, e)
	}
	return out, nil
}

// adapter/checkpoint/git.go
package checkpoint

import (
	"context"
	"errors"
	"fmt"
	"io"
	"io/fs"
	"os"
	"path/filepath"
	"strings"

	"github.com/arbitron/arbitron/domain/runner"
	"github.com/google/uuid"
)

// materializeLinkedGit turns a linked worktree's (or --separate-git-dir)
// gitdir into a self-contained `.git` directory under tmp: the common dir
// (objects, refs, packed-refs, config, info…) minus other worktrees, overlaid
// with the worktree's own HEAD/index/logs/refs, with core.bare=false and no
// core.worktree. Repositories whose objects are not on local disk fail closed.
func (s *Store) materializeLinkedGit(ctx context.Context, gitdir, tmp string) (string, error) {
	common := gitdir
	if b, err := os.ReadFile(filepath.Join(gitdir, "commondir")); err == nil {
		c := strings.TrimSpace(string(b))
		if !filepath.IsAbs(c) {
			c = filepath.Join(gitdir, c)
		}
		common = filepath.Clean(c)
	}
	if _, err := os.Stat(filepath.Join(common, "objects", "info", "alternates")); err == nil {
		return "", fmt.Errorf("%w: repository uses objects/info/alternates", runner.ErrUnsupportedGit)
	}
	if cfg, err := os.ReadFile(filepath.Join(common, "config")); err == nil &&
		strings.Contains(strings.ToLower(string(cfg)), "partialclone") {
		return "", fmt.Errorf("%w: repository is a partial clone", runner.ErrUnsupportedGit)
	}
	dst := filepath.Join(tmp, "git-"+uuid.NewString(), ".git")
	if err := os.MkdirAll(dst, 0o700); err != nil {
		return "", err
	}
	if err := copyTree(common, dst, map[string]bool{"worktrees": true}); err != nil {
		return "", fmt.Errorf("%w: copy common dir: %v", runner.ErrUnsupportedGit, err)
	}
	if common != gitdir {
		if err := copyTree(gitdir, dst, map[string]bool{"commondir": true, "gitdir": true, "locked": true}); err != nil {
			return "", fmt.Errorf("%w: copy worktree dir: %v", runner.ErrUnsupportedGit, err)
		}
	}
	cfg := filepath.Join(dst, "config")
	if _, err := s.git(ctx, "config", "--file", cfg, "core.bare", "false"); err != nil {
		return "", fmt.Errorf("%w: %v", runner.ErrUnsupportedGit, err)
	}
	// Exit status 5 (key not set) is expected for most repositories.
	_, _ = s.git(ctx, "config", "--file", cfg, "--unset-all", "core.worktree")
	if _, err := s.git(ctx, "--git-dir", dst, "rev-parse", "--verify", "--quiet", "HEAD"); err != nil {
		return "", fmt.Errorf("%w: materialized repository has no resolvable HEAD: %v", runner.ErrUnsupportedGit, err)
	}
	return dst, nil
}

func copyTree(src, dst string, skipTop map[string]bool) error {
	return filepath.WalkDir(src, func(p string, d fs.DirEntry, werr error) error {
		if werr != nil {
			return werr
		}
		rel, err := filepath.Rel(src, p)
		if err != nil || rel == "." {
			return err
		}
		top := strings.SplitN(filepath.ToSlash(rel), "/", 2)[0]
		if skipTop[top] {
			if d.IsDir() {
				return filepath.SkipDir
			}
			return nil
		}
		if strings.HasSuffix(d.Name(), ".lock") {
			return nil
		}
		target := filepath.Join(dst, rel)
		info, err := os.Lstat(p)
		if err != nil {
			return err
		}
		mode := info.Mode()
		switch {
		case mode.IsDir():
			return os.MkdirAll(target, mode.Perm()|0o700)
		case mode.IsRegular():
			return copyFile(p, target, mode.Perm())
		case mode&fs.ModeSymlink != 0:
			l, err := os.Readlink(p)
			if err != nil {
				return err
			}
			if err := os.Remove(target); err != nil && !errors.Is(err, fs.ErrNotExist) {
				return err
			}
			return os.Symlink(l, target)
		default:
			return nil // sockets/fifos (fsmonitor) are runtime artefacts
		}
	})
}

func copyFile(src, dst string, perm fs.FileMode) error {
	if err := os.Remove(dst); err != nil && !errors.Is(err, fs.ErrNotExist) {
		return err
	}
	in, err := os.Open(src)
	if err != nil {
		return err
	}
	defer in.Close()
	out, err := os.OpenFile(dst, os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0o600)
	if err != nil {
		return err
	}
	if _, err := io.Copy(out, in); err != nil {
		out.Close()
		return err
	}
	if err := out.Close(); err != nil {
		return err
	}
	return os.Chmod(dst, perm)
}

package sqlite

import (
	"context"
	"database/sql"
	"errors"
	"fmt"
	"time"

	"github.com/arbitron/arbitron/domain/runner"
	"github.com/google/uuid"

var _ runner.CheckpointCatalog = (*CheckpointCatalogRepo)(nil)

// CheckpointCatalogRepo is the publication index of workspace checkpoints.
// Hand-written SQL (pattern: AuditRepo). Publish checks the version CAS and the
// ledger job fence in the same transaction as the insert.
type CheckpointCatalogRepo struct {
	db  *sql.DB
	now func() time.Time
}

// NewCheckpointCatalogRepo creates the repo.
func NewCheckpointCatalogRepo(db *sql.DB) *CheckpointCatalogRepo {
	return &CheckpointCatalogRepo{db: db, now: func() time.Time { return time.Now().UTC() }}
}

func (r *CheckpointCatalogRepo) Publish(ctx context.Context, ref runner.CheckpointRef, archivePath string, sizeBytes int64, f *runner.Fence) error {
	tx, err := r.db.BeginTx(ctx, nil)
	if err != nil {
		return fmt.Errorf("checkpoint catalog: begin: %w", err)
	}
	defer tx.Rollback()
	var latest int64
	if err := tx.QueryRowContext(ctx,
		`SELECT COALESCE(MAX(version), 0) FROM workspace_checkpoints WHERE run_id = ?`,
		uuidToString(ref.RunID)).Scan(&latest); err != nil {
		return fmt.Errorf("checkpoint catalog: latest: %w", err)
	}
	if ref.Version != latest+1 {
		return fmt.Errorf("%w: run %s publishing v%d over v%d", runner.ErrCheckpointConflict, ref.RunID, ref.Version, latest)
	}
	var jobID sql.NullString
	var jobGen sql.NullInt64
	if f != nil {
		if err := checkJobFenceTx(ctx, tx, *f); err != nil {
			return err
		}
		jobID = sql.NullString{String: uuidToString(f.JobID), Valid: true}
		jobGen = sql.NullInt64{Int64: f.Generation, Valid: true}
	}
	if _, err := tx.ExecContext(ctx,
		`INSERT INTO workspace_checkpoints (run_id, version, manifest_hash, archive_path, size_bytes, job_id, job_generation, created_at)
		 VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
		uuidToString(ref.RunID), ref.Version, ref.ManifestHash, archivePath, sizeBytes, jobID, jobGen, timeToString(r.now())); err != nil {
		return fmt.Errorf("checkpoint catalog: insert: %w", err)
	}
	if err := tx.Commit(); err != nil {
		return fmt.Errorf("checkpoint catalog: commit: %w", err)
	}
	return nil
}

// checkJobFenceTx is the in-transaction form of the ledger fence: the job must
// still be offered or running on f.RunnerID at f.Generation. Shared with
// StepCommitRepo (P3 T5), which additionally accepts 'completed'.
func checkJobFenceTx(ctx context.Context, tx *sql.Tx, f runner.Fence, extraStates ...string) error {
	var state string
	var runnerID sql.NullString
	var gen int64
	err := tx.QueryRowContext(ctx, `SELECT state, runner_id, generation FROM runner_jobs WHERE id = ?`,
		uuidToString(f.JobID)).Scan(&state, &runnerID, &gen)
	if errors.Is(err, sql.ErrNoRows) {
		return fmt.Errorf("%w: job %s not found", runner.ErrFenced, f.JobID)
	}
	if err != nil {
		return fmt.Errorf("fence check: %w", err)
	}
	live := state == "offered" || state == "running"
	for _, s := range extraStates {
		if state == s {
			live = true
		}
	}
	if !live || runnerID.String != string(f.RunnerID) || gen != f.Generation {
		return fmt.Errorf("%w: job %s is %s on %q gen %d, fence is %q gen %d",
			runner.ErrFenced, f.JobID, state, runnerID.String, gen, f.RunnerID, f.Generation)
	}
	return nil
}

func (r *CheckpointCatalogRepo) Latest(ctx context.Context, runID uuid.UUID) (runner.CheckpointRef, string, bool, error) {
	var v int64
	var hash, path string
	err := r.db.QueryRowContext(ctx,
		`SELECT version, manifest_hash, archive_path FROM workspace_checkpoints WHERE run_id = ? ORDER BY version DESC LIMIT 1`,
		uuidToString(runID)).Scan(&v, &hash, &path)
	if errors.Is(err, sql.ErrNoRows) {
		return runner.CheckpointRef{}, "", false, nil
	}
	if err != nil {
		return runner.CheckpointRef{}, "", false, fmt.Errorf("checkpoint catalog: latest: %w", err)
	}
	return runner.CheckpointRef{RunID: runID, Version: v, ManifestHash: hash}, path, true, nil
}

func (r *CheckpointCatalogRepo) Get(ctx context.Context, runID uuid.UUID, version int64) (runner.CheckpointRef, string, error) {
	var hash, path string
	err := r.db.QueryRowContext(ctx,
		`SELECT manifest_hash, archive_path FROM workspace_checkpoints WHERE run_id = ? AND version = ?`,
		uuidToString(runID), version).Scan(&hash, &path)
	if errors.Is(err, sql.ErrNoRows) {
		return runner.CheckpointRef{}, "", fmt.Errorf("%w: run %s v%d", runner.ErrCheckpointNotFound, runID, version)
	}
	if err != nil {
		return runner.CheckpointRef{}, "", fmt.Errorf("checkpoint catalog: get: %w", err)
	}
	return runner.CheckpointRef{RunID: runID, Version: version, ManifestHash: hash}, path, nil
}

func (r *CheckpointCatalogRepo) DeleteRun(ctx context.Context, runID uuid.UUID) error {
	if _, err := r.db.ExecContext(ctx, `DELETE FROM workspace_checkpoints WHERE run_id = ?`, uuidToString(runID)); err != nil {
		return fmt.Errorf("checkpoint catalog: delete: %w", err)
	}
	return nil
}
