# Remote Runners — Phase 1 Task 15: Runner Service and operator surface (REST + MCP)

**Goal:** Operators and users can read the pool (runners, sources, slots, assignments, events) and admins can change step limits, cordon/uncordon, and probe a source. The same rules apply on REST and MCP: reads for any authenticated user, admin-only writes, every write audited, and runs the caller cannot view never appear.

**Architecture:**
- `domain/runner/service.go:Service` owns authz (identity from context), visibility filtering (`RunVisibility`), audit (`AuditRecorder`), operator pool events (`EventRecorder`) and dispatcher wake-ups (`PoolNotifier`).
- `runner.NewInstanceVisibility` answers visibility with the engine's own `canView` predicate through the new exported `engine.CanViewInstance`.
- The adapters are thin. `adapter/mcp/tools_runner.go` (`Deps.Runners`, registered in `RegisterAllTools`) and `adapter/http/runner_handler.go` (`RouterDeps.Runners`, routes in `NewRouterFromDeps`) map the service to identical camelCase DTOs (§14).

**Tech Stack:** Go 1.25, testify, chi v5, mark3labs mcp-go.

package runner

import (
	"context"

	"github.com/arbitron/arbitron/domain/engine"
	"github.com/arbitron/arbitron/domain/instance"
	"github.com/google/uuid"
)

// RunVisibility answers whether the context identity may see a run.
type RunVisibility interface {
	CanView(ctx context.Context, runID uuid.UUID) bool
}

// NewInstanceVisibility checks runs against the instance store with the
// engine's own read predicate. Unknown runs and read errors are hidden.
func NewInstanceVisibility(repo instance.InstanceRepository) RunVisibility {
	if repo == nil {
		panic("runner: NewInstanceVisibility requires an instance repository")
	}
	return instanceVisibility{repo: repo}
}

type instanceVisibility struct{ repo instance.InstanceRepository }

func (v instanceVisibility) CanView(ctx context.Context, runID uuid.UUID) bool {
	inst, err := v.repo.Get(ctx, runID)
	if err != nil || inst == nil {
		return false
	}
	return engine.CanViewInstance(ctx, inst)
}

import (
	"context"
	"encoding/json"
	"fmt"
	"sort"
	"time"

	"github.com/arbitron/arbitron/domain/identity"

// AuditRecorder records admin actions (*audit.Service satisfies it).
type AuditRecorder interface {
	Record(ctx context.Context, action, target, detail string)
}

// ServiceDeps wires the operator-facing pool service.
type ServiceDeps struct {
	Registry   Registry
	Ledger     Ledger
	Events     EventLog
	Recorder   *EventRecorder
	Audit      AuditRecorder
	Visibility RunVisibility
	Monitor    *HealthMonitor
	Telemetry  TelemetrySource
	Notifier   *PoolNotifier
	Clock      func() time.Time
}

// RunnerView is one runner as the pool page lists it.
type RunnerView struct {
	Runner          Runner
	Sources         []Source
	SlotsUsed       int
	Queued          int
	Telemetry       *Telemetry
	HomedWorkspaces int
}

// RunnerDetail adds what is running there now.
type RunnerDetail struct {
	RunnerView
	Assignments     []AssignmentView
	RecentTelemetry []Telemetry
}

// AssignmentView is one active job; RunID nil + Hidden when the caller cannot view the run.
type AssignmentView struct {
	JobID    uuid.UUID
	RunID    *uuid.UUID
	NodeID   string
	State    JobState
	SourceID SourceID
	Hidden   bool
}

const (
	svcDefaultEventLimit = 100
	svcMaxEventLimit     = 500
)

// Service is the single authority for pool reads and admin writes; REST and MCP
// are thin adapters over it so the two surfaces cannot drift.
type Service struct{ d ServiceDeps }

// NewService validates the wiring (missing dependencies panic at boot).
func NewService(d ServiceDeps) *Service {
	if d.Registry == nil || d.Ledger == nil || d.Events == nil || d.Recorder == nil || d.Audit == nil ||
		d.Visibility == nil || d.Monitor == nil || d.Notifier == nil {
		panic("runner: NewService requires Registry, Ledger, Events, Recorder, Audit, Visibility, Monitor and Notifier")
	}
	if d.Telemetry == nil {
		d.Telemetry = EmptyTelemetry{}
	}
	if d.Clock == nil {
		d.Clock = time.Now
	}
	return &Service{d: d}
}

func svcCaller(ctx context.Context) (identity.Identity, error) {
	id, ok := identity.FromContext(ctx)
	if !ok {
		return identity.Identity{}, identity.ErrUnauthenticated
	}
	return id, nil
}

func svcAdmin(ctx context.Context) (identity.Identity, error) {
	id, err := svcCaller(ctx)
	if err != nil {
		return id, err
	}
	if !id.IsAdmin() {
		return id, identity.ErrForbidden
	}
	return id, nil
}

// List returns every runner (built-in first) to any authenticated caller.
func (s *Service) List(ctx context.Context) ([]RunnerView, error) {
	if _, err := svcCaller(ctx); err != nil {
		return nil, err
	}
	snap, active, err := s.state(ctx)
	if err != nil {
		return nil, err
	}
	tel := s.d.Telemetry.Latest()
	out := make([]RunnerView, 0, len(snap.Runners))
	for _, r := range snap.Runners {
		out = append(out, svcView(r, snap, active, tel))
	}
	sort.SliceStable(out, func(i, j int) bool {
		if out[i].Runner.BuiltIn != out[j].Runner.BuiltIn {
			return out[i].Runner.BuiltIn
		}
		return out[i].Runner.ID < out[j].Runner.ID
	})
	return out, nil
}

// Get returns one runner with its active assignments, hiding runs the caller cannot view.
func (s *Service) Get(ctx context.Context, id RunnerID) (RunnerDetail, error) {
	if _, err := svcCaller(ctx); err != nil {
		return RunnerDetail{}, err
	}
	rn, _, err := s.d.Registry.GetRunner(ctx, id)
	if err != nil {
		return RunnerDetail{}, err
	}
	snap, active, err := s.state(ctx)
	if err != nil {
		return RunnerDetail{}, err
	}
	jobs, err := s.d.Ledger.ListActiveByRunner(ctx, id)
	if err != nil {
		return RunnerDetail{}, fmt.Errorf("runner: list assignments: %w", err)
	}
	d := RunnerDetail{RunnerView: svcView(rn, snap, active, s.d.Telemetry.Latest())}
	for _, j := range jobs {
		if j.State != JobOffered && j.State != JobRunning {
			continue
		}
		a := AssignmentView{JobID: j.ID, State: j.State, SourceID: j.SourceID}
		if s.d.Visibility.CanView(ctx, j.Key.RunID) {
			run := j.Key.RunID
			a.RunID, a.NodeID = &run, j.NodeID
		} else {
			a.Hidden = true
		}
		d.Assignments = append(d.Assignments, a)
	}
	return d, nil
}

// Events returns recent pool events, newest first, dropping run-scoped events
// of runs the caller cannot view. limit <= 0 uses the default; it is capped.
func (s *Service) Events(ctx context.Context, limit int) ([]PoolEvent, error) {
	if _, err := svcCaller(ctx); err != nil {
		return nil, err
	}
	switch {
	case limit <= 0:
		limit = svcDefaultEventLimit
	case limit > svcMaxEventLimit:
		limit = svcMaxEventLimit
	}
	evs, err := s.d.Events.List(ctx, limit)
	if err != nil {
		return nil, fmt.Errorf("runner: list events: %w", err)
	}
	out := make([]PoolEvent, 0, len(evs))
	for _, ev := range evs {
		if ev.RunID != nil && !s.d.Visibility.CanView(ctx, *ev.RunID) {
			continue
		}
		out = append(out, ev)
	}
	return out, nil
}

// Update changes the step limit and/or labels (admin only, audited).
func (s *Service) Update(ctx context.Context, id RunnerID, p RunnerConfigPatch) (Runner, error) {
	who, err := svcAdmin(ctx)
	if err != nil {
		return Runner{}, err
	}
	if p.StepLimit != nil && !ValidStepLimit(*p.StepLimit) {
		return Runner{}, fmt.Errorf("%w: %d", ErrInvalidStepLimit, *p.StepLimit)
	}
	r, err := s.d.Registry.UpdateConfig(ctx, id, p)
	if err != nil {
		return Runner{}, err
	}
	detail := map[string]any{}
	if p.StepLimit != nil {
		detail["stepLimit"] = *p.StepLimit
	}
	if p.Labels != nil {
		detail["labels"] = p.Labels
	}
	s.d.Audit.Record(ctx, audit.ActionRunnerUpdate, string(id), svcAuditDetail(detail))
	s.operatorEvent(ctx, EventRunnerUpdated, id, who, detail)
	s.d.Notifier.Notify()
	return r, nil
}

// Cordon stops new placements on the runner (admin only, audited).
func (s *Service) Cordon(ctx context.Context, id RunnerID) error {
	return s.setCordon(ctx, id, true)
}

// Uncordon allows placements again (admin only, audited) and wakes waiters.
func (s *Service) Uncordon(ctx context.Context, id RunnerID) error {
	return s.setCordon(ctx, id, false)
}

func (s *Service) setCordon(ctx context.Context, id RunnerID, on bool) error {
	who, err := svcAdmin(ctx)
	if err != nil {
		return err
	}
	if err := s.d.Registry.SetCordoned(ctx, id, on); err != nil {
		return err
	}
	action, kind := audit.ActionRunnerUncordon, EventRunnerUncordoned
	if on {
		action, kind = audit.ActionRunnerCordon, EventRunnerCordoned
	}
	s.d.Audit.Record(ctx, action, string(id), svcAuditDetail(map[string]any{"cordoned": on}))
	s.operatorEvent(ctx, kind, id, who, map[string]any{"cordoned": on})
	s.d.Notifier.Notify()
	return nil
}

// ProbeSource runs the read-only probe now (admin only, audited).
func (s *Service) ProbeSource(ctx context.Context, id SourceID) (SourceHealth, error) {
	if _, err := svcAdmin(ctx); err != nil {
		return SourceHealth{}, err
	}
	h, err := s.d.Monitor.ProbeNow(ctx, id)
	if err != nil {
		return SourceHealth{}, err
	}
	s.d.Audit.Record(ctx, audit.ActionRunnerSourceProbe, string(id),
		svcAuditDetail(map[string]any{"auth": string(h.Auth), "capacity": string(h.Capacity)}))
	return h, nil
}

func (s *Service) state(ctx context.Context) (PoolSnapshot, []Job, error) {
	snap, err := s.d.Registry.Snapshot(ctx)
	if err != nil {
		return PoolSnapshot{}, nil, fmt.Errorf("runner: snapshot: %w", err)
	}
	active, err := s.d.Ledger.ListActive(ctx)
	if err != nil {
		return PoolSnapshot{}, nil, fmt.Errorf("runner: list jobs: %w", err)
	}
	return snap, active, nil
}

func (s *Service) operatorEvent(ctx context.Context, kind string, id RunnerID, who identity.Identity, detail map[string]any) {
	s.d.Recorder.Record(ctx, PoolEvent{
		ID: uuid.New(), At: s.d.Clock(), Kind: kind, RunnerID: id,
		Cause: Cause{Kind: CauseOperator, Actor: who.Username}, Detail: detail,
	})
}

func svcView(r Runner, snap PoolSnapshot, active []Job, tel map[RunnerID]Telemetry) RunnerView {
	v := RunnerView{Runner: r, SlotsUsed: snap.SlotsInUse[r.ID]}
	for _, src := range snap.Sources {
		if src.RunnerID == r.ID {
			v.Sources = append(v.Sources, src)
		}
	}
	sort.Slice(v.Sources, func(i, j int) bool { return v.Sources[i].ID < v.Sources[j].ID })
	for _, j := range active {
		if j.State == JobQueued && j.Wait != nil && j.Wait.Kind == WaitSlot && svcServes(v.Sources, j.Group) {
			v.Queued++
		}
	}
	if t, ok := tel[r.ID]; ok {
		t := t
		v.Telemetry = &t
	}
	for _, h := range snap.Homes {
		if h.RunnerID == r.ID {
			v.HomedWorkspaces++
		}
	}
	return v
}

// svcServes reports whether any of the runner's sources serves group g.
func svcServes(srcs []Source, g ModelGroup) bool {
	for _, s := range srcs {
		if s.Kind != g.Kind {
			continue
		}
		if g.ConnectionRef != "" && s.CredentialRef != g.ConnectionRef {
			continue
		}
		if g.Kind == CredNone {
			return true
		}
		for _, f := range s.Families {
			if f == g.Family {
				return true
			}
		}
	}
	return false
}

func svcAuditDetail(m map[string]any) string {
	b, err := json.Marshal(m)
	if err != nil {
		return "{}"
	}
	return string(b)
}

		return

	r.Body = http.MaxBytesReader(w, r.Body, maxRunnerBodyBytes)

		return

		writeRunnerError(w, err)
		return
	}
	WriteJSON(w, http.StatusOK, toRunnerViewResponse(runner.RunnerView{Runner: rn}))

func (h *RunnerHandler) cordon(w http.ResponseWriter, r *http.Request, on bool) {

	if !ok {
		return
	}
	var err error
	if on {
		err = h.svc.Cordon(r.Context(), runner.RunnerID(id))
	} else {
		err = h.svc.Uncordon(r.Context(), runner.RunnerID(id))
	}
	if err != nil {
		writeRunnerError(w, err)
		return
	}
	w.WriteHeader(http.StatusNoContent)

func (h *RunnerHandler) ProbeSource(w http.ResponseWriter, r *http.Request) {
	id, ok := runnerParam(w, r, "sourceId")
	if !ok {
		return
	}
	hl, err := h.svc.ProbeSource(r.Context(), runner.SourceID(id))
	if err != nil {
		writeRunnerError(w, err)
		return
	}

// runnerParam reads a URL-escaped path parameter (source IDs contain "/",
// sent as %2F; chi matches on the raw path, so the value arrives escaped).
func runnerParam(w http.ResponseWriter, r *http.Request, name string) (string, bool) {
	v, err := url.PathUnescape(chi.URLParam(r, name))
	if err != nil || v == "" {
		WriteError(w, http.StatusBadRequest, "BAD_REQUEST", "invalid "+name)
		return "", false
	}
	return v, true
}

func writeRunnerError(w http.ResponseWriter, err error) {
	switch {
	case errors.Is(err, identity.ErrUnauthenticated):
		WriteError(w, http.StatusUnauthorized, "UNAUTHENTICATED", "authentication required")
	case errors.Is(err, identity.ErrForbidden):
		WriteError(w, http.StatusForbidden, "FORBIDDEN", "admin only")
	case errors.Is(err, runner.ErrRunnerNotFound):
		WriteError(w, http.StatusNotFound, "NOT_FOUND", "runner not found")
	case errors.Is(err, runner.ErrSourceNotFound):
		WriteError(w, http.StatusNotFound, "NOT_FOUND", "source not found")
	case errors.Is(err, runner.ErrInvalidStepLimit):
		WriteError(w, http.StatusBadRequest, "INVALID_STEP_LIMIT", "stepLimit must be between 1 and 256")
	default:
		WriteError(w, http.StatusInternalServerError, "INTERNAL", err.Error())
	}
}

import (

// RunnerService is the pool service surface (*runner.Service satisfies it).
type RunnerService interface {
	List(ctx context.Context) ([]runner.RunnerView, error)
	Get(ctx context.Context, id runner.RunnerID) (runner.RunnerDetail, error)
	Events(ctx context.Context, limit int) ([]runner.PoolEvent, error)
	Update(ctx context.Context, id runner.RunnerID, p runner.RunnerConfigPatch) (runner.Runner, error)
	Cordon(ctx context.Context, id runner.RunnerID) error
	Uncordon(ctx context.Context, id runner.RunnerID) error
	ProbeSource(ctx context.Context, id runner.SourceID) (runner.SourceHealth, error)
}

// RegisterRunnerTools registers the runner-pool tools. Authz, visibility and
// audit live in runner.Service (shared with REST), so the surfaces never drift.
func RegisterRunnerTools(srv *Server) {
	srv.AddTool(gomcp.NewTool("runner_list",
		gomcp.WithDescription("List runners in the pool with their credential sources, slot use and health."),
	), runnerListHandler(srv))
	srv.AddTool(gomcp.NewTool("runner_get",
		gomcp.WithDescription("Get one runner with its active assignments (runs you cannot view are hidden)."),
		gomcp.WithString("id", gomcp.Required(), gomcp.Description("Runner id, e.g. builtin")),
	), runnerGetHandler(srv))
	srv.AddTool(gomcp.NewTool("runner_events",
		gomcp.WithDescription("Recent pool events, newest first (events of runs you cannot view are omitted)."),
		gomcp.WithNumber("limit", gomcp.Description("Max events (default 100, max 500)")),
	), runnerEventsHandler(srv))
	srv.AddTool(gomcp.NewTool("runner_update",
		gomcp.WithDescription("Admin: change a runner's step limit (1-256) and/or labels."),
		gomcp.WithString("id", gomcp.Required(), gomcp.Description("Runner id")),
		gomcp.WithNumber("stepLimit", gomcp.Description("Concurrent step limit, 1-256")),
		gomcp.WithObject("labels", gomcp.Description("Replacement label map (string values)")),
	), runnerUpdateHandler(srv))
	srv.AddTool(gomcp.NewTool("runner_cordon",
		gomcp.WithDescription("Admin: stop placing new steps on a runner."),
		gomcp.WithString("id", gomcp.Required(), gomcp.Description("Runner id")),
	), runnerCordonHandler(srv, true))
	srv.AddTool(gomcp.NewTool("runner_uncordon",
		gomcp.WithDescription("Admin: allow placements on a runner again."),
		gomcp.WithString("id", gomcp.Required(), gomcp.Description("Runner id")),
	), runnerCordonHandler(srv, false))
	srv.AddTool(gomcp.NewTool("runner_source_probe",
		gomcp.WithDescription("Admin: probe a credential source's auth/capacity now (read-only, no tokens)."),
		gomcp.WithString("sourceId", gomcp.Required(), gomcp.Description("Source id, e.g. builtin/anthropic_subscription/-")),
	), runnerProbeHandler(srv))
}

func runnerErrorResult(err error) *gomcp.CallToolResult {
	switch {
	case errors.Is(err, identity.ErrUnauthenticated):
		return gomcp.NewToolResultError("authentication required")
	case errors.Is(err, identity.ErrForbidden):
		return gomcp.NewToolResultError("admin only")
	case errors.Is(err, runner.ErrRunnerNotFound):
		return gomcp.NewToolResultError("runner not found")
	case errors.Is(err, runner.ErrSourceNotFound):
		return gomcp.NewToolResultError("source not found")
	case errors.Is(err, runner.ErrInvalidStepLimit):
		return gomcp.NewToolResultError("stepLimit must be between 1 and 256")
	default:
		return gomcp.NewToolResultError(fmt.Sprintf("runner operation failed: %s", err))
	}
}

var errRunnerUnavailable = gomcp.NewToolResultError("runner service not available")

func runnerArg(request gomcp.CallToolRequest, name string) (string, *gomcp.CallToolResult) {
	v, _ := request.GetArguments()[name].(string)
	if v == "" {
		return "", gomcp.NewToolResultError(name + " is required")
	}
	return v, nil
}

func runnerListHandler(srv *Server) server.ToolHandlerFunc {
	return func(ctx context.Context, _ gomcp.CallToolRequest) (*gomcp.CallToolResult, error) {
		if srv.deps.Runners == nil {
			return errRunnerUnavailable, nil
		}
		views, err := srv.deps.Runners.List(ctx)
		if err != nil {
			return runnerErrorResult(err), nil
		}
		out := make([]runnerViewJSON, 0, len(views))
		for _, v := range views {
			out = append(out, toRunnerViewJSON(v))
		}
		return MarshalResult(map[string]any{"runners": out})
	}
}

func runnerGetHandler(srv *Server) server.ToolHandlerFunc {
	return func(ctx context.Context, request gomcp.CallToolRequest) (*gomcp.CallToolResult, error) {
		if srv.deps.Runners == nil {
			return errRunnerUnavailable, nil
		}
		id, bad := runnerArg(request, "id")
		if bad != nil {
			return bad, nil
		}
		d, err := srv.deps.Runners.Get(ctx, runner.RunnerID(id))
		if err != nil {
			return runnerErrorResult(err), nil
		}
		return MarshalResult(toRunnerDetailJSON(d))
	}
}

func runnerEventsHandler(srv *Server) server.ToolHandlerFunc {
	return func(ctx context.Context, request gomcp.CallToolRequest) (*gomcp.CallToolResult, error) {
		if srv.deps.Runners == nil {
			return errRunnerUnavailable, nil
		}
		limit := 0
		if v, ok := request.GetArguments()["limit"].(float64); ok {
			if v < 0 || v != math.Trunc(v) {
				return gomcp.NewToolResultError("limit must be a non-negative integer"), nil
			}
			limit = int(v)
		}
		evs, err := srv.deps.Runners.Events(ctx, limit)
		if err != nil {
			return runnerErrorResult(err), nil
		}
		views := make([]runner.PoolEventView, 0, len(evs))
		for _, ev := range evs {
			views = append(views, runner.NewPoolEventView(ev))
		}
		return MarshalResult(map[string]any{"events": views})
	}

func runnerUpdateHandler(srv *Server) server.ToolHandlerFunc {
	return func(ctx context.Context, request gomcp.CallToolRequest) (*gomcp.CallToolResult, error) {
		if srv.deps.Runners == nil {
			return errRunnerUnavailable, nil
		}
		id, bad := runnerArg(request, "id")
		if bad != nil {
			return bad, nil
		}
		args := request.GetArguments()
		var p runner.RunnerConfigPatch
		if v, ok := args["stepLimit"].(float64); ok {
			if v != math.Trunc(v) {
				return gomcp.NewToolResultError("stepLimit must be an integer"), nil
			}
			n := int(v)
			p.StepLimit = &n
		}
		if raw, ok := args["labels"].(map[string]any); ok {
			p.Labels = make(map[string]string, len(raw))
			for k, v := range raw {
				s, ok := v.(string)
				if !ok {
					return gomcp.NewToolResultError("labels values must be strings"), nil
				}
				p.Labels[k] = s
			}
		}
		r, err := srv.deps.Runners.Update(ctx, runner.RunnerID(id), p)
		if err != nil {
			return runnerErrorResult(err), nil
		}
		return MarshalResult(toRunnerViewJSON(runner.RunnerView{Runner: r}))
	}
}

func runnerCordonHandler(srv *Server, on bool) server.ToolHandlerFunc {

	}
}
