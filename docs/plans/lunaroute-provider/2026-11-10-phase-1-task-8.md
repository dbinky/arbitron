**Goal:** Carry curated, evidence-referenced per-model LunaRoute facts (task kinds, compatible backends, context/output limits, supported efforts) on the already-loaded `CapabilityRules`, with every absent fact explicitly `unknown`, loaded from a new `lunaroute_models` YAML section with per-model override replacement — and prove the composition root feeds the loaded table into `ProfileService`.

**Architecture:** New `domain/provider/lunaroute_capability.go` defines `Support`, `SupportFact`, `LimitFact`, `EffortFact`, `LunaRouteModel` (+ accessors that default to unknown), the `LunaRouteCapabilities` read port, and `CapabilityRules.WithLunaRouteModels / LunaRouteModel / LunaRouteModels`. `adapter/config/capabilities.go` parses and validates the section (unknown task/backend/effort keys, non-positive limits and known facts without evidence are rejected at load). The embedded default ships the LR-02 release set: `glm-5.3`, `glm-5.3-flash`, `deepseek-4.1-flash-1017` and `claude-sonnet-5-lr1`, with doc-sourced static facts. No new root wiring path: `cmd/arbitron/main.go` already passes `capabilityRules` into `ProfileService`, `Factory` and the claude configs.

**Tech Stack:** Go 1.25, `go.yaml.in/yaml/v3`, testify.

**Spec:** spec §Model discovery ("Unknown information is explicitly marked; availability alone does not imply task or backend compatibility") · **Design:** phase-1 §3.6, §5 Deferred "Config schema"

## Global Constraints
- The table is the ONLY source of capability knowledge; `/v1/models` decides availability only.
- Facts are independent: `supported` / `unsupported` / `unknown`. An omitted backend is `unknown`, never "unsupported". A supported backend is not an allowlist.
- Every known fact carries a non-empty `evidence` reference (path/anchor into `docs/evidence/lunaroute-provider/`).
- Effort keys: `chat` or a backend (`claude-code`, `opencode`, `codex`, `pi`); levels exclude `default`.
- Overlay semantics: an override file's `lunaroute_models.<id>` fully replaces that model's embedded entry; other models keep theirs.
- **LR-02 decided** (`docs/designs/2026-11-10-lunaroute-provider-decisions.md`). The embedded section ships one entry per release model (`glm-5.3`, `glm-5.3-flash`, `deepseek-4.1-flash-1017`, `claude-sonnet-5-lr1`).
  - **Static facts** (`context_tokens`, `output_tokens`, `reasoning`, `efforts`, `request_settings`) are copied exactly from `docs/evidence/lunaroute-provider/model-capabilities.md`, which P1-T11 Step 1 writes from LunaRoute's model docs (`https://docs.lunaroute.com/models/` and per-model pages). Each fact's `evidence` is that note's anchor.
  - A fact the docs do not publish is **omitted** (unknown). There are no placeholder or "conservative" values.
  - **`tasks`/`backends`** are omitted (unknown) until a completed live run (P2-T8 chat, P3-T5 claude-code/opencode, P4-T1 codex/pi) writes `supported` with its observation path as evidence.
  - Unit tests still use override files for distinctive fixture facts.

## Review Focus
- A model entry with `context_tokens: {tokens: 0}` → load error (non-positive known limit).
- `support: unknown` with no evidence → allowed (explicit unknown).
- `support: supported` without evidence → load error.
- Unknown backend key (`backends: {cursor: …}`) → load error naming the model.
- Existing backend-only override files keep loading unchanged (no `lunaroute_models` key).

package provider

import (
	"sort"

	"github.com/arbitron/arbitron/domain/workflow"
)

// Support is one independent capability fact.
type Support string

const (
	SupportUnknown     Support = "unknown"
	SupportSupported   Support = "supported"
	SupportUnsupported Support = "unsupported"
)

// SupportFact is a support verdict plus the evidence reference that justifies a
// known verdict. A zero value reads as unknown.
type SupportFact struct {
	Support  Support
	Evidence string
}

// LimitFact is a token limit; Tokens == 0 means unknown.
type LimitFact struct {
	Tokens   int
	Evidence string
}

// Known reports whether the limit is a recorded fact.
func (l LimitFact) Known() bool { return l.Tokens > 0 }

// EffortFact lists the non-default effort levels known to be supported on one
// axis. Known=false is unknown; Known=true with no Levels means "known: none".
type EffortFact struct {
	Known    bool
	Levels   []Effort
	Evidence string
}

// LunaRouteChatAxis keys the chat-task effort fact; agent efforts are keyed by
// backend name.
const LunaRouteChatAxis = "chat"

// LunaRouteBackends is the agent-backend vocabulary the table may describe.
var LunaRouteBackends = []string{workflow.BackendClaudeCode, workflow.BackendOpencode, workflow.BackendCodex, workflow.BackendPi}

// LunaRouteModel is the curated record of known facts for one LunaRoute id.
// Absent entries are unknown; one successful task proves only that tested
// combination.
type LunaRouteModel struct {
	ID            string
	Tasks         map[ProfileKind]SupportFact
	Backends      map[string]SupportFact
	ContextTokens LimitFact
	OutputTokens  LimitFact
	Efforts       map[string]EffortFact
	Reasoning     SupportFact // independent pi metadata, not inferred from efforts
	RequestSettings map[string]SupportFact // temperature/output_schema conflicts
}

func normalized(f SupportFact, ok bool) SupportFact {
	if !ok || f.Support == "" {
		return SupportFact{Support: SupportUnknown}
	}
	return f
}

// Task returns the fact for a task kind (unknown when absent).
func (m LunaRouteModel) Task(k ProfileKind) SupportFact {
	f, ok := m.Tasks[k]
	return normalized(f, ok)
}

// Backend returns the fact for an agent backend (unknown when absent).
func (m LunaRouteModel) Backend(b string) SupportFact {
	f, ok := m.Backends[b]
	return normalized(f, ok)
}

// Effort returns the effort fact for an axis ("chat" or a backend).
func (m LunaRouteModel) Effort(axis string) EffortFact { return m.Efforts[axis] }

// LunaRouteCapabilities is the domain read port over the curated table.
type LunaRouteCapabilities interface {
	LunaRouteModel(id string) (LunaRouteModel, bool)
}

var _ LunaRouteCapabilities = CapabilityRules{}

// WithLunaRouteModels returns a copy of r carrying the LunaRoute table. The map
// is taken as-is; callers must not mutate it afterwards.
func (r CapabilityRules) WithLunaRouteModels(m map[string]LunaRouteModel) CapabilityRules {
	r.lunaroute = m
	return r
}

// LunaRouteModel looks up one model id exactly.
func (r CapabilityRules) LunaRouteModel(id string) (LunaRouteModel, bool) {
	m, ok := r.lunaroute[id]
	return m, ok
}

// LunaRouteModels returns the table sorted by id (read-only copy of the slice).
func (r CapabilityRules) LunaRouteModels() []LunaRouteModel {
	out := make([]LunaRouteModel, 0, len(r.lunaroute))
	for _, m := range r.lunaroute {
		out = append(out, m)
	}
	sort.Slice(out, func(i, j int) bool { return out[i].ID < out[j].ID })
	return out
}

func LoadCapabilityRules(overridePath string) (provider.CapabilityRules, error) {
	base, err := parseCapFile(defaultCapabilitiesYAML)
	if err != nil {
		return provider.CapabilityRules{}, fmt.Errorf("embedded capabilities: %w", err)
	}
	if overridePath != "" {
		raw, err := os.ReadFile(overridePath)
		switch {
		case err == nil:
			ov, perr := parseCapFile(raw)
			if perr != nil {
				return provider.CapabilityRules{}, fmt.Errorf("override capabilities %q: %w", overridePath, perr)
			}
			for k, v := range ov.backends {
				base.backends[k] = v // per-backend replace
			}
			for k, v := range ov.lunaroute {
				base.lunaroute[k] = v // per-model replace
			}
		case os.IsNotExist(err):
			// Missing override is not an error; the embedded default stands.
		default:
			return provider.CapabilityRules{}, fmt.Errorf("reading %q: %w", overridePath, err)
		}
	}
	return provider.NewCapabilityRules(base.backends).WithLunaRouteModels(base.lunaroute), nil
}

func parseSupport(model, field, key string, s supportYAML) (provider.SupportFact, error) {
	switch provider.Support(s.Support) {
	case provider.SupportUnknown, "":
		return provider.SupportFact{Support: provider.SupportUnknown}, nil
	case provider.SupportSupported, provider.SupportUnsupported:
		if s.Evidence == "" {
			return provider.SupportFact{}, fmt.Errorf("lunaroute_models/%s/%s/%s: a known fact needs an evidence reference", model, field, key)
		}
		return provider.SupportFact{Support: provider.Support(s.Support), Evidence: s.Evidence}, nil
	default:
		return provider.SupportFact{}, fmt.Errorf("lunaroute_models/%s/%s/%s: support must be supported, unsupported or unknown, got %q", model, field, key, s.Support)
	}
}

func parseLimit(model, field string, l *limitYAML) (provider.LimitFact, error) {
	if l == nil {
		return provider.LimitFact{}, nil
	}
	if l.Tokens <= 0 {
		return provider.LimitFact{}, fmt.Errorf("lunaroute_models/%s/%s: a known limit must be positive (omit the field when unknown)", model, field)
	}
	if l.Evidence == "" {
		return provider.LimitFact{}, fmt.Errorf("lunaroute_models/%s/%s: a known limit needs an evidence reference", model, field)
	}
	return provider.LimitFact{Tokens: l.Tokens, Evidence: l.Evidence}, nil
}

func parseLunaRouteModels(in map[string]lunaRouteModelYAML) (map[string]provider.LunaRouteModel, error) {
	out := make(map[string]provider.LunaRouteModel, len(in))
	for id, y := range in {
		m := provider.LunaRouteModel{ID: id}
		if len(y.Tasks) > 0 {
			m.Tasks = map[provider.ProfileKind]provider.SupportFact{}
			for k, s := range y.Tasks {
				kind := provider.ProfileKind(k)
				if kind != provider.ProfileKindChat && kind != provider.ProfileKindAgent {
					return nil, fmt.Errorf("lunaroute_models/%s/tasks: unknown task kind %q (chat or agent)", id, k)
				}
				f, err := parseSupport(id, "tasks", k, s)
				if err != nil {
					return nil, err
				}
				m.Tasks[kind] = f
			}
		}
		if len(y.Backends) > 0 {
			m.Backends = map[string]provider.SupportFact{}
			for b, s := range y.Backends {
				if !slices.Contains(provider.LunaRouteBackends, b) {
					return nil, fmt.Errorf("lunaroute_models/%s/backends: unknown backend %q", id, b)
				}
				f, err := parseSupport(id, "backends", b, s)
				if err != nil {
					return nil, err
				}
				m.Backends[b] = f
			}
		}
		var err error
		if m.ContextTokens, err = parseLimit(id, "context_tokens", y.ContextTokens); err != nil {
			return nil, err
		}
		if m.OutputTokens, err = parseLimit(id, "output_tokens", y.OutputTokens); err != nil {
			return nil, err
		}
		if len(y.Efforts) > 0 {
			m.Efforts = map[string]provider.EffortFact{}
			for axis, e := range y.Efforts {
				if axis != provider.LunaRouteChatAxis && !slices.Contains(provider.LunaRouteBackends, axis) {
					return nil, fmt.Errorf("lunaroute_models/%s/efforts: unknown axis %q (chat or a backend)", id, axis)
				}
				if e.Evidence == "" {
					return nil, fmt.Errorf("lunaroute_models/%s/efforts/%s: a known effort fact needs an evidence reference", id, axis)
				}
				levels := make([]provider.Effort, 0, len(e.Levels))
				for _, l := range e.Levels {
					v, perr := provider.ParseEffort(l)
					if perr != nil || v == provider.EffortDefault {
						return nil, fmt.Errorf("lunaroute_models/%s/efforts/%s: invalid level %q", id, axis, l)
					}
					levels = append(levels, v)
				}
				m.Efforts[axis] = provider.EffortFact{Known: true, Levels: levels, Evidence: e.Evidence}
			}
		}
		if m.Reasoning, err = parseSupport(id, "reasoning", "", y.Reasoning); err != nil {
			return nil, err
		}
		m.RequestSettings = map[string]provider.SupportFact{}
		for setting, value := range y.RequestSettings {
			if setting != "temperature" && setting != "output_schema" {
				return nil, fmt.Errorf("lunaroute_models/%s/request_settings: unknown setting %q", id, setting)
			}
			fact, err := parseSupport(id, "request_settings", setting, value)
			if err != nil { return nil, err }
			m.RequestSettings[setting] = fact
		}
		out[id] = m
	}
	return out, nil
}
