# LunaRoute Phase 4 · Task 5 — Wire pi on LunaRoute: factory injects the capability table, enable pi, root sink test

**Dependencies:** P4-T4; metadata injection and pi enablement land together.

**Shared contract:** Read [README.md](README.md) for execution order, production provenance and the five-category RED → GREEN → REFACTOR gate. All required coverage/provenance sections below belong to this task's RED/GREEN cycle and commit; read the complete document before Step 1.

**Goal:** Feed the composition root's loaded capability table into every pi adapter the factory builds, enable `pi` for LunaRoute routes, and prove through the real root that an `agent_call` on a LunaRoute pi profile launches `pi` with the private agent dir, a `models.json` whose limits come from the operator's capabilities file, the `lr_` key only in env — and that the dir is gone afterwards.

**Architecture:** `adapter/provider/factory.go` pi arm passes `CapabilityRules: f.deps.CapabilityRules` (already set from `capabilityRules` in `cmd/arbitron/main.go:1062`; no new root field). `domain/provider/lunaroute_validate.go:lunaRouteEnabledBackends` gains `pi` in the same commit, so a pi LunaRoute profile becomes savable only once its adapter is fed. The Phase 2 wrapper (`wrapLunaRouteAgent`) runs `CheckLunaRouteLaunch` before the pi adapter, which repeats the limits check with the injected rules.

**Tech Stack:** Go 1.25, testify, POSIX `sh` fake binary.

**Spec:** spec §Chat and agent workflows (Pi) · **Design:** phase-4 §3.3 "Policy wiring", §5 Deferred "Wiring"

## Global Constraints
- Non-LunaRoute pi construction semantics unchanged (rules are carried but unread for non-LunaRoute bindings).
- The root test feeds limits ONLY through `capabilitiespath` (the production data source) and asserts them in the file the subprocess reads (the sink); a missing injection makes the run fail closed with `ErrLunaRouteLimitsUnknown`.
- The fake `pi` exits non-zero if: the key var is wrong/missing, `OPENAI_API_KEY` leaks, the agent dir is missing or is `~/.pi/agent`, `models.json` is missing or contains the key, the argv contract differs, or stdin is empty.

## Review Focus
- After this task all four backends are enabled; any remaining test asserting "LunaRoute on pi not yet supported" must switch to `DisableLunaRouteBackendsForTest`.
- A pi LunaRoute profile for a model without known limits must be refused at `profile_create` (domain rule), not only at launch.
- The user's `~/.pi/agent/settings.json` content is unchanged after the run.

```go
package main

import (
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"testing"

	"github.com/arbitron/arbitron/domain/instance"
	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
)

const piRootKey = "lr_live_pi_root"

func TestLunaRoutePiWiring(t *testing.T) {
	dir := t.TempDir()
	argvLog, dirsLog, capture := filepath.Join(dir, "argv"), filepath.Join(dir, "dirs"), filepath.Join(dir, "models.json")
	userHome := filepath.Join(dir, "home")
	settings := filepath.Join(userHome, ".pi", "agent", "settings.json")
	require.NoError(t, os.MkdirAll(filepath.Dir(settings), 0o700))
	require.NoError(t, os.WriteFile(settings, []byte(`{"defaultProvider":"openai","defaultModel":"gpt-conflict"}`), 0o600))
	caps := filepath.Join(dir, "caps.yaml")
	require.NoError(t, os.WriteFile(caps, []byte("lunaroute_models:\n  glm-5.3:\n    context_tokens: {tokens: 202752, evidence: \"test-fixture\"}\n    output_tokens: {tokens: 32768, evidence: \"test-fixture\"}\n    reasoning: {support: unsupported, evidence: \"test-fixture#reasoning\"}\n  deepseek-4.1-flash-1017: {}\n"), 0o600))

	bin := filepath.Join(dir, "pi")
	require.NoError(t, os.WriteFile(bin, []byte(fmt.Sprintf(`#!/bin/sh
if [ "$1" = '--version' ]; then echo '0.87.0'; exit 0; fi
printf '%%s\n' "$*" >> %[1]q
printf '%%s\n' "$PI_CODING_AGENT_DIR" >> %[2]q
[ "$LUNAROUTE_API_KEY" = %[4]q ] || exit 40
[ -z "$OPENAI_API_KEY" ] || exit 41
d="$PI_CODING_AGENT_DIR"
[ -n "$d" ] && [ -d "$d" ] || exit 42
[ "$d" != "$HOME/.pi/agent" ] || exit 43
[ -f "$d/models.json" ] || exit 44
for expected in '"baseUrl": "https://gw.lunaroute.com/v1"' '"api": "openai-completions"' '"apiKey": "$LUNAROUTE_API_KEY"' '"authHeader": true' '"id": "glm-5.3"' '"contextWindow": 202752' '"maxTokens": 32768' '"reasoning": false'; do
  grep -Fq "$expected" "$d/models.json" || exit 54
done
cp "$d/models.json" %[3]q
if grep -q %[4]q "$d/models.json"; then exit 45; fi
case " $* " in *' --print --mode json --no-session --model lunaroute/glm-5.3 '*) ;; *) exit 46 ;; esac
case "$*" in *%[4]s*) exit 47 ;; esac
[ "$(cat)" = hello ] || exit 48
printf '%%s\n' '{"type":"session","id":"01a0c55a-44ef-71da-96be-a1ab13d85460"}'
printf '%%s\n' '{"type":"message_end","message":{"role":"assistant","provider":"lunaroute","model":"glm-5.3","stopReason":"stop","content":[{"type":"text","text":"done"}],"usage":{"input":1,"output":1}}}'
`, argvLog, dirsLog, capture, piRootKey)), 0o700))
	t.Setenv("PATH", dir+string(os.PathListSeparator)+os.Getenv("PATH"))
	t.Setenv("HOME", userHome)
	t.Setenv("OPENAI_API_KEY", "sk-ambient-must-not-reach")

	ds := bootSinkRoot(t, "capabilitiespath: "+caps+"\n", func(ws string) { writeAgentWorkflow(t, ws, "lr-pi", "lr-pi-agent") })
	call := mcpToolCaller(t, ds)
	ctx := adminCtxForLocalUser(t, ds)

	isErr, text := call("connection_create", map[string]any{"name": "lr", "provider": "lunaroute", "base_url": "", "api_key": piRootKey})
	require.False(t, isErr, text)

	// a model without known limits is refused at save for pi
	isErr, text = call("profile_create", map[string]any{"alias": "lr-pi-unknown", "kind": "agent", "connection_ref": "lr", "model": "deepseek-4.1-flash-1017", "backend": "pi"})
	assert.True(t, isErr)
	assert.Contains(t, text, "pi on LunaRoute needs known context and output limits")

	isErr, text = call("profile_create", map[string]any{"alias": "lr-pi-agent", "kind": "agent", "connection_ref": "lr", "model": "glm-5.3", "backend": "pi"})
	require.False(t, isErr, "pi LunaRoute profile saves once enabled and fed: %s", text)

	id, _, err := ds.eng.Trigger(ctx, "lr-pi", map[string]any{}, "normal")
	require.NoError(t, err)
	inst := waitTerminal(t, ds, id)
	if inst.State != instance.StateCompleted {
		var nerr string
		_ = ds.db.QueryRow(`SELECT COALESCE(error,'') FROM node_executions WHERE instance_id = ? AND error IS NOT NULL`, id.String()).Scan(&nerr)
		t.Fatalf("run ended %s: %s", inst.State, nerr)
	}
	assert.Equal(t, map[string]string{"lr-pi-agent": "glm-5.3"}, inst.ModelPins)

	raw, err := os.ReadFile(capture)
	require.NoError(t, err)
	var def struct {
		Providers map[string]struct {
			BaseURL string `json:"baseUrl"`
			API     string `json:"api"`
			APIKey  string `json:"apiKey"`
			Models  []struct {
				ID            string `json:"id"`
				ContextWindow int    `json:"contextWindow"`
				MaxTokens     int    `json:"maxTokens"`
				Reasoning bool `json:"reasoning"`
			} `json:"models"`
		} `json:"providers"`
	}
	require.NoError(t, json.Unmarshal(raw, &def))
	lr := def.Providers["lunaroute"]
	assert.Equal(t, "https://gw.lunaroute.com/v1", lr.BaseURL)
	assert.Equal(t, "openai-completions", lr.API)
	assert.Equal(t, "$LUNAROUTE_API_KEY", lr.APIKey, "LR-08 decided: pi env interpolation, never a literal name")
	require.Len(t, lr.Models, 1)
	assert.Equal(t, "glm-5.3", lr.Models[0].ID)
	assert.Equal(t, 202752, lr.Models[0].ContextWindow, "limits flowed capabilities file → root → factory → pi")
	assert.Equal(t, 32768, lr.Models[0].MaxTokens)
	assert.False(t, lr.Models[0].Reasoning, "independent known-false fact reached the sink")

	argv := readArgvLog(t, argvLog)
	require.Len(t, argv, 1)
	assert.NotContains(t, argv[0], piRootKey)
	dirs := readArgvLog(t, dirsLog)
	require.Len(t, dirs, 1)
	_, statErr := os.Stat(dirs[0])
	assert.True(t, os.IsNotExist(statErr), "private agent dir removed after the run")
	b, err := os.ReadFile(settings)
	require.NoError(t, err)
	assert.Contains(t, string(b), "gpt-conflict", "user ~/.pi untouched")
}
```
