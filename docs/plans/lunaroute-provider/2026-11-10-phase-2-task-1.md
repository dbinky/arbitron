# LunaRoute Phase 2 · Task 1 — LunaRoute model probe (`GET <root>/v1/models`) + no-redirect client + CLI `connection validate`

**Dependencies:** P1 complete; LR-01/LR-04 verification in P1-T11 before final protocol fixtures.

**Shared contract:** Read [README.md](README.md) for execution order, production provenance and the five-category RED → GREEN → REFACTOR gate. All required coverage/provenance sections below belong to this task's RED/GREEN cycle and commit; read the complete document before Step 1.

**Goal:** Authenticate against the hosted gateway and list available model ids through the existing `ConnectionProbe` port, with every non-2xx / transport / decode failure classified by the phase-1 LunaRoute classifier, no redirects followed with the credential, and the hosted-endpoint + `lr_` guards enforced inside the probe itself.

**Architecture:** New `adapter/probe/lunaroute.go` (`newLunaRouteProbe(d Deps) *lunaRouteProbe`) registered as the `domprovider.CanonicalLunaRoute` arm of `ProbeFactory.ProbeFor`. URL from `provider.LunaRouteOpenAIBaseURL` + `/models`; key from `provider.GuardLunaRouteReveal(b).RevealKey` (re-applied even though resolver/CLI bindings are already guarded — CLI and discovery bindings bypass the resolver). New shared helper `procenv.NoRedirectClient` (`adapter/procenv/httpx.go`) clones the injected client with `CheckRedirect → http.ErrUseLastResponse`. The listing decides availability only — no limits/capabilities are parsed. `cmd/arbitron/connection.go:runConnectionValidate` gains LunaRoute support through the factory with no code change.

**Tech Stack:** Go 1.25 `net/http`, `encoding/json`, testify.

**Spec:** spec §Model discovery, §Failure behavior · **Design:** phase-2 §3.1, §6 (ProbeSelector)

## Global Constraints
- Exactly one `/v1` in the URL: `https://gw.lunaroute.com/v1/models` for every accepted stored spelling.
- Header: `Authorization: Bearer <lr_ key>` (LR-01 candidate). No other credential header.
- Errors are `*provider.LunaRouteFailure` from `provider.ClassifyLunaRoute` with `Operation: provider.LunaRouteListModels` and `Secrets: []string{key}` — never the generic `classifyStatus`.
- A bare 404 → `ErrLunaRouteCauseUnknown` (failed listing), never "model unavailable".
- Malformed / missing `data` array / oversized body on a 2xx → `ErrLunaRouteServiceFailure` (Malformed), never an empty successful catalog.
- A non-`lr_` stored key or non-hosted base URL fails BEFORE any request is sent.
- Tests inject an `http.RoundTripper` that answers at `https://gw.lunaroute.com`; the production endpoint rule is never weakened for localhost.
- **ESCALATION LR-01:** hosted root / `/v1/models` path / Bearer header are idea-doc candidates; verified only by Phase 1 Task 11's live test.
- **ESCALATION LR-04:** the `{"data":[{"id":…}]}` envelope and error bodies are unverified offline; Phase 2 Task 8 records them. Mapping beyond HTTP status stays off until then.

## Review Focus
- 302/307 with a `Location` header: the credential must not be replayed; the response is classified (cause unknown) and the transport sees exactly one request.
- Upstream 401 body that echoes the key: the error text must not contain it.
- `{"object":"list"}` (no `data`) is malformed, not "zero models".
- `{"data":[]}` is reachable with zero models (CLI prints "reachable, no models").
- Entries with empty `id` are skipped, not listed as `""`.

// NoRedirectClient returns a shallow copy of c (a fresh client when c is nil)
// that never follows redirects: a credential-bearing request must not be
// replayed to whatever Location a server names. The redirect response itself
// is returned to the caller. c is not modified.
func NoRedirectClient(c *http.Client) *http.Client {
	var out http.Client
	if c != nil {
		out = *c
	}
	out.CheckRedirect = func(*http.Request, []*http.Request) error { return http.ErrUseLastResponse }
	return &out
}

// lrGateway answers ONLY at the hosted root and routes on path + auth header,

	reqs    []*http.Request

	body    string

func (g *lrGateway) RoundTrip(req *http.Request) (*http.Response, error) {
	g.reqs = append(g.reqs, req)
	if g.err != nil {
		return nil, g.err
	}
	if req.URL.Scheme != "https" || req.URL.Host != "gw.lunaroute.com" {
		return nil, errors.New("fake gateway: wrong host " + req.URL.String())
	}
	status, body := g.status, g.body

	h := g.header
	if h == nil {
		h = http.Header{"Content-Type": []string{"application/json"}}

package probe

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"net/http"

	"github.com/arbitron/arbitron/adapter/procenv"
	domprovider "github.com/arbitron/arbitron/domain/provider"
)

// lunaRouteProbe lists the models a LunaRoute connection can reach with
// GET <root>/v1/models. It enforces the hosted endpoint and lr_ key guards
// itself (CLI and discovery bindings bypass the profile resolver), never
// follows redirects with the credential, and classifies every failure with
// the shared LunaRoute classifier. The listing decides availability only.
type lunaRouteProbe struct {
	httpClient       *http.Client
	timeout          func(context.Context) (context.Context, context.CancelFunc)
	maxResponseBytes int64
}

var _ domprovider.ConnectionProbe = (*lunaRouteProbe)(nil)

func newLunaRouteProbe(d Deps) *lunaRouteProbe {
	return &lunaRouteProbe{
		httpClient: procenv.NoRedirectClient(d.HTTPClient),
		timeout: func(ctx context.Context) (context.Context, context.CancelFunc) {
			return procenv.WithTimeout(ctx, d.Timeout)
		},
		maxResponseBytes: d.MaxResponseBytes,
	}
}

// lunaRouteModelList is the assumed OpenAI-style envelope (UNRESOLVED
// DEPENDENCY LR-04). Data is a pointer so a missing array is detectable.
type lunaRouteModelList struct {
	Data *[]struct {
		ID string `json:"id"`
	} `json:"data"`
}

func (p *lunaRouteProbe) Probe(ctx context.Context, b domprovider.ResolvedBinding) (domprovider.ProbeResult, error) {
	ctx, cancel := p.timeout(ctx)
	defer cancel()

	base, err := domprovider.LunaRouteOpenAIBaseURL(b.BaseURL)
	if err != nil {
		return domprovider.ProbeResult{}, err
	}
	key, err := domprovider.GuardLunaRouteReveal(b).RevealKey(ctx)
	if err != nil {
		return domprovider.ProbeResult{}, err
	}
	outcome := func(status int, body string, transport, malformed error) error {
		return domprovider.ClassifyLunaRoute(domprovider.LunaRouteOutcome{
			Operation: domprovider.LunaRouteListModels, Status: status, Body: body,
			Code: domprovider.LunaRouteErrorCode(domprovider.LunaRouteListModels, body),
			Transport: transport, Malformed: malformed, Secrets: []string{key},
		})
	}
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, base+"/models", nil)
	if err != nil {
		return domprovider.ProbeResult{}, fmt.Errorf("probe: build request: %w", err)
	}
	req.Header.Set("Authorization", "Bearer "+key)
	req.Header.Set("Accept", "application/json")

	resp, err := p.httpClient.Do(req)
	if err != nil {
		return domprovider.ProbeResult{}, outcome(0, "", err, nil)
	}
	defer func() { _ = resp.Body.Close() }()
	raw, readErr := io.ReadAll(boundedReader(resp.Body, p.maxResponseBytes))
	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		return domprovider.ProbeResult{}, outcome(resp.StatusCode, string(raw), nil, nil)
	}
	if readErr != nil {
		return domprovider.ProbeResult{}, outcome(resp.StatusCode, "", nil, readErr)
	}
	var list lunaRouteModelList
	if err := json.Unmarshal(raw, &list); err != nil {
		return domprovider.ProbeResult{}, outcome(resp.StatusCode, "", nil, err)
	}
	if list.Data == nil {
		return domprovider.ProbeResult{}, outcome(resp.StatusCode, "", nil, errors.New("model listing has no data array"))
	}
	ids := make([]string, 0, len(*list.Data))
	for _, m := range *list.Data {
		if m.ID != "" {
			ids = append(ids, m.ID)
		}
	}
	return domprovider.ProbeResult{Reachable: true, Models: ids}, nil
}

package main

import (
	"bytes"
	"context"
	"io"
	"net/http"
	"strings"
	"testing"

	adaptprobe "github.com/arbitron/arbitron/adapter/probe"
	"github.com/arbitron/arbitron/domain/provider"
	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
)

type cliLRStore struct {
	provider.ConnectionStore
	conn provider.Connection
	key  string
}

func (s cliLRStore) Get(_ context.Context, name string) (provider.Connection, error) {
	if name != s.conn.Name {
		return provider.Connection{}, provider.ErrConnectionNotFound
	}
	return s.conn, nil
}
func (s cliLRStore) Reveal(context.Context, string) (string, error) { return s.key, nil }

type cliLRGateway struct{ urls []string }

func (g *cliLRGateway) RoundTrip(r *http.Request) (*http.Response, error) {
	g.urls = append(g.urls, r.URL.String())
	status, body := 200, `{"data":[{"id":"glm-5.3"},{"id":"deepseek-4.1-flash"}]}`
	if r.Header.Get("Authorization") != "Bearer lr_cli_key" || r.URL.String() != "https://gw.lunaroute.com/v1/models" {
		status, body = 401, `{"error":"bad key"}`
	}
	return &http.Response{StatusCode: status, Body: io.NopCloser(strings.NewReader(body)), Header: http.Header{}, Request: r}, nil
}

func TestRunConnectionValidate_LunaRoute(t *testing.T) {
	g := &cliLRGateway{}
	sel := adaptprobe.NewProbeFactory(adaptprobe.Deps{HTTPClient: &http.Client{Transport: g}})
	conn := provider.Connection{Name: "lr", Provider: "lunaroute", BaseURL: "https://gw.lunaroute.com"}

	var out bytes.Buffer
	require.NoError(t, runConnectionValidate(context.Background(), validateDeps{store: cliLRStore{conn: conn, key: "lr_cli_key"}, selector: sel}, "lr", &out))
	assert.Contains(t, out.String(), `connection "lr" (lunaroute): reachable. 2 model(s):`)
	assert.Contains(t, out.String(), "  - glm-5.3\n")
	assert.Equal(t, []string{"https://gw.lunaroute.com/v1/models"}, g.urls)

	out.Reset()
	err := runConnectionValidate(context.Background(), validateDeps{store: cliLRStore{conn: conn, key: "lr_wrong"}, selector: sel}, "lr", &out)
	require.ErrorIs(t, err, provider.ErrLunaRouteCredentialsRejected)
	assert.NotContains(t, err.Error(), "lr_wrong")

	err = runConnectionValidate(context.Background(), validateDeps{store: cliLRStore{conn: conn, key: "sk-or-v1-foreign"}, selector: sel}, "lr", &out)
	require.ErrorIs(t, err, provider.ErrLunaRouteStoredCredential, "CLI bindings are guarded too")
	assert.Len(t, g.urls, 2, "the foreign key was never sent")
}
