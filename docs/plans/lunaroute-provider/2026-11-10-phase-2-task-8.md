// errorCodes keeps only string code/type fields of an {"error":{...}} body —
// never messages, which may echo request data.
func errorCodes(body []byte) map[string]string {
	out := map[string]string{}
	var m map[string]any
	if json.Unmarshal(body, &m) != nil {
		return out
	}
	obj, _ := m["error"].(map[string]any)
	for _, k := range []string{"code", "type"} {
		if s, ok := obj[k].(string); ok {
			out["error."+k] = s
		}
	}
	return out
}

type rawErrorObservation struct {
	Case        string            `json:"case"`
	Path        string            `json:"path"`
	Status      int               `json:"status"`
	ContentType string            `json:"content_type"`
	BodyShape   map[string]string `json:"body_shape"`
	Codes       map[string]string `json:"codes"`
	Classified  string            `json:"classified_as"`
}

// classifiedAs labels the outcome with a short name (sentinel messages mention
// "lr_" and would be refused by writeObservation).
func classifiedAs(err error) string {
	labels := []string{"credentials_rejected", "model_unavailable", "busy", "service_failure", "cause_unknown", "request_refused"}
	for i, s := range provider.LunaRouteGatewaySentinels() {
		if errors.Is(err, s) {
			return labels[i]
		}
	}
	if err == nil {
		return "success"
	}
	return "unclassified"
}

func rawPost(t *testing.T, key, path string, body any) (int, string, []byte) {
	t.Helper()
	raw, err := json.Marshal(body)
	require.NoError(t, err)
	req, err := http.NewRequest(http.MethodPost, hostedRoot+path, bytes.NewReader(raw))
	require.NoError(t, err)
	req.Header.Set("Authorization", "Bearer "+key)
	req.Header.Set("Content-Type", "application/json")
	resp, err := liveClient().Do(req)
	require.NoError(t, err)
	defer resp.Body.Close()
	out, readErr := io.ReadAll(io.LimitReader(resp.Body, (1<<20)+1))
	require.NoError(t, readErr)
	require.LessOrEqual(t, len(out), 1<<20, "raw error observation exceeds cap")
	return resp.StatusCode, resp.Header.Get("Content-Type"), out
}

func TestLiveDiscoveryAndChat(t *testing.T) {
	key := liveKey(t)
	model := os.Getenv("ARBITRON_LUNAROUTE_E2E_CHAT_MODEL")
	if model == "" {
		if os.Getenv("ARBITRON_LUNAROUTE_RELEASE_GATE") == "1" { t.Fatal("ARBITRON_LUNAROUTE_E2E_CHAT_MODEL is required") }
		t.Skip("choose an explicit reviewed ARBITRON_LUNAROUTE_E2E_CHAT_MODEL; release acceptance remains pending")
	}
	rules, err := adapterconfig.LoadCapabilityRules(os.Getenv("ARBITRON_LUNAROUTE_E2E_CAPABILITIES"))
	require.NoError(t, err)
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Minute)
	defer cancel()

	client := &http.Client{Timeout: 120 * time.Second}
	binding := func(k, m string) provider.ResolvedBinding {
		return provider.GuardLunaRouteReveal(provider.ResolvedBinding{Kind: provider.ProfileKindChat, Provider: "lunaroute",
			BaseURL: provider.LunaRouteHostedRoot, Model: m, ConnectionRef: "live",
			RevealKey: func(context.Context) (string, error) { return k, nil }})
	}

	// Discovery through the production probe.
	probes := adaptprobe.NewProbeFactory(adaptprobe.Deps{HTTPClient: client, Timeout: 60 * time.Second, MaxResponseBytes: 8 << 20})
	probe, err := probes.ProbeFor(binding(key, ""))
	require.NoError(t, err)
	listing, err := probe.Probe(ctx, binding(key, ""))
	require.NoError(t, err)
	require.True(t, listing.Reachable)
	require.Contains(t, listing.Models, model, "the chosen chat model must be listed; no fallback model is tried")

	// Representative chat task through the production factory arm.
	factory := providersadapter.NewFactory(providersadapter.Deps{HTTPClient: client, HTTPTimeout: 120 * time.Second, MaxResponseBytes: 8 << 20, CapabilityRules: rules})
	chat, err := factory.ChatProviderFor(binding(key, model))
	require.NoError(t, err)
	started := time.Now()
	resp, err := chat.Chat(ctx, engine.ChatRequest{Prompt: "Reply with exactly the single word: pong", MaxTokens: 64})
	require.NoError(t, err, "representative chat task must finish successfully")
	output, ok := resp.Output.(string)
	require.True(t, ok, "chat adapter returns text output")
	completed := strings.Contains(strings.ToLower(output), "pong")
	assert.True(t, completed)

	// Invalid key → classified credentials through the production probe.
	invalid := "lr_" + "invalid_for_arbitron_evidence"
	_, badErr := probe.Probe(ctx, binding(invalid, ""))
	require.ErrorIs(t, badErr, provider.ErrLunaRouteCredentialsRejected)

	// Optional negative inference: a reviewed id genuinely absent from this connection.
	unknownModel := os.Getenv("ARBITRON_LUNAROUTE_E2E_UNAVAILABLE_MODEL")
	var unknownErr error
	if unknownModel != "" {
		_, err := modelidentity.NewVersionResolver().SpecificVersion(ctx, modelidentity.VersionRequest{Model: unknownModel, RouteProvider: "lunaroute"})
		require.NoError(t, err, "a local grammar refusal cannot verify a gateway error")
		require.NotContains(t, listing.Models, unknownModel)
		unknownChat, err := factory.ChatProviderFor(binding(key, unknownModel))
		require.NoError(t, err)
		_, unknownErr = unknownChat.Chat(ctx, engine.ChatRequest{Prompt: "ping", MaxTokens: 8})
		require.Error(t, unknownErr, "an unavailable model never silently succeeds")
	}

	// Raw error bodies (codes/shapes only) for LR-04.
	var raws []rawErrorObservation
	reqBad, _ := http.NewRequest(http.MethodGet, hostedRoot+"/v1/models", nil)
	reqBad.Header.Set("Authorization", "Bearer "+invalid)
	r, err := liveClient().Do(reqBad)
	require.NoError(t, err, "required raw credential observation failed")
	{
		b, readErr := io.ReadAll(io.LimitReader(r.Body, (1<<20)+1))
		require.NoError(t, readErr)
		require.LessOrEqual(t, len(b), 1<<20)
		require.Equal(t, http.StatusUnauthorized, r.StatusCode)
		r.Body.Close()
		raws = append(raws, rawErrorObservation{Case: "invalid key", Path: "/v1/models", Status: r.StatusCode, ContentType: r.Header.Get("Content-Type"), BodyShape: shape(b), Codes: errorCodes(b), Classified: classifiedAs(badErr)})
	}
	if unknownModel != "" {
		status, ct, body := rawPost(t, key, "/v1/chat/completions", map[string]any{"model": unknownModel, "messages": []map[string]string{{"role": "user", "content": "ping"}}, "max_tokens": 8})
		require.True(t, status < 200 || status >= 300, "negative inference must actually fail at the gateway")
		raws = append(raws, rawErrorObservation{Case: "unknown model", Path: "/v1/chat/completions", Status: status, ContentType: ct, BodyShape: shape(body), Codes: errorCodes(body), Classified: classifiedAs(unknownErr)})
	}


	listed := slices.Clone(listing.Models)
	slices.Sort(listed)
	writeObservation(t, key, "lr04-discovery-chat.json", map[string]any{
		"recorded_at":        time.Now().UTC().Format(time.RFC3339),
		"root":               hostedRoot,
		"listing_model_ids":  listed,
		"chat": map[string]any{
			"model":            model,
			"response_model":   resp.Model,
			"completed":        completed,
			"tokens_in":        resp.TokensIn,
			"tokens_out":       resp.TokensOut,
			"latency_ms":       time.Since(started).Milliseconds(),
			"scope":            "one chat task on this model only (LR-02 scoped observation); proves no limits and no agent backend",
		},
		"error_bodies": raws,
	})
}
