## Global Constraints
- Base URL: `provider.LunaRouteMessagesBaseURL(binding.BaseURL)` (the root; Claude Code appends `/v1/messages`). Never `/v1`.
- Key (LR-01 decided, Bearer on every sink): `ANTHROPIC_AUTH_TOKEN` (Claude Code sends `Authorization: Bearer`) through `procenv.CredentialEnv` with `KeyVar: lunaRouteAuthTokenVar`, revealed through `provider.GuardLunaRouteReveal`; empty key rejected. `ANTHROPIC_API_KEY` is stripped from the child env (ambient or inherited) and never set for a LunaRoute binding.
- Supporting selectors set to the binding model: `ANTHROPIC_MODEL`, `ANTHROPIC_DEFAULT_HAIKU_MODEL`, `ANTHROPIC_SMALL_FAST_MODEL`, `ANTHROPIC_DEFAULT_SONNET_MODEL`, `ANTHROPIC_DEFAULT_OPUS_MODEL`, `CLAUDE_CODE_SUBAGENT_MODEL`; `CLAUDE_CODE_GATEWAY_HINT_HEADERS=1`; `CLAUDE_CODE_MAX_CONTEXT_TOKENS` / `CLAUDE_CODE_MAX_OUTPUT_TOKENS` from `rules.LunaRouteModel(model)`.
- Never emitted for LunaRoute: `CLAUDE_CODE_AUTO_COMPACT_WINDOW`, `CLAUDE_CODE_DISABLE_1M_CONTEXT`.
- Effort: default → nothing; `none` → `MAX_THINKING_TOKENS=0`; graded → `CLAUDE_CODE_EFFORT_LEVEL=<level>` verbatim; only when the table's `claude-code` effort fact lists the level. `CapabilityRules.Emit` (which floors) is NOT used for LunaRoute.
- Fail-closed order (no reveal on any of these): endpoint → context tier → limits → effort → reveal.
- Every var this branch sets is stripped from the base env first, so exactly one entry each.
- **LR-05 decided** (`docs/designs/2026-11-10-lunaroute-provider-decisions.md`): the selector, hint and limit names above are the contract. Every LunaRoute claude-code launch (`AgentAdapter.RunAgent`, `StreamAgentAdapter`, `AgentAdapter.ProbeExecutedModel`) also passes `--setting-sources ""`, so no user, project or local `settings.json` can override the env. `CLAUDE.md` memory still loads, and non-LunaRoute argv is unchanged. Names live in ONE const/var block. If P3-T5 shows the installed binary (verified-on 2.1.322) ignores a selector, that selector is removed in one edit. If a helper category cannot be routed, P3-T3 enablement stops and LR-05 becomes `escalated`.
- **LR-02 decided:** limits come only from `lunaroute_models` entries (the release set's doc-sourced facts, P1-T8). A model with an unknown limit fails closed here (and at save, Phase 1 Task 9).

import (
	"context"
	"fmt"
	"slices"
	"strconv"

	"github.com/arbitron/arbitron/adapter/procenv"
	"github.com/arbitron/arbitron/domain/provider"
	"github.com/arbitron/arbitron/domain/workflow"
)

// LunaRoute Claude Code env contract — the ONE place these names live.
// LR-05 (decided 2026-11-10, docs/designs/2026-11-10-lunaroute-provider-decisions.md):
// these names plus --setting-sources "" are the contract; P3-T5 verifies main
// AND supporting calls on the installed binary and removes any ignored name here.
const (
	lunaRouteAuthTokenVar  = "ANTHROPIC_AUTH_TOKEN" // LR-01: Claude Code sends Authorization: Bearer
	lunaRouteHintVar       = "CLAUDE_CODE_GATEWAY_HINT_HEADERS"
	lunaRouteMaxContextVar = "CLAUDE_CODE_MAX_CONTEXT_TOKENS"
	lunaRouteMaxOutputVar  = "CLAUDE_CODE_MAX_OUTPUT_TOKENS"
	claudeEffortLevelVar   = "CLAUDE_CODE_EFFORT_LEVEL"
	claudeMaxThinkingVar   = "MAX_THINKING_TOKENS"
)

// lunaRouteModelSelectorVars are the main and every supporting-model selector;
// each is set to the binding's model so titles, summaries, compaction and
// subagents never ask the gateway for a model it does not serve.
var lunaRouteModelSelectorVars = []string{
	"ANTHROPIC_MODEL",
	"ANTHROPIC_DEFAULT_HAIKU_MODEL",
	"ANTHROPIC_SMALL_FAST_MODEL",
	"ANTHROPIC_DEFAULT_SONNET_MODEL",
	"ANTHROPIC_DEFAULT_OPUS_MODEL",
	"CLAUDE_CODE_SUBAGENT_MODEL",
}

// lunaRouteSuppressedVars are Claude-native context-tier switches. They are
// not model token limits and are never emitted for a LunaRoute binding.
var lunaRouteSuppressedVars = []string{"CLAUDE_CODE_AUTO_COMPACT_WINDOW", "CLAUDE_CODE_DISABLE_1M_CONTEXT"}

// lunaRouteAgentEnv builds the claude-cli env for a LunaRoute binding. Every
// refusal happens before the key is revealed; no partial env is returned.
func lunaRouteAgentEnv(ctx context.Context, binding provider.ResolvedBinding, rules provider.CapabilityRules) ([]string, error) {
	refuse := func(reason error, detail string) error {
		return &provider.LunaRouteRefusal{Reason: reason, Model: binding.Model, Backend: workflow.BackendClaudeCode, Detail: detail}
	}
	base, err := provider.LunaRouteMessagesBaseURL(binding.BaseURL)
	if err != nil {
		return nil, err
	}
	if binding.ContextWindow != "" && binding.ContextWindow != provider.ContextDefault {
		return nil, refuse(provider.ErrLunaRouteAxisUnsupported, fmt.Sprintf("context %q is a Claude-native switch, not a LunaRoute model limit; set context to default", binding.ContextWindow))
	}
	m, _ := rules.LunaRouteModel(binding.Model)
	if !m.ContextTokens.Known() || !m.OutputTokens.Known() {
		return nil, refuse(provider.ErrLunaRouteLimitsUnknown, "claude-code on LunaRoute needs known context and output limits for this model in the capability table (lunaroute_models)")
	}
	effortEnv, err := lunaRouteEffortEnv(binding.Effort, m, refuse)
	if err != nil {
		return nil, err
	}
	guarded := provider.GuardLunaRouteReveal(binding)
	env, err := procenv.CredentialEnv(ctx, procenv.CredentialSpec{
		Runner:     "claudecli",
		KeyVar:     lunaRouteAuthTokenVar, // LR-01: Bearer
		BaseURLVar: anthropicBaseURLVar,
		BaseURL:    base,
		Reveal:     guarded.RevealKey,
		EmptyKey:   procenv.EmptyKeyReject,
	})
	if err != nil {
		return nil, err
	}
	strip := slices.Concat(lunaRouteModelSelectorVars, lunaRouteSuppressedVars,
		[]string{anthropicKeyVar, lunaRouteHintVar, lunaRouteMaxContextVar, lunaRouteMaxOutputVar, claudeEffortLevelVar, claudeMaxThinkingVar})
	for _, name := range strip {
		env = procenv.RemoveVar(env, name)
	}
	for _, name := range lunaRouteModelSelectorVars {
		env = append(env, name+"="+binding.Model)
	}
	env = append(env,
		lunaRouteHintVar+"=1",
		lunaRouteMaxContextVar+"="+strconv.Itoa(m.ContextTokens.Tokens),
		lunaRouteMaxOutputVar+"="+strconv.Itoa(m.OutputTokens.Tokens),
	)
	return append(env, effortEnv...), nil
}

// lunaRouteEffortEnv emits effort exactly (claude-code sends every valid level
// verbatim) and only when the table records support for it on claude-code.
func lunaRouteEffortEnv(e provider.Effort, m provider.LunaRouteModel, refuse func(error, string) error) ([]string, error) {
	if e == "" || e == provider.EffortDefault {
		return nil, nil
	}
	if !e.Valid() {
		return nil, provider.ErrInvalidEffort
	}
	if f := m.Effort(workflow.BackendClaudeCode); !f.Known || !slices.Contains(f.Levels, e) {
		return nil, refuse(provider.ErrLunaRouteAxisUnsupported, fmt.Sprintf("effort %q is not known to be supported for this model on claude-code; set effort to default or record evidence in the capability table", e))
	}
	if e == provider.EffortNone {
		return []string{claudeMaxThinkingVar + "=0"}, nil
	}
	return []string{claudeEffortLevelVar + "=" + string(e)}, nil
}
