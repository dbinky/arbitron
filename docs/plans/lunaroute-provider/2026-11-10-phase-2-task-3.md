# LunaRoute Phase 2 · Task 3 — OpenAI-compatible chat adapter: optional error-classification and preflight hooks

**Goal:** Give `adapter/openrouter` two optional, provider-neutral seams — a structured error-classification hook (status, bounded body, transport error) applied to every immediate and streamed failure of `Chat` and `StreamChat`, and a request preflight run before any I/O — with byte-identical behaviour when both are nil.

**Architecture:** `adapter/openrouter/provider.go:Config` gains `ClassifyError func(status int, body string, transport error) error` and `Preflight func(req engine.ChatRequest) error`. `statusError` reads at most 4096 bytes of the (already bounded) body for the hook instead of parsing a formatted string; a new `transportError` routes `httpClient.Do` failures through the hook. The LunaRoute factory arm (Task 4) injects the LunaRoute classifier and launch check; no other provider sets them.

**Tech Stack:** Go 1.25, testify.

**Spec:** spec §Failure behavior · **Design:** phase-2 §3.4 "Implementation seam"

## Global Constraints
- Nil hooks: errors and requests identical to today (`ErrUpstreamStatus`, `openrouter: chat call: …`, `openrouter: stream call: …`).
- `Preflight` runs first in `Chat` and `StreamChat` — before body build, key reveal or any request.
- `ClassifyError` receives the raw (bounded, ≤4096 bytes) body; it is responsible for redaction. Its return value is returned verbatim (identity preserved).
- Streamed status errors arrive on the channel as `ChatChunk{Error: <hook result>}`; immediate transport errors are returned from `StreamChat` itself.
- Scanner/SSE parse errors mid-stream are not changed (not gateway outcomes).

## Review Focus
- A 2 MiB error body is truncated to 4096 bytes before reaching the hook.
- StreamChat preflight failure must not leak the timeout context (cancel called).
- A preflight error is returned even when the binding is pinned with a model mismatch (preflight precedes the evidence check).
- Existing openrouter tests stay green unchanged.

```go
var errHooked = errors.New("hooked")

type hookCall struct {
	status    int
	body      string
	transport error
}

func hookedProvider(ft *fakeTransport, calls *[]hookCall, preflight func(engine.ChatRequest) error) *Provider {
	return New(Config{
		Binding:    staticBinding("https://gw.example/v1", "m"),
		HTTPClient: &http.Client{Transport: ft},
		ClassifyError: func(status int, body string, transport error) error {
			*calls = append(*calls, hookCall{status, body, transport})
			return errHooked
		},
		Preflight: preflight,
	})
}

func TestClassifyError_ChatStatus(t *testing.T) {
	var calls []hookCall
	ft := &fakeTransport{resp: jsonResponse(503, `{"error":"down"}`)}
	_, err := hookedProvider(ft, &calls, nil).Chat(context.Background(), engine.ChatRequest{Prompt: "hi"})
	require.ErrorIs(t, err, errHooked)
	require.Len(t, calls, 1)
	assert.Equal(t, 503, calls[0].status)
	assert.Equal(t, `{"error":"down"}`, calls[0].body)
	assert.Nil(t, calls[0].transport)
}

func TestClassifyError_BodyBounded(t *testing.T) {
	var calls []hookCall
	ft := &fakeTransport{resp: jsonResponse(500, strings.Repeat("z", 2<<20))}
	_, err := hookedProvider(ft, &calls, nil).Chat(context.Background(), engine.ChatRequest{Prompt: "hi"})
	require.ErrorIs(t, err, errHooked)
	assert.Len(t, calls[0].body, 4096)
}

func TestClassifyError_ChatTransport(t *testing.T) {
	var calls []hookCall
	boom := errors.New("connection reset")
	ft := &fakeTransport{err: boom}
	_, err := hookedProvider(ft, &calls, nil).Chat(context.Background(), engine.ChatRequest{Prompt: "hi"})
	require.ErrorIs(t, err, errHooked)
	require.Len(t, calls, 1)
	assert.Equal(t, 0, calls[0].status)
	assert.ErrorIs(t, calls[0].transport, boom)

func TestClassifyError_StreamImmediateAndChannel(t *testing.T) {
	var calls []hookCall
	boom := errors.New("dial tcp: refused")
	_, err := hookedProvider(&fakeTransport{err: boom}, &calls, nil).StreamChat(context.Background(), engine.ChatRequest{Prompt: "hi"})
	require.ErrorIs(t, err, errHooked, "immediate transport error")

	calls = nil
	ch, err := hookedProvider(&fakeTransport{resp: sseResponse(429, `{"error":"busy"}`)}, &calls, nil).StreamChat(context.Background(), engine.ChatRequest{Prompt: "hi"})
	require.NoError(t, err)
	var chunkErr error
	for c := range ch {
		if c.Error != nil {
			chunkErr = c.Error
		}
	}
	require.ErrorIs(t, chunkErr, errHooked, "status error on the channel")
	require.Len(t, calls, 1)
	assert.Equal(t, 429, calls[0].status)

func TestPreflight_BlocksBeforeAnyIO(t *testing.T) {
	var calls []hookCall
	refused := errors.New("refused by preflight")
	var seen engine.ChatRequest
	pf := func(r engine.ChatRequest) error { seen = r; return refused }
	ft := &fakeTransport{resp: jsonResponse(200, `{}`)}
	p := hookedProvider(ft, &calls, pf)
	_, err := p.Chat(context.Background(), engine.ChatRequest{Prompt: "hi", MaxTokens: 77})
	require.ErrorIs(t, err, refused)
	assert.Equal(t, 77, seen.MaxTokens, "preflight sees the request's settings")

	require.ErrorIs(t, err, refused)
	assert.Nil(t, ft.gotReq, "no request was built or sent")
```

```go
// classifyBodyMax bounds the error body handed to ClassifyError.
const classifyBodyMax = 4096

// transportError maps an httpClient.Do failure through the classification hook
// when one is configured.
func (p *Provider) transportError(op string, err error) error {
	if p.classify != nil {
		return p.classify(0, "", err)
	}
	return fmt.Errorf("openrouter: %s call: %w", op, err)
}
```
Replace `statusError` with:
```go
// statusError reads the bounded error body and either hands it to the
// classification hook or wraps ErrUpstreamStatus with a control-scrubbed
// snippet. It never inspects or logs the request's key.
func (p *Provider) statusError(status int, body io.Reader) error {
	if p.classify != nil {
		raw, _ := io.ReadAll(io.LimitReader(body, classifyBodyMax))
		return p.classify(status, string(raw), nil)
	}
	return fmt.Errorf("%w: status=%d body=%s", ErrUpstreamStatus, status, procenv.StatusSnippet(body))
}
```
