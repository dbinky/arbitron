# LunaRoute Phase 1 · Task 4 — Connection write seam: repo enforcement, guarded update, MCP mapping, root wiring

**Dependencies:** P1-T3.

**Shared contract:** Read [README.md](README.md) for execution order, production provenance and the five-category RED → GREEN → REFACTOR gate. All required coverage/provenance sections below belong to this task's RED/GREEN cycle and commit; read the complete document before Step 1.

**Goal:** Enforce the Task-3 credential rule at the single connection write seam (`ConnectionRepo.Create/Update`), make LunaRoute metadata-only updates race-safe against concurrent provider switches, map the write sentinels to actionable MCP text, and prove it through the production MCP server built by the composition root.

**Architecture:** `ConnectionRepo.Create` calls `provider.PrepareLunaRouteCreate` after `ValidateConnection`; `Update` keeps the `GetConnectionMeta` row it already reads (today discarded) and passes `row.Provider` to `provider.PrepareLunaRouteUpdate`. A LunaRoute metadata-only update uses a new sqlc query `UpdateConnectionMetadataIfProvider` (`WHERE name = ? AND provider = ?`) so a provider switch between read and write fails with `ErrConnectionChangedConcurrently` instead of reusing a foreign key. `adapter/mcp/tools_connection.go:connErrorResult` maps the sentinels; `connection_create` echoes the stored (normalized) row for LunaRoute.

**Tech Stack:** Go 1.25, SQLite, sqlc v1.31.1 (`~/go/bin/sqlc` or `go run github.com/sqlc-dev/sqlc/cmd/sqlc@v1.31.1`), mcp-go, testify.

**Spec:** spec §Connection setup, §Failure behavior · **Design:** phase-1 §3.3, §4 (connection write), §5 Deferred "Repo change"

## Global Constraints
- The plaintext key is inspected only for shape; never logged, never echoed.
- Non-LunaRoute create/update behaviour (including `rows == 0 → ErrConnectionNotFound`) is unchanged.
- Validation order on update: `ValidateConnection` → existence read (`GetConnectionMeta`) → `PrepareLunaRouteUpdate` → `ValidateConnectionURL` → write.
- No schema/migration change.

## Review Focus
- Concurrent provider switch between the meta read and the metadata write (tested with a test-only hook).
- Update of a non-existent LunaRoute connection must still return `ErrConnectionNotFound`, not a key error.
- MCP `connection_create` error text for a foreign key must not contain the key.
- MCP response for a LunaRoute create shows the normalized root even when `base_url` was empty.
- `connection_update` with `api_key: ""` means "omitted" (existing handler behaviour) → switch-to-lunaroute without key still rejected.

func (r *ConnectionRepo) Update(ctx context.Context, c provider.Connection, apiKey *string) error {
	if err := provider.ValidateConnection(c); err != nil {
		return err
	}
	// Pre-check existence so a missing connection maps to the sentinel. The row
	// is KEPT: its stored provider decides whether a LunaRoute update may keep
	// the sealed key.
	stored, err := r.queries.GetConnectionMeta(ctx, c.Name)
	if err != nil {
		if errors.Is(err, sql.ErrNoRows) {
			return fmt.Errorf("connection %q: %w", c.Name, provider.ErrConnectionNotFound)
		}
		return fmt.Errorf("checking connection %q: %w", c.Name, err)
	}
	if r.afterMetaRead != nil {
		r.afterMetaRead()
	}
	c, err = provider.PrepareLunaRouteUpdate(c, apiKey, stored.Provider)
	if err != nil {
		return err
	}

	// SECURITY (SSRF / credential exfiltration): re-validate the BaseURL on every
	// update so a connection cannot be repointed at a non-https or internal host.
	if err := provider.ValidateConnectionURL(c, r.urlPolicy); err != nil {
		return err
	}

	extrasJSON, err := extrasToJSON(c.Extras)
	if err != nil {
		return fmt.Errorf("marshalling extras: %w", err)
	}

	if apiKey == nil {
		if provider.IsLunaRoute(c.Provider) {
			// Keep the stored key only if the row still has the provider we
			// validated against: a concurrent switch must not let a foreign key
			// ride along under provider=lunaroute.
			rows, err := r.queries.UpdateConnectionMetadataIfProvider(ctx, generated.UpdateConnectionMetadataIfProviderParams{
				Provider:   c.Provider,
				BaseURL:    c.BaseURL,
				Extras:     extrasJSON,
				Name:       c.Name,
				Provider_2: stored.Provider,
			})
			if err != nil {
				return fmt.Errorf("updating connection %q: %w", c.Name, err)
			}
			if rows == 0 {
				return r.missingOrChanged(ctx, c.Name)
			}
			return nil
		}
		rows, err := r.queries.UpdateConnectionMetadata(ctx, generated.UpdateConnectionMetadataParams{
			Provider: c.Provider,
			BaseURL:  c.BaseURL,
			Extras:   extrasJSON,
			Name:     c.Name,
		})
		if err != nil {
			return fmt.Errorf("updating connection %q: %w", c.Name, err)
		}
		if rows == 0 {
			return fmt.Errorf("connection %q: %w", c.Name, provider.ErrConnectionNotFound)
		}
		return nil
	}

	// *apiKey is consumed only by Seal here; never logged or placed elsewhere.
	cipher, nonce, err := r.crypto.Seal([]byte(*apiKey))
	if err != nil {
		return fmt.Errorf("sealing api key for connection %q: %w", c.Name, err)
	}
	rows, err := r.queries.UpdateConnectionWithKey(ctx, generated.UpdateConnectionWithKeyParams{
		Provider:     c.Provider,
		BaseURL:      c.BaseURL,
		Extras:       extrasJSON,
		APIKeyCipher: cipher,
		APIKeyNonce:  nonce,
		Name:         c.Name,
	})
	if err != nil {
		return fmt.Errorf("updating connection %q: %w", c.Name, err)
	}
	if rows == 0 {
		return fmt.Errorf("connection %q: %w", c.Name, provider.ErrConnectionNotFound)
	}
	return nil
}

// missingOrChanged explains a guarded update that matched no row: the
// connection was deleted (not found) or changed under us (retry).
func (r *ConnectionRepo) missingOrChanged(ctx context.Context, name string) error {
	if _, err := r.queries.GetConnectionMeta(ctx, name); err != nil {
		if errors.Is(err, sql.ErrNoRows) {
			return fmt.Errorf("connection %q: %w", name, provider.ErrConnectionNotFound)
		}
		return fmt.Errorf("checking connection %q: %w", name, err)
	}
	return fmt.Errorf("connection %q: %w", name, provider.ErrConnectionChangedConcurrently)
}

	t.Helper()
	srv := adaptmcp.NewServerWithDeps("root", "test", ds.mcpDeps())
	adaptmcp.RegisterAllTools(srv)
	return func(name string, args map[string]any) (bool, string) {
		t.Helper()
		req, err := json.Marshal(map[string]any{"jsonrpc": "2.0", "id": 1, "method": "tools/call", "params": map[string]any{"name": name, "arguments": args}})

		data, err := json.Marshal(srv.MCPServer().HandleMessage(ctx, req))

			Result gomcp.CallToolResult `json:"result"`
			Error  any                  `json:"error"`

		text := ""

				text = tc.Text
			}
		}
