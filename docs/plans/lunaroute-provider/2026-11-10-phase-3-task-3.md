const lrRootKey = "lr_live_root_key"

// agentListingGateway is the in-test LunaRoute gateway behind the provider
// transport. It honours the load-bearing inputs: only GET
// https://gw.lunaroute.com/v1/models with "Bearer <lrRootKey>" is answered with
// the listing; any other URL is 404 and any other credential 401.
type agentListingGateway struct {
	mu       sync.Mutex
	models   []string
	requests []string // "METHOD URL auth=ok|bad" — never the key itself
}

func (g *agentListingGateway) RoundTrip(r *http.Request) (*http.Response, error) {
	auth := "bad"
	if r.Header.Get("Authorization") == "Bearer "+lrRootKey {
		auth = "ok"
	}
	g.mu.Lock()
	g.requests = append(g.requests, r.Method+" "+r.URL.String()+" auth="+auth)
	models := append([]string(nil), g.models...)
	g.mu.Unlock()
	reply := func(status int, body string) (*http.Response, error) {
		return &http.Response{StatusCode: status, Header: http.Header{"Content-Type": {"application/json"}},
			Body: io.NopCloser(strings.NewReader(body)), Request: r}, nil
	}
	if r.Method != http.MethodGet || r.URL.String() != "https://gw.lunaroute.com/v1/models" {
		return reply(404, `{"error":"not found"}`)
	}
	if auth != "ok" {
		return reply(401, `{"error":"invalid key"}`)
	}
	data := make([]map[string]string, 0, len(models))
	for _, m := range models {
		data = append(data, map[string]string{"id": m})
	}
	body, _ := json.Marshal(map[string]any{"object": "list", "data": data})
	return reply(200, string(body))
}

func (g *agentListingGateway) seen() []string {
	g.mu.Lock()
	defer g.mu.Unlock()
	return append([]string(nil), g.requests...)
}

// writeLunaRouteCaps writes a capabilities override with FIXTURE limits for

func TestLunaRouteAgentProfileSaveWiring(t *testing.T) {
	gw := &agentListingGateway{models: []string{"glm-5.3"}}
	ds := bootLunaRouteRoot(t, "capabilitiespath: "+writeLunaRouteCaps(t)+"\n", gw, func(string) {})
	call := mcpToolCaller(t, ds)
	isErr, text := call("connection_create", map[string]any{"name": "lr", "provider": "lunaroute", "base_url": "", "api_key": lrRootKey})
	require.False(t, isErr, text)

	save := func(alias string, extra map[string]any) (bool, string) {
		args := map[string]any{"alias": alias, "kind": "agent", "connection_ref": "lr"}
		for k, v := range extra {
			args[k] = v
		}
		return call("profile_create", args)
	}
	isErr, text = save("lr-oc", map[string]any{"model": "glm-5.3", "backend": "opencode"})
	require.False(t, isErr, text)
	isErr, text = save("lr-cc", map[string]any{"model": "glm-5.3", "backend": "claude-code"})
	require.False(t, isErr, text)

	isErr, text = save("lr-cc-unknown", map[string]any{"model": "deepseek-4.1-flash-1017", "backend": "claude-code"})
	assert.True(t, isErr)
	assert.Contains(t, text, "claude-code on LunaRoute needs known context and output limits")
	isErr, text = save("lr-cc-200k", map[string]any{"model": "glm-5.3", "backend": "claude-code", "context_window": "200k"})
	assert.True(t, isErr)
	assert.Contains(t, text, "Claude-native switch")
	isErr, text = save("lr-oc-effort", map[string]any{"model": "glm-5.3", "backend": "opencode", "effort": "low"})
	assert.True(t, isErr)
	assert.Contains(t, text, "no exact emission on opencode")
	for _, b := range []string{"pi", "codex"} {
		isErr, text = save("lr-"+b, map[string]any{"model": "glm-5.3", "backend": b})
		assert.True(t, isErr)
		assert.Contains(t, text, "LunaRoute on "+b+" not yet supported")
	}
	assert.NotContains(t, fmt.Sprint(gw.seen()), lrRootKey)
	assert.Empty(t, gw.seen(), "profile save performs no gateway I/O")
}

// fakeLunaRouteClaude fails with a distinct exit code unless the composition
// root delivered every LunaRoute env var exactly; a FAIL file makes it emulate
// an opaque upstream failure whose stderr echoes the key.
func fakeLunaRouteClaude(t *testing.T, dir string) (bin, argvLog, failFlag string) {
	t.Helper()
	bin = filepath.Join(dir, "fake-claude.sh")
	argvLog = filepath.Join(dir, "argv.log")
	failFlag = filepath.Join(dir, "FAIL")
	script := strings.NewReplacer("@ARGV@", argvLog, "@FAIL@", failFlag, "@KEY@", lrRootKey).Replace(`#!/bin/sh
input=$(cat)
[ "$input" = hello ] || exit 54
printf '%s\n' "$*" >> '@ARGV@'
check() { eval "v=\${$1-__unset__}"; if [ "$v" != "$2" ]; then echo "env mismatch $1" >&2; exit "$3"; fi; }
check ANTHROPIC_BASE_URL https://gw.lunaroute.com 41
check ANTHROPIC_AUTH_TOKEN '@KEY@' 42
check ANTHROPIC_API_KEY __unset__ 52
prev=; iso=0; for a in "$@"; do if [ "$prev" = "--setting-sources" ] && [ -z "$a" ]; then iso=1; fi; prev=$a; done; [ "$iso" = 1 ] || exit 53
for v in ANTHROPIC_MODEL ANTHROPIC_DEFAULT_HAIKU_MODEL ANTHROPIC_SMALL_FAST_MODEL ANTHROPIC_DEFAULT_SONNET_MODEL ANTHROPIC_DEFAULT_OPUS_MODEL CLAUDE_CODE_SUBAGENT_MODEL; do check "$v" glm-5.3 43; done
check CLAUDE_CODE_GATEWAY_HINT_HEADERS 1 44
check CLAUDE_CODE_MAX_CONTEXT_TOKENS 202752 45
check CLAUDE_CODE_MAX_OUTPUT_TOKENS 32768 46
check CLAUDE_CODE_AUTO_COMPACT_WINDOW __unset__ 47
check CLAUDE_CODE_DISABLE_1M_CONTEXT __unset__ 48
case " $* " in *" --model glm-5.3 "*) ;; *) exit 49 ;; esac
case "$*" in *'@KEY@'*) exit 50 ;; esac
if [ -f '@FAIL@' ]; then echo "upstream rejected key @KEY@" >&2; exit 1; fi
echo '{"type":"result","is_error":false,"result":"lunaroute ok","session_id":"s","usage":{"input_tokens":1,"output_tokens":1},"modelUsage":{"glm-5.3":{}}}'
`)
	require.NoError(t, os.WriteFile(bin, []byte(script), 0o700))
	return bin, argvLog, failFlag
}

func TestLunaRouteClaudeCodeAgentSinkWiring(t *testing.T) {
	bin, argvLog, failFlag := fakeLunaRouteClaude(t, t.TempDir())
	gw := &agentListingGateway{models: []string{"glm-5.3"}}
	cfg := "capabilitiespath: " + writeLunaRouteCaps(t) + "\nllm:\n  binarypath: \"" + bin + "\"\n"
	ds := bootLunaRouteRoot(t, cfg, gw, func(ws string) {
		writeWorkflow(t, ws, "lr-cc", `{"name":"lr-cc","nodes":[{"id":"work","type":"agent_call","entrypoint":true,"config":{"model_profile":"lr-cc","templateFile":"templates/p.md"}}],"edges":[]}`)
		writePinTemplate(t, ws, "lr-cc")
	})
	ctx := adminCtxForLocalUser(t, ds)
	require.NoError(t, ds.connSvc.Create(ctx, provider.Connection{Name: "lr", Provider: "lunaroute"}, lrRootKey))
	_, _, err := ds.profileSvc.Create(ctx, provider.Profile{Alias: "lr-cc", Kind: provider.ProfileKindAgent, Backend: "claude-code", ConnectionRef: "lr", Model: "glm-5.3"})
	require.NoError(t, err)

	id, _, err := ds.eng.Trigger(ctx, "lr-cc", nil, "normal")
	require.NoError(t, err)
	inst := waitTerminal(t, ds, id)
	require.Equal(t, instance.StateCompleted, inst.State, "subprocess rejected the env: %s", nodeError(t, ds, id))
	assert.Equal(t, map[string]string{"lr-cc": "glm-5.3"}, inst.ModelPins)
	require.Len(t, readArgvLog(t, argvLog), 1)
	assert.Empty(t, gw.seen(), "success is a pure pass-through: no diagnosis probe")

	// (d) opaque failure → one bounded post-failure observation, no key anywhere.
	require.NoError(t, os.WriteFile(failFlag, nil, 0o600))

	require.NoError(t, err)
	require.Equal(t, instance.StateFailed, waitTerminal(t, ds, id2).State)
	nerr := nodeError(t, ds, id2)
	require.NotEmpty(t, nerr)
	assert.NotContains(t, nerr, lrRootKey)
	assert.NotContains(t, nerr, "lr_live")
	assert.Equal(t, []string{"GET https://gw.lunaroute.com/v1/models auth=ok"}, gw.seen(),
		"exactly one diagnosis listing, through the guarded binding's lr_ key")
}
