# LunaRoute Phase 1 · Task 5 — Guarded reveal at every binding producer

**Dependencies:** P1-T4.

**Shared contract:** Read [README.md](README.md) for execution order, production provenance and the five-category RED → GREEN → REFACTOR gate. All required coverage/provenance sections below belong to this task's RED/GREEN cycle and commit; read the complete document before Step 1.

**Goal:** Wrap the lazy `RevealKey` of every LunaRoute binding so that, before any adapter can place a key in env/config/headers, the hosted endpoint is re-checked and a revealed key that is not a valid `lr_` key fails closed — and apply it at all three production binding producers.

**Architecture:** `provider.GuardLunaRouteReveal(b ResolvedBinding) ResolvedBinding` (new `domain/provider/lunaroute_reveal.go`) is a no-op for non-LunaRoute bindings. It is applied in `adapter/provider/dynamic.go:DynamicProfileResolver.Resolve` (execution bindings), `adapter/provider/path_inventory.go:routeBinding` (inventory/probe bindings) and `cmd/arbitron/connection.go:bindingFromConnection` (CLI `connection validate`). Construction stays lazy: nothing is revealed until an adapter calls `RevealKey`.

**Tech Stack:** Go 1.25, testify.

**Spec:** spec §Connection setup ("Credentials remain protected… rejected rather than forwarded") · **Design:** phase-1 §3.4, §5 Deferred "Binding producers"

## Global Constraints
- Non-LunaRoute bindings are returned byte-identical (same closure).
- Reveal errors propagate without partial values; the guard's own errors carry no key material.
- The `routeBinding` codex no-op reveal is NOT changed here (Phase 4 Task 3 narrows it); a LunaRoute+codex inventory binding therefore reveals `""` and the guard turns that into `ErrLunaRouteStoredCredential` — fail closed until Phase 4.
- Wiring test must drive both production producers through the composition root (`initDomain`), not construct them directly.

## Review Focus
- Stored connection whose provider was flipped to `lunaroute` out-of-band (legacy row / manual DB edit) while holding an `sk-…` key.
- Stored LunaRoute row whose base URL was tampered to a non-hosted host: reveal must fail before the store is asked for the key.
- Nil `RevealKey` on a LunaRoute binding (hand-built binding) must error, not panic and not return `""`.
- Context cancellation from the inner reveal must still satisfy `errors.Is(err, context.Canceled)`.

`domain/provider/lunaroute_reveal.go`:
```go
package provider

import "context"

// GuardLunaRouteReveal wraps a LunaRoute binding's lazy RevealKey so that every
// reveal first re-checks the hosted endpoint, then fails closed on a revealed
// value that is not a valid lr_ key. It reveals nothing itself; construction
// stays lazy. Non-LunaRoute bindings are returned unchanged.
func GuardLunaRouteReveal(b ResolvedBinding) ResolvedBinding {
	if !IsLunaRoute(b.Provider) {
		return b
	}
	inner := b.RevealKey
	baseURL := b.BaseURL
	b.RevealKey = func(ctx context.Context) (string, error) {
		if _, err := NormalizeLunaRouteBaseURL(baseURL); err != nil {
			return "", err
		}
		if inner == nil {
			return "", ErrLunaRouteStoredCredential
		}
		key, err := inner(ctx)
		if err != nil {
			return "", LocalLunaRouteError("reveal", err, key)
		}
		if !ValidLunaRouteKey(key) {
			return "", ErrLunaRouteStoredCredential
		}
		return key, nil
	}
	return b
}
```

```go
package main

import (
	"context"
	"testing"

	providersadapter "github.com/arbitron/arbitron/adapter/provider"
	adaptsqlite "github.com/arbitron/arbitron/adapter/sqlite"
	"github.com/arbitron/arbitron/domain/provider"
	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
)

// Both production binding producers built by the composition root guard a
// LunaRoute reveal: the execution resolver (ds.profileResolver) and the
// execution-path inventory (ds.pathInventory). The stored rows are tampered
// out-of-band (the write seam would refuse them) to model legacy/foreign data.
func TestLunaRouteRevealGuardWiring(t *testing.T) {
	ds := bootSinkRoot(t, "", func(string) {})
	ctx := adminCtxForLocalUser(t, ds)

	// A foreign key smuggled under provider=lunaroute.
	require.NoError(t, ds.connSvc.Create(ctx, provider.Connection{Name: "lr-foreign", Provider: "openrouter", BaseURL: "https://gw.lunaroute.com"}, "sk-or-v1-foreign"))
	_, err := ds.db.ExecContext(ctx, `UPDATE connections SET provider = 'lunaroute' WHERE name = 'lr-foreign'`)
	require.NoError(t, err)
	// A valid LunaRoute row whose endpoint was tampered.
	require.NoError(t, ds.connSvc.Create(ctx, provider.Connection{Name: "lr-moved", Provider: "lunaroute"}, "lr_live_moved"))
	_, err = ds.db.ExecContext(ctx, `UPDATE connections SET base_url = 'https://evil.example/v1' WHERE name = 'lr-moved'`)
	require.NoError(t, err)
	// A good row.
	require.NoError(t, ds.connSvc.Create(ctx, provider.Connection{Name: "lr-good", Provider: "lunaroute"}, "lr_live_good"))

	// Profiles are inserted below the service (phase 1 enables no LunaRoute
	// backend, so the service would refuse them); the resolver only reads.
	repo := adaptsqlite.NewProfileRepo(ds.db)
	for alias, conn := range map[string]string{"p-foreign": "lr-foreign", "p-moved": "lr-moved", "p-good": "lr-good"} {
		require.NoError(t, repo.Create(ctx, provider.Profile{Alias: alias, Kind: provider.ProfileKindAgent, Backend: "opencode", ConnectionRef: conn, Model: "glm-5.3"}))
	}
	resolver, ok := ds.profileResolver.(*providersadapter.DynamicProfileResolver)
	require.True(t, ok, "root wires the dynamic resolver")
	require.NoError(t, resolver.Reload(ctx))

	reveal := func(alias string) (string, error) {
		b, err := ds.profileResolver.Resolve(ctx, alias)
		require.NoError(t, err, "resolution itself stays lazy and succeeds")
		return b.RevealKey(ctx)
	}
	key, err := reveal("p-foreign")
	require.ErrorIs(t, err, provider.ErrLunaRouteStoredCredential)
	assert.Empty(t, key)
	assert.NotContains(t, err.Error(), "sk-or-v1-foreign")
	_, err = reveal("p-moved")
	require.ErrorIs(t, err, provider.ErrLunaRouteEndpoint)
	key, err = reveal("p-good")
	require.NoError(t, err)
	assert.Equal(t, "lr_live_good", key)

	snap, err := ds.pathInventory.Snapshot(ctx)
	require.NoError(t, err)
	seen := map[string]bool{}
	for _, p := range snap.Paths {
		switch p.ConnectionRef {
		case "lr-foreign":
			_, err := p.Binding.RevealKey(ctx)
			require.ErrorIs(t, err, provider.ErrLunaRouteStoredCredential)
			seen[p.ConnectionRef] = true
		case "lr-good":
			k, err := p.Binding.RevealKey(ctx)
			require.NoError(t, err)
			assert.Equal(t, "lr_live_good", k)
			seen[p.ConnectionRef] = true
		}
	}
	assert.True(t, seen["lr-foreign"] && seen["lr-good"], "inventory produced LunaRoute paths through routeBinding")
}
```
