# opencode 1.17.x Adapter Modernization Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Update `adapter/opencode` from the opencode 0.4.x CLI contract to the 1.17.x contract so agent-kind bindings (starting with `coder-ds4`: `ollama-cloud`/`deepseek-v4-flash`) run successfully.

**Architecture:** Per mission, the adapter emits `opencode run <prompt> -m <opencodeId>/<model> --dangerously-skip-permissions --format json`, sets only credential env var(s), and injects one generated `OPENCODE_CONFIG_CONTENT` JSON (a `provider` block when a custom/no-built-in endpoint applies, plus a `permission` block from the posture). Secrets stay in env and are referenced from the JSON via `{env:VAR}`.

**Tech Stack:** Go, `adapter/opencode`, the shared `adapter/claudecli` `CommandRunner` seam + fake, `domain/provider`, `adapter/procenv`. Reference spec: `docs/superpowers/specs/2026-06-15-opencode-1.17-adapter-modernization-design.md`.

```go
package opencode

import "github.com/arbitron/arbitron/domain/provider"

// providerDescriptor is the per-provider opencode 1.17.x contract: the `-m`
// prefix, the credential env var carrying the secret (empty for keyless/aws),
// the npm package + whether a config `provider` block is always required, and
// the AWS-quartet flag for bedrock. It is the single opencode-side vocabulary
// shared by args.go (`-m`), env.go (credential env), and config.go (provider block).
type providerDescriptor struct {
	opencodeID   string // `-m <opencodeID>/<model>` prefix
	credEnvVar   string // env var holding the secret key ("" => keyless or aws)
	npmPackage   string // npm package when a provider{} block is emitted
	alwaysConfig bool   // emit a provider{} block even without a custom BaseURL
	keyless      bool   // may launch without a key (local ollama)
	aws          bool   // bedrock AWS-quartet credential model
}

// providerDescriptors is keyed by the DOMAIN-CANONICAL provider id
// (provider.Canonicalize output): the alias collapse (anthropic/anthropic-api,
// bedrock/amazon-bedrock, the azure trio) lives once in Canonicalize, so this
// table lists canonical ids only. Note "ollama-cloud" is canonically DISTINCT
// from "ollama" (url_policy loopback boundary) and is listed explicitly.
var providerDescriptors = map[string]providerDescriptor{
	provider.CanonicalAnthropicAPI: {opencodeID: "anthropic", credEnvVar: "ANTHROPIC_API_KEY", npmPackage: "@ai-sdk/anthropic"},
	"openai":                       {opencodeID: "openai", credEnvVar: "OPENAI_API_KEY", npmPackage: "@ai-sdk/openai"},
	"litellm":                      {opencodeID: "litellm", credEnvVar: "OPENAI_API_KEY", npmPackage: "@ai-sdk/openai-compatible", alwaysConfig: true},
	"openrouter":                   {opencodeID: "openrouter", credEnvVar: "OPENROUTER_API_KEY", npmPackage: "@openrouter/ai-sdk-provider"},
	provider.CanonicalAzure:        {opencodeID: "azure", credEnvVar: "AZURE_API_KEY", npmPackage: "@ai-sdk/azure", alwaysConfig: true},
	provider.CanonicalBedrock:      {opencodeID: "amazon-bedrock", aws: true},
	"ollama":                       {opencodeID: "ollama", npmPackage: "@ai-sdk/openai-compatible", alwaysConfig: true, keyless: true},
	"ollama-cloud":                 {opencodeID: "ollama-cloud", credEnvVar: "OLLAMA_API_KEY"},
}

// descriptorFor canonicalizes an arbitron provider string and returns its
// opencode 1.17.x descriptor. ok is false for an unknown/empty provider so
// callers fail closed.
func descriptorFor(arbitronProvider string) (providerDescriptor, bool) {
	d, ok := providerDescriptors[provider.Canonicalize(arbitronProvider)]
	return d, ok
}
```

Replace the body of `opencodeEnv` so it resolves the descriptor and sets ONLY the credential var (no base-url var). Keep the AWS-quartet (`bedrockEnv`) and keyless paths. Keep `ErrUnknownProvider`, `ErrBedrockConfig`, `bedrockEnv`, `rejectBaseNames`.

```go
func opencodeEnv(ctx context.Context, binding provider.ResolvedBinding) ([]string, error) {
	d, ok := descriptorFor(binding.Provider)
	if !ok {
		return nil, fmt.Errorf("%w: %q", ErrUnknownProvider, binding.Provider)
	}
	if d.aws {
		return bedrockEnv(ctx, binding)
	}
	// Credential key only. The base URL is delivered via the generated
	// OPENCODE_CONFIG_CONTENT (config.go), NEVER an env var — opencode 1.17.x
	// ignores per-provider *_BASE_URL on its routing path.
	emptyKey := procenv.EmptyKeyReject
	if d.keyless {
		emptyKey = procenv.EmptyKeyAllowBaseURLOnly
	}
	if d.credEnvVar == "" && d.keyless {
		// Keyless (local ollama): no credential var; base env only.
		return procenv.BaseEnv(), nil
	}
	return procenv.CredentialEnv(ctx, procenv.CredentialSpec{
		Runner:   fmt.Sprintf("opencode: provider %q", binding.Provider),
		KeyVar:   d.credEnvVar,
		BaseURLVar: "", // never set a base-url env var on 1.17.x
		BaseURL:  "",
		Reveal:   binding.RevealKey,
		EmptyKey: emptyKey,
	})
}
```

```go
package opencode

import "strings"

// claudeToOpencodeTool maps Claude-style tool names (used in arbitron profile
// AllowedTools) to opencode 1.17.x tool names. Names with no 1.17.x equivalent
// (MultiEdit, TodoRead) are absent and dropped by mapTools.
var claudeToOpencodeTool = map[string]string{
	"read":      "read",
	"write":     "write",
	"edit":      "edit",
	"grep":      "grep",
	"glob":      "glob",
	"bash":      "bash",
	"ls":        "list",
	"webfetch":  "webfetch",
	"websearch": "websearch",
	"todowrite": "todowrite",
	"task":      "task",
}

// mapTools translates an arbitron AllowedTools list to opencode tool names,
// preserving order and de-casing. Unknown names are returned in `dropped` so the
// caller can log them once (never silently swallow a requested tool).
func mapTools(allowed []string) (mapped []string, dropped []string) {
	for _, a := range allowed {
		key := strings.ToLower(strings.TrimSpace(a))
		if oc, ok := claudeToOpencodeTool[key]; ok {
			mapped = append(mapped, oc)
		} else {
			dropped = append(dropped, a)
		}
	}
	return mapped, dropped
}
```

```go
package opencode

import (
	"encoding/json"
	"fmt"

	"github.com/arbitron/arbitron/domain/provider"
)

const opencodeConfigSchema = "https://opencode.ai/config.json"

// ErrNoEffectiveTools is returned when a scoped AllowedTools list maps to zero
// opencode tools — a tool-less run is fail-closed, never silently launched.
var ErrNoEffectiveTools = fmt.Errorf("opencode: scoped allowlist maps to no opencode tools")

type opencodeConfig struct {
	Schema     string                    `json:"$schema"`
	Provider   map[string]providerBlock  `json:"provider,omitempty"`
	Permission map[string]string         `json:"permission,omitempty"`
}

type providerBlock struct {
	NPM     string         `json:"npm,omitempty"`
	Name    string         `json:"name,omitempty"`
	Options map[string]any `json:"options,omitempty"`
}

// buildConfig generates the OPENCODE_CONFIG_CONTENT JSON for one run: a
// permission block from the posture (with tool names mapped) and, when the
// provider has no built-in or carries a custom BaseURL, a provider block whose
// apiKey references the credential env var via {env:VAR} (secret stays in env).
func buildConfig(binding provider.ResolvedBinding, post Posture) (string, error) {
	d, ok := descriptorFor(binding.Provider)
	if !ok {
		return "", fmt.Errorf("%w: %q", ErrUnknownProvider, binding.Provider)
	}

	cfg := opencodeConfig{Schema: opencodeConfigSchema}

	// Permission block.
	switch {
	case len(post.AllowedTools) > 0:
		mapped, dropped := mapTools(post.AllowedTools)
		if len(dropped) > 0 {
			// Visible, not silent. slog is fine; keep it INFO-level.
			logDroppedTools(binding.Provider, dropped)
		}
		if len(mapped) == 0 {
			return "", ErrNoEffectiveTools
		}
		perm := map[string]string{"*": "deny"}
		for _, tool := range mapped {
			perm[tool] = "allow"
		}
		cfg.Permission = perm
	case post.AllowAll:
		cfg.Permission = map[string]string{"*": "allow"}
	default:
		// Closed default: omit permission; opencode's own default applies under
		// --dangerously-skip-permissions.
	}

	// Provider block (only when needed).
	if d.alwaysConfig || binding.BaseURL != "" {
		opts := map[string]any{}
		if binding.BaseURL != "" {
			opts["baseURL"] = binding.BaseURL
		}
		if d.credEnvVar != "" {
			opts["apiKey"] = fmt.Sprintf("{env:%s}", d.credEnvVar)
		}
		// azure structured options.
		if d.opencodeID == "azure" {
			if rn := binding.Extras["resource-name"]; rn != "" {
				opts["resourceName"] = rn
			}
			if av := binding.Extras["api-version"]; av != "" {
				opts["apiVersion"] = av
			}
		}
		// Do NOT emit an empty options object that says nothing.
		if len(opts) > 0 || d.npmPackage != "" {
			cfg.Provider = map[string]providerBlock{
				d.opencodeID: {NPM: d.npmPackage, Options: opts},
			}
		}
	}

	out, err := json.Marshal(cfg)
	if err != nil {
		return "", fmt.Errorf("opencode: marshal config: %w", err)
	}
	return string(out), nil
}
```

```go
// parseOutput folds opencode's `--format json` NDJSON stream into an AgentResult.
// Non-JSON lines (human log output) are skipped. Text events accumulate into the
// summary; an error event is surfaced into the summary and a step so failures are
// visible. Tolerant: junk/empty yields an empty result, never a panic.
func parseOutput(lines []string) (summary string, steps []agentStep, tokensOut int) {
	var text strings.Builder
	for _, l := range lines {
		s := strings.TrimSpace(l)
		if s == "" || s[0] != '{' {
			continue
		}
		var ev ocEvent
		if err := json.Unmarshal([]byte(s), &ev); err != nil {
			continue
		}
		switch ev.Type {
		case "text": // adjust to captured assistant-text type(s)
			if ev.Text != "" {
				text.WriteString(ev.Text)
				steps = append(steps, agentStep{Kind: "assistant", Summary: ev.Text})
			}
		case "error":
			if ev.Error != nil {
				msg := ev.Error.Data.Message
				steps = append(steps, agentStep{Kind: "error", Summary: msg})
				if text.Len() == 0 {
					text.WriteString(msg)
				}
			}
		}
		// if ev.Tokens != nil { tokensOut += ev.Tokens.Output }
	}
	return text.String(), steps, tokensOut
```
